הַשְּׁלָטִים הַסּוֹדִיִּים שֶׁל תָּמָר

תָּמָר נוֹסַעַת עִם אַבָּא לְתֵל אָבִיב, וּמְגַלָּה עוֹלָם שָׁלֵם שֶׁל רָאשֵׁי תֵּבוֹת וְקִצּוּרִים

📖 קְרִיאָה

בְּבֹקֶר יוֹם שִׁשִּׁי תָּמָר וְאַבָּא שֶׁלָּהּ עָלוּ עַל הָרַכֶּבֶת לְבַקֵּר אֶת הַדּוֹד אָסָף, שֶׁעוֹבֵד כְּרוֹפֵא יְלָדִים בְּבֵית חוֹלִים בְּתֵל אָבִיב. תָּמָר הִתְרַגְּשָׁה מְאֹד, כִּי זֶה הָיָה הַמַּסָּע הָרִאשׁוֹן שֶׁלָּהּ בָּרַכֶּבֶת לְבַד עִם אַבָּא.

כְּשֶׁהָרַכֶּבֶת הִתְקָרְבָה לַתַּחֲנָה, תָּמָר רָאֲתָה שֶׁלֶט גָּדוֹל שֶׁכָּתוּב עָלָיו: "תַּחֲנַת רַכֶּבֶת — תֵּל אָבִיב הַשָּׁלוֹם". מִתַּחַת לַשֵּׁם, בְּאוֹתִיּוֹת קְטַנּוֹת, הָיָה כָּתוּב גַּם ת״א. "אַבָּא, מָה זֶה ת״א?" שָׁאֲלָה תָּמָר בְּתִמָּהוֹן.

אַבָּא חִיֵּךְ וְהֵסְבִּיר: "זֶה רָאשֵׁי תֵּבוֹת — אוֹתִיּוֹת רִאשׁוֹנוֹת שֶׁל כַּמָּה מִלִּים, שֶׁבָּאוֹת בִּמְקוֹם הַמִּלִּים הַמְּלֵאוֹת. ת״א הוּא קִצּוּר שֶׁל 'תֵּל אָבִיב'. שִׂים לֵב — יֵשׁ גֵּרְשַׁיִם לִפְנֵי הָאוֹת הָאַחֲרוֹנָה."

הֵם יָרְדוּ מֵהָרַכֶּבֶת וְהָלְכוּ בָּרְחוֹב. תָּמָר רָאֲתָה שֶׁלֶט עִם הַכְּתוֹבֶת "בֵּית סֵפֶר יְסוֹדִי — הַדֶּקֶל", וְלְצִדּוֹ שֶׁלֶט קָטָן יוֹתֵר שֶׁכָּתוּב עָלָיו בֵּי״ס הַדֶּקֶל. "גַּם זֶה רָאשֵׁי תֵּבוֹת?" שָׁאֲלָה תָּמָר בְּגַאֲוָה, כְּאִלּוּ הֵבִינָה אֶת הַסּוֹד. "נָכוֹן!" אָמַר אַבָּא, "בֵּי״ס הוּא קִצּוּר שֶׁל 'בֵּית סֵפֶר'."

בְּדֶרֶךְ אֲגָב, עָבְרוּ לְיַד שֶׁלֶט רְחוֹב שֶׁכָּתוּב עָלָיו רְח׳ הֶרְצְל 14. "וְזֶה?" שָׁאֲלָה תָּמָר. "זֶה לֹא בְּדִיּוּק כְּמוֹ ת״א וּבֵי״ס," אָמַר אַבָּא. "רְח׳ הוּא קִצּוּר שֶׁל מִלָּה אַחַת — 'רְחוֹב'. קִצּוּר שֶׁל מִלָּה אַחַת מְסֻמָּן בְּגֶרֶש אֶחָד, וְלֹא בִּגְרְשַׁיִם."

הֵם הִגִּיעוּ לְבֵית הַחוֹלִים, וּבַכְּנִיסָה הָיָה תָּלוּי שֶׁלֶט: ד״ר אָסָף לֵוִי — רוֹפֵא יְלָדִים. תָּמָר הִתְפַּלְּאָה: "ד״ר זֶה גַּם רָאשֵׁי תֵּבוֹת? אֲבָל זוֹ רַק מִלָּה אַחַת — דּוֹקְטוֹר!" אַבָּא צָחַק: "אַתְּ צוֹדֶקֶת, זֶה יוֹצֵא דֹּפֶן. ד״ר הוּא קִצּוּר שֶׁל מִלָּה אַחַת, אֲבָל מִסִּבָּה הִיסְטוֹרִית הוּא נִכְתָּב בִּגְרְשַׁיִם, כְּמוֹ תְּאָרִים רְשׁמִיִּים אֲחֵרִים."

הַדּוֹד אָסָף קִבֵּל אוֹתָם בְּחִיּוּךְ גָּדוֹל. בְּדֶלֶת הַמִּשְׂרָד שֶׁלּוֹ הָיָה תָּלוּי גַּם שֶׁלֶט עַל חַיָּלִים שֶׁבָּאוּ לְבַקֵּר חוֹלִים: חַיָּלֵי צה״ל מְבַקְּרִים הַיּוֹם. תָּמָר זָכְרָה אֶת הַכְּלָל וְנִסְּתָה לְבַדֵּהּ: "צה״ל זֶה רָאשֵׁי תֵּבוֹת שֶׁל... צָבָא הֲגָנָה לְיִשְׂרָאֵל?" "מְצֻיָּן!" קָרָא אַבָּא בְּהִתְלַהֲבוּת.

בַּדֶּרֶךְ חֲזָרָה לַתַּחֲנָה, עָבְרוּ לְיַד סְנִיף דֹּאַר, וְעַל קֻפְסָה קְטַנָּה הָיָה כָּתוּב ת.ד. 120. "מָה זֶה עִם הַנְּקֻדּוֹת?" שָׁאֲלָה תָּמָר. "זֶה ת.ד. — תֵּבַת דֹּאַר," הֵסְבִּיר אַבָּא. "יֵשׁ גַּם קִצּוּרִים שֶׁמְּסֻמָּנִים בִּנְקֻדּוֹת בִּמְקוֹם גֶּרֶשׁ."

תָּמָר חָשְׁבָה לְרֶגַע וְאָמְרָה: "אִם כָּךְ, רָאשֵׁי תֵּבוֹת בָּאִים מִכַּמָּה מִלִּים וּמְסֻמָּנִים בִּגְרְשַׁיִם, וְקִצּוּר שֶׁל מִלָּה אַחַת מְסֻמָּן בְּגֶרֶש — אֲבָל יֵשׁ גַּם יוֹצְאֵי דֹּפֶן כְּמוֹ ד״ר." אַבָּא הִנְהֵן בְּרֹאשׁוֹ בְּהַעֲרָכָה: "בְּדִיּוּק כָּכָה. אַתְּ מַתְחִילָה לְהָבִין אֶת הַכְּלָל."

בָּרַכֶּבֶת בַּחֲזָרָה הַבַּיְתָה, תָּמָר פָּתְחָה מַחְבֶּרֶת קְטַנָּה וְהֵחֵלָּה לִכְתֹּב בָּהּ אֶת כָּל הַקִּצּוּרִים שֶׁפָּגְשָׁה בַּיּוֹם: ת״א, בֵּי״ס, רְח׳, ד״ר, צה״ל, ת.ד. "אֲנִי הוֹלֶכֶת לִהְיוֹת בַּלָּשִׁית שֶׁל רָאשֵׁי תֵּבוֹת," אָמְרָה בְּחִיּוּךְ. "מֵעַתָּה אֲחַפֵּשׂ קִצּוּרִים בְּכָל שֶׁלֶט בָּעִיר!"

📚 דִּקְדּוּק — רָאשֵׁי תֵּבוֹת וְקִצּוּרִים

מָתַי גֶּרֶשׁ וּמָתַי גֵּרְשַׁיִם?

רָאשֵׁי תֵּבוֹת נוֹצָרִים מֵאוֹתִיּוֹת רִאשׁוֹנוֹת שֶׁל כַּמָּה מִלִּים, וּמְסֻמָּנִים בְּגֵּרְשַׁיִם (״) לִפְנֵי הָאוֹת הָאַחֲרוֹנָה — לְמָשָׁל ת״א (תֵּל אָבִיב). קִצּוּר הוּא חֵלֶק מִמִּלָּה אַחַת שֶׁהוּשְׁמַט סוֹפָהּ, וּמְסֻמָּן בְּגֶּרֶשׁ (׳) אֶחָד — לְמָשָׁל רְח׳ (רְחוֹב).

כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים שְׁלֵמִים. בְּכָל מִשְׁפָּט הִשְׁתַּמְּשׁוּ בְּקִצּוּר אוֹ בְּרָאשֵׁי תֵּבוֹת אֶחָד מֵהַשִּׁיעוּר, כְּמוֹ בַּדֻּגְמָה: עִיר הָאָבִיב הִיא תֵּל אָבִיב, וּבְקִצּוּר קוֹרְאִים לָהּ ת״א.
