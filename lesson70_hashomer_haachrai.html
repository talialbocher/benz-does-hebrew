📖 קְרִיאָה

בְּיוֹם חֲמִישִׁי בָּעֶרֶב, אַבָּא וְאִמָּא הִתְכּוֹנְנוּ לָצֵאת לַחֲתֻנָּה שֶׁל דּוֹדָה רוּת.

"תּוֹמֶר, אַתָּה כְּבָר בֶּן עֶשֶׂר וְיוֹדֵעַ לִשְׁמֹר עַל נֹגָה לְשָׁעָה אוֹ שְׁתַּיִם," אָמְרָה אִמָּא. "אֲנִי בּוֹטַחַת בְּךָ."

לִפְנֵי שֶׁיָּצְאוּ, אִמָּא נָתְנָה לְתוֹמֶר וּלְנֹגָה הוֹרָאוֹת בְּטִיחוּת: "אַל תִּפְתְּחוּ אֶת הַדֶּלֶת לְאִישׁ בְּלִי שֶׁאֲדַבֵּר אֲלֵיכֶם בַּטֵּלֵפוֹן קֹדֶם."

"וְעוֹד דָּבָר," הוֹסִיף אַבָּא, "אַל תַּדְלִיקוּ אֶת הַתַּנּוּר, וְאַל תְּשַׂחֲקוּ בְּגַפְרוּרִים אֲפִלּוּ לֹא לְרֶגַע."

"וּאַל תֵּצְאוּ לַחָצֵר בְּלִי לְהוֹדִיעַ לִי," סִיְּמָה אִמָּא וְחִבְּקָה אֶת שְׁנֵיהֶם.

תּוֹמֶר חִיֵּךְ וְהִבְטִיח: "לֹא אֶפְתַּח אֶת הַדֶּלֶת לְאִישׁ, אֲנַחְנוּ לֹא נַדְלִיק אֶת הַתַּנּוּר, וְגַם לֹא נֵצֵא מֵהַבַּיִת. אַתֶּם יְכוֹלִים לִסְמֹךְ עָלַי."

גַּם נֹגָה הִצְטָרְפָה בְּקוֹל רָצִינִי: "גַּם אֲנִי לֹא אֶפְתַּח לְאִישׁ!"

אַחֲרֵי שֶׁאַבָּא וְאִמָּא יָצְאוּ, תּוֹמֶר וְנֹגָה שִׂחֲקוּ בַּמִּשְׂחָקִים שֶׁלָּהֶם וְצִיְּרוּ תְּמוּנוֹת יָפוֹת בַּסָּלוֹן.

לְפֶתַע, פַּעֲמוֹן הַדֶּלֶת צִלְצֵל. נֹגָה קָפְצָה מִיָּד: "בּוֹא נִפְתַּח, אוּלַי זֶה אַבָּא וְאִמָּא שֶׁשָּׁכְחוּ מַשֶּׁהוּ!"

"אַל תִּפְתְּחִי, נֹגָה!" אָמַר תּוֹמֶר בְּקוֹל בָּטוּחַ. "קֹדֶם נִבְדֹּק מִי זֶה."

הֵם הִסְתַּכְּלוּ מִבַּעַד לַחַלּוֹן וְרָאוּ אֶת הַשְּׁכֵנָה, גְּבֶרֶת כֹּהֵן, מַחֲזִיקָה אֶת הַכַּדּוּר הָאָדֹם שֶׁל נֹגָה.

תּוֹמֶר לֹא פָּתַח אֶת הַדֶּלֶת — הוּא הִתְקַשֵּׁר קֹדֶם לְאִמָּא. "זוֹ גְּבֶרֶת כֹּהֵן, הִיא מַחֲזִירָה לְנֹגָה אֶת הַכַּדּוּר," הוּא הִסְבִּיר.

"מְצֻיָּן, תּוֹמֶר! אַתֶּם יְכוֹלִים לִפְתֹּחַ לָהּ," אָמְרָה אִמָּא בַּטֵּלֵפוֹן.

כְּשֶׁאַבָּא וְאִמָּא חָזְרוּ מֵהַחֲתֻנָּה, נֹגָה סִפְּרָה לָהֶם בְּהִתְרַגְּשׁוּת עַל הַפַּעֲמוֹן וְעַל הַכַּדּוּר.

אִמָּא חִבְּקָה אֶת תּוֹמֶר וְאָמְרָה: "אֲנִי כָּל כָּךְ גֵּאָה בְּךָ! עָשִׂית בְּדִיּוּק כְּמוֹ שֶׁבִּקַּשְׁנוּ — שָׁמַרְתָּ עַל הַהוֹרָאוֹת וְהָיִית אַחְרָאִי מְאֹד."

💡 שִׂימוּ לֵב: מִלַּת "אַל" מוֹפִיעָה כְּשֶׁמַּזְהִירִים אוֹ אוֹסְרִים עַל מִישֶׁהוּ לַעֲשׂוֹת דָּבָר מָה (אַזְהָרָה אוֹ פְּקֻדָּה), וּמִלַּת "לֹא" מוֹפִיעָה כְּשֶׁמַּצְהִירִים שֶׁמַּשֶּׁהוּ לֹא יִקְרֶה. שְׁתֵּי הַמִּלִּים בָּאוֹת לִפְנֵי פֹּעַל בִּזְמַן עָתִיד — לֹא לִפְנֵי צוּרַת הַצִּוּוּי!

📚 דִּקְדּוּק — שְׁלִילַת צִוּוּי וְעָתִיד

"אַל" וְ"לֹא" — שְׁתֵּי מִלִּים שׁוֹנוֹת לִשְׁלִילָה!

צִוּוּי חִיּוּבִי (פְּקֻדָּה) נִבְנֶה מִצּוּרַת הַצִּוּוּי. אֲבָל כְּשֶׁרוֹצִים לוֹמַר "אַל תַּעֲשֶׂה אֶת זֶה!" — אֵין שׁוֹלְלִים אֶת צוּרַת הַצִּוּוּי. בִּמְקוֹם זֶה אוֹמְרִים "אַל" וּמוֹסִיפִים אֶת הַפֹּעַל בִּזְמַן עָתִיד:

הַפֹּעַלצִוּוּי חִיּוּבִישְׁלִילַת צִוּוּי (אַל + עָתִיד)
לִפְתֹּחַפְּתַח!אַל תִּפְתַּח!
לָלֶכֶתלֵךְ!אַל תֵּלֵךְ!
לָרוּץרוּץ!אַל תָּרוּץ!

וְיֵש עוֹד צוּרַת שְׁלִילָה: "לֹא" + עָתִיד. הִיא לֹא פְּקֻדָּה אֶלָּא הַצְהָרָה עַל מַה שֶּׁלֹּא יִקְרֶה — לְמָשָׁל: אֲנִי לֹא אֵלֵךְ, הוּא לֹא יָבוֹא, אֲנַחְנוּ לֹא נֵצֵא.

🔑 הַטְּרִיק: גַּם "אַל" וְגַם "לֹא" מַגִּיעִים לִפְנֵי צוּרַת הֶעָתִיד (תִּפְתַּח, תֵּלֵךְ, נֵצֵא) — וְלֹא לִפְנֵי צוּרַת הַצִּוּוּי (פְּתַח, לֵךְ)!
⚠️ הֶבְדֵּל חָשׁוּב: "אַל" הִיא בְּדֶרֶךְ כְּלָל אַזְהָרָה אוֹ פְּקֻדָּה לְמִישֶׁהוּ שֶׁעוֹמְדִים לְדַבֵּר אֵלָיו (אַתָּה / אַתְּ / אַתֶּם). "לֹא" הִיא הַצְהָרָה כְּלָלִית עַל מַה שֶּׁיִּהְיֶה אוֹ לֹא יִהְיֶה, וִיכוֹלָה לְהוֹפִיעַ בְּכָל גּוּף — גַּם כְּשֶׁמְּדַבְּרִים עַל מִישֶׁהוּ אַחֵר.
