מַחְבֶּרֶת רָאשֵׁי הַתֵּבוֹת

כִּתָּה ה׳ נוֹסַעַת לְתֵל אָבִיב וְלוֹמֶדֶת לִקְרֹא אֶת הַשְּׁלָטִים בָּעִיר

📖 קְרִיאָה

בְּיוֹם רִאשׁוֹן בַּבֹּקֶר עָלְתָה כִּתָּה ה׳ עַל אוֹטוֹבּוּס גָּדוֹל לְטִיּוּל בְּתֵל אָבִיב, וְכֻלָּם הִתְרַגְּשׁוּ מְאֹד.

הַמַּדְרִיךְ, אִישׁ נָעִים בְּשֵׁם דָּנִי, עָמַד מוּל הַכִּתָּה וְהֵרִים שֶׁלֶט עִם הַכְּתֻבָּה "ת״א — הָעִיר שֶׁלֹּא נִרְדָּמָת".

מַאיָה הֵרִימָה אֶת הַיָּד וְשָׁאֲלָה: "מָה זֶה ת״א? יֵשׁ שָׁם סִימָן מוּזָר בְּתוֹךְ הַמִּלָּה!"

דָּנִי חִיֵּךְ וְהֶסְבִּיר: "ת״א הוּא רֹאשׁ תֵּבוֹת — אוֹתִיּוֹת רִאשׁוֹנוֹת שֶׁל שְׁתֵּי מִלִּים: תֵּל אָבִיב. כְּשֶׁלּוֹקְחִים אוֹת רִאשׁוֹנָה מִכָּל מִלָּה בְּצֵרוּף, שָׁמִים גֵּרְשַׁיִם לִפְנֵי הָאוֹת הָאַחֲרוֹנָה."

הַכִּתָּה יָרְדָה מֵהָאוֹטוֹבּוּס לְיַד בִּנְיָן גָּדוֹל, וְעַל הַשַּׁעַר הָיָה כָּתוּב "בֵּית הַחַיָּל — צה״ל".

"צה״ל," אָמַר דָּנִי, "הוּא רֹאשׁ תֵּבוֹת שֶׁל צְבָא הַגַּנָּה לְיִשְׂרָאֵל. שָׁלוֹשׁ מִלִּים — צָבָא, הַגַּנָּה, יִשְׂרָאֵל — הִתְקַצְּרוּ לְאַרְבַּע אוֹתִיּוֹת."

בְּדֶרֶךְ הֵם רָאוּ אַמְבּוּלַנְס לָבָן עִם כִּתּוּב אָדֹם — "מד״א" — וּמַאיָה כָּתְבָה בִּמְהִירוּת בַּמַּחְבֶּרֶת שֶׁלָּהּ: מָגֵן דָּוִד אָדֹם.

כְּשֶׁעָבְרוּ לְיַד בִּנְיָן הַמּוּזֵיאוֹן, מַאיָה שָׂמָה לֵב לְשֶׁלֶט קָטָן: "עַמ׳ 3 — כְּנִיסָה לְתַצוּגָה". "זֶה דּוֹמֶה לְקֹדֶם, אֲבָל יֵשׁ כָּאן רַק סִימָן אֶחָד," הִיא אָמְרָה.

דָּנִי הֶסְבִּיר שֶׁזֶּה סוּג שׁוֹנֶה: "עַמ׳ הוּא קִצּוּר שֶׁל מִלָּה אַחַת — עַמּוּד. כְּשֶׁמְּקַצְּרִים מִלָּה אַחַת וּמַשְׁאִירִים רַק חֵלֶק מִמֶּנָּה, שָׁמִים גֶּרֶש אֶחָד בְּסוֹף הַחֵלֶק שֶׁנִּשְׁאַר."

בְּסוֹף הַיּוֹם נִכְנְסוּ לְבִנְיָן עִם שֶׁלֶט גָּדוֹל: "כָּאן מִתְכַּנְּסִים נְצִיגֵי הָאו״ם", וְדָנִי סִפֵּר שֶׁאו״ם הוּא רֹאשׁ תֵּבוֹת שֶׁל הָאֻמּוֹת הַמְּאֻחָדוֹת, אִרְגּוּן עוֹלָמִי שֶׁמְּחַבֵּר בֵּין מְדִינוֹת.

מַאיָה סָגְרָה אֶת הַמַּחְבֶּרֶת שֶׁלָּהּ, מְלֵאָה בְּרָאשֵׁי תֵּבוֹת וְקִצּוּרִים, וְאָמְרָה לְחַבְרָה: "הַיּוֹם לָמַדְתִּי שֶׁכָּל שֶׁלֶט בָּעִיר מַסְתִּיר כְּלָל דִּקְדּוּקִי קָטָן."

בָּאוֹטוֹבּוּס בַּחֲזָרָה הוֹסִיף דָּנִי עוֹד דֻּגְמָאוֹת: "רֹאשׁ הַמֶּמְשָׁלָה נִקְרָא בְּקִצּוּר רה״מ, וְהַמִּלִּים בֵּית סֵפֶר מִתְקַצְּרוֹת לְבִי״ס." כֻּלָּם כָּתְבוּ בִּמְהִירוּת בַּמַּחְבָּרוֹת שֶׁלָּהֶם, לִפְנֵי שֶׁהָאוֹטוֹבּוּס הִגִּיעַ לְבֵית הַסֵּפֶר.

📚 דִּקְדּוּק — רָאשֵׁי תֵּבוֹת וְקִצּוּרִים

מָה הַהֶבְדֵּל בֵּין רֹאשׁ תֵּבוֹת לְקִצּוּר?

רֹאשׁ תֵּבוֹת נוֹצָר מֵהָאוֹת הָרִאשׁוֹנָה (אוֹ הָרִאשׁוֹנוֹת) שֶׁל כַּמָּה מִלִּים, וְכָתוּב בְּגֵּרְשַׁיִם (״) לִפְנֵי הָאוֹת הָאַחֲרוֹנָה. קִצּוּר נוֹצָר כְּשֶׁמְּקַצְּרִים מִלָּה אַחַת וּמַשְׁאִירִים רַק חֵלֶק מִמֶּנָּה, וְכָתוּב בְּגֶּרֶש (׳) אֶחָד בְּסוֹף הַחֵלֶק שֶׁנִּשְׁאַר.
