יוֹם הַיְּצִירָה שֶׁל כִּתָּה ה׳

📖 קְרִיאָה

בַּבֹּקֶר הִכְרִיזָה הַמּוֹרָה בְּכִתָּה ה׳: "מָחָר יִהְיֶה יוֹם יְצִירָה — תַּחֲנַת בִּשּׁוּל וְתַחֲנַת יְצִירָה! כָּל תַּחֲנָה תִּכְתֹּב דַּף הוֹרָאוֹת לְכָל הַכִּתָּה."

תָּמָר, שֶׁאוֹהֶבֶת לְבַשֵּׁל עִם אִמָּא שֶׁלָּהּ, הִתְנַדְּבָה לִכְתֹּב מִתְכּוֹן לְעֻגִיּוֹת שׁוֹקוֹלָד. אוֹפִיר, שֶׁאוֹהֵב לְהָכִין דְּבָרִים בַּיָּדַיִם, הִתְנַדֵּב לִכְתֹּב הוֹרָאוֹת הַרְכָּבָה לְמָטוֹס נְיָר.

"שִׂימוּ לֵב," אָמְרָה הַמּוֹרָה, "כְּדֵי שֶׁכָּל הַכִּתָּה תֵּדַע בְּדִיּוּק מָה לַעֲשׂוֹת, כּוֹתְבִים בִּלְשׁוֹן צִוּוּי — הַלָּשׁוֹן שֶׁכְּבָר לָמַדְנוּ, כְּמוֹ 'כִּתְבוּ!' וְ'עֲשׂוּ!'."

תָּמָר פָּתְחָה דַּף חָדָשׁ וְכָתְבָה בְּרֹאשׁוֹ: "עֻגִיּוֹת שׁוֹקוֹלָד — זְמַן הֲכָנָה: עֶשְׂרִים דַּקּוֹת. מַסְפִּיק לְעֶשְׂרִים עֻגִיּוֹת." מִתַּחַת כָּתְבָה רְשִׁימַת מַצְרָכִים: שְׁתֵּי כּוֹסוֹת קֶמַח, כּוֹס סֻכָּר אַחַת, בֵּיצָה אַחַת וַחֲבִילַת שׁוֹקוֹלָד קְטַנָּה.

אַחַר כָּךְ כָּתְבָה תָּמָר אֶת הוֹרָאוֹת הַהֲכָנָה, שָׁלָב אַחַר שָׁלָב: "1. עַרְבְּבוּ אֶת הַקֶּמַח וְהַסֻּכָּר בִּקְעָרָה גְּדוֹלָה. 2. הוֹסִיפוּ אֶת הַבֵּיצָה וְעַרְבְּבוּ הֵיטֵב. 3. שִׁבְרוּ אֶת הַשּׁוֹקוֹלָד לַחֲתִיכוֹת וְעַרְבְּבוּ בְּתוֹךְ הַבָּצֵק. 4. גַּלְגְּלוּ כַּדּוּרִים קְטַנִּים וְשִׂימוּ עַל תַּבְנִית. 5. אֱפוּ בְּתַנּוּר חָם עֶשְׂרִים דַּקּוֹת."

אוֹפִיר, בַּתַּחֲנָה הַשְּׁנִיָּה, כָּתַב כּוֹתֶרֶת: "אֵיךְ מְקַפְּלִים מָטוֹס נְיָר" וְרָשַׁם חוֹמָרִים: דַּף נְיָר מַלְבֵּנִי אֶחָד. בְּלִי כַּמֻּיּוֹת וּבְלִי זְמַן הֲכָנָה — רַק דַּף אֶחָד.

אוֹפִיר כָּתַב אֶת הַשְּׁלָבִים הַמְּמֻסְפָּרִים: "1. קַפְּלוּ אֶת הַדַּף לַחֲצִי לְאֹרֶךְ וּפִתְחוּ שׁוּב. 2. קַפְּלוּ אֶת שְׁתֵּי הַפִּנּוֹת הָעֶלְיוֹנוֹת אֶל הַקַּו הָאֶמְצָעִי. 3. קַפְּלוּ אֶת הַצְּדָדִים פְּנִימָה שׁוּב. 4. קַפְּלוּ אֶת שְׁתֵּי הַכְּנָפַיִם כְּלַפֵּי חוּץ. 5. הַחְזִיקוּ מִלְּמַטָּה וְשַׁגְּרוּ אֶת הַמָּטוֹס!"

הַמּוֹרָה קָרְאָה אֶת שְׁנֵי הַדַּפִּים וְחִיְּכָה: "יֵשׁ פֹּה הֶבְדֵּל יָפֶה. בְּמִתְכּוֹן חוֹבָה לְכַלֵּל מַצְרָכִים בִּכְמֻיּוֹת מְדֻיָּקוֹת וּזְמַן הֲכָנָה — כִּי אַתֶּם מַכִינִים מַשֶּׁהוּ שֶׁמִּתְעָרֵב וּמִשְׁתַּנֶּה. בְּהוֹרָאוֹת הַרְכָּבָה כּוֹתְבִים רַק חוֹמָרִים וּשְׁלָבִים — כִּי הַנְּיָר לֹא מִתְעָרֵב, הוּא רַק מִתְקַפֵּל."

"וּבִשְׁנֵיהֶם," הוֹסִיפָה הַמּוֹרָה, "כּוֹתְבִים בִּלְשׁוֹן צִוּוּי בִּזְמַן רַבִּים — 'עַרְבְּבוּ', 'קַפְּלוּ' — כִּי הַדַּף מְיֹעָד לְכָל מִי שֶׁיִּקְרָא אוֹתוֹ, וְלֹא רַק לְאָדָם אֶחָד."

אַחַר הַצָּהֳרַיִם הֵכִינָה כָּל הַכִּתָּה עֻגִיּוֹת שׁוֹקוֹלָד לְפִי הַמִּתְכּוֹן שֶׁל תָּמָר, וְקִפְּלָה מְטוֹסֵי נְיָר לְפִי הַהוֹרָאוֹת שֶׁל אוֹפִיר. הָעֻגִיּוֹת יָצְאוּ טְעִימוֹת, וְהַמְּטוֹסִים עָפוּ רָחוֹק בֶּחָצֵר.

תָּמָר וְאוֹפִיר הֵבִינוּ: כְּשֶׁרוֹצִים שֶׁמִּישֶׁהוּ יְבַצֵּעַ פְּעֻלָּה לְפִי סֵדֶר מְדֻיָּק, כּוֹתְבִים בִּלְשׁוֹן צִוּוּי וּמְחַלְּקִים אֶת הַהֶסְבֵּר לִשְׁלָבִים מְמֻסְפָּרִים.
