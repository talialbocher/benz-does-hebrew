📖 קְרִיאָה — הַסֵּפֶר הַמְּדַבֵּר שֶׁל אָדִי

אָדִי אָהַב לְבַקֵּר בַּסִּפְרִיָּה הַיְּשָׁנָה שֶׁבִּרְחוֹבוֹ. יוֹם אֶחָד הוּא מָצָא עַל הַמַּדָּף סֵפֶר עִם כְּרִיכָה כְּחֻלָּה וּכְתָב זָהָב. כַּאֲשֶׁר פָּתַח אָדִי אֶת הַסֵּפֶר, הוּא שָׁמַע קוֹל מַפְתִּיעַ.

"שָׁלוֹם, אָדִי," אָמַר הַסֵּפֶר. "אֲנִי הַסֵּפֶר הַמְּדַבֵּר! אֲנִי אֲלַמֵּד אוֹתְךָ סוֹד מְרַתֵּק עַל מִלִּים עִבְרִיּוֹת."

אָדִי בָּלַע רֹק בְּהַפְתָּעָה. "סֵפֶר מְדַבֵּר?" הוּא לָחַשׁ. "זֶה לֹא יֵאָמֵן!"

"הַרְבֵּה מִלִּים בְּעִבְרִית," הִסְבִּיר הַסֵּפֶר, "נִרְאוֹת אוֹתוֹ דָּבָר אֲבָל מַשְׁמָעוּתָן שׁוֹנָה לְגַמְרֵי. הַכֹּל תָּלוּי בַּהֶקְשֵׁר — בַּמִּשְׁפָּטִים שֶׁסָּבִיב הַמִּלָּה."

הַסֵּפֶר פָּתַח דַּף וְאָדִי קָרָא בְּקוֹל: "עַיִן."

"עַיִן?" שָׁאַל אָדִי. "זֹאת הָעַיִן שֶׁרוֹאָה?"

"נָכוֹן מְאֹד!" אָמַר הַסֵּפֶר. "אֲבָל עַיִן יְכוֹלָה גַּם לִהְיוֹת מַעְיָן מַיִם בָּהָר. הַכֹּל תָּלוּי בַּהֶקְשֵׁר!"

אָדִי הִנְהֵן בְּרֹאשׁוֹ. הַסֵּפֶר הוֹסִיף: "וְהַמִּלָּה כּוֹכָב — הִיא גַּם כּוֹכָב בַּשָּׁמַיִם וְגַם כּוֹכָב קוֹלְנוֹעַ מְפֻרְסָם."

אָדִי פָּתַח אֶת מַחְבַּרְתּוֹ וְצִיֵּר שְׁתֵּי תְּמוּנוֹת לְכָל מִלָּה. לַמִּלָּה קֶשֶׁת הוּא צִיֵּר קֶשֶׁת צִבְעוֹנִית בַּשָּׁמַיִם וְגַם קֶשֶׁת שֶׁיּוֹרָה חִצִּים. לַמִּלָּה גַּן הוּא צִיֵּר גַּן פְּרָחִים יָפֶה וְגַם גַּן יְלָדִים עִם יְלָדִים מְשַׂחֲקִים.

"אַתָּה מְצֻיָּן!" קָרָא הַסֵּפֶר בְּשִׂמְחָה. "וְלַמִּלָּה רֶגֶל — מָה אַתָּה יוֹדֵעַ?"

"רֶגֶל זֹאת חֵלֶק בַּגּוּף," עָנָה אָדִי בְּבִטָּחוֹן.

"נָכוֹן! אֲבָל גַּם רֶגֶל הָהָר — הַחֵלֶק הַנָּמוּךְ שֶׁל הָהָר — גַּם זוֹ רֶגֶל!" אָמַר הַסֵּפֶר.

אָדִי הֻפְתַּע. הוּא לֹא חָשַׁב שֶׁמִּלִּים מֻכָּרוֹת כָּל כָּךְ יְכוֹלוֹת לְהַפְתִּיעַ אוֹתוֹ. הוּא סָגַר אֶת הַמַּחְבֶּרֶת וְחִיֵּךְ. יוֹם אֶחָד, הוּא יִהְיֶה מֻמְחֶה לְמִלִּים!

🔮 הֲבָנַת הַנִּקְרָא — כַּדּוּר הַבְּדֹלַח

הַכַּדּוּר הַמִּסְתּוֹרִי חוֹשֵׂף שְׁאֵלוֹת עַל הַסִּיפּוּר. לְחַץ עָלָיו כְּדֵי לִגְלוֹת אֶת הַשְּׁאֵלָה, בְּחַר תְּשׁוּבָה וּלְחַץ "אַשֵּׁר"!

📚 דִּקְדּוּק — מִלִּים בַּעֲלוֹת מַשְׁמָעוּיוֹת רַבּוֹת

מַה זֶּה מִלָּה בַּעֲלַת מַשְׁמָעוּיוֹת רַבּוֹת?

בְּעִבְרִית יֵשׁ מִלִּים שֶׁנִּרְאוֹת אוֹתוֹ דָּבָר אֲבָל יֵשׁ לָהֶן יוֹתֵר מִמַּשְׁמָעוּת אַחַת. כְּדֵי לְהָבִין אֵיזוֹ מַשְׁמָעוּת נְכוֹנָה, צָרִיךְ לִקְרֹא אֶת הַהֶקְשֵׁר — הַמִּשְׁפָּטִים שֶׁלְּפָנֶיהָ וְאַחֲרֶיהָ.

לְדֻגְמָה: הַמִּלָּה עַיִן יְכוֹלָה לְקַבֵּל מַשְׁמָעוּת שׁוֹנָה לְגַמְרֵי בְּמִשְׁפָּטִים שׁוֹנִים:

מִלָּה מַשְׁמָעוּת 1 דֻּגְמָה 1 מַשְׁמָעוּת 2 דֻּגְמָה 2
עַיִן אֵיבַר הָרְאִיָּה יֵשׁ לָהּ עֵינַיִם יָפוֹת. מַעְיָן מַיִם הֵם שָׁתוּ מִן הָעַיִן שֶׁבַּהָר.
כּוֹכָב גֶּרֶם שָׁמַיִם רָאִינוּ כּוֹכָבִים בַּלַּיְלָה. שַׂחְקָן מְפֻרְסָם הִיא כּוֹכֶבֶת קוֹלְנוֹעַ.
קֶשֶׁת קֶשֶׁת בֶּעָנָן אַחֲרֵי הַגֶּשֶׁם רָאִינוּ קֶשֶׁת בַּשָּׁמַיִם. כְּלִי לִירִיַּת חִצִּים הַיֶּלֶד יָרָה חֵץ מִן הַקֶּשֶׁת.
רֶגֶל חֵלֶק בַּגּוּף כָּאֲבָה לִי הָרֶגֶל. חֵלֶק תַּחְתּוֹן שֶׁל הָהָר עָמַדְנוּ בְּרֶגֶל הָהָר.
אוֹר מַה שֶּׁמֵּאִיר אֶת הַחֹשֶׁךְ הִדְלַקְנוּ אֶת הָאוֹר בַּחֶדֶר. שֵׁם פְּרָטִי אוֹר הוּא הַחָבֵר הַכִּי טוֹב שֶׁלִּי.

⚡ טִיפ חָשׁוּב!

כְּשֶׁאַתֶּם פּוֹגְשִׁים מִלָּה לֹא מוּבֶנֶת — קִרְאוּ אֶת כָּל הַמִּשְׁפָּט! הַהֶקְשֵׁר יַסְבִּיר לָכֶם אֵיזוֹ מַשְׁמָעוּת נְכוֹנָה.

🪔 תַּרְגּוּל דִּקְדּוּק — מְנוֹרַת הַקְּסָמִים

עִם כָּל תְּשׁוּבָה נְכוֹנָה — הַמְּנוֹרָה תַּבְהִיק יוֹתֵר! קְרָא אֶת הַמִּשְׁפָּט, מְצָא אֶת מַשְׁמָעוּת הַמִּלָּה הַמּוּדְגֶּשֶׁת וּבְחַר אֶת הַתְּשׁוּבָה הַנְּכוֹנָה.

✍️ כְּתִיבָה יְצִירָתִית
כְּתֹב שְׁנֵי מִשְׁפָּטִים לְכָל מִלָּה — מִשְׁפָּט אֶחָד לְכָל מַשְׁמָעוּת:

1. עַיִן — כְּתֹב מִשְׁפָּט עִם הָעַיִן שֶׁרוֹאָה וּמִשְׁפָּט עִם עַיִן שֶׁהִיא מַעְיָן מַיִם.

2. כּוֹכָב — כְּתֹב מִשְׁפָּט עִם כּוֹכָב בַּשָּׁמַיִם וּמִשְׁפָּט עִם כּוֹכָב מְפֻרְסָם.

3. בְּחַר מִלָּה מַפְתִּיעָה מִשֶּׁלְּךָ — מִלָּה שֶׁיֵּשׁ לָהּ יוֹתֵר מִמַּשְׁמָעוּת אַחַת, וּכְתֹב שְׁנֵי מִשְׁפָּטִים.
📝 סִיכּוּם הַשִּׁיעוּר

בַּשִּׁיעוּר הַזֶּה לָמַדְנוּ שֶׁבְּעִבְרִית יֵשׁ מִלִּים בַּעֲלוֹת מַשְׁמָעוּיוֹת רַבּוֹת. כְּדֵי לְהָבִין אֵיזוֹ מַשְׁמָעוּת נְכוֹנָה — מִסְתַּכְּלִים עַל הַהֶקְשֵׁר.
