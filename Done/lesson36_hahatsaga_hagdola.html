הַהַצָּגָה הַגְּדוֹלָה

רוֹנִי הָיָה לָחוּץ מְאֹד. לִפְנֵי שֶׁלִפְנֵי שֶׁ = before — פְּעֻלָּה שֶׁקָּרְתָה קֹדֶםהַהַצָּגָה הִתְחִילָה, הוּא חָזַר עַל הַמִּלִּים שֶׁלּוֹ מֵאָה פְּעָמִים. "אֲנִי אֶזְכֹּר אֶת הַכֹּל," הוּא אָמַר לְעַצְמוֹ, אֲבָל רַגְלָיו רָעֲדוּ בְּכָל זֹאת.

הַמּוֹרָה שָׁרָה הִסְתַּכְּלָה עָלָיו וְחִיְּכָה. "רוֹנִי, תִּנְשֹׁם עָמֹק," הִיא לָחֲשָׁה. כְּשֶׁכְּשֶׁ = when — שְׁתֵּי פְּעֻלּוֹת בְּאוֹתוֹ זְמַןהַוִּילוֹן נִפְתַּח, כָּל הַקָּהָל שָׁתַק. רוֹנִי עָמַד עַל הַבָּמָה וְהִרְגִּישׁ שֶׁלִּבּוֹ פּוֹעֵם חָזָק.

אַחֲרֵי שֶׁאַחֲרֵי שֶׁ = after — פְּעֻלָּה שֶׁקָּרְתָה אַחַר כָּךְהוּא לָקַח נְשִׁימָה עֲמֻקָּה, רוֹנִי פָּתַח אֶת פִּיו וְהִתְחִיל לְדַבֵּר בְּקוֹל רָם וּבָרוּר. הַמִּלִּים יָצְאוּ בְּסֵדֶר מוּשְׁלָם. הוּא לֹא שָׁכַח דָּבָר!

בְּזְמַן שֶׁבְּזְמַן שֶׁ = while — שְׁתֵּי פְּעֻלּוֹת בְּמַקְבִּילרוֹנִי דִּיבֵּר, חֲבֵרָיו הִקְשִׁיבוּ בְּשֶׁקֶט וְהַמּוֹרָה חִיְּכָה אֵלָיו. הוּא הִרְגִּישׁ שֶׁכּוֹחוֹת חֲדָשִׁים עוֹלִים בּוֹ. כְּשֶׁכְּשֶׁ = whenהַהַצָּגָה נִגְמְרָה, כָּל הַקָּהָל מָחָא כַּפַּיִם בְּשִׂמְחָה רַבָּה.

אַחֲרֵי שֶׁאַחֲרֵי שֶׁ = after — קֹדֶם יָרַד, אַחַר כָּךְ חִיֵּךְרוֹנִי יָרַד מִן הַבָּמָה, הוּא חִיֵּךְ חִיּוּךְ גָּדוֹל. הַפַּחַד שֶׁהָיָה לוֹ לִפְנֵי שֶׁלִפְנֵי שֶׁ = before — הַפַּחַד הָיָה לִפְנֵי תְּחִילַּת הַהַצָּגָההַהַצָּגָה הִתְחִילָה נֶעֱלַם לְגַמְרֵי. הוּא הֵבִין: כְּשֶׁכְּשֶׁ = when — הַהַצְלָחָה קּוֹרֵית בְּזְמַן שֶׁמִּתְאַמְּצִיםמִּתְאַמְּצִים וּמִתְאַמְּנִים, אֶפְשָׁר לְהַצְלִיחַ.
