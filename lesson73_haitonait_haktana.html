📚 סִיפּוּרִים בְּעִבְרִית — שִׁיעוּר 73

הָעִתּוֹנָאִית הַקְּטַנָּה

נֹעָה מוֹצִיאָה אֶת הָעִתּוֹן הָרִאשׁוֹן שֶׁל הַכִּתָּה

📖 קְרִיאָה

בְּיוֹם שְׁלִישִׁי בַּבֹּקֶר אָמְרָה הַמּוֹרָה דָּפְנָה לַכִּתָּה חֲדָשׁוֹת מְרַגְּשׁוֹת: הַכִּתָּה תּוֹצִיא עִתּוֹן כִּתָּתִי מִשֶּׁלָּהּ! נֹעָה, שֶׁאוֹהֶבֶת לִכְתֹּב יוֹתֵר מִכָּל דָּבָר אַחֵר, הִתְנַדְּבָה מַיָּד לִהְיוֹת הָעוֹרֶכֶת הָרָאשִׁית.

הַמּוֹרָה דָּפְנָה הִסְבִּירָה שֶׁבָּעִתּוֹן יִהְיוּ כְּתָבוֹת שׁוֹנוֹת: רֵאָיוֹן עִם תַּלְמִיד, סִפּוּר עַל אֵירוּעַ בַּכִּתָּה וְגַם צִיּוּר אוֹ בְּדִיחָה. "כָּל תַּלְמִיד יָכוֹל לְהַצִּיעַ כְּתָבָה," אָמְרָה, "וְנֹעָה תְּסַדֵּר אֶת הַכֹּל לְעִתּוֹן אֶחָד."

נֹעָה חָשְׁבָה זְמַן רַב עַל שֵׁם לָעִתּוֹן, וְלַבַּסּוֹף בָּחֲרָה בַּשֵּׁם "קוֹל הַכִּתָּה". הִיא צִיְּרָה כּוֹתֶרֶת גְּדוֹלָה וְצִבְעוֹנִית, וְתִכְנְנָה שֶׁהָעִתּוֹן יֵצֵא בְּכָל יוֹם שִׁשִּׁי.

הַכְּתָבָה הָרִאשׁוֹנָה שֶׁל נֹעָה הָיְתָה רֵאָיוֹן עִם יוֹתָם, שֶׁאוֹסֵף בּוּלִים כְּבָר שָׁלוֹש שָׁנִים. "יֵשׁ לִי יוֹתֵר מֵאַלְפַּיִם בּוּלִים מִכָּל הָעוֹלָם," סִפֵּר יוֹתָם בְּהִתְלַהֲבוּת. נֹעָה כָּתְבָה כָּל מִלָּה שֶׁהוּא אָמַר, וְהוֹסִיפָה גַּם תְּמוּנָה שֶׁל הַבּוּלִים הַיְּקָרִים בְּיוֹתֵר.

כְּתָבָה אַחֶרֶת בָּעִתּוֹן הָיְתָה עַל הַשּׁוֹעֵר הֶחָדָשׁ שֶׁל בֵּית הַסֵּפֶר. נֹעָה שָׁאֲלָה אוֹתוֹ: "מָה אַתָּה אוֹהֵב בְּיוֹתֵר בָּעֲבוֹדָה שֶׁלְּךָ?" הַשּׁוֹעֵר חִיֵּךְ וְעָנָה: "אֲנִי אוֹהֵב בְּעִיקָּר לְטַפֵּל בַּגִּנָּה וְלִרְאוֹת אֵיךְ הַפְּרָחִים פּוֹרְחִים."

גַּם רוֹעִי, הָאָח הַקָּטָן שֶׁל נֹעָה, רָצָה לְהִשְׁתַּתֵּף. הוּא צִיֵּר צִיּוּר מַצְחִיק שֶׁל נֹעָה עִם בּוּעַת דִּבּוּר, וּבְתוֹכָהּ כָּתַב: "אֲנִי הָעוֹרֶכֶת הָרָאשִׁית — אֲבָל אֲנִי גַּם צְרִיכָה לִישֹׁן!" כֻּלָּם צָחֲקוּ מֵהַבְּדִיחָה הַחֲמוּדָה.

כַּאֲשֶׁר נֹעָה הֵבִיאָה אֶת כָּל הַכְּתָבוֹת אֶל הַמּוֹרָה דָּפְנָה, שֶׁשִׁמְּשָׁה כְּעוֹרֶכֶת הָעִתּוֹן, הַמּוֹרָה שָׂמָה לֵב לְמַשֶּׁהוּ חָשׁוּב. "כְּשֶׁאַתְּ מְצַטֶּטֶת מַה שֶּׁמִּישֶׁהוּ אָמַר," הִסְבִּירָה, "צָרִיךְ לָשִׂים אֶת הַדִּבּוּר בֵּין מִירְכָּאוֹת — כָּכָה: "אֲנִי אוֹסֵף בּוּלִים." וְאִם אַתְּ רוֹצָה לְהוֹסִיף מֵידָע נוֹסָף שֶׁאֵינוֹ חֵלֶק מֶרְכָּזִי מִן הַמִּשְׁפָּט, אֶפְשָׁר לָשִׂים אוֹתוֹ בְּסוֹגְרַיִם."

נֹעָה הֵבִינָה אֶת הַהֶסְבֵּר, וְתִקְּנָה כָּל מָקוֹם שֶׁבּוֹ הָיְתָה צְרִיכָה מִירְכָּאוֹת אוֹ סוֹגְרַיִם. לְמָשָׁל, הִיא כָּתְבָה: "הַשּׁוֹעֵר הֶחָדָשׁ (שְׁמוֹ אוּרִי) אוֹהֵב מְאֹד לְטַפֵּל בַּגִּנָּה." כְּשֶׁכָּל הַכְּתָבוֹת הָיוּ מוּכָנוֹת, נֹעָה הֶרְאֲתָה אֶת הָעִתּוֹן לְעוֹרֶכֶת — הַמּוֹרָה דָּפְנָה — וְהִיא חִיְּכָה בְּגַאֲוָה.

נֹעָה תִּקְּנָה עוֹד כַּמָּה דְּבָרִים קְטַנִּים, וְלַבַּסּוֹף הִדְפִּיסָה עֶשְׂרִים עִתּוֹנִים. בְּיוֹם שִׁשִּׁי בַּבֹּקֶר חִלְּקָה נֹעָה אֶת הָעִתּוֹנִים לְכָל הַכִּתָּה, וְכֻלָּם קָרְאוּ בְּשִׂמְחָה אֶת "קוֹל הַכִּתָּה" הָרִאשׁוֹן.

בַּהֲפָסָקָה, יוֹתָם נִגַּשׁ אֶל נֹעָה וְאָמַר: "תּוֹדָה שֶׁכָּתַבְתְּ עָלַי! עַכְשָׁו כָּל הַכִּתָּה יוֹדַעַת עַל אֹסֶף הַבּוּלִים שֶׁלִּי." נֹעָה חִיְּכָה וְהֵשִׁיבָה: "זֶה הַתַּפְקִיד שֶׁל עִתּוֹנָאִית — לְסַפֵּר סִיפּוּרִים שֶׁל אֲחֵרִים." הִיא יָדְעָה שֶׁבָּעִתּוֹן הַבָּא יִהְיוּ עוֹד כְּתָבוֹת מְעַנְיְנוֹת.

📚 דִּקְדּוּק — מִירְכָּאוֹת וְסוֹגְרַיִם

מָתַי שָׂמִים מִירְכָּאוֹת, וּמָתַי שָׂמִים סוֹגְרַיִם?

מִירְכָּאוֹת (" ") הֵן סִימָן פִּסּוּק שֶׁמַּקִּיף מִלִּים. שָׂמִים אוֹתָן בִּשְׁנֵי מִקְרִים עִקָּרִיִּים:

מָתַי?דֻּגְמָה
צִטּוּט מְדֻיָּק שֶׁל דִּבְרֵי מִישֶׁהוּ"אֲנִי הוֹלֵךְ הַבַּיְתָה," אָמַר דָּן.
שֵׁם שֶׁל סֵפֶר, שִׁיר, סֶרֶט אוֹ עִתּוֹןקָרָאתִי אֶת הַסֵּפֶר "הַנָּסִיךְ הַקָּטָן".

סוֹגְרַיִם ( ) מְשַׁמְּשִׁים לְהוֹסָפַת מֵידָע נוֹסָף, שֶׁאֵינוֹ חֵלֶק מֶרְכָּזִי מִן הַמִּשְׁפָּט:

דָּוִד (אֲחִי הַגָּדוֹל שֶׁל יוֹתָם) הִגִּיעַ מְאֻחָר. הַטִּיּוּל יִתְקַיֵּם בְּיוֹם רִאשׁוֹן (אִם לֹא יֵרֵד גֶּשֶׁם).

כְּלָל חָשׁוּב: הַנְּקֻדָּה שֶׁל סוֹף הַמִּשְׁפָּט בָּאָה אַחֲרֵי הַסּוֹגְרַיִם הַסּוֹגְרִים.

כָּל תְּשׁוּבָה נְכוֹנָה מוֹסִיפָה בּוּל פִּסּוּק אֶחָד לָאֹסֶף!

כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים שְׁלֵמִים. הִשְׁתַּמְּשׁוּ נָכוֹן בְּמִירְכָּאוֹת וּבְסוֹגְרַיִם, כְּמוֹ בַּדֻּגְמָה: דָּן אָמַר: "אֲנִי אוֹהֵב לְשַׂחֵק כַּדּוּרֶגֶל" (הוּא מְשַׂחֵק כָּל יוֹם שְׁלִישִׁי).
