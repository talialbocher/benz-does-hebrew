מָא לְיָאנְג וְהַמִּכְחוֹל הַקָּסוּם

אַגָּדַת עַם סִינִית עַל יֶלֶד עָנִי וּמִכְחוֹל שֶׁכָּל מָה שֶׁמְּצַיְּרִים בּוֹ נַעֲשֶׂה אֲמִתִּי

📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת, בִּכְפָר קָטָן בְּסִין, חַי יֶלֶד יָתוֹם וְעָנִי בְּשֵׁם מָא לְיָאנְג. בַּבֹּקֶר הוּא אָסַף עֵצִים בֶּהָרִים, וּבָעֶרֶב עָזַר לַשְּׁכֵנִים בַּעֲבוֹדַת הַשָּׂדֶה. יוֹתֵר מִכֹּל אָהַב מָא לְיָאנְג לְצַיֵּר, וְחָלַם לִהְיוֹת צַיָּר. אֲבָל לֹא הָיָה לוֹ כֶּסֶף לִקְנוֹת מִכְחוֹל, וְלָכֵן צִיֵּר בְּמַקֵּל עַל הַחוֹל, וּבְפֶחָם עַל אַבְנֵי הַנָּהָר.

יוֹם אֶחָד הִגִּיעַ לַכְּפָר צַיָּר מְפֻרְסָם מִן הָעִיר. מָא לְיָאנְג רָאָה אוֹתוֹ מְצַיֵּר לְיַד הַנָּהָר, וְשָׁאַל בְּנִימוּס: "אוּלַי תַּשְׁאִיל לִי אֶת הַמִּכְחוֹל שֶׁלְּךָ לְרֶגַע?" אֲבָל הַצַּיָּר הָיָה אִישׁ קַמְצָן וְרַגְזָן. "מִכְחוֹל בִּשְׁבִיל יֶלֶד עָנִי?" צָעַק. "לֵךְ מִכָּאן!" מָא לְיָאנְג לֹא וִתֵּר. הוּא הִמְשִׁיךְ לְצַיֵּר בַּחוֹל כָּל יוֹם, עַד שֶׁהַצִּפֳּרִים שֶׁצִּיֵּר נִרְאוּ כְּאִלּוּ הֵן עוֹמְדוֹת לָעוּף.

לַיְלָה אֶחָד הוֹפִיעַ לְפָנָיו בַּחֲלוֹם אִישׁ זָקֵן עִם זָקָן לָבָן וְאָרֹךְ. הוּא נָתַן לְמָא לְיָאנְג מִכְחוֹל שֶׁל זָהָב וְאָמַר: "זֶהוּ מִכְחוֹל קָסוּם. צַיֵּר בּוֹ רַק בִּשְׁבִיל אֲנָשִׁים עֲנִיִּים." כְּשֶׁהִתְעוֹרֵר מָא לְיָאנְג, הַמִּכְחוֹל הָיָה בְּיָדוֹ! הוּא צִיֵּר צִפּוֹר עַל אֶבֶן — וְהַצִּפּוֹר נִפְנְפָה בִּכְנָפֶיהָ, צִיְּצָה וְעָפָה אֶל הַשָּׁמַיִם.

מֵאָז צִיֵּר מָא לְיָאנְג בִּשְׁבִיל כָּל הָעֲנִיִּים בַּכְּפָר. בְּעוֹנַת הֶחָרִישׁ צִיֵּר מַחְרֵשָׁה וְשׁוֹר חָזָק לְאִכָּר זָקֵן שֶׁלֹּא הָיָה לוֹ כֶּסֶף. לִפְנֵי הַקָּצִיר צִיֵּר מַגְרֵפוֹת לַקּוֹצְרִים, וּלְדַיָּג עָנִי צִיֵּר רֶשֶׁת חֲדָשָׁה. כְּשֶׁיַּלְדָּה קְטַנָּה חָלְתָה בְּקַדַּחַת, הוּא צִיֵּר לָהּ שְׂמִיכָה חַמָּה וּקְעָרָה שֶׁל מָרָק חַם, וּלְמָחֳרָת הִיא כְּבָר צָחֲקָה וְשִׂחֲקָה בֶּחָצֵר.

הַשְּׁמוּעָה עַל הַמִּכְחוֹל הַקָּסוּם עָבְרָה מִכְּפָר לִכְפָר, עַד שֶׁהִגִּיעָה אֶל בַּעַל אֲחֻזָּה עָשִׁיר — אִישׁ חַמְדָן שֶׁאָהַב זָהָב יוֹתֵר מִכָּל דָּבָר. הוּא שָׁלַח אֶת הַשּׁוֹמְרִים שֶׁלּוֹ, וְהֵם הֵבִיאוּ אֶת מָא לְיָאנְג אֶל בֵּיתוֹ. "צַיֵּר לִי הַר שֶׁל זָהָב!" צִוָּה הֶעָשִׁיר. "אֲנִי מְצַיֵּר רַק בִּשְׁבִיל עֲנִיִּים," עָנָה מָא לְיָאנְג בְּשֶׁקֶט. הֶעָשִׁיר הִתְרַגֵּז מְאוֹד, וְנָעַל אוֹתוֹ בַּמַּחְסָן הַקַּר שֶׁמֵּאֲחוֹרֵי הַמִּטְבָּח.

בַּלַּיְלָה, כְּשֶׁכֻּלָּם יָשְׁנוּ, הוֹצִיא מָא לְיָאנְג אֶת הַמִּכְחוֹל שֶׁהִסְתִּיר בְּבִגְדוֹ, וְצִיֵּר עַל הַקִּיר מַפְתֵּחַ גָּדוֹל. הַמַּפְתֵּחַ נָפַל הַיְשֵׁר לְתוֹךְ יָדוֹ. הוּא פָּתַח בּוֹ אֶת הַמַּנְעוּל, יָצָא בְּשֶׁקֶט וְצִיֵּר סוּס לָבָן. אַחַר כָּךְ קָפַץ עַל הַסּוּס וְדָהַר אֶל הַיָּם.

בַּבֹּקֶר גִּלָּה הֶעָשִׁיר שֶׁהַמַּחְסָן רֵיק. הוּא רָדַף אַחֲרֵי הַיֶּלֶד עִם כָּל הַשּׁוֹמְרִים עַד חוֹף הַיָּם, וְשָׁם תָּפַס אוֹתוֹ. "צַיֵּר לִי אִי שֶׁל זָהָב בְּאֶמְצַע הַיָּם," דָּרַשׁ, "וּסְפִינָה גְּדוֹלָה שֶׁתָּבִיא אוֹתִי אֵלָיו!" הַפַּעַם מָא לְיָאנְג הִסְכִּים. הוּא צִיֵּר יָם כָּחֹל וְרָחָב, וּבְאֶמְצָעוֹ אִי שֶׁמַּבְרִיק כְּמוֹ זָהָב. אַחַר כָּךְ צִיֵּר סְפִינָה גְּדוֹלָה, וְהֶעָשִׁיר וְהַשּׁוֹמְרִים עָלוּ עָלֶיהָ בְּשִׂמְחָה.

"רוּחַ! אֲנִי צָרִיךְ רוּחַ!" צָעַק הֶעָשִׁיר. מָא לְיָאנְג צִיֵּר קַו אֶחָד שֶׁל רוּחַ, וְהַסְּפִינָה הִתְחִילָה לָזוּז. "עוֹד! מַהֵר יוֹתֵר!" צָעַק הֶעָשִׁיר הַחַמְדָן. מָא לְיָאנְג צִיֵּר עוֹד קַוִּים וְעוֹד קַוִּים, עַד שֶׁקָּמָה סוּפָה חֲזָקָה. הַגַּלִּים נָשְׂאוּ אֶת הַסְּפִינָה הַרְחֵק מִן הַחוֹף, אֶל אֶרֶץ רְחוֹקָה — וְהֶעָשִׁיר לֹא חָזַר עוֹד.

וּמָא לְיָאנְג? הוּא חָזַר אֶל הַכְּפָר שֶׁלּוֹ. אוֹמְרִים שֶׁעַד הַיּוֹם הוּא נוֹדֵד מִכְּפָר לִכְפָר וּמְצַיֵּר בִּשְׁבִיל מִי שֶׁצָּרִיךְ: מַחְרֵשָׁה לְאִכָּר, רֶשֶׁת לְדַיָּג וּמְנוֹרָה לְיֶלֶד שֶׁרוֹצֶה לִקְרֹא בַּלַּיְלָה.

עַל פִּי אַגָּדַת הָעַם הַסִּינִית ״מָא לְיָאנְג וְהַמִּכְחוֹל הַקָּסוּם״ (神笔马良), בְּעִבּוּד חָפְשִׁי וּמְקֻצָּר.

שִׁשָּׁה מִשְׁקָלִים — מִי, בְּמָה, אֵיפֹה וְאֵיךְ

בַּשִּׁיעוּרִים 130–137 לָמַדְנוּ שִׁשָּׁה מִשְׁקָלִים. מִשְׁקָל הוּא כְּמוֹ תַּבְנִית: מַכְנִיסִים לְתוֹכָהּ אֶת אוֹתִיּוֹת הַשֹּׁרֶשׁ, וְהַתַּבְנִית רוֹמֶזֶת לָנוּ מָה הַמִּלָּה מְצַיֶּנֶת — אָדָם, כְּלִי, מָקוֹם, מַחֲלָה, תְּכוּנָה אוֹ עוֹנָה. בַּסִּפּוּר עַל מָא לְיָאנְג יֵשׁ מִלִּים מִכָּל שֵׁשֶׁת הַמִּשְׁקָלִים!

שִׂימוּ לֵב:

1. שֹׁרֶשׁ אֶחָד — כַּמָּה מִשְׁקָלִים. אוֹתוֹ שֹׁרֶשׁ יָכוֹל לְהִכָּנֵס לְכַמָּה תַּבְנִיּוֹת, וּבְכָל תַּבְנִית יֵשׁ לוֹ מַשְׁמָעוּת אַחֶרֶת: חָרִישׁ ← מַחְרֵשָׁה (הָעוֹנָה וְהַכְּלִי) טַבָּח ← מִטְבָּח (הָאָדָם וְהַמָּקוֹם) קָצִיר ← מַקְצֵרָה (הָעוֹנָה וְהַכְּלִי)

2. מַ־ אוֹ מִ־? בַּכְּלִי (מַקְטֵל) יֵשׁ פַּתָּח תַּחַת הַ־מ׳ וְצֵירֵי בָּאוֹת הַשְּׁנִיָּה שֶׁל הַשֹּׁרֶשׁ: מַפְתֵּחַ. בַּמָּקוֹם (מִקְטָל) יֵשׁ חִירִיק תַּחַת הַ־מ׳ וְקָמָץ בָּאוֹת הַשְּׁנִיָּה שֶׁל הַשֹּׁרֶשׁ: מִטְבָּח. אֲבָל כְּשֶׁהָאוֹת הָרִאשׁוֹנָה בַּשֹּׁרֶשׁ גְּרוֹנִית, גַּם הַמָּקוֹם מְקַבֵּל פַּתָּח: מַחְסָן — וְאָז מִסְתַּכְּלִים עַל הַקָּמָץ בָּאוֹת הַשְּׁנִיָּה.

3. קַטָּל אוֹ קַטְלָן? קַטָּל הוּא בְּדֶרֶךְ כְּלָל מִקְצוֹעַ (נַגָּר, דַּיָּג), וְקַטְלָן הוּא בְּדֶרֶךְ כְּלָל תְּכוּנָה (קַמְצָן, רַגְזָן). בַּסִּפּוּר, הַצַּיָּר מִן הָעִיר הוּא גַּם צַיָּר (מִקְצוֹעַ) וְגַם קַמְצָן (תְּכוּנָה)!

4. אוֹתִיּוֹת גְּרוֹנִיּוֹת (א, ה, ח, ע) מְשַׁנּוֹת לִפְעָמִים אֶת הַנִּקּוּד: מַחְסָן (פַּתָּח בִּמְקוֹם חִירִיק) כַּעֲסָן (חֲטַף פַּתָּח) צַהֶבֶת (בְּלִי דָּגֵשׁ)

מַלְכּוֹדוֹת: לֹא כָּל מִלָּה שֶׁנִּרְאֵית דּוֹמָה שַׁיֶּכֶת לַמִּשְׁקָל. מִכְחוֹל הוּא כְּלִי, אֲבָל יֵשׁ בּוֹ חִירִיק וְחוֹלָם — זֶה מִשְׁקָל אַחֵר, לֹא מַקְטֵל. מִכְתָּב בָּנוּי כְּמוֹ מִקְטָל, אֲבָל הוּא לֹא מָקוֹם. שֻׁלְחָן נִגְמָר בְּ־ָן, אֲבָל הוּא לֹא תְּכוּנָה.

טִיפּ: כְּדֵי לְזַהוֹת מִשְׁקָל — מוֹצְאִים קֹדֶם אֶת הַשֹּׁרֶשׁ. מָה שֶׁנִּשְׁאָר (הַ־מ׳ בַּהַתְחָלָה, הַ־ן אוֹ הַ־ת בַּסּוֹף וְהַתְּנוּעוֹת) — זֶה הַמִּשְׁקָל: מַחְרֵשָׁה ← ח־ר־שׁ + מַקְטֵלָה

✏️ מִשְׂחַק דִּקְדּוּק — הַסְּפִינָה וְהָרוּחַ
⛵ הַסְּפִינָה וְהָרוּחַ
עַל כָּל תְּשׁוּבָה נְכוֹנָה מָא לְיָאנְג מְצַיֵּר עוֹד קַו שֶׁל רוּחַ — וְהַסְּפִינָה שֶׁל הֶעָשִׁיר מִתְרַחֶקֶת!
⛵ הַסְּפִינָה עוֹמֶדֶת לְיַד הַחוֹף...

🎨 תַּפְזֹרֶת הַמִּכְחוֹל
מִצְאוּ שְׁמוֹנֶה מִלִּים מִן הַסִּפּוּר (הַמִּלִּים כְּתוּבוֹת בְּלִי נִקּוּד): כֵּלִים, מְקוֹמוֹת, תְּכוּנָה, עוֹנָה וּמַחֲלָה. לַחֲצוּ עַל הָאוֹת הָרִאשׁוֹנָה וְאַחַר כָּךְ עַל הָאוֹת הָאַחֲרוֹנָה שֶׁל הַמִּלָּה. עַל כָּל מִלָּה — עוֹד מְשִׁיכַת מִכְחוֹל!
✍️ כְּתִיבָה
✍️ מַטְלַת כְּתִיבָה

דַּמְיְנוּ שֶׁמָּא לְיָאנְג הִשְׁאִיל לָכֶם אֶת הַמִּכְחוֹל הַקָּסוּם לְיוֹם אֶחָד. כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים עַל מָה שֶׁהֱיִיתֶם מְצַיְּרִים וּבִשְׁבִיל מִי — וּבְכָל מִשְׁפָּט מִלָּה מִמִּשְׁקָל אַחֵר. דֻּגְמָה: הָיִיתִי מְצַיֵּר מַחְרֵשָׁה חֲדָשָׁה לְאִכָּר שֶׁאֵין לוֹ כֶּסֶף.
