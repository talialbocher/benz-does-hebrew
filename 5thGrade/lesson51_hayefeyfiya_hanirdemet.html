📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת מְאֹד חָיוּ בְּמַמְלָכָה רְחוֹקָה מֶלֶךְ וּמַלְכָּה שֶׁרָצוּ יוֹתֵר מִכֹּל דָּבָר אַחֵר בָּעוֹלָם - יֶלֶד. שָׁנִים רַבּוֹת חִכּוּ, עַד שֶׁסּוֹף סוֹף נוֹלְדָה לָהֶם בַּת קְטַנָּה וְיָפָה. שִׂמְחָתָם הָיְתָה כֹּה גְּדוֹלָה, שֶׁהֵם הֶחְלִיטוּ לְחַגֵּג אֶת יוֹם הֻלַּדְתָּהּ בְּנֶשֶׁף מְפֹאָר שֶׁכָּל הַמַּמְלָכָה תִּזְכֹּר.

לַנֶּשֶׁף הֻזְמְנוּ שְׁתֵּים־עֶשְׂרֵה פֵּיוֹת טוֹבוֹת שֶׁחָיוּ בַּמַּמְלָכָה, כְּדֵי שֶׁכָּל אַחַת מֵהֶן תְּבָרֵךְ אֶת הַתִּינֹקֶת בְּמַתָּנָה קְסוּמָה. פֵּיָה אַחַת בֵּרְכָה אוֹתָהּ בְּיֹפִי, שְׁנִיָּה בְּטוּב לֵב, שְׁלִישִׁית בְּקוֹל שִׁירָה נִפְלָא - וְכָךְ הָלְאָה, עַד שֶׁכִּמְעַט כָּל הַמַּתָּנוֹת הַטּוֹבוֹת בָּעוֹלָם נִתְּנוּ לַנְּסִיכָה הַקְּטַנָּה.

אַךְ בַּמַּמְלָכָה חָיְתָה גַּם פֵּיָה נוֹסֶפֶת, זְקֵנָה וְעַצְבָּנִית, שֶׁאִישׁ לֹא זָכַר בְּמֶשֶׁךְ שָׁנִים רַבּוֹת. בָּאֲרָמוֹן הָיוּ רַק שְׁתֵּים־עֶשְׂרֵה צַלָּחוֹת זָהָב, וְלָכֵן הִיא לֹא הֻזְמְנָה לַנֶּשֶׁף. הַפֵּיָה הַזְּקֵנָה נִפְגְּעָה עַד עֹמֶק לִבָּהּ, וּבְלִי אַזְהָרָה הוֹפִיעָה בָּאוּלָם בְּזַעַם גָּדוֹל. "מַדּוּעַ לֹא הֻזְמַנְתִּי?!" צָעֲקָה, וְהֵטִילָה עַל הַתִּינֹקֶת קְלָלָה נוֹרָאָה: בְּיוֹם הֻלֶּדֶת הַשֵּׁשׁ־עֶשְׂרֵה שֶׁלָּהּ תִּדְקֹר הַנְּסִיכָה אֶת אֶצְבָּעָהּ בְּפֶּלֶךְ שֶׁל גַּלְגַּל טְוִיָּה, וְתָמוּת בּוֹ בַּמָּקוֹם.

כָּל הָאוּלָם קָפָא מֵאֵימָה. אֲבָל פֵּיָה צְעִירָה אַחַת עוֹד לֹא הִסְפִּיקָה לָתֵת אֶת מַתָּנָתָהּ, וּמִהֲרָה קָדִימָה. הִיא לֹא הָיְתָה חֲזָקָה מַסְפִּיק כְּדֵי לְבַטֵּל קְלָלָה שֶׁל פֵּיָה חֲזָקָה מִמֶּנָּה, אַךְ יָכְלָה לְהָקֵל עָלֶיהָ קְצָת. בַּהֲקָלָה קְטַנָּה אֲבָל חֲשׁוּבָה, הִיא שִׁנְּתָה אֶת הַגְּזֵרָה: הַנְּסִיכָה לֹא תָּמוּת, אֶלָּא תֵּרָדֵם שֵׁנָה עֲמֻקָּה בַּת מֵאָה שָׁנִים - עַד שֶׁמִּישֶׁהוּ יַגִּיעַ וְיָעִיר אוֹתָהּ.

הַמֶּלֶךְ, מְפֻחָד מְאֹד, פָּקַד לֶאֱסֹר כָּל טְוִיָּה בְּגַלְגַּל וּבְפֶלֶךְ בְּכָל הַמַּמְלָכָה, וְצִוָּה לִשְׂרֹף כָּל פֶּלֶךְ שֶׁנִּמְצָא בַּבָּתִּים. שָׁנִים רַבּוֹת עָבְרוּ בְּשֶׁקֶט, וְהַנְּסִיכָה גָּדְלָה לְנַעֲרָה סַקְרָנִית, טוֹבַת לֵב וּמְלֵאַת חַיִּים - שֶׁכְּמוּבָן לֹא יָדְעָה כְּלוּם עַל הַקְּלָלָה שֶׁרָבְצָה עָלֶיהָ.

בְּיוֹם הֻלַּדְתָּהּ הַשֵּׁשׁ־עֶשְׂרֵה, שׁוֹטְטָה הַנְּסִיכָה בֵּין חַדְרֵי הַמִּגְדָּל הַיָּשָׁן, וְהִגִּיעָה לְחֶדֶר קָטָן שֶׁמֵּעוֹלָם לֹא רָאֲתָה. שָׁם יָשְׁבָה אִשָּׁה זְקֵנָה שֶׁלֹּא שָׁמְעָה מֵעוֹלָם עַל אִסּוּר הַמֶּלֶךְ, וְטָוְתָה חוּטִים בְּפֶלֶךְ יָשָׁן. הַנְּסִיכָה, שֶׁמֵּעוֹלָם לֹא רָאֲתָה כְּלִי כָּזֶה, בִּקְּשָׁה לְנַסּוֹת - וּבְרֶגַע שֶׁנָּגְעָה בַּפֶּלֶךְ, נִדְקְרָה בְּאֶצְבָּעָהּ וְנָפְלָה מִיָּד אַרְצָה, שְׁקוּעָה בְּשֵׁנָה עֲמֻקָּה.

הַפֵּיָה הַצְּעִירָה, שֶׁלֹּא שָׁכְחָה אֶת הַנְּסִיכָה הַקְּטַנָּה, שָׁמְעָה מִיָּד מָה קָרָה וּמִהֲרָה לָאֲרָמוֹן. כְּדֵי שֶׁהַנְּסִיכָה לֹא תִּתְעוֹרֵר לְבַדָּהּ, בְּעוֹלָם זָר וּמְבֻגָּר מִדַּי, הִיא בִּצְּעָה הַרְדָּמָה קְסוּמָה עַל כָּל הָאֲרָמוֹן: הַמֶּלֶךְ וְהַמַּלְכָּה, הַמְּשָׁרְתִים, הַסּוּסִים בָּאֻרְוָה, וַאֲפִלּוּ הָאֵשׁ בַּתַּנּוּר - כֻּלָּם נִרְדְּמוּ בְּבַת אַחַת, בְּדִיּוּק כְּמוֹ הַנְּסִיכָה.

בְּמֶשֶׁךְ מֵאָה הַשָּׁנִים שֶׁחָלְפוּ, גֶּדֶר עֲבֻתָּה שֶׁל שִׂיחֵי קוֹצִים הֵחֵלָּה בִּצְמִיחָה סְבִיב הָאֲרָמוֹן, גָּבְהָה וְהִתְעַבְּתָה מִשָּׁנָה לְשָׁנָה, עַד שֶׁהִסְתִּירָה אֶת הַמִּגְדָּלִים כֻּלָּם מֵעֵין רוֹאֶה. סִפּוּרִים עַל נְסִיכָה יְשֵׁנָה מֵאֲחוֹרֵי הַקּוֹצִים הִתְפַּשְּׁטוּ בֵּין הָאֲנָשִׁים, וְכַמָּה נְסִיכִים נִסּוּ לְפַלֵּס דֶּרֶךְ בַּגָּדֵר - אַךְ הַקּוֹצִים תָּפְסוּ אוֹתָם וְהֵם לֹא הִצְלִיחוּ לְעוֹלָם.

בְּדִיּוּק כְּשֶׁמָּלְאוּ מֵאָה שָׁנִים, הִגִּיעַ לַמָּקוֹם נָסִיךְ צָעִיר שֶׁשָּׁמַע אֶת הָאַגָּדָה. הַפַּעַם, לְתַדְהֵמַת כֻּלָּם, נִפְתַּח לְפָנָיו מַסְלוּל בֵּין הַקּוֹצִים כְּאִלּוּ מֵאֵלָיו - הָעֵת הַנְּכוֹנָה הִגִּיעָה. הַנָּסִיךְ עָבַר בֵּין הַמְּשָׁרְתִים הַיְּשֵׁנִים וְעָלָה בַּמַּדְרֵגוֹת אֶל הַמִּגְדָּל, עַד שֶׁמָּצָא אֶת הַנְּסִיכָה שׁוֹכֶבֶת בְּשֵׁנָה שְׁלֵוָה, יָפָה כְּמוֹ בַּחֲלוֹם.

מֻקְסָם מִיָּפְיָהּ, הוּא הִתְכּוֹפֵף וְנָשַׁק בַּעֲדִינוּת לְמִצְחָהּ. בְּאוֹתוֹ הָרֶגַע פָּקְחָה הַנְּסִיכָה אֶת עֵינֶיהָ, וְעִמָּהּ הִתְעוֹרֵר כָּל הָאֲרָמוֹן: הַמֶּלֶךְ וְהַמַּלְכָּה הִבִּיטוּ סְבִיבָם בְּתִמָּהוֹן, הַמְּשָׁרְתִים חָזְרוּ לַעֲבוֹדָתָם, וְהָאֵשׁ בַּתַּנּוּר שָׁבָה לִבְעֹר. הַהִתְעוֹרְרוּת הַגְּדוֹלָה עָבְרָה בְּכָל הָאֲרָמוֹן כְּמוֹ גַּל שֶׁל אוֹר.

הַנְּסִיכָה וְהַנָּסִיךְ הִתְחַתְּנוּ בַּחֲתֻנָּה שִׂמְחָה, וְחָיוּ מְאֻשָּׁרִים שָׁנִים רַבּוֹת. וּמֵאָז, כְּשֶׁמְּסַפְּרִים אֶת סִפּוּרָהּ, זוֹכְרִים גַּם אֶת הַפֵּיָה הַצְּעִירָה: הִיא לֹא הִצְלִיחָה לְבַטֵּל אֶת הָרַע לְגַמְרֵי, אֲבָל בְּאַהֲבָה וּבְחָכְמָה יָדְעָה לְהָקֵל עָלָיו - וְלִפְעָמִים זֶה בְּדִיּוּק מַסְפִּיק.

מְסֻפָּר עַל־פִּי אַגָּדַת "הַיְּפֵיפִיָּה הַנִּרְדֶּמֶת" (La Belle au Bois Dormant) מֵאֵת שַׁארְל פֶּרוֹ (צָרְפַת, 1697) וְעַל־פִּי "דֹּרֶנְרֶשֶׁן" מֵאֵת הָאַחִים גְּרִים (גֶּרְמַנְיָה, 1812) - מְעֻבָּד

מָה זֶה שֵׁם הַפְּעֻלָּה?

שֵׁם הַפְּעֻלָּה הוּא מִלָּה שֶׁנִּגְזֶרֶת מִפֹּעַל, אֲבָל בְּשׁוֹנֶה מִשֵּׁם הַפֹּעַל (שֶׁלָּמַדְנוּ בַּשִּׁיעוּר הַקּוֹדֵם, כְּמוֹ "לִבְרֹחַ") - שֵׁם הַפְּעֻלָּה הוּא שֵׁם עֶצֶם אֲמִתִּי לְכָל דָּבָר. הוּא יָכוֹל לְקַבֵּל ה' הַיְּדִיעָה ("הַטְּוִיָּה"), רִבּוּי ("שְׁתֵּי טְוִיּוֹת"), וְאַף לְהוֹפִיעַ בִּסְמִיכוּת ("טְוִיַּת הַזְּקֵנָה") - בְּדִיּוּק כְּמוֹ שֻׁלְחָן אוֹ עֵץ. שֵׁם הַפֹּעַל, לְעֻמַּת זֹאת, לְעוֹלָם לֹא מִתְנַהֵג כְּמוֹ שֵׁם עֶצֶם.

שֹׁרֶשׁ אֶחָד, בִּנְיָן אַחֵר, מַשְׁמָעוּת שׁוֹנָה לְגַמְרֵי

שִׂימוּ לֵב לְמִלִּים "קְלָלָה" וְ"הֲקָלָה" בַּטַּבְלָה - שְׁתֵּיהֶן בָּאוֹת מֵאוֹתוֹ שֹׁרֶשׁ בְּדִיּוּק (ק-ל-ל), אֲבָל מִבִּנְיָנִים שׁוֹנִים: בְּבִנְיַן פִּעֵל "קִלֵּל" פֵּרוּשׁוֹ "אָמַר דְּבָרִים רָעִים עַל מִישֶׁהוּ" (וּמִזֶּה - "קְלָלָה"), וּבְבִנְיַן הִפְעִיל "הֵקֵל" פֵּרוּשׁוֹ "עָשָׂה מַשֶּׁהוּ קַל וְנוֹחַ יוֹתֵר" (וּמִזֶּה - "הֲקָלָה"). הַבִּנְיָן יָכוֹל לְשַׁנּוֹת אֶת מַשְׁמָעוּת הַשֹּׁרֶשׁ לְגַמְרֵי!

🔑 כְּלָל זָהָב: כְּדֵי לִבְדֹּק אִם מִלָּה הִיא שֵׁם פְּעֻלָּה (וְלֹא שֵׁם הַפֹּעַל) - נַסּוּ לְהוֹסִיף לָהּ ה' הַיְּדִיעָה אוֹ רִבּוּי. "הַטְּוִיָּה שֶׁל הַזְּקֵנָה" עוֹבֵד מְצֻיָּן; "הַלִּטְווֹת שֶׁל הַזְּקֵנָה" - לֹא קַיָּם בִּכְלָל בְּעִבְרִית.

כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים מִשֶּׁלָּכֶם, כָּל אֶחָד עִם שֵׁם פְּעֻלָּה אַחֵר מִן הָרְשִׁימָה: קְלָלָה, הֲקָלָה, טְוִיָּה, הַרְדָּמָה, צְמִיחָה, הִתְעוֹרְרוּת (בַּחֲרוּ חָמֵשׁ מִתּוֹךְ הַשֵּׁשׁ) - בְּדִיּוּק כְּמוֹ בַּסִּפּוּר עַל הַיְּפֵיפִיָּה הַנִּרְדֶּמֶת. לְדֻגְמָה: "הַפֵּיָה הַצְּעִירָה הֵבִיאָה הֲקָלָה גְּדוֹלָה לְכָל הָאֲרָמוֹן."
