מֵי הַחַיִּים

אַגָּדָה שֶׁל הָאַחִים גְּרִים עַל מֶלֶךְ חוֹלֶה וּשְׁלוֹשָׁה בָּנִים
🎯 כִּיתָה ה׳  |  שְׁמוֹת מַחֲלוֹת — מִשְׁקַל קַטֶּלֶת

📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת חַי מֶלֶךְ זָקֵן, וְהָיוּ לוֹ שְׁלוֹשָׁה בָּנִים. בְּחֹרֶף קַר אֶחָד חָלוּ רַבִּים בָּעִיר בְּשַׁפַּעַת, וְגַם הַמֶּלֶךְ חָלָה. בַּהַתְחָלָה חָשְׁבוּ כֻּלָּם שֶׁיֵּשׁ לוֹ רַק נַזֶּלֶת קַלָּה. אֲבָל אַחַר כָּךְ עָלְתָה לוֹ קַדַּחַת גְּבוֹהָה, וּפָנָיו הִצְהִיבוּ כְּמוֹ אֵצֶל חוֹלֵה צַהֶבֶת.

רוֹפְאִים בָּאוּ מִכָּל הַמַּמְלָכָה. "יֵשׁ לַמֶּלֶךְ דַּלֶּקֶת קָשָׁה," אָמְרוּ בְּצַעַר, "וְאֵין לָנוּ תְּרוּפָה בִּשְׁבִילָהּ." שְׁלוֹשֶׁת הַבָּנִים יָשְׁבוּ בַּגַּן וּבָכוּ. פִּתְאוֹם נִגַּשׁ אֲלֵיהֶם אִישׁ זָקֵן וְאָמַר: "יֵשׁ תְּרוּפָה אַחַת — מֵי הַחַיִּים. מִי שֶׁשּׁוֹתֶה מֵהֶם, מַבְרִיא מִיָּד. אֲבָל קָשֶׁה מְאוֹד לִמְצֹא אוֹתָם."

הַבֵּן הַבְּכוֹר אָמַר בְּלִבּוֹ: "אִם אָבִיא אֶת הַמַּיִם, אַבָּא יִתֵּן לִי אֶת הַמַּמְלָכָה." הוּא רָכַב עַל סוּסוֹ, וּבַדֶּרֶךְ פָּגַשׁ גַּמָּד קָטָן. "לְאָן אַתָּה מְמַהֵר?" שָׁאַל הַגַּמָּד. "זֶה לֹא עִנְיָנְךָ, גַּמָּד טִפֵּשׁ!" עָנָה הַנָּסִיךְ בְּגַאֲוָה וְהִמְשִׁיךְ לִרְכֹּב. הַגַּמָּד כָּעַס, וְהַדֶּרֶךְ שֶׁל הַנָּסִיךְ נַעֲשְׂתָה צָרָה וְצָרָה, עַד שֶׁהוּא נִתְקַע בֵּין הֶהָרִים וְלֹא יָכוֹל הָיָה לָזוּז — לֹא קָדִימָה וְלֹא אֲחוֹרָה. גַּם לַבֵּן הַשֵּׁנִי קָרָה בְּדִיּוּק אוֹתוֹ דָּבָר.

אָז יָצָא לַדֶּרֶךְ הַבֵּן הַצָּעִיר. כְּשֶׁפָּגַשׁ אֶת הַגַּמָּד, הוּא עָצַר אֶת הַסּוּס וְעָנָה בְּנִימוּס: "אָבִי חוֹלֶה, וַאֲנִי מְחַפֵּשׂ אֶת מֵי הַחַיִּים." הַגַּמָּד חִיֵּךְ. "מִפְּנֵי שֶׁדִּבַּרְתָּ אֵלַי יָפֶה, אֶעֱזֹר לְךָ," אָמַר, וְנָתַן לוֹ שַׁרְבִיט שֶׁל בַּרְזֶל וְשַׂק קָטָן וּבוֹ לֶחֶם. "הַמַּיִם נוֹבְעִים בֶּחָצֵר שֶׁל טִירָה מְכֻשֶּׁפֶת. בַּשַּׁרְבִיט תִּפְתַּח אֶת הַשַּׁעַר, וְאֶת הַלֶּחֶם תִּתֵּן לִשְׁנֵי הָאֲרָיוֹת שֶׁשּׁוֹמְרִים עָלֶיהָ. אֲבָל זְכֹר: עָלֶיךָ לָצֵאת מִשָּׁם לִפְנֵי חֲצוֹת!"

הַנָּסִיךְ עָשָׂה כָּל מָה שֶׁאָמַר הַגַּמָּד. בְּתוֹךְ הַטִּירָה פָּגַשׁ נְסִיכָה יָפָה, וְהִיא שָׂמְחָה מְאוֹד לִרְאוֹת אוֹתוֹ, כִּי כְּשֶׁהוּא בָּא לַטִּירָה, הוּא שִׁחְרֵר אוֹתָהּ מִן הַכִּשּׁוּף. הִיא נָתְנָה לוֹ חֶרֶב וְלֶחֶם שֶׁאֵינוֹ נִגְמָר לְעוֹלָם, וְאָמְרָה: "שׁוּב אֵלַי בְּעוֹד שָׁנָה, וְנִתְחַתֵּן." הַנָּסִיךְ מִהֵר אֶל הַמַּעְיָן, מִלֵּא כַּד בְּמֵי הַחַיִּים, וְיָצָא מִן הַשַּׁעַר בְּדִיּוּק כְּשֶׁהַשָּׁעוֹן צִלְצֵל שְׁתֵּים עֶשְׂרֵה.

בַּדֶּרֶךְ הַבַּיְתָה עָבַר הַנָּסִיךְ בִּשְׁתֵּי מַמְלָכוֹת. בַּמַּמְלָכָה הָרִאשׁוֹנָה הָיָה רָעָב, וְהַנָּסִיךְ הִשְׁאִיל לַמֶּלֶךְ אֶת הַלֶּחֶם שֶׁאֵינוֹ נִגְמָר, עַד שֶׁכָּל הָעָם שָׂבַע. בַּמַּמְלָכָה הַשְּׁנִיָּה הָיְתָה מִלְחָמָה, וְהוּא הִשְׁאִיל לַמֶּלֶךְ אֶת הַחֶרֶב, וְהָאוֹיְבִים בָּרְחוּ. אַחַר כָּךְ בִּקֵּשׁ מִן הַגַּמָּד לְשַׁחְרֵר אֶת אֶחָיו. הַגַּמָּד שִׁחְרֵר אוֹתָם, אֲבָל הִזְהִיר: "הִזָּהֵר מֵהֶם — יֵשׁ לָהֶם לֵב רַע."

הַנָּסִיךְ סִפֵּר לְאֶחָיו עַל הַכֹּל. בַּלַּיְלָה, כְּשֶׁהוּא יָשַׁן, לָקְחוּ הָאַחִים אֶת מֵי הַחַיִּים מִן הַכַּד שֶׁלּוֹ, וּמִלְּאוּ אוֹתוֹ בְּמֵי יָם מְלוּחִים. בָּאַרְמוֹן נָתַן הַבֵּן הַצָּעִיר לְאָבִיו לִשְׁתּוֹת מִן הַכַּד. הַמֶּלֶךְ טָעַם אֶת מֵי הַיָּם, וּמִיָּד הִתְחִיל לְהִשְׁתַּעֵל וְהִרְגִּישׁ רַע עוֹד יוֹתֵר.

"הוּא רוֹצֶה לְהַרְעִיל אוֹתְךָ!" קָרְאוּ הָאַחִים, וְנָתְנוּ לַמֶּלֶךְ אֶת מֵי הַחַיִּים הָאֲמִתִּיִּים. הַמֶּלֶךְ שָׁתָה — וְהַקַּדַּחַת יָרְדָה, הַדַּלֶּקֶת נֶעֶלְמָה, וְהוּא קָם מִמִּטָּתוֹ בָּרִיא וְחָזָק כְּמוֹ בִּצְעִירוּתוֹ. אֲבָל הוּא כָּעַס מְאוֹד עַל בְּנוֹ הַצָּעִיר, וְגֵרֵשׁ אוֹתוֹ מִן הָאַרְמוֹן.

עָבְרוּ כַּמָּה חֳדָשִׁים. יוֹם אֶחָד הִגִּיעוּ לָאַרְמוֹן שְׁלִיחִים מִשְּׁתֵּי הַמַּמְלָכוֹת, עִם עֲגָלוֹת מְלֵאוֹת זָהָב. "זוֹ תּוֹדָה לַנָּסִיךְ הַצָּעִיר," אָמְרוּ, "הוּא הִצִּיל אוֹתָנוּ מֵרָעָב וּמִמִּלְחָמָה." הַמֶּלֶךְ הֵבִין שֶׁטָּעָה, וְשָׁלַח לְהַחֲזִיר אֶת בְּנוֹ הַבַּיְתָה.

בֵּינְתַיִם סָלְלָה הַנְּסִיכָה דֶּרֶךְ שֶׁל זָהָב אֶל הַטִּירָה שֶׁלָּהּ. "רַק מִי שֶׁיִּרְכַּב בְּאֶמְצַע הַדֶּרֶךְ — הוּא הֶחָתָן הָאֲמִתִּי," אָמְרָה לַשּׁוֹמְרִים. הָאַחִים הַגְּדוֹלִים לֹא רָצוּ לְקַלְקֵל אֶת הַזָּהָב, וְרָכְבוּ בְּצַד הַדֶּרֶךְ — וְהַשּׁוֹמְרִים שָׁלְחוּ אוֹתָם הַבַּיְתָה. הַבֵּן הַצָּעִיר חָשַׁב רַק עַל הַנְּסִיכָה וְלֹא שָׂם לֵב לַזָּהָב. הוּא רָכַב יָשָׁר בְּאֶמְצַע הַדֶּרֶךְ, וְהַשַּׁעַר נִפְתַּח לְפָנָיו. הֵם הִתְחַתְּנוּ, וְהַמֶּלֶךְ הַבָּרִיא בָּא לַחֲתֻנָּה. וְהָאַחִים הַגְּדוֹלִים? הֵם הִפְלִיגוּ לְאֶרֶץ רְחוֹקָה, וְלֹא חָזְרוּ עוֹד.

עַל פִּי ״מֵי הַחַיִּים״ (Das Wasser des Lebens) שֶׁל הָאַחִים גְּרִים (גֶּרְמַנְיָה, 1815), בְּעִבּוּד חָפְשִׁי וּמְקֻצָּר.

קַטֶּלֶת — הַמִּשְׁקָל שֶׁל שְׁמוֹת הַמַּחֲלוֹת

בַּשִּׁיעוּרִים הָאַחֲרוֹנִים לָמַדְנוּ שְׁלוֹשָׁה מִשְׁקָלִים: קַטָּל — הָאָדָם שֶׁעוֹשֶׂה (טַבָּח, שִׁיעוּר 130), מַקְטֵל — הַכְּלִי (מַפְתֵּחַ, שִׁיעוּר 132), וּמִקְטָל — הַמָּקוֹם (מִטְבָּח, שִׁיעוּר 133). הַיּוֹם נִלְמַד מִשְׁקָל שֶׁרֹב הַמִּלִּים בּוֹ הֵן שְׁמוֹת שֶׁל מַחֲלוֹת: מִשְׁקַל קַטֶּלֶת.

אֵיךְ בּוֹנִים? הָאוֹת הָרִאשׁוֹנָה בַּשֹּׁרֶשׁ מְקַבֶּלֶת פַּתָּח, הָאוֹת הַשְּׁנִיָּה מְקַבֶּלֶת דָּגֵשׁ וּסְגוֹל, הָאוֹת הַשְּׁלִישִׁית מְקַבֶּלֶת סְגוֹל, וּבַסּוֹף בָּאָה ת׳: ד־ל־ק ← דַּלֶּקֶת.

1. שְׁמוֹת הַמַּחֲלוֹת בְּמִשְׁקָל הַזֶּה הֵם בִּלְשׁוֹן נְקֵבָה, וּבְדֶרֶךְ כְּלָל מִשְׁתַּמְּשִׁים בָּהֶם בְּיָחִיד: הַשַּׁפַּעַת עָבְרָה יֵשׁ לוֹ נַזֶּלֶת קַלָּה

2. כְּשֶׁהָאוֹת הַשְּׁנִיָּה בַּשֹּׁרֶשׁ הִיא גְּרוֹנִית (כְּמוֹ ה, ח, ע), אֵין בָּהּ דָּגֵשׁ: צַהֶבֶת שַׁעֶלֶת שַׁחֶפֶת

3. כְּשֶׁהָאוֹת הַשְּׁלִישִׁית הִיא ח אוֹ ע, בָּאִים שְׁנֵי פַּתָּחִים בִּמְקוֹם שְׁנֵי הַסְּגוֹלִים (כְּמוֹ שֶׁלָּמַדְנוּ עַל אוֹתִיּוֹת גְּרוֹנִיּוֹת בְּשִׁיעוּר 126): קַדַּחַת שַׁפַּעַת

4. לֹא רַק מַחֲלוֹת: יֵשׁ גַּם כַּמָּה מִלִּים אֲחֵרוֹת בְּמִשְׁקַל קַטֶּלֶת: רַכֶּבֶת (ר־כ־ב, רוֹכְבִים בָּהּ) כַּוֶּרֶת (בַּיִת לִדְבוֹרִים) כַּסֶּפֶת (שׁוֹמְרִים בָּהּ כֶּסֶף)

מַלְכּוֹדוֹת: לֹא כָּל שֵׁם שֶׁל מַחֲלָה אוֹ שֶׁל סִימָן לְמַחֲלָה בָּנוּי בְּמִשְׁקָל הַזֶּה: שִׁעוּל חֹם הִצְטַנְּנוּת. וְלֹא כָּל מִלָּה שֶׁנִּגְמֶרֶת בְּ־ֶת הִיא בְּמִשְׁקַל קַטֶּלֶת: מַחְבֶּרֶת (מַתְחִילָה בְּ־מ׳) דֶּלֶת

טִיפּ: כְּדֵי לְזַהוֹת קַטֶּלֶת — מְחַפְּשִׂים פַּתָּח בָּאוֹת הָרִאשׁוֹנָה, דָּגֵשׁ בַּשְּׁנִיָּה (אִם הִיא לֹא גְּרוֹנִית), וְ־ֶת בַּסּוֹף. כְּדֵי לִמְצֹא אֶת הַשֹּׁרֶשׁ — מוֹחֲקִים אֶת הַ־ת׳ בַּסּוֹף: נַזֶּלֶת ← נ־ז־ל.

מִצְאוּ שְׁמוֹנָה שְׁמוֹת שֶׁל מַחֲלוֹת בְּמִשְׁקַל קַטֶּלֶת (הַמִּלִּים כְּתוּבוֹת בְּלִי נִקּוּד). לַחֲצוּ עַל הָאוֹת הָרִאשׁוֹנָה וְאַחַר כָּךְ עַל הָאוֹת הָאַחֲרוֹנָה שֶׁל הַמִּלָּה. עַל כָּל מִלָּה — עוֹד טִפָּה שֶׁל מֵי הַחַיִּים!
✍️ כְּתִיבָה
✍️ מַטְלַת כְּתִיבָה

דַּמְיְנוּ שֶׁאַתֶּם הָרוֹפְאִים שֶׁל הַמֶּלֶךְ, וְאַתֶּם כּוֹתְבִים יוֹמָן. כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים עִם מִלִּים בְּמִשְׁקַל קַטֶּלֶת, וְהַרְאוּ בַּמִּשְׁפָּט מָה מַרְגִּישׁ הַחוֹלֶה. דֻּגְמָה: הַבֹּקֶר עָלְתָה לַמֶּלֶךְ קַדַּחַת גְּבוֹהָה, וְהוּא לֹא יָכוֹל הָיָה לָקוּם מִן הַמִּטָּה.
