הֶנְס וּמַזָּלוֹ

אֵיךְ גּוּשׁ זָהָב הָפַךְ לְסוּס, לְפָרָה, לַחֲזִיר, לַאֲוָזָה וְלִשְׁתֵּי אֲבָנִים - וְלִבְסוֹף לַמַּזָּל הֲכִי גָּדוֹל מִכֻּלָּם

📖 קְרִיאָה

שֶׁבַע שָׁנִים עָבַד הֶנְס אֵצֶל אֻמָּן זָקֵן בְּעִיר רְחוֹקָה, בְּלִי לָשׁוּב וְלוּ לְיוֹם אֶחָד לְבֵית אִמּוֹ. יוֹם אֶחָד אָמַר לְאֻמָּנוֹ: "שֶׁבַע שָׁנִים עָבַדְתִּי בֶּאֱמוּנָה. הִגִּיעַ הַזְּמַן שֶׁאֵלֵךְ הַבַּיְתָה, אֶל אִמִּי." הָאֻמָּן שָׂמַח מְאֹד מֵעֲבוֹדָתוֹ הַנֶּאֱמָנָה שֶׁל הֶנְס, וְנָתַן לוֹ שְׂכַר שֶׁבַע שְׁנוֹת עֲבוֹדָה בְּבַת אַחַת: גּוּשׁ זָהָב גָּדוֹל כְּרֹאשׁוֹ שֶׁל הֶנְס. הֶנְס שָׂם אֶת הַגּוּשׁ הַכָּבֵד עַל כְּתֵפוֹ וְיָצָא לַדֶּרֶךְ, שָׂמֵחַ וְעָיֵף כְּאֶחָד.

אַחֲרֵי שָׁעָה שֶׁל הֲלִיכָה הִרְגִּישׁ הֶנְס שֶׁהַזָּהָב הוֹלֵךְ וְנַעֲשֶׂה כָּבֵד יוֹתֵר וְיוֹתֵר, וּמַשָּׂא כָּזֶה קָשֶׁה לָשֵׂאת בַּדֶּרֶךְ הָאֲרֻכָּה. אָז חָלַף לְיָדוֹ פָּרָשׁ עַל סוּס יָפֶה, שֶׁדָּהַר בְּקַלּוּת וּבְשִׂמְחָה. "כַּמָּה נָעִים לִרְכֹּב עַל סוּס!" קָרָא הֶנְס בְּקוֹל. הַפָּרָשׁ שָׁמַע וְהִצִּיעַ: "אִם הַנְּסִיעָה עַל סוּס נִרְאֵית לְךָ טוֹבָה יוֹתֵר מִלָּשֵׂאת מַשָּׂא כָּבֵד, בּוֹא נַחְלִיף." הֶנְס שָׂמַח מְאֹד וְהֶחְלִיף אֶת גּוּשׁ הַזָּהָב שֶׁלּוֹ בַּסּוּס, בְּלִי לְהַסֵּס אֲפִלּוּ רֶגַע. "אֵיזֶה מַזָּל יֵשׁ לִי!" קָרָא בְּשִׂמְחָה, וְעָלָה עַל גַּב הַסּוּס.

אַךְ הַסּוּס הָיָה פִּרְאִי וּמָהִיר מִדַּי, וְאַחֲרֵי כַּמָּה דַּקּוֹת הִשְׁלִיךְ אֶת הֶנְס אֶל הַקַּרְקַע. עוֹבֵר אֹרַח שֶׁהוֹבִיל פָּרָה שְׁמֵנָה רָאָה אֶת הַנְּפִילָה וְאָמַר: "סוּס פִּרְאִי כָּזֶה מַתְאִים רַק לְרוֹכְבִים מְנֻסִּים. פָּרָה, לְעֻמַּת זֹאת, נוֹתֶנֶת חָלָב, חֶמְאָה וּגְבִינָה בְּכָל יוֹם." הֶנְס שָׂמַח לִשְׁמֹעַ עַל כָּל הָאֹכֶל שֶׁהַפָּרָה תִּתֵּן לוֹ, וּמִיָּד הֶחְלִיף אֶת הַסּוּס בַּפָּרָה. "שׁוּב הִצְלַחְתִּי!" חָשַׁב לְעַצְמוֹ. "אֵין מִי שֶׁמַּזָּלוֹ טוֹב מִשֶּׁלִּי."

כְּשֶׁהִגִּיעַ הֶנְס לָאָחוּ הִרְגִּישׁ צָמֵא, וְנִסָּה לַחְלֹב אֶת הַפָּרָה. אַךְ הַפָּרָה הָיְתָה זְקֵנָה וְלֹא נָתְנָה טִפַּת חָלָב, וּבִבְעִיטָה חֲזָקָה כִּמְעַט הִפִּילָה אוֹתוֹ שׁוּב. קַצָּב שֶׁהוֹבִיל חֲזִיר שָׁמֵן רָאָה אֶת הַתְּקָרִית וְאָמַר: "פָּרָה זְקֵנָה כָּזֹאת שָׁוָה רַק לִבְשַׂר צָלִי, וְגַם אָז לֹא בְּשַׂר מְשֻׁבָּח. חֲזִיר, לְעֻמַּת זֹאת, נוֹתֵן נַקְנִיקִיּוֹת טְעִימוֹת." הֶנְס הֶחְלִיף בְּשִׂמְחָה אֶת הַפָּרָה בַּחֲזִיר, וְהִמְשִׁיךְ בְּדַרְכּוֹ כְּשֶׁהוּא מוֹשֵׁךְ אוֹתוֹ בְּחֶבֶל.

בַּדֶּרֶךְ פָּגַשׁ הֶנְס נַעַר שֶׁנָּשָׂא אֲוָזָה לְבָנָה וּשְׁמֵנָה מִתַּחַת לִזְרוֹעוֹ. הַנַּעַר סִפֵּר לוֹ שֶׁבַּכְּפָר הַסָּמוּךְ נִגְנַב לְאַחֲרוֹנָה חֲזִיר, וְשׁוֹטֵר מְחַפֵּשׂ אַחַר הַגַּנָּב. הֶנְס נִבְהַל וְלֹא רָצָה שֶׁיַּחְשְׁדוּ בּוֹ לַשָּׁוְא, וְהִצִּיעַ מִיָּד לְהַחְלִיף אֶת הַחֲזִיר בָּאֲוָזָה. "אֲוָזָה כָּזֹאת שָׁוָה הַרְבֵּה," חָשַׁב הֶנְס, "נוֹצוֹת רַכּוֹת לְכַר, וּבָשָׂר טָעִים לַאֲרוּחַת חַג." הַנַּעַר הִסְכִּים, וְהֶנְס הִמְשִׁיךְ בְּדַרְכּוֹ, כְּשֶׁהוּא נוֹשֵׂא אֶת הָאֲוָזָה בְּגַאֲוָה.

בִּכְנִיסָה לַכְּפָר הָאַחֲרוֹן לִפְנֵי בֵּיתוֹ יָשַׁב מְלַטֵּשׁ סַכִּינִים לְיַד אֶבֶן הַשְּׁחָזָה שֶׁלּוֹ, שָׁר לוֹ לְעַצְמוֹ בְּשִׂמְחָה. "אֵיזֶה מִקְצוֹעַ טוֹב יֵשׁ לְךָ!" אָמַר הֶנְס. "אַתָּה תָּמִיד שָׂמֵחַ." עָנָה לוֹ הַמְּלַטֵּשׁ: "מִי שֶׁיֵּשׁ לוֹ אֶבֶן רֵחַיִם טוֹבָה כְּמוֹ שֶׁלִּי, לְעוֹלָם לֹא יֶחְסַר לוֹ כֶּסֶף - כָּל בֹּקֶר מוֹצְאִים בְּתוֹכָהּ מַטְבְּעוֹת זָהָב!" הֶנְס, שֶׁחָשַׁב עַל כָּל הַזָּהָב שֶׁיּוּכַל לִמְצֹא, הֶחְלִיט לְהַחְלִיף אֶת הָאֲוָזָה בְּאֶבֶן רֵחַיִם כְּבֵדָה - וַאֲפִלּוּ קִבֵּל בְּמַתָּנָה אֶבֶן שָׂדֶה פְּשׁוּטָה כְּתוֹסֶפֶת.

עַכְשָׁו הָיָה עַל הֶנְס לָשֵׂאת שְׁתֵּי אֲבָנִים כְּבֵדוֹת עַל כְּתֵפָיו, וְהוּא הִתְקַדֵּם לְאַט וּבִכְבֵדוּת. הַשֶּׁמֶשׁ הִכְבִּידָה, וְהָרַגְלַיִם כָּאֲבוּ. הֶנְס הִתְעַיֵּף מְאֹד וְהִרְגִּישׁ צָמֵא, וּכְשֶׁרָאָה בְּאֵר עֲמֻקָּה לְיַד הַדֶּרֶךְ, הִנִּיחַ אֶת שְׁתֵּי הָאֲבָנִים עַל שְׂפַת הַבְּאֵר כְּדֵי לִשְׁתּוֹת מַיִם קָרִים.

בְּשָׁעָה שֶׁהִתְכּוֹפֵף לִשְׁתּוֹת, נָגְעָה יָדוֹ בְּאַחַת מִן הָאֲבָנִים - וְזוֹ הֶחֱלִיקָה וְנָפְלָה יָשָׁר לְתוֹךְ הַבְּאֵר, וּמִיָּד גָּרְרָה אַחֲרֶיהָ גַּם אֶת הַשְּׁנִיָּה. הֶנְס הִבִּיט בַּמַּיִם הַעֲמֻקִּים בְּתַדְהֵמָה - וְאָז, לְמַרְבֵּה הַפֶּלֶא, קָפַץ בְּשִׂמְחָה. "עַכְשָׁו אֲנִי הָאִישׁ הֲכִי בַּר־מַזָּל בָּעוֹלָם!" קָרָא בְּקוֹל רָם. "סוֹף־סוֹף אֲנִי חָפְשִׁי מֵהַמַּשָּׂא הַכָּבֵד, בְּלִי שֶׁאֶצְטָרֵךְ אֲפִלּוּ לְהִצְטַעֵר עַל כָּךְ."

הֶנְס הֵקֵל אֶת צְעָדָיו וְהִמְשִׁיךְ בְּדַרְכּוֹ, קַל וְשָׂמֵחַ, עַד שֶׁהִגִּיעַ בְּסוֹף הַיּוֹם לְבֵית אִמּוֹ. אָמְנָם לֹא נוֹתְרָה בְּיָדוֹ אַף לֹא מַטְבֵּעַ אֶחָד מִכָּל שְׂכַר שֶׁבַע הַשָּׁנִים שֶׁלּוֹ, אֲבָל הוּא הִרְגִּישׁ עָשִׁיר יוֹתֵר מִכָּל מֶלֶךְ. "אִמָּא!" קָרָא בְּשִׂמְחָה כְּשֶׁפָּתַח אֶת דֶּלֶת הַבַּיִת. "אֵין בָּעוֹלָם אָדָם בַּר־מַזָּל מִמֶּנִּי!" וְאִמּוֹ, שֶׁרָאֲתָה אֶת הַחִיּוּךְ הָרָחָב עַל פָּנָיו, יָדְעָה שֶׁהוּא צוֹדֵק - כִּי הָאֹשֶׁר הָאֲמִתִּי, הִתְבָּרֵר, אֵינוֹ תָּלוּי כְּלָל בְּמַה שֶּׁיֵּשׁ בַּיָּדַיִם.

מְסֻפָּר עַל־פִּי אַגָּדַת "הֶנְס בְּמַזָּלוֹ" (Hans im Glück) מֵאֵת הָאַחִים גְּרִים (גֶּרְמַנְיָה, פֻּרְסְמָה לָרִאשׁוֹנָה בְּ-1812), מְעֻבֶּדֶת וּמְסֻפֶּרֶת מֵחָדָשׁ - יְצִירָה מוּכֶּרֶת חֲדָשָׁה שֶׁטֶּרֶם שִׁמְּשָׁה בַּסִּדְרָה, נוֹסֶפֶת כָּעֵת לַמַּאֲגָר בְּחֵלֶק ג.2.

📚 דִּקְדּוּק — שֵׁם עֶצֶם מוּחָשׁ לְעֻמַּת מוּפְשָׁט

שֵׁם עֶצֶם מוּחָשׁ וְשֵׁם עֶצֶם מוּפְשָׁט

שֵׁם עֶצֶם מוּחָשׁ הוּא שֵׁם שֶׁל דָּבָר שֶׁאֶפְשָׁר לִתְפֹּס בַּחוּשִׁים - לִרְאוֹת אוֹתוֹ, לָגַעַת בּוֹ, לִשְׁמֹעַ, לְהָרִיחַ אוֹ לִטְעֹם. הֶנְס בַּסִּפּוּר הֶחְלִיף כָּל הַזְּמַן חֲפָצִים מוּחָשִׁים: זָהָב, סוּס, פָּרָה, חֲזִיר, אֲוָזָה וְאֶבֶן רֵחַיִם - כֻּלָּם דְּבָרִים שֶׁאֶפְשָׁר לְמַשֵּׁשׁ בַּיָּדַיִם.

שֵׁם עֶצֶם מוּפְשָׁט, לְעֻמַּת זֹאת, הוּא שֵׁם שֶׁל רַעְיוֹן, רֶגֶשׁ אוֹ מַצָּב - מַשֶּׁהוּ שֶׁאִי אֶפְשָׁר לִרְאוֹת אוֹ לָגַעַת בּוֹ בְּעַצְמוֹ, אֶלָּא רַק לְהַרְגִּישׁ אוֹתוֹ אוֹ לַחְשֹׁב עָלָיו. הַמִּלָּה הַמֶּרְכָּזִית בְּכָל הַסִּפּוּר - מַזָּל - הִיא בְּדִיּוּק שֵׁם עֶצֶם כָּזֶה: אִי אֶפְשָׁר לִרְאוֹת אוֹ לְהַחְזִיק "מַזָּל" בַּיָּד, אֲבָל אֶפְשָׁר בְּהֶחְלֵט לְהַרְגִּישׁ אוֹתוֹ. גַּם שִׂמְחָה, עֲיֵפוּת וְאֹשֶׁר הֵם שֵׁמוֹת עֶצֶם מוּפְשָׁטִים.

בְּדִיקָה פְּשׁוּטָה

אִם אֶפְשָׁר לְצַיֵּר תְּמוּנָה בְּרוּרָה שֶׁל הַדָּבָר עַצְמוֹ - זֶה מוּחָשׁ. אִם צָרִיךְ לְצַיֵּר סֵמֶל אוֹ מַצָּב כְּדֵי לְרַמֵּז עָלָיו (כְּמוֹ לֵב לְאַהֲבָה, אוֹ פָּנִים מְחַיְּכוֹת לְשִׂמְחָה) - זֶה מוּפְשָׁט.

הַרְבֵּה שֵׁמוֹת עֶצֶם מוּפְשָׁטִים נִבְנִים בְּעֶזְרַת הַסִּיּוֹמֶת ־וּת: עָיֵף ← עֲיֵפוּת, יָדִיד ← יְדִידוּת, בּוֹגֵר ← בַּגְרוּת. אֲבָל לֹא כָּל שֵׁם עֶצֶם מוּפְשָׁט נִבְנֶה כָּךְ - יֵשׁ גַּם מִלִּים כְּמוֹ מַזָּל, פַּחַד וְאַהֲבָה שֶׁאֵין לָהֶן סִיּוֹמֶת מְיֻחֶדֶת.
