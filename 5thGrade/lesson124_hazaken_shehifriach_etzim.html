📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת, בִּכְפָר קָטָן בְּיַפָּן, גָּרוּ זָקֵן וּזְקֵנָה טוֹבֵי לֵב. יְלָדִים לֹא הָיוּ לָהֶם, אֲבָל הָיָה לָהֶם כֶּלֶב לָבָן וְנֶאֱמָן בְּשֵׁם שִׁירוֹ. בְּכָל עֶרֶב יָשְׁבוּ שְׁלָשְׁתָּם לְיַד הָאָח, וְהַזְּקֵנָה הָיְתָה אוֹמֶרֶת: ״אֵין לָנוּ הַרְבֵּה, אֲבָל יֵשׁ לָנוּ זֶה אֶת זֶה.״ בַּבַּיִת הַסָּמוּךְ גָּר שָׁכֵן חַמְדָן וְקַמְצָן, שֶׁתָּמִיד הֵצִיץ אֲלֵיהֶם מֵעֵבֶר לַגָּדֵר.

יוֹם אֶחָד יָצָא הַזָּקֵן לַשָּׂדֶה, וְשִׁירוֹ רָץ אַחֲרָיו. פִּתְאוֹם עָצַר הַכֶּלֶב לְיַד עֵץ גָּבוֹהַּ, נָבַח בְּקוֹל וְחָפַר בָּאֲדָמָה בְּרַגְלָיו. הַזָּקֵן לֹא יָדַע מָה הַכֶּלֶב רוֹצֶה, אֲבָל הוּא הֵבִיא מַעְדֵּר וְהִתְחִיל לַחְפֹּר. הַבּוֹר הָלַךְ וְהֶעְמִיק, וְהַזָּקֵן יָרַד לְתוֹכוֹ. שָׁם, בְּתוֹךְ הָאֲדָמָה, מָצָא אוֹצָר: תֵּבָה מְלֵאָה מַטְבְּעוֹת זָהָב!

הַשָּׁכֵן הֶחַמְדָן רָאָה הַכֹּל מִבַּעַד לַגָּדֵר. ״גַּם אֲנִי אֵלֵךְ לַשָּׂדֶה עִם הַכֶּלֶב הַזֶּה, וְגַם אֲנִי אֶמְצָא זָהָב!״ אָמַר לְעַצְמוֹ. לְמָחֳרָת לָקַח אֶת שִׁירוֹ בְּלִי רְשׁוּת וּמָשַׁךְ אוֹתוֹ בְּחֶבֶל. ״לֵךְ מַהֵר וּמְצָא לִי זָהָב!״ צָעַק. הַכֶּלֶב הַמְּבֹהָל עָצַר לְיַד שִׂיחַ, וְהַשָּׁכֵן חָפַר וְחָפַר, אֲבָל מָצָא רַק אֲבָנִים, שִׁבְרֵי כֵּלִים וּבֹץ מַסְרִיחַ. בְּכַעַס גָּדוֹל גֵּרֵשׁ אֶת הַכֶּלֶב וְצָעַק: ״צֵא מִכָּאן וְאַל תָּשׁוּב!״ וְשִׁירוֹ בָּרַח אֶל הַיַּעַר.

הַזָּקֵן וְהַזְּקֵנָה שָׁמְעוּ מָה קָרָה, וְחִפְּשׂוּ אֶת שִׁירוֹ יָמִים רַבִּים, אֲבָל לֹא מָצְאוּ אוֹתוֹ. לַיְלָה אֶחָד רָאָה הַזָּקֵן אֶת שִׁירוֹ בַּחֲלוֹם. ״אַל תִּדְאַג לִי״, אָמַר הַכֶּלֶב. ״לֵךְ אֶל הָעֵץ שֶׁלְּיָדוֹ חָפַרְנוּ, וַעֲשֵׂה מִמֶּנּוּ מַכְתֵּשׁ. אֲנִי אָשׁוּב אֲלֵיכֶם בַּיּוֹם שֶׁכָּל הָעֵצִים יִפְרְחוּ.״ הַזָּקֵן כָּרַת אֶת הָעֵץ וּבָנָה מִמֶּנּוּ מַכְתֵּשׁ גָּדוֹל. כְּשֶׁיָּשְׁבוּ הַשְּׁנַיִם וְכָתְשׁוּ בּוֹ אֹרֶז, הָפַךְ הָאֹרֶז לְמַטְבְּעוֹת זָהָב!

הַשָּׁכֵן הֶחַמְדָן שׁוּב הֵצִיץ, וּבָא לְבַקֵּשׁ: ״הַשְׁאִילוּ לִי אֶת הַמַּכְתֵּשׁ. גַּם אֲנִי רוֹצֶה לָשֶׁבֶת וְלִכְתֹּשׁ אֹרֶז.״ הַזָּקֵן הַטּוֹב לֹא יָדַע לְסָרֵב, וְנָתַן לוֹ. אֲבָל בַּבַּיִת שֶׁל הַשָּׁכֵן הָפַךְ הָאֹרֶז לְבֹץ שָׁחֹר וּמַסְרִיחַ. הַשָּׁכֵן הִתְרַגֵּז, שָׁבַר אֶת הַמַּכְתֵּשׁ וְשָׂרַף אוֹתוֹ בָּאָח. כְּשֶׁהַזָּקֵן בָּא לְקַחַת אֶת הַמַּכְתֵּשׁ בַּחֲזָרָה, נִשְׁאַר מִמֶּנּוּ רַק אֵפֶר אָפֹר.

הַזָּקֵן אָסַף אֶת הָאֵפֶר לְתוֹךְ סַל וְהָלַךְ הַבַּיְתָה בְּעֶצֶב. בַּדֶּרֶךְ נָשְׁבָה רוּחַ, וּמְעַט אֵפֶר עָף עַל עֵץ דֻּבְדְּבָן יָבֵשׁ שֶׁעָמַד לְיַד הַדֶּרֶךְ. וְהִנֵּה פֶּלֶא: עַל הָעֲנָפִים הַיְּבֵשִׁים פָּרְחוּ פִּתְאוֹם פְּרָחִים וְרֻדִּים! ״עַכְשָׁו אֲנִי יוֹדֵעַ מָה עָלַי לַעֲשׂוֹת״, אָמַר הַזָּקֵן. ״מָחָר אֵצֵא אֶל הַדֶּרֶךְ הַגְּדוֹלָה, וַאֲפַזֵּר אֶת הָאֵפֶר עַל כָּל הָעֵצִים הַיְּבֵשִׁים.״

לְמָחֳרָת טִפֵּס הַזָּקֵן עַל עֵץ יָבֵשׁ לְיַד הַדֶּרֶךְ. בְּדִיּוּק אָז עָבַר שָׁם הָאָצִיל שֶׁל הָאֵזוֹר עִם חַיָּלָיו. ״מָה אַתָּה עוֹשֶׂה שָׁם לְמַעְלָה?״ שָׁאַל הָאָצִיל. ״אֲנִי הַזָּקֵן שֶׁמַּפְרִיחַ עֵצִים״, עָנָה. הוּא פִּזֵּר אֵפֶר מִן הַסַּל, וּמִיָּד הִתְכַּסָּה הָעֵץ כֻּלּוֹ בִּפְרָחִים. הָאָצִיל הִתְפַּעֵל מְאֹד. ״רֵד אֵלַי, זָקֵן טוֹב״, אָמַר, וְנָתַן לוֹ בְּגָדִים יָפִים וְשַׂק מָלֵא מַתָּנוֹת.

הַשָּׁכֵן הֶחַמְדָן שָׁמַע עַל כָּךְ, וְאָסַף אֶת מְעַט הָאֵפֶר שֶׁנִּשְׁאַר בָּאָח שֶׁלּוֹ. ״גַּם אֲנִי אֵדַע לְהַפְרִיחַ עֵצִים, וְגַם לִי יִתְּנוּ מַתָּנוֹת!״ אָמַר. הוּא יָשַׁב עַל עֵץ לְיַד הַדֶּרֶךְ וְחִכָּה. כְּשֶׁהָאָצִיל עָבַר, זָרַק הַשָּׁכֵן אֵפֶר בְּכָל כֹּחוֹ, אֲבָל שׁוּם פֶּרַח לֹא פָּרַח. הָרוּחַ נָשְׂאָה אֶת הָאֵפֶר יָשָׁר אֶל עֵינֵי הָאָצִיל! ״רַמַּאי!״ צָעַק הָאָצִיל. ״רֵד מִן הָעֵץ, וְצֵא מִן הָאֵזוֹר שֶׁלִּי!״ הַחַיָּלִים גֵּרְשׁוּ אֶת הַשָּׁכֵן, וְהוּא לֹא חָזַר עוֹד לַכְּפָר.

וּמָה קָרָה לְשִׁירוֹ? בָּאָבִיב, כְּשֶׁכָּל עֲצֵי הַדֻּבְדְּבָן בַּכְּפָר פָּרְחוּ, יָצָא כֶּלֶב לָבָן מִן הַיַּעַר וְרָץ אֶל הַבַּיִת. הַזְּקֵנָה בָּכְתָה מִשִּׂמְחָה. ״מֵעַכְשָׁו לֹא תֵּצֵא לְבַד״, אָמַר הַזָּקֵן, ״וּבְכָל עֶרֶב נֵשֵׁב שְׁלָשְׁתֵּנוּ יַחַד לְיַד הָאָח.״ וְכָךְ הָיָה: הַלֵּב הַטּוֹב הֵבִיא לָהֶם שִׂמְחָה, וְהַחַמְדָנוּת הֵבִיאָה לַשָּׁכֵן רַק בּוּשָׁה.

אַגָּדַת עַם יַפָּנִית, ״הָנָסָקָה גִּ׳יסָן״ (הַזָּקֵן שֶׁהִפְרִיחַ עֵצִים), בְּעִבּוּד חָפְשִׁי עִם סוֹף מְרֻכָּךְ.

📚 דִּקְדּוּק — פְּעָלִים בְּגִזְרַת פ״י

הָאוֹת י׳ שֶׁנֶּעֱלֶמֶת

בַּסִּפּוּר פָּגַשְׁנוּ פְּעָלִים כְּמוֹ יָשַׁב, יָרַד, יָדַע, יָצָא. הָאוֹת הָרִאשׁוֹנָה בַּשֹּׁרֶשׁ שֶׁלָּהֶם הִיא י׳ (י־שׁ־ב, י־ר־ד, י־ד־ע, י־צ־א), וְלָכֵן הֵם נִקְרָאִים פְּעָלִים בְּגִזְרַת פ״י. בֶּעָבָר (וְגַם בַּהֹוֶה: יוֹשֵׁב) הַיּ׳ נִשְׁמֶרֶת וְרוֹאִים אוֹתָהּ: יָשַׁב יָשַׁבְתִּי יָרַדְנוּ. אֲבָל בֶּעָתִיד, בַּצִּוּוּי וּבְשֵׁם הַפֹּעַל הַיּ׳ נֶעֱלֶמֶת.

🔑 הַכְּלָל: בֶּעָבָר רוֹאִים אֶת הַיּ׳ (יָשַׁב). בֶּעָתִיד הַיּ׳ נֶעֱלֶמֶת, וְתַחַת אוֹתִיּוֹת הֶעָתִיד יֵשׁ צֵירֵי: אֵשֵׁב, תֵּרֵד, נֵלֵךְ. בַּצִּוּוּי נִשְׁאֶרֶת מִלָּה קְצָרָה: שֵׁב! רֵד! לֵךְ! וְשֵׁם הַפֹּעַל בָּנוּי בְּדֶרֶךְ כְּלָל כָּךְ: לָ + שְׁתֵּי הָאוֹתִיּוֹת הָאַחֲרוֹנוֹת שֶׁל הַשֹּׁרֶשׁ + ת: לָשֶׁבֶת, לָרֶדֶת, לָלֶכֶת, לָלֶדֶת (וּבִגְלַל הָע׳: לָדַעַת).

הָלַךְ: הָאוֹרֵחַ הַמְּפֻרְסָם

בַּשֹּׁרֶשׁ ה־ל־ךְ אֵין י׳ בִּכְלָל, וּבֶעָבָר אוֹמְרִים הָלַכְתִּי. אֲבָל בֶּעָתִיד, בַּצִּוּוּי וּבְשֵׁם הַפֹּעַל הוּא מִתְנַהֵג בְּדִיּוּק כְּמוֹ פ״י: הָאוֹת הָרִאשׁוֹנָה נֶעֱלֶמֶת: אֵלֵךְ לֵךְ! לָלֶכֶת. לָכֵן אוֹמְרִים ״אֵלֵךְ״ וְלֹא ״אֶהֱלֹךְ״.

יָצָא: פ״י וְגַם ל״א

בַּפֹּעַל יָצָא יֵשׁ שְׁתֵּי גְּזָרוֹת יַחַד: הַיּ׳ בַּהַתְחָלָה (פ״י) וְהָא׳ בַּסּוֹף (ל״א, שֶׁלָּמַדְנוּ בְּשִׁיעוּר 119). הַיּ׳ נֶעֱלֶמֶת כָּרָגִיל: אֵצֵא תֵּצֵא צֵא!. גַּם בְּשֵׁם הַפֹּעַל מוֹסִיפִים ת׳, אֲבָל בִּגְלַל הָא׳ הַנִּקּוּד שׁוֹנֶה: לָצֵאת, וְלֹא ״לָצֶאֶת״.

מָה דּוֹמֶה לְגִזְרַת פ״נ?

בְּשִׁיעוּר 122 רָאִינוּ שֶׁבְּגִזְרַת פ״נ הַנּ׳ נֶעֱלֶמֶת בֶּעָתִיד (נָפַל ← אֶפֹּל), וּבְכַמָּה פְּעָלִים גַּם בְּשֵׁם הַפֹּעַל (נָתַן ← לָתֵת). גַּם בְּפ״י אוֹת מִן הַשֹּׁרֶשׁ נֶעֱלֶמֶת! הַהֶבְדֵּל: בְּפ״נ הָאוֹת הַבָּאָה מְקַבֶּלֶת דָּגֵשׁ (אֶפֹּל), וּבְפ״י אֵין דָּגֵשׁ, וְיֵשׁ צֵירֵי בַּהַתְחָלָה (אֵשֵׁב).

יוֹצְאִים מִן הַכְּלָל

• לֹא כָּל פֹּעַל בְּפ״י מְאַבֵּד אֶת הַיּ׳. יֵשׁ פְּעָלִים שֶׁהַיּ׳ שֶׁלָּהֶם נִשְׁאֶרֶת גַּם בֶּעָתִיד, וְתַחַת אוֹת הֶעָתִיד יֵשׁ חִירִיק: יָשַׁן ← אִישַׁן יָבֵשׁ ← יִיבַשׁ. כְּשֶׁרוֹאִים י׳ בֶּעָתִיד, כְּדַאי לִבְדֹּק אִם זֶה פֹּעַל כָּזֶה.

• אַל תִּתְבַּלְבְּלוּ בֵּין יָשַׁב (פ״י, שֹׁרֶשׁ י־שׁ־ב) לְבֵין שָׁב (ע״ו, שֹׁרֶשׁ שׁ־ו־ב, שֶׁלָּמַדְנוּ בְּשִׁיעוּר 121): ״הוּא יָשַׁב לְיַד הָאָח״, אֲבָל ״אַל תָּשׁוּב!״
