📖 קְרִיאָה

פַּעַם, בִּכְפָר קָטָן בְּהֹדּוּ, חָיוּ שִׁשָּׁה אֲנָשִׁים עִוְרִים שֶׁלֹּא רָאוּ מֵעוֹלָם דָּבָר בְּעֵינֵיהֶם, אֲבָל יָדְעוּ לְהַכִּיר אֶת הָעוֹלָם הֵיטֵב בְּעֶזְרַת הַיָּדַיִם וְהָאָזְנַיִם. יוֹם אֶחָד הִגִּיעַ אֶל הַכְּפָר סוֹחֵר עוֹבֵר אֹרַח, וְעִמּוֹ חַיָּה עֲנָקִית וּמֻזָּרָה שֶׁאִישׁ מֵאַנְשֵׁי הַכְּפָר לֹא רָאָה כָּמוֹהָ מֵעוֹלָם: פִּיל.

הַסּוֹחֵר הִסְכִּים שֶׁכָּל אֶחָד מֵהָעִוְרִים יִגַּע בַּפִּיל, כְּדֵי לְהָבִין סוֹף סוֹף אֵיךְ נִרְאֵית הַחַיָּה הַמֻּפְלָאָה הַזֹּאת. הָעִוֵּר הָרִאשׁוֹן נִגַּשׁ וְהִנִּיחַ אֶת יָדוֹ עַל צִדּוֹ הָרָחָב שֶׁל הַפִּיל. "אֲהָה!" קָרָא בְּשִׂמְחָה, "הַפִּיל דּוֹמֶה בְּדִיּוּק לְקִיר חָלָק וְעָבֶה!"

הָעִוֵּר הַשֵּׁנִי מִשֵּׁשׁ אֶת הַשֵּׁן הַחַדָּה וְהַקָּשָׁה שֶׁל הַפִּיל, וְקָבַע בְּבִטָּחוֹן: "אַתָּה טוֹעֶה! הַפִּיל דּוֹמֶה לַחֲנִית חַדָּה וּמְחֻדֶּדֶת." הָעִוֵּר הַשְּׁלִישִׁי תָּפַס בְּיָדוֹ אֶת הַחֶדֶק הַמִּתְפַּתֵּל שֶׁל הַפִּיל, וְצָעַק: "לֹא וְלֹא! זֶה בִּכְלָל נָחָשׁ אָרֹךְ וְגָמִישׁ!"

הָעִוֵּר הָרְבִיעִי חִבֵּק אֶת רַגְלוֹ הָעֲבָה שֶׁל הַפִּיל, וְאָמַר בְּבִטְחָה: "כֻּלְּכֶם טוֹעִים - זֶה בְּדִיּוּק כְּמוֹ גֶּזַע עֵץ עָבֶה וְחָזָק." הָעִוֵּר הַחֲמִישִׁי מִשֵּׁשׁ אֶת הָאֹזֶן הָרְחָבָה וְהַדַּקָּה, וְקָבַע: "לֹא, לֹא - זֶה מַנִּיף גָּדוֹל שֶׁמְּנַעְנְעִים בּוֹ בַּיָּד." וְהָעִוֵּר הַשִּׁשִּׁי, שֶׁתָּפַס בְּזְנָבוֹ הַדַּק שֶׁל הַפִּיל, סִכֵּם בְּבִטָּחוֹן גָּמוּר: "כֻּלְּכֶם לֹא צוֹדְקִים כְּלָל - זֶה סְתָם חֶבֶל דַּק וּמִתְפַּתֵּל."

מִיָּד פָּרַץ בֵּין שֵׁשֶׁת הָעִוְרִים וִכּוּחַ סוֹעֵר. כָּל אֶחָד הָיָה בָּטוּחַ שֶׁהוּא הַיָּחִיד שֶׁצּוֹדֵק, וְשֶׁכָּל הָאֲחֵרִים טוֹעִים לְגַמְרֵי. הֵם צָעֲקוּ זֶה עַל זֶה בְּקוֹלֵי קוֹלוֹת, וְאִישׁ מֵהֶם לֹא הָיָה מוּכָן לְהַאֲמִין לְתֵאוּר שֶׁל הָאֲחֵרִים.

בְּדִיּוּק אָז עָבַר בַּמָּקוֹם זָקֵן חָכָם, שֶׁשָּׁמַע אֶת הַצְּעָקוֹת מֵרָחוֹק וְנֶעֱצַר לִשְׁאֹל מָה קָרָה. כְּשֶׁכָּל אֶחָד מֵהָעִוְרִים סִפֵּר לוֹ בְּנִפְרָד מָה בְּדִיּוּק מִשֵּׁשׁ, חִיֵּךְ הַזָּקֵן בְּעֶדְנָה.

"אַתֶּם כֻּלְּכֶם צוֹדְקִים," אָמַר לָהֶם, "וְגַם כֻּלְּכֶם טוֹעִים בְּעֵת וּבְעוֹנָה אַחַת. כָּל אֶחָד מִכֶּם נָגַע רַק בְּחֵלֶק אֶחָד קָטָן שֶׁל הַפִּיל - צַד, שֵׁן, חֶדֶק, רֶגֶל, אֹזֶן אוֹ זָנָב - וְחָשַׁב שֶׁזֶּה כָּל הַפִּיל כֻּלּוֹ. אֲבָל הַפִּיל הָאֲמִתִּי אֵינוֹ רַק חֵלֶק אֶחָד - הוּא כָּל הַחֲלָקִים יַחַד."

הָעִוְרִים הִבִּיטוּ זֶה בְּזֶה בְּשֶׁקֶט, וּלְאַט לְאַט הֵבִינוּ אֶת דִּבְרֵי הֶחָכָם. "אִם נַחְבֹּר יַחַד אֶת מָה שֶׁכָּל אֶחָד מֵאִתָּנוּ גִּלָּה," אָמַר הָעִוֵּר הָרִאשׁוֹן בְּהִתְלַהֲבוּת, "נוּכַל סוֹף סוֹף לְהָבִין אֵיךְ בֶּאֱמֶת נִרְאֶה הַפִּיל בִּשְׁלֵמוּתוֹ!" מֵאָז יָשְׁבוּ שֵׁשֶׁת הָעִוְרִים יַחַד לְעִתִּים קְרוֹבוֹת, וְלִפְנֵי שֶׁקָּבְעוּ מַסְקָנָה עַל מַשֶּׁהוּ חָדָשׁ - הִקְפִּידוּ תָּמִיד לִשְׁמֹעַ קֹדֶם אֶת כָּל הַצְּדָדִים.

מְסֻפָּר עַל־פִּי מָשָׁל הֹדִּי עַתִּיק (מָסֹרֶת בּוּדְהִיסְטִית-הִינְדּוּאִית), הַמֻּכָּר בָּעוֹלָם גַּם בִּזְכוּת שִׁירוֹ שֶׁל הַמְּשׁוֹרֵר הָאֲמֵרִיקָנִי גִ'ון גּוֹדְפְרִי סַקְס (1872), מְעֻבָּד.

📚 דִּקְדּוּק — רַעֲיוֹן מֶרְכָּזִי וְסִכּוּם (חֲזָרָה)

תַּזְכֹּרֶת: רַעֲיוֹן מֶרְכָּזִי לְעֻמַּת פְּרָט

פְּרָט הוּא אֵרוּעַ בּוֹדֵד בָּעֲלִילָה. רַעֲיוֹן מֶרְכָּזִי הוּא הַמַּסְקָנָה הַכְּלָלִית שֶׁאֶפְשָׁר לְהוֹצִיא מִכָּל הַסִּפּוּר יַחַד - וְכִמְעַט תָּמִיד אֶפְשָׁר לְהַחִיל אוֹתָהּ גַּם עַל מַצָּבִים אֲחֵרִים בַּחַיִּים, לֹא רַק עַל הַדְּמֻיּוֹת בַּסִּפּוּר עַצְמוֹ.

לֹא כָּל סִפּוּר אוֹמֵר אֶת הָרַעֲיוֹן שֶׁלּוֹ בְּמִלִּים מְפֹרָשׁוֹת

בַּסִּפּוּר עַל הָעִוְרִים וְהַפִּיל, אַף אֶחָד לֹא אָמַר בְּמִלִּים "צָרִיךְ לִרְאוֹת אֶת הַתְּמוּנָה הַשְּׁלֵמָה" - זֶה נִשְׁאָר רַעֲיוֹן מְרֻמָּז (לֹא גָּלוּי), שֶׁצָּרִיךְ לְהָסִיק בְּעַצְמְכֶם מִתּוֹךְ הַמַּעֲשִׂים וְהַתּוֹצָאָה. לְעֻמַּת זֹאת, בְּמָשָׁל כְּמוֹ "הָאוֹצָר בַּכֶּרֶם" (שִׁיעוּר 29) הַלֶּקַח מֻסְבָּר בְּמִלִּים מְפֹרָשׁוֹת בְּסוֹף הַסִּפּוּר עַצְמוֹ. שְׁנֵי הַסּוּגִים תְּקִינִים - צָרִיךְ פָּשׁוּט לָדַעַת לְחַפֵּשׂ גַּם רַעֲיוֹן שֶׁלֹּא נֶאֱמַר בְּמִלִּים מְפֹרָשׁוֹת.
