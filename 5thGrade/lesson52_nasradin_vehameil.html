📖 קְרִיאָה

בְּעִיר קְטַנָּה בְּתוּרְכִּיָּה, לִפְנֵי מֵאוֹת שָׁנִים, חַי אִישׁ חָכָם וּמַצְחִיק בְּשֵׁם נַסְרָדִּין חוֹגָ'ה. הוּא לֹא הָיָה עָשִׁיר וְלֹא לָבַשׁ בְּגָדִים מְפֹאָרִים, אֲבָל בְּכָל הָאֵזוֹר יָדְעוּ שֶׁמִּי שֶׁמַּקְשִׁיב לוֹ בֶּאֱמֶת - לוֹמֵד מַשֶּׁהוּ חָדָשׁ עַל הַחַיִּים.

יוֹם אֶחָד עָרַךְ אֶחָד מֵעֲשִׁירֵי הָעִיר סְעֻדָּה גְּדוֹלָה בְּבֵיתוֹ, וְהִזְמִין אֵלֶיהָ אֶת כָּל נִכְבְּדֵי הַמָּקוֹם. גַּם נַסְרָדִּין הֻזְמַן. הַהַזְמָנָה הִגִּיעָה אֵלָיו בְּדִיּוּק כְּשֶׁסִּיֵּם יוֹם עֲבוֹדָה אָרֹךְ בַּשָּׂדֶה, וְהוּא מִהֵר לְשָׁם כְּמוֹ שֶׁהָיָה - בְּבִגְדֵי עֲבוֹדָה יְשָׁנִים וּמְאֻבָּקִים.

כְּשֶׁנִּכְנַס לָאוּלָם, אִישׁ לֹא בֵּרַךְ אוֹתוֹ לְשָׁלוֹם. הָאוֹרְחִים הַמְּפֹאָרִים הֶעֱלִימוּ עַיִן מִמֶּנּוּ, כְּאִלּוּ לֹא רָאוּ אוֹתוֹ כְּלָל. מִישֶׁהוּ הֶרְאָה לוֹ בִּתְנוּעַת יָד קְצָרָה מָקוֹם צַר בִּקְצֵה הַשֻּׁלְחָן, לְיַד הַדֶּלֶת, וְהַמְּשָׁרְתִים שֶׁעָבְרוּ עִם הַמַּגָּשִׁים הִמְשִׁיכוּ הָלְאָה בְּלִי לַעֲצֹר לְיָדוֹ.

נַסְרָדִּין יָשַׁב שָׁם שָׁעָה אֲרֻכָּה וּבָחַן אֶת הַכֹּל בְּשֶׁקֶט. הוּא לֹא כָּעַס וְלֹא הֵרִים אֶת קוֹלוֹ - הוּא רַק חָשַׁב. וּכְשֶׁקָּם לְבַסּוֹף וְיָצָא מִן הַבַּיִת, שׁוּב אִישׁ לֹא שָׂם לֵב שֶׁהוּא הָלַךְ.

בְּבֵיתוֹ פָּתַח נַסְרָדִּין אֶת הָאָרוֹן וְהוֹצִיא מִמֶּנּוּ אֶת מְעִיל הַפַּרְוָה הַיָּקָר שֶׁלּוֹ - בֶּגֶד כָּבוֹד שֶׁלָּבַשׁ רַק בַּחַגִּים. הוּא נִעֵר מִמֶּנּוּ אֶת הָאָבָק, חָגַר חֲגוֹרָה רְחָבָה, וְחָזַר אֶל הַסְּעֻדָּה בְּדִיּוּק כְּפִי שֶׁיָּצָא מִמֶּנָּה - אוֹתוֹ אִישׁ, אוֹתָן מִלִּים, אוֹתָהּ חָכְמָה.

הַפַּעַם הִשְׁתַּנָּה הַכֹּל. הַמְּאָרֵחַ מִהֵר אֵלָיו בְּעַצְמוֹ, לָחַץ אֶת יָדוֹ בִּשְׁתֵּי יָדָיו, וְהוֹשִׁיב אוֹתוֹ בְּרֹאשׁ הַשֻּׁלְחָן. הַמְּשָׁרְתִים הִגִּישׁוּ לוֹ בְּיָד רְחָבָה קְעָרוֹת מָרָק, צַלָּחוֹת אֹרֶז וּמַגָּשִׁים שֶׁל פֵּרוֹת, וְכָל מִי שֶׁיָּשַׁב סְבִיבוֹ נָתַן לוֹ אֹזֶן קַשֶּׁבֶת לְכָל מִלָּה שֶׁאָמַר.

וְאָז, לְעֵינֵי כֻּלָּם, עָשָׂה נַסְרָדִּין מַשֶּׁהוּ מוּזָר: הוּא הֵרִים אֶת שַׁרְווּל מְעִילוֹ, טָבַל אוֹתוֹ עָמֹק בְּתוֹךְ קְעָרַת הַמָּרָק, וְאָמַר בְּקוֹל בָּרוּר: "אֱכֹל, מְעִילִי, אֱכֹל!"

הַמְּאָרֵחַ נִבְהַל. "מָה אַתָּה עוֹשֶׂה?" קָרָא. "אַל תַּעֲשֶׂה מֵעַכְבָּר פִּיל, יְדִידִי - אוּלַי לֹא קִבַּלְתָּ קֹדֶם מָקוֹם טוֹב, אֲבָל אֵין שׁוּם סִבָּה לְהַכְתִּים מְעִיל יָפֶה כָּל כָּךְ!"

נַסְרָדִּין הִנִּיחַ אֶת הַשַּׁרְווּל וְחִיֵּךְ. "בָּאתִי לְכָאן קֹדֶם בִּבְגָדַי הַיְּשָׁנִים," אָמַר בְּשֶׁקֶט, "וְאִישׁ לֹא הִבִּיט בִּי. חָזַרְתִּי בַּמְּעִיל הַזֶּה - וּפִתְאֹם יֵשׁ לִי מָקוֹם, יֵשׁ לִי אֹכֶל וְיֵשׁ לִי כָּבוֹד. אִם כָּךְ, בָּרוּר שֶׁהַסְּעֻדָּה אֵינָהּ שֶׁלִּי אֶלָּא שֶׁל הַמְּעִיל - וְנִמְצָא שֶׁהוּא זֶה שֶׁצָּרִיךְ לֶאֱכֹל."

בָּאוּלָם הִשְׂתָּרְרָה דְּמָמָה. הָאוֹרְחִים הִשְׁתַּתְּקוּ וְהִסְמִיקוּ מֵרֹב בּוּשָׁה, וְהַמְּאָרֵחַ הוֹרִיד אֶת עֵינָיו וּבִקֵּשׁ סְלִיחָה. "אֶת זֶה לֹא לָמַדְתִּי מִסֵּפֶר," הוֹסִיף נַסְרָדִּין בְּרֹךְ, "כִּי אֵין חָכָם כְּבַעַל הַנִּסָּיוֹן. וְאִם תִּזְכְּרוּ מִמֶּנִּי דָּבָר אֶחָד, שֶׁיִּהְיֶה זֶה: אַל תִּסְתַּכֵּל בַּקַּנְקַן אֶלָּא בְּמַה שֶּׁיֵּשׁ בּוֹ."

מְסֻפָּר עַל־פִּי מַעֲשִׂיָּה עֲמָמִית מְפֻרְסֶמֶת עַל נַסְרָדִּין חוֹגָ'ה (הַמְּכֻנֶּה גַּם גֻ'חָא), מִן הַפּוֹלְקְלוֹר הַתּוּרְכִּי וְהַמִּזְרַח־תִּיכוֹנִי - מְעֻבָּד. הַפִּתְגָּם "אַל תִּסְתַּכֵּל בַּקַּנְקַן אֶלָּא בְּמַה שֶּׁיֵּשׁ בּוֹ" - מִשְׁנָה, מַסֶּכֶת אָבוֹת, פֶּרֶק ד' (דִּבְרֵי רַבִּי מֵאִיר).

      כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים מִשֶּׁלָּכֶם, כָּל אֶחָד עִם נִיב אוֹ פִּתְגָּם אַחֵר מִן הַשִּׁיעוּר: הֶעֱלִים עַיִן, בְּיָד רְחָבָה, אֹזֶן קַשֶּׁבֶת, עָשָׂה מֵעַכְבָּר פִּיל, אֵין חָכָם כְּבַעַל הַנִּסָּיוֹן, אַל תִּסְתַּכֵּל בַּקַּנְקַן (בַּחֲרוּ חֲמִשָּׁה מִתּוֹךְ הַשִּׁשָּׁה). שִׂימוּ לֵב לְהִשְׁתַּמֵּשׁ בַּמַּשְׁמָעוּת הַמֻּשְׁאֶלֶת, לֹא בַּמִּלּוּלִית. לְדֻגְמָה: "הַמּוֹרָה נָתְנָה לִי אֹזֶן קַשֶּׁבֶת כְּשֶׁהָיָה לִי יוֹם קָשֶׁה."
