הַחוֹלֶבֶת וְהַדְּלִי

📖 קְרִיאָה

בִּכְפָר קָטָן לְיַד הַנָּהָר גָּרָה חוֹלֶבֶת צְעִירָה בִּשְׁמָהּ פֶּרֶט. בְּבֹקֶר אֶחָד קָמָה פֶּרֶט עִם הַשַּׁחַר, חָלְבָה אֶת הַפָּרָה וּמִלְּאָה דְּלִי גָּדוֹל בְּחָלָב טָרִי. אַחַר כָּךְ שָׂמָה אֶת הַדְּלִי עַל רֹאשָׁהּ וְהָלְכָה אֶל הַשּׁוּק בָּעִיר, כְּדֵי לִמְכֹּר אֶת הֶחָלָב.

הַדֶּרֶךְ הָיְתָה אֲרֻכָּה. פֶּרֶט שָׁרָה שִׁיר שָׂמֵחַ וְגַם הִתְחִילָה לַחֲלֹם. ״אֶמְכֹּר אֶת הֶחָלָב וְאֶקְנֶה מֵאָה בֵּיצִים,״ אָמְרָה לְעַצְמָהּ. ״מֵהַבֵּיצִים יֵצְאוּ אֶפְרוֹחִים, וְהֵם יִגְדְּלוּ וְיִהְיוּ לְתַרְנְגֹלוֹת. אֶמְכֹּר אוֹתָן וְאֶקְנֶה שִׂמְלָה חֲדָשָׁה. כֻּלָּם יִסְתַּכְּלוּ עָלַי, וַאֲנִי אֵלֵךְ בְּגַאֲוָה.״

פִּתְאֹם זָקְפָה פֶּרֶט אֶת רֹאשָׁהּ בְּגַאֲוָה, כְּאִלּוּ כֻּלָּם כְּבָר מַבִּיטִים בָּהּ. הַדְּלִי זָז מִמְּקוֹמוֹ, נָפַל לָאָרֶץ, וְכָל הֶחָלָב נִשְׁפַּךְ עַל הַדֶּרֶךְ. פֶּרֶט עָמְדָה וּבָכְתָה. הַשִּׂמְלָה, הָאֶפְרוֹחִים וְהַבֵּיצִים נֶעֶלְמוּ בְּרֶגַע.

לְבַסּוֹף שָׁבָה פֶּרֶט הַבַּיְתָה עִם דְּלִי רֵיק. אִמָּהּ בָּאָה לִקְרָאתָהּ וְשָׁמְעָה אֶת כָּל הַסִּפּוּר. הָאֵם חִיְּכָה וְאָמְרָה: ״אֵין סוֹפְרִים אֶפְרוֹחִים לִפְנֵי שֶׁהַבֵּיצִים בָּקְעוּ.״ מֵאָז הוֹסִיפָה פֶּרֶט לַחֲלֹם, אֲבָל תָּמִיד שָׁמְרָה עַל הַדְּלִי.

מָשָׁל מֻכָּר מִן הַמָּסֹרֶת הָאִיזוֹפִּית (יָוָן הָעַתִּיקָה), וְזָכָה לְגִרְסָה מְפֻרְסֶמֶת מֵאֵת לָה פּוֹנְטֶן (צָרְפַת, הַמֵּאָה ה־17). בְּעִבּוּד חָפְשִׁי.

הָאוֹת הָאֶמְצָעִית שֶׁנֶּעֱלֶמֶת

בַּסִּפּוּר פָּגַשְׁנוּ פְּעָלִים כְּמוֹ קָמָה, שָׂמָה, גָּרָה, שָׁרָה, זָז, שָׁבָה, בָּאָה. בְּשֹׁרֶשׁ שֶׁלָּהֶם הָאוֹת הָאֶמְצָעִית הִיא ו׳ (ק־ו־מ, שׂ־ו־מ, ג־ו־ר) אוֹ י׳ (שׁ־י־ר). לָכֵן קוֹרְאִים לָהֶם פְּעָלִים בְּגִזְרַת ע״ו. בְּשֵׁם הַפֹּעַל הָאוֹת הָאֶמְצָעִית נִרְאֵית: לָקוּם, לָשִׂים, לָגוּר, לָשִׁיר. אֲבָל בְּעָבָר הִיא נֶעֱלֶמֶת: אוֹמְרִים קָם וְלֹא ״קוּם״, וְקָמָה וְלֹא ״קוּמָה״.

סַפְּרוּ אֶת הַסִּפּוּר בְּמִלִּים שֶׁלָּכֶם. כִּתְבוּ חָמֵשׁ שׁוּרוֹת, בְּכָל אַחַת פֹּעַל בְּגִזְרַת ע״ו בְּעָבָר, וְשִׂימוּ לֵב שֶׁאֵין ו׳ אוֹ י׳ בָּאֶמְצַע. לְמָשָׁל: ״פֶּרֶט קָמָה עִם הַשַּׁחַר, וְשָׂמָה אֶת הַדְּלִי עַל רֹאשָׁהּ.״
