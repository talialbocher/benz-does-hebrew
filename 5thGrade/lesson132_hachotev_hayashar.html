📖 קְרִיאָה

בִּכְפָר קָטָן לְיַד יַעַר גָּדוֹל גָּר פַּעַם חוֹטֵב עֵצִים עָנִי וְיָשָׁר. כָּל בֹּקֶר הוּא נָעַל אֶת דֶּלֶת הַבִּקְתָּה בְּמַפְתֵּחַ בַּרְזֶל יָשָׁן, הֶעֱמִיס עַל גַּבּוֹ אֶת הַגַּרְזֶן וְיָצָא אֶל הַיַּעַר.

בַּסַּל שֶׁלּוֹ הָיוּ כָּל הַכֵּלִים שֶׁהָיָה צָרִיךְ: מַזְמֵרָה לִגְזֹם עֲנָפִים קְטַנִּים, מַגְרֵפָה לֶאֱסֹף עָלִים יְבֵשִׁים, וְקֻפְסָה קְטַנָּה שֶׁל מַסְמְרִים לְתַקֵּן אֶת הָעֲגָלָה. אֲבָל הַכְּלִי הֶחָשׁוּב מִכֻּלָּם הָיָה הַגַּרְזֶן. בְּלִי גַּרְזֶן לֹא יָכוֹל הָיָה לַחְטֹב עֵצִים, וּבְלִי עֵצִים לֹא הָיָה לוֹ כֶּסֶף לִקְנוֹת לֶחֶם.

יוֹם אֶחָד עָמַד הַחוֹטֵב עַל גְּדַת נָהָר עָמֹק וְחָטַב עֵץ זָקֵן שֶׁנָּטָה מֵעַל הַמַּיִם. יָדָיו הָיוּ עֲיֵפוֹת וּרְטֻבּוֹת מִזֵּעָה. פִּתְאוֹם הֶחֱלִיק הַגַּרְזֶן מִבֵּין אֶצְבְּעוֹתָיו, עָף בָּאֲוִיר — וְנָפַל לְתוֹךְ הַנָּהָר. "אוֹי לִי!" קָרָא הַחוֹטֵב. "מָה אֶעֱשֶׂה עַכְשָׁו?" הוּא יָשַׁב עַל שְׂפַת הַמַּיִם וּבָכָה.

פִּתְאוֹם הִתְחִילוּ הַמַּיִם לְבַעְבֵּעַ, וּמִתּוֹכָם עָלָה זְקַן הַנָּהָר — אִישׁ מוּזָר בַּעַל זָקָן לָבָן וְאָרֹךְ, שֶׁנָּטְפוּ מִמֶּנּוּ טִפּוֹת מַיִם. "לָמָּה אַתָּה בּוֹכֶה?" שָׁאַל. הַחוֹטֵב סִפֵּר לוֹ עַל הַגַּרְזֶן שֶׁנָּפַל.

זְקַן הַנָּהָר צָלַל אֶל הַמְּצוּלוֹת וְעָלָה עִם גַּרְזֶן שֶׁל זָהָב, שֶׁנָּצַץ בַּשֶּׁמֶשׁ. "הַאִם זֶה הַגַּרְזֶן שֶׁלְּךָ?" שָׁאַל. הַחוֹטֵב הִבִּיט בַּגַּרְזֶן הַיָּפֶה וְנֶאֱנַח. "לֹא," עָנָה, "הַגַּרְזֶן שֶׁלִּי פָּשׁוּט הַרְבֵּה יוֹתֵר."

הַזָּקֵן צָלַל שׁוּב וְהֶעֱלָה גַּרְזֶן שֶׁל כֶּסֶף. "אוּלַי זֶה?" שָׁאַל. "גַּם זֶה לֹא שֶׁלִּי," אָמַר הַחוֹטֵב. בַּפַּעַם הַשְּׁלִישִׁית עָלָה הַזָּקֵן עִם גַּרְזֶן בַּרְזֶל יָשָׁן, שֶׁהָיְתָה לוֹ יָדִית עֵץ סְדוּקָה. "זֶה הוּא!" קָרָא הַחוֹטֵב בְּשִׂמְחָה. "זֶה הַגַּרְזֶן שֶׁלִּי!"

זְקַן הַנָּהָר חִיֵּךְ. "מִפְּנֵי שֶׁאָמַרְתָּ אֱמֶת," אָמַר, "קַח לְךָ אֶת שְׁלוֹשֶׁת הַגַּרְזִנִּים." הַחוֹטֵב חָזַר הַבַּיְתָה, פָּתַח אֶת הַדֶּלֶת בַּמַּפְתֵּחַ הַיָּשָׁן וְסִפֵּר לְאִשְׁתּוֹ אֶת כָּל הַסִּפּוּר.

בַּכֶּסֶף שֶׁקִּבֵּל עַל גַּרְזֶן הַכֶּסֶף קָנָה לְאִשְׁתּוֹ מַסְרֵק חָדָשׁ, וּלְעַצְמוֹ מַחְרֵשָׁה לַחֲרֹשׁ אֶת הַשָּׂדֶה הַקָּטָן שֶׁלָּהֶם. אֶת גַּרְזֶן הַזָּהָב שָׁמַר לִשְׁעַת צָרָה, וְאֶת גַּרְזֶן הַבַּרְזֶל הַיָּשָׁן הִמְשִׁיךְ לָקַחַת אִתּוֹ אֶל הַיַּעַר בְּכָל בֹּקֶר.

בַּבַּיִת הַסָּמוּךְ גָּר שָׁכֵן קַנְאִי וְחַמְדָן. כְּשֶׁשָּׁמַע אֶת הַסִּפּוּר, מִהֵר אֶל הַנָּהָר וְזָרַק אֶת הַגַּרְזֶן שֶׁלּוֹ לַמַּיִם בְּכַוָּנָה. אַחַר כָּךְ יָשַׁב עַל הַגָּדָה וְהֶעֱמִיד פָּנִים שֶׁהוּא בּוֹכֶה.

זְקַן הַנָּהָר עָלָה מִן הַמַּיִם עִם גַּרְזֶן שֶׁל זָהָב. "הַאִם זֶה שֶׁלְּךָ?" שָׁאַל. "כֵּן, כֵּן! זֶה שֶׁלִּי!" צָעַק הַשָּׁכֵן וְהוֹשִׁיט אֶת יָדָיו. אֲבָל הַזָּקֵן הִבִּיט בּוֹ בְּעֵינַיִם חֲמוּרוֹת וְנֶעְלַם בְּתוֹךְ הַמַּיִם יַחַד עִם גַּרְזֶן הַזָּהָב. גַּם אֶת הַגַּרְזֶן הַיָּשָׁן שֶׁל הַשָּׁכֵן הוּא לֹא הֶחֱזִיר.

הַשָּׁכֵן חָזַר הַבַּיְתָה בְּיָדַיִם רֵיקוֹת. לְמָחֳרָת לֹא הָיָה לוֹ בַּמֶּה לַחְטֹב עֵצִים, וְהוּא נֶאֱלַץ לְבַקֵּשׁ מִן הַחוֹטֵב לְהַשְׁאִיל לוֹ גַּרְזֶן. מוּסַר הַהַשְׂכֵּל: הַיֹּשֶׁר מִשְׁתַּלֵּם, וְהַחַמְדָן מְאַבֵּד גַּם אֶת מָה שֶׁהָיָה לוֹ.

עַל פִּי מָשָׁל הַמְּיֻחָס לְאִיזוֹפּוֹס (יָוָן הָעַתִּיקָה), ״הַחוֹטֵב וְהֶרְמֵס״, בְּעִבּוּד חָפְשִׁי. בַּמָּשָׁל הַמְּקוֹרִי מִי שֶׁעוֹלֶה מִן הַמַּיִם הוּא הָאֵל הֶרְמֵס.

מַקְטֵל וּמַקְטֵלָה — הַכְּלִי שֶׁבְּעֶזְרָתוֹ עוֹשִׂים אֶת הַפְּעֻלָּה

בְּשִׁיעוּר 130 לָמַדְנוּ אֶת מִשְׁקַל קַטָּל — הָאָדָם שֶׁעוֹשֶׂה אֶת הַפְּעֻלָּה (נַגָּר, צַיָּר). הַיּוֹם נִלְמַד מִשְׁקָל שֶׁל הַכְּלִי שֶׁבְּעֶזְרָתוֹ עוֹשִׂים אֶת הַפְּעֻלָּה: מַקְטֵל וּמַקְטֵלָה.

אֵיךְ בּוֹנִים? מוֹסִיפִים מַ לִפְנֵי הַשֹּׁרֶשׁ. הָאוֹת הָרִאשׁוֹנָה בַּשֹּׁרֶשׁ מְקַבֶּלֶת שְׁוָא, וְהָאוֹת הַשְּׁנִיָּה מְקַבֶּלֶת צֵירֵי: פ־ת־ח ← מַפְתֵּחַ. בְּמַקְטֵלָה מוֹסִיפִים גַּם ־ָה בַּסּוֹף: ג־ר־ף ← מַגְרֵפָה.

שִׂימוּ לֵב:

1. מַקְטֵל הוּא בְּדֶרֶךְ כְּלָל שֵׁם בְּזָכָר (מַפְתֵּחַ, מַסְרֵק), וּמַקְטֵלָה — שֵׁם בִּנְקֵבָה (מַגְרֵפָה, מַצְלֵמָה).

2. כְּשֶׁהָאוֹת הַשְּׁנִיָּה בַּשֹּׁרֶשׁ הִיא בּ אוֹ פּ, הִיא מְקַבֶּלֶת דָּגֵשׁ וְנִשְׁמַעַת b אוֹ p: מַחְבֵּט מַצְפֵּן

3. כְּשֶׁהַשֹּׁרֶשׁ מִסְתַּיֵּם בְּ־ח, יֵשׁ בְּמַקְטֵל פַּתָּח גְּנוּבָה בַּסּוֹף: מַפְתֵּחַ (בְּמַקְטֵלָה אֵין: מַקְדֵּחָה)

4. רַבִּים: מַקְטֵלָה ← ־וֹת, וְהַצֵּירֵי נִשְׁאָר: מַגְרֵפוֹת מַצְלֵמוֹת. מַקְטֵל ← בְּדֶרֶךְ כְּלָל ־ִים, וְהַצֵּירֵי הוֹפֵךְ לִשְׁוָא: מַסְמְרִים מַחְבְּטִים. יוֹצְאֵי דֹּפֶן: מַפְתְּחוֹת מַזְלְגוֹת — זָכָר, אֲבָל בָּרַבִּים ־וֹת.

מַלְכּוֹדוֹת: לֹא כָּל כְּלִי בָּנוּי בְּמִשְׁקָל הַזֶּה — גַּרְזֶן פַּטִּישׁ סַכִּין. וְלֹא כָּל מִלָּה שֶׁמַּתְחִילָה בְּ־מַ הִיא מַקְטֵל: בַּמִּלָּה מַלְכָּה הַ־מ׳ הִיא חֵלֶק מֵהַשֹּׁרֶשׁ מ־ל־ךְ.

טִיפּ: כְּדֵי לִמְצֹא אֶת הַשֹּׁרֶשׁ — מוֹחֲקִים אֶת הַ־מַ בַּהַתְחָלָה (וְאֶת הַ־ָה בַּסּוֹף): מַצְלֵמָה ← צ־ל־ם.

כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים עִם שְׁמוֹת כֵּלִים בְּמִשְׁקַל מַקְטֵל אוֹ מַקְטֵלָה. נַסּוּ לְהַרְאוֹת בַּמִּשְׁפָּט מָה עוֹשִׂים בַּכְּלִי. דֻּגְמָה: הַחוֹטֵב נָעַל אֶת הַבִּקְתָּה בְּמַפְתֵּחַ יָשָׁן.
