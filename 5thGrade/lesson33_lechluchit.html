📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת חָיְתָה בִּכְפָר נַעֲרָה טוֹבַת לֵב, שֶׁאִמָּהּ נִפְטְרָה כְּשֶׁהָיְתָה קְטַנָּה. אָבִיהָ נָשָׂא לְאִשָּׁה אַלְמָנָה קְשׁוּחָה, שֶׁהֵבִיאָה אִתָּהּ לַבַּיִת שְׁתֵּי בָּנוֹת מִשֶּׁלָּהּ - שְׁתֵּי אֲחָיוֹת חוֹרְגוֹת עַצְלָנִיּוֹת וְרָעוֹת לֵב.

מֵהַיּוֹם הָרִאשׁוֹן הֵטִילוּ הָאֵם הַחוֹרֶגֶת וּשְׁתֵּי בְּנוֹתֶיהָ עַל הַנַּעֲרָה אֶת כָּל הָעֲבוֹדוֹת הַקָּשׁוֹת בַּבַּיִת - לְנַקּוֹת, לְבַשֵּׁל, לְכַבֵּס - בְּעוֹד הֵן עַצְמָן יָשְׁבוּ בְּנַחַת. בַּלֵּילוֹת יָשְׁנָה הַנַּעֲרָה לְיַד הָאָח הַכְּבוּיָה, וּבְגָדֶיהָ תָּמִיד הָיוּ מְכֻסִּים בְּאֵפֶר, וּלְכֵן קָרְאוּ לָהּ כֻּלָּם "לִכְלוּכִית".

יוֹם אֶחָד הִגִּיעַ לַכְּפָר שְׁלִיחַ הַמֶּלֶךְ, וְתָקַע בַּחֲצוֹצְרָה בָּרְחוֹב הָרָאשִׁי. הוּא הִכְרִיז בְּקוֹל רָם עַל מוֹדָעָה שֶׁנִּתְלְתָה עַל שַׁעַר הָאַרְמוֹן: הַמֶּלֶךְ מַזְמִין אֶת כָּל הָעֲלָמוֹת הַצְּעִירוֹת בַּמַּמְלָכָה לְנֶשֶׁף גָּדוֹל בָּאַרְמוֹן, כְּדֵי שֶׁהַנָּסִיךְ יִבְחַר לוֹ כַּלָּה.

בְּעִקְבוֹת הַמּוֹדָעָה קִיבֵּל כָּל בֵּית אֲצֻלָּה בַּכְּפָר הַזְמָנָה אִישִׁית וּמְהֻדֶּרֶת, עִם שֵׁם הַמִּשְׁפָּחָה כָּתוּב עָלֶיהָ בִּכְתַב יָד מְהֻדָּר, תַּאֲרִיךְ הַנֶּשֶׁף, הַשָּׁעָה וּמְקוֹם הָאֵרוּעַ - אַרְמוֹן הַמֶּלֶךְ. הָאֵם הַחוֹרֶגֶת קָרְאָה אֶת הַהַזְמָנָה בְּהִתְרַגְּשׁוּת, אַךְ הוֹדִיעָה לְלִכְלוּכִית שֶׁהַהַזְמָנָה מְיֹעֶדֶת רַק לָהּ וְלִבְנוֹתֶיהָ.

בְּלֵיל הַנֶּשֶׁף, כְּשֶׁהָאֵם הַחוֹרֶגֶת וּשְׁתֵּי הַבָּנוֹת יָצְאוּ לַדֶּרֶךְ בְּכִּרְכָּרָה מְפוֹאֶרֶת, יָשְׁבָה לִכְלוּכִית לְבַדָּהּ וּבָכְתָה. לְפֶתַע הוֹפִיעָה לְפָנֶיהָ זְקֵנָה מוּזָרָה, שֶׁהִכְרִיזָה שֶׁהִיא פֵּיָה סַנְדְּקִית, וּבְיָדָהּ מַטֵּה קְסָמִים.

הַפֵּיָה נָגְעָה בִּדְלַעַת גְּדוֹלָה בַּמַּטֶּה, שֶׁהָפְכָה מִיָּד לְכִרְכָּרָה נוֹצֶצֶת; הִיא נָגְעָה בִּבְגָדֶיהָ הַיְּשָׁנִים וְהַבָּלִים שֶׁל לִכְלוּכִית, שֶׁהָפְכוּ לְשִׂמְלָה מַרְהִיבָה; וּלְרַגְלֶיהָ הוֹפִיעוּ שְׁתֵּי נַעֲלֵי זְכוּכִית קְטַנּוֹת וְנוֹצְצוֹת. "אֲבָל זִכְרִי," אָמְרָה הַפֵּיָה, "בַּחֲצוֹת הַקֶּסֶם נִגְמָר - עָלַיִךְ לַחֲזֹר הַבַּיְתָה לִפְנֵי אָז."

כְּשֶׁהִגִּיעָה לִכְלוּכִית לָאַרְמוֹן, כָּל הַפַּמַּלְיָה עָצְרָה נְשִׁימָתָהּ: אַף אֶחָד לֹא זִיהָה אֶת הַנַּעֲרָה הַיְּפֵהפִיָּה בַּשִּׂמְלָה הַנּוֹצֶצֶת. הַנָּסִיךְ עַצְמוֹ בִּיקֵּשׁ לִרְקֹד אִתָּהּ, וְלֹא הִרְפָּה מִמֶּנָּה כָּל הָעֶרֶב.

לִכְלוּכִית כִּמְעַט שָׁכְחָה אֶת הַזְּמַן, עַד שֶׁשָּׁמְעָה אֶת פַּעֲמוֹן הַשָּׁעוֹן מַתְחִיל לְצַלְצֵל אֶת חֲצוֹת. הִיא בָּרְחָה בִּמְהִירוּת מֵהָאַרְמוֹן בְּמוֹרַד הַמַּדְרֵגוֹת, וְאִיבְּדָה בְּדַרְכָּהּ אַחַת מִנַּעֲלֵי הַזְּכוּכִית שֶׁלָּהּ. הַנָּסִיךְ רָדַף אַחֲרֶיהָ, אַךְ הִיא כְּבָר נֶעֶלְמָה בַּחֲשֵׁכָה.

לְמָחֳרָת בַּבֹּקֶר יָצָא כָּרוֹז הַמֶּלֶךְ לִרְחוֹבוֹת הַמַּמְלָכָה, וְתָקַע שׁוּב בַּחֲצוֹצְרָה. הַפַּעַם הִכְרִיז הַכָּרוֹז עַל מוֹדָעָה חֲדָשָׁה: הַנָּסִיךְ יִישָׂא לְאִשָּׁה אֶת הַנַּעֲרָה שֶׁכַּף רַגְלָהּ תַּתְאִים בְּדִיּוּק לְנַעַל הַזְּכוּכִית שֶׁנִּמְצְאָה עַל מַדְרֵגוֹת הָאַרְמוֹן - וּלְשֵׁם כָּךְ יַעַבְרוּ שְׁלִיחֵי הַמֶּלֶךְ מִבַּיִת לְבַיִת בְּכָל הַמַּמְלָכָה.

כְּשֶׁהִגִּיעוּ שְׁלִיחֵי הַמֶּלֶךְ לְבֵית הָאֵם הַחוֹרֶגֶת, נִיסּוּ שְׁתֵּי הָאֲחָיוֹת הַחוֹרְגוֹת בְּכֹחַ לִדְחֹס אֶת כַּפּוֹת רַגְלֵיהֶן לַנַּעַל הַקְּטַנָּה - אַךְ הִיא לֹא הִתְאִימָה לְאַף אַחַת מֵהֶן. לְבַסּוֹף בִּיקְּשָׁה לִכְלוּכִית לְנַסּוֹת גַּם הִיא, לַמְרוֹת לַעַג הָאֵם הַחוֹרֶגֶת. הַנַּעַל הֶחֱלִיקָה לְרַגְלָהּ בְּקַלּוּת, כְּאִלּוּ נִתְפְּרָה בִּמְיֻחָד בִּשְׁבִילָהּ.

הַנָּסִיךְ זִיהָה מִיָּד אֶת פָּנֶיהָ שֶׁל הַנַּעֲרָה מֵהַנֶּשֶׁף, וְשָׂמַח מְאֹד. הוּא נָשָׂא אֶת לִכְלוּכִית לְאִשָּׁה, וּמֵאָז חָיוּ הַשְּׁנַיִם בְּאֹשֶׁר וּבְשָׁלוֹם בָּאַרְמוֹן - רָחוֹק מֵהָאֵם הַחוֹרֶגֶת וּמֵאֲחָיוֹתֶיהָ הָרָעוֹת.

עִבּוּד שֶׁל סִפּוּר הָעָם "לִכְלוּכִית", מֻכָּר בְּעִקָּר בְּגִרְסָאוֹת שֶׁל הָאַחִים גְּרִים (גֶּרְמַנְיָה) וְשֶׁל שַׁארְל פֶּרוֹ (צָרְפַת)

חמישה חלקים קבועים בהזמנה

הזמנה היא טקסט קצר שמזמין אדם מסוים, או קבוצת אנשים ידועה, לאירוע - יום הולדת, חתונה, מסיבה. בדיוק כמו מכתב אישי (שיעור 32), גם הזמנה בנויה מחלקים קבועים, תמיד באותו סדר:

מה ההבדל בין הזמנה למודעה?

לפעמים לא מזמינים אדם מסוים, אלא רוצים להודיע לכולם על משהו - בדיוק כמו המודעה שהכריז שליח המלך בכיכר הכפר. אז לא כותבים הזמנה, אלא מודעה או כרוז: הודעה המיועדת לקהל רחב ולא-מסוים, בטון רשמי וישיר יותר מהזמנה אישית.
