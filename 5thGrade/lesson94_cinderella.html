📖 קְרִיאָה

הָיְתָה פַּעַם נַעֲרָה טוֹבַת לֵב וִיפֵהפִיָּה בְּשֵׁם סִינְדֶּרֶלָה. אִמָּהּ מֵתָה, וְאָבִיהָ נָשָׂא אִשָּׁה חֲדָשָׁה, אֵם חוֹרֶגֶת קַרַת לֵב, שֶׁהָיוּ לָהּ שְׁתֵּי בָּנוֹת יְהִירוֹת וְגַאֲוְתָנִיּוֹת.

אַחֲרֵי שֶׁאָבִיהָ מֵת, הָאֵם הַחוֹרֶגֶת וּבְנוֹתֶיהָ יָשְׁבוּ בַּסָּלוֹן בִּבְגָדִים יָפִים, וְסִינְדֶּרֶלָה עָבְדָה מִבֹּקֶר עַד לַיְלָה. הִיא טִאטְאָה, שָׁטְפָה וְנִקְּתָה, וְלָנָה סָמוּךְ לַתַּנּוּר, בֵּין הַגֶּחָלִים וְהאֵפֶר. לָכֵן לָעֲגוּ לָהּ וְקָרְאוּ לָהּ "בַּת הָאֵפֶר".

יוֹם אֶחָד הוֹדִיעַ הַמֶּלֶךְ: "בָּעֶרֶב יֵעָרֵךְ נֶשֶׁף גָּדוֹל בָּאַרְמוֹן, וְכָל הַנְּעָרוֹת מוּזְמָנוֹת!" הַבָּנוֹת הִתְרַגְּשׁוּ מְאֹד, וְסִינְדֶּרֶלָה עָזְרָה לָהֶן לְהִתְלַבֵּשׁ. כְּשֶׁבִּקְּשָׁה לָבוֹא גַּם הִיא, צָחֲקָה הָאֵם הַחוֹרֶגֶת: "אַתְּ? בִּבְגָדִים מְלֻכְלָכִים? אַתְּ נִשְׁאֶרֶת בַּבַּיִת!"

כְּשֶׁהַמִּשְׁפָּחָה יָצְאָה, סִינְדֶּרֶלָה יָשְׁבָה בַּגִּנָּה וּבָכְתָה. פִּתְאֹם הוֹפִיעָה לְפָנֶיהָ פֵּיָה טוֹבָה. "אַל תִּבְכִּי, יַלְדָּתִי," אָמְרָה הַפֵּיָה, "אֲנִי אֶשְׁלַח אוֹתָךְ לַנֶּשֶׁף!"

הַפֵּיָה הֵנִיפָה אֶת שַׁרְבִיטָהּ. הַדְּלַעַת הַגְּדוֹלָה בַּגִּנָּה הָפְכָה לְכַּרְכָּרָה זְהֻבָּה, שְׁלוֹשָׁה עַכְבָּרִים הָפְכוּ לְסוּסִים לְבָנִים, וּבִגְדֵי הָעֲבוֹדָה שֶׁל סִינְדֶּרֶלָה הָפְכוּ לְשִׂמְלָה מַבְרִיקָה וּלְנַעֲלֵי זְכוּכִית קְטַנּוֹת. "אֲבָל זִכְרִי," הִזְהִירָה הַפֵּיָה, "הַקֶּסֶם נֶעֱלָם בַּחֲצוֹת. עַד אָז עָלַיִךְ לַחֲזֹר הַבַּיְתָה!"

כְּשֶׁסִּינְדֶּרֶלָה נִכְנְסָה לָאוּלָם, כֻּלָּם הִפְסִיקוּ לְדַבֵּר. הַנָּסִיךְ הִתְקָרֵב אֵלֶיהָ, הִזְמִין אוֹתָהּ לִרְקֹד, וְכָל הָעֶרֶב לֹא עָזַב אֶת יָדָהּ. הִיא הָיְתָה כֹּה מְאֻשֶּׁרֶת, שֶׁכִּמְעַט שָׁכְחָה אֶת הַשָּׁעָה.

פִּתְאֹם צִלְצֵל הַשָּׁעוֹן הַגָּדוֹל: "בּוֹם! בּוֹם!" סִינְדֶּרֶלָה נִבְהֲלָה וְנִמְלְטָה מִן הָאוּלָם. עַל הַמַּדְרֵגוֹת נָפְלָה מִמֶּנָּה אַחַת מִנַּעֲלֵי הַזְּכוּכִית, אֲבָל לֹא הָיָה לָהּ זְמַן לַעֲצֹר. הַנָּסִיךְ מָצָא אֶת הַנַּעַל וְשָׁמַר אוֹתָהּ אֶצְלוֹ.

בַּבֹּקֶר יָצָא הַנָּסִיךְ לָעִיר, וְהִצִּיעַ אֶת הַנַּעַל לְכָל נַעֲרָה. בְּבֵיתָהּ שֶׁל סִינְדֶּרֶלָה נִסּוּ שְׁתֵּי הַבָּנוֹת לִנְעֹל אוֹתָהּ, אֲבָל הַנַּעַל הָיְתָה קְטַנָּה מִדַּי לָרַגְלַיִם שֶׁלָּהֶן.

"אוּלַי תְּנַסֶּה גַּם הִיא?" שָׁאַל הַנָּסִיךְ כְּשֶׁרָאָה אֶת סִינְדֶּרֶלָה עוֹמֶדֶת בַּפִּנָּה. הָאֵם הַחוֹרֶגֶת צָחֲקָה, אֲבָל הַנָּסִיךְ הִתְעַקֵּשׁ. סִינְדֶּרֶלָה נָעֲלָה אֶת הַנַּעַל — וְהִיא הִתְאִימָה בְּדִיּוּק!

הַנָּסִיךְ וְסִינְדֶּרֶלָה הִתְחַתְּנוּ, וְסִינְדֶּרֶלָה סָלְחָה לְאִמָּהּ הַחוֹרֶגֶת. מֵאָז חָיוּ בְּשִׂמְחָה.

הַסִּפּוּר מְבֻסָּס עַל הַמַּעֲשִׂיָּה הַמְּפֻרְסֶמֶת "סִינְדֶּרֶלָה" מֵאֵת שַׁארְל פֶּרוֹ (צָרְפַת, 1697), מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ.

📚 דִּקְדּוּק — מִלִּים נִרְדָּפוֹת וּמִלִּים הֲפוּכוֹת

אוֹתָהּ מַשְׁמָעוּת, אוֹ הֵפֶךְ? וְהַאִם הֵן בֶּאֱמֶת זֵהוֹת?

מִלִּים נִרְדָּפוֹת הֵן מִלִּים שֶׁהַמַּשְׁמָעוּת שֶׁלָּהֶן קְרוֹבָה מְאֹד. מִלִּים הֲפוּכוֹת הֵן מִלִּים שֶׁהַמַּשְׁמָעוּת שֶׁלָּהֶן הִפּוּכָה. אֲבָל יֵשׁ שְׁלוֹשָׁה דְּבָרִים חֲשׁוּבִים לָדַעַת:

1. נִרְדָּפוֹת לֹא תָּמִיד זֵהוֹת. לְכָל מִלָּה יֵשׁ "גָּוֶן" מִשֶּׁלָּהּ, וְלָכֵן בּוֹחֲרִים אֶת הַמִּלָּה לְפִי הַהֶקְשֵׁר.

בַּסִּפּוּרנִרְדֶּפֶתמָה הַהֶבְדֵּל?
צָחֲקָהלָעֲגָהלוֹעֲגִים כְּדֵי לִפְגֹּעַ; צוֹחֲקִים גַּם מִשִּׂמְחָה
נִמְלְטָהרָצָהנִמְלָטִים מִפַּחַד אוֹ מִסַּכָּנָה; רָצִים גַּם סְתָם בִּשְׁבִיל הַכֵּיף
יְהִירָהגַּאֲוְתָנִיתקְרוֹבוֹת מְאֹד, וּמַתְאִימוֹת לְאוֹתָהּ דְּמוּת

2. כַּמָּה סוּגִים שֶׁל הֲפָכִים. יֵשׁ הֵפֶךְ שֶׁל "אוֹ זֶה אוֹ זֶה", וְיֵשׁ הֵפֶךְ שֶׁל שְׁנֵי קְצָווֹת, וּבֵינֵיהֶם דַּרְגּוֹת.

פְּעֻלָּה בְּכִוּוּן הֶפְכִּינִכְנְסָה ↔ יָצְאָה
שְׁנֵי קְצָווֹת שֶׁל גֹּדֶלגְּדוֹלָה ↔ קְטַנָּה (וּבֵינֵיהֶן: בֵּינוֹנִית)
מַצָּבמְלֻכְלָכִים ↔ נְקִיִּים

3. הַהֶקְשֵׁר קוֹבֵעַ אֶת הַהֵפֶךְ. הַמִּלָּה קַר יְכוֹלָה לְתָאֵר טֶמְפֶּרָטוּרָה (הֵפֶךְ: חַם) אוֹ אָדָם (קַר לֵב — הֵפֶךְ: טוֹב לֵב). לְכֵן לֹא כָּל הַהֲפָכִים שֶׁל מִלָּה מַתְאִימִים בְּכָל מִשְׁפָּט.

מִבְחַן הַהַחְלָפָה: מַחֲלִיפִים אֶת הַמִּלָּה בַּמִּשְׁפָּט. אִם הַמִּשְׁפָּט נִשְׁאַר עִם אוֹתָהּ מַשְׁמָעוּת — הַמִּלָּה נִרְדֶּפֶת. אִם הַמַּשְׁמָעוּת מִתְהַפֶּכֶת — הַמִּלָּה הֲפוּכָה.

כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים. בְּכָל מִשְׁפָּט שִׁלְבוּ אֶת שְׁתֵּי הַמִּלִּים שֶׁכְּתוּבוֹת בַּשּׁוּרָה. לְדֻגְמָה: סִינְדֶּרֶלָה הָיְתָה טוֹבַת לֵב, וְאִמָּהּ הַחוֹרֶגֶת הָיְתָה קַרַת לֵב.
