הֶעָצִיץ הָרֵיק

אַגָּדַת עַם סִינִית עַל זֶרַע אֶחָד, עָצִיץ אֶחָד וְלֵב יָשָׁר

📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת, בְּסִין הָרְחוֹקָה, חַי קֵיסָר שֶׁאָהַב פְּרָחִים יוֹתֵר מִכֹּל. בְּגַן הָאַרְמוֹן שֶׁלּוֹ פָּרְחוּ אַדְמוֹנִיּוֹת, חַרְצִיּוֹת וּפִרְחֵי שָׁקֵד, וְכָל מִי שֶׁעָבַר שָׁם עָצַר וְהִסְתַּכֵּל בָּהֶם בְּהִתְפַּעֲלוּת.

אֲבָל הַקֵּיסָר הִזְדַּקֵּן, וְלֹא הָיוּ לוֹ יְלָדִים. "מִי יִמְלֹךְ אַחֲרַי?" שָׁאַל אֶת עַצְמוֹ. יוֹם אֶחָד הוּא הוֹדִיעַ: "כָּל יֶלֶד וְכָל יַלְדָּה בַּמַּמְלָכָה מֻזְמָנִים לָאַרְמוֹן. כָּל אֶחָד יְקַבֵּל מִמֶּנִּי זֶרַע אֶחָד. מִי שֶׁיָּבִיא לִי בְּעוֹד שָׁנָה אֶת הַפֶּרַח הַיָּפֶה בְּיוֹתֵר — יִהְיֶה הַקֵּיסָר הַבָּא."

יְלָדִים מִכָּל הַמַּמְלָכָה הִזְדָּרְזוּ לְהַגִּיעַ. בֵּינֵיהֶם הָיָה גַּם פִּינְג, נַעַר עָנִי שֶׁהָיָה יָדוּעַ בַּכְּפָר כְּגַנָּן מֻצְלָח: כָּל צֶמַח שֶׁטִּפֵּל בּוֹ — צָמַח וְגָדַל. פִּינְג קִבֵּל אֶת הַזֶּרַע, רָץ הַבַּיְתָה, מִלֵּא עָצִיץ בַּאֲדָמָה טוֹבָה וְשָׁתַל אוֹתוֹ בִּזְהִירוּת.

בְּכָל יוֹם הִשְׁתַּדֵּל פִּינְג לְטַפֵּל בֶּעָצִיץ: הוּא הִשְׁקָה אוֹתוֹ בְּמַיִם צְלוּלִים, הֶעֱמִיד אוֹתוֹ בַּשֶּׁמֶשׁ וְהִשְׁתַּמֵּשׁ בָּאֲדָמָה הַשְּׁחֹרָה וְהַטּוֹבָה בְּיוֹתֵר שֶׁמָּצָא. אֲבָל שָׁבוּעַ עָבַר, וְעוֹד שָׁבוּעַ, וְדָבָר לֹא צָמַח.

פִּינְג הֶעֱבִיר אֶת הַזֶּרַע לְעָצִיץ גָּדוֹל יוֹתֵר, עִם אֲדָמָה חֲדָשָׁה. הוּא חִכָּה חֳדָשִׁים. הָאָבִיב עָבַר, הַקַּיִץ עָבַר — וְהֶעָצִיץ נִשְׁאַר רֵיק. "אֲנִי לֹא מֵבִין," לָחַשׁ, "מָה אֲנִי עוֹשֶׂה לֹא נָכוֹן?"

כְּשֶׁהִגִּיעַ הַיּוֹם שֶׁקָּבַע הַקֵּיסָר, הָרְחוֹבוֹת הִתְמַלְּאוּ בִּילָדִים שֶׁנָּשְׂאוּ עֲצִיצִים מְלֵאִים פְּרָחִים: אֲדֻמִּים, צְהֻבִּים, סְגֻלִּים — כָּל אֶחָד יָפֶה מֵהַשֵּׁנִי. פִּינְג הִתְבַּיֵּשׁ. הוּא רָצָה לְהִסְתַּתֵּר בַּבַּיִת וְלֹא לָלֶכֶת בִּכְלָל.

אֲבָל אָבִיו הִנִּיחַ יָד עַל כְּתֵפוֹ וְאָמַר: "עָשִׂיתָ כְּכָל יְכָלְתְּךָ, וְזֶה מַה שֶּׁחָשׁוּב. לֵךְ וְהָבֵא לַקֵּיסָר אֶת מַה שֶּׁיֵּשׁ לְךָ." פִּינְג לָקַח אֶת הֶעָצִיץ הָרֵיק וְהִצְטָרֵף לַיְּלָדִים שֶׁהָלְכוּ לָאַרְמוֹן.

בַּחֲצַר הָאַרְמוֹן הִצְטוֹפְפוּ מֵאוֹת יְלָדִים. הַקֵּיסָר עָבַר בֵּינֵיהֶם לְאַט וְהִסְתַּכֵּל בַּפְּרָחִים — אֲבָל פָּנָיו הָיוּ רְצִינִיִּים, וְהוּא לֹא חִיֵּךְ אֲפִלּוּ פַּעַם אַחַת. פִּתְאוֹם עָצַר מוּל פִּינְג. "מַדּוּעַ הֵבֵאתָ לִי עָצִיץ רֵיק?" שָׁאַל.

פִּינְג הִשְׁפִּיל אֶת עֵינָיו. "אֲנִי מִצְטַעֵר, אֲדוֹנִי הַקֵּיסָר," אָמַר. "שָׁתַלְתִּי אֶת הַזֶּרַע וְטִפַּלְתִּי בּוֹ כָּל יוֹם, אֲבָל הוּא לֹא צָמַח." הַיְּלָדִים סְבִיבוֹ צָחֲקוּ.

אָז חִיֵּךְ הַקֵּיסָר חִיּוּךְ רָחָב, הֵרִים אֶת הֶעָצִיץ הָרֵיק לְמַעְלָה וְקָרָא: "מָצָאתִי אֶת הַקֵּיסָר הַבָּא!" כֻּלָּם הִשְׁתּוֹמְמוּ. "אֶת כָּל הַזְּרָעִים שֶׁנָּתַתִּי לָכֶם בִּשַּׁלְתִּי קֹדֶם," הִסְבִּיר הַקֵּיסָר. "זֶרַע מְבֻשָּׁל אֵינוֹ יָכוֹל לִצְמֹחַ. רַק פִּינְג הָיָה אַמִּיץ דַּיּוֹ לְהָבִיא לִי אֶת הָאֱמֶת."

וְכָךְ, בִּזְכוּת עָצִיץ רֵיק וְלֵב יָשָׁר, הָפַךְ הַנַּעַר הֶעָנִי לַקֵּיסָר הַבָּא שֶׁל סִין. וְאוֹמְרִים שֶׁגַּם כְּשֶׁהִזְדַּקֵּן, הִקְפִּיד לְהַשְׁאִיר בְּאַרְמוֹנוֹ עָצִיץ רֵיק אֶחָד — כְּדֵי לִזְכֹּר שֶׁהָאֱמֶת יְקָרָה יוֹתֵר מִכָּל פֶּרַח.

מָקוֹר: אַגָּדַת עַם סִינִית מֻכֶּרֶת, "הֶעָצִיץ הָרֵיק" (The Empty Pot), בְּעִבּוּד חָפְשִׁי. הַמִּלִּים הַמְּסֻמָּנוֹת הֵן פְּעָלִים בְּבִנְיַן הִתְפַּעֵל שֶׁבָּהֶם הַ־ת׳ "מַחְלִיפָה מָקוֹם".

בְּשִׁיעוּר 15 לָמַדְנוּ אֶת בִּנְיַן הִתְפַּעֵל: הִתְ + שֹׁרֶשׁ — הִתְלַבֵּשׁ, הִתְבַּיֵּשׁ, הִתְפַּלֵּא. אֲבָל בַּסִּפּוּר מוֹפִיעִים פְּעָלִים שֶׁנִּרְאִים קְצָת אַחֶרֶת: הִסְתַּכֵּל, הִשְׁתַּדֵּל, הִצְטָרֵף, הִזְדַּקֵּן. מָה קָרָה לָהֶם?
