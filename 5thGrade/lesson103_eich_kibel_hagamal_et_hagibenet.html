📖 קְרִיאָה

בַּיָּמִים הָרִאשׁוֹנִים שֶׁל הָעוֹלָם, כְּשֶׁהַכֹּל עוֹד הָיָה חָדָשׁ, עָבְדוּ הַחַיּוֹת לְצַד הָאָדָם. הַסּוּס סָחַב אֶת הָעֲגָלוֹת, הַשּׁוֹר חָרַשׁ אֶת הַשָּׂדֶה, תֶּלֶם אַחַר תֶּלֶם, וְהַכֶּלֶב שָׁמַר עַל הַצֹּאן.

רַק הַגָּמָל לֹא עָזַר לְאִישׁ. הוּא יָשַׁב לְבַדּוֹ בְּלֵב הַמִּדְבָּר, לָעַס אֶת הַקּוֹצִים הַיְּבֵשִׁים וְלֹא עָבַד כְּלָל.

בַּבֹּקֶר בָּא הַסּוּס וְקָרָא: ״גָּמָל, בּוֹא וַעֲזֹר לָנוּ לִסְחֹב אֶת הָעֲגָלוֹת!״ ״הֻמְפְּ!״ עָנָה הַגָּמָל, וְהִמְשִׁיךְ לִלְעֹס.

בַּצָּהֳרַיִם בָּא הַכֶּלֶב. ״גָּמָל, בּוֹא וַעֲזֹר לָנוּ לִשְׁמֹר עַל הַצֹּאן!״ ״הֻמְפְּ!״ עָנָה הַגָּמָל שׁוּב, וְלֹא זָז מִמְּקוֹמוֹ.

לִפְנֵי הָעֶרֶב בָּא הַשּׁוֹר. ״גָּמָל, בּוֹא וַעֲזֹר לָנוּ לַחֲרֹשׁ אֶת הַשָּׂדֶה!״ ״הֻמְפְּ!״ אָמַר הַגָּמָל בַּפַּעַם הַשְּׁלִישִׁית, וְעָצַם אֶת עֵינָיו.

הַסּוּס, הַכֶּלֶב וְהַשּׁוֹר הָלְכוּ אֶל הָאָדָם. ״הַגָּמָל לֹא רוֹצֶה לַעֲבֹד, וַאֲנַחְנוּ עוֹבְדִים גַּם בִּמְקוֹמוֹ!״ הִתְלוֹנְנוּ. הָאָדָם נֶאֱנַח, אֲבָל לֹא אָמַר דָּבָר.

בָּעֶרֶב הִגִּיעַ לַמִּדְבָּר הַגִּ'ינִי שֶׁל כָּל הַחוֹלוֹת. הוּא גָּלַשׁ בְּתוֹךְ עָנָן שֶׁל אָבָק וְשָׁמַע אֶת הַחַיּוֹת מִתְלוֹנְנוֹת. הוּא עָצַר לְיַד הַגָּמָל וְשָׁאַל: ״לָמָּה אַתָּה לֹא עוֹבֵד?״

״הֻמְפְּ!״ עָנָה הַגָּמָל, בְּלִי לְהַסְתַּכֵּל עָלָיו. ״הֻמְפְּ?״ חָזַר הַגִּ'ינִי. ״אָז אֲנִי אֶתֵּן לְךָ הֻמְפְּ מַמָּשִׁי!״ הוּא מָחָא כַּף, וּפִתְאוֹם צָמְחָה עַל גַּבּוֹ שֶׁל הַגָּמָל גִּבֶּנֶת גְּדוֹלָה.

״זֹאת הַגִּבֶּנֶת שֶׁלְּךָ,״ אָמַר הַגִּ'ינִי. ״הִיא תִּשְׁמֹר בִּשְׁבִילְךָ אֹכֶל וּמַיִם, וְאַתָּה תּוּכַל לַעֲבֹד שְׁלוֹשָׁה יָמִים בְּלִי לֶאֱכֹל.״

הַגָּמָל הִסְתַּכֵּל עַל גַּבּוֹ וְהֵבִין שֶׁבִּזְבֵּז הַרְבֵּה זְמַן. הוּא הָלַךְ אֶל הַסּוּס, אֶל הַכֶּלֶב וְאֶל הַשּׁוֹר וְאָמַר: ״אֲנִי מוּכָן לַעֲבֹד.״

מֵאָז הַגָּמָל נוֹשֵׂא אֶת הַמַּשָּׂאוֹת וְעוֹבֵר אֶת הַמִּדְבָּר בְּלִי לְהִתְלוֹנֵן, וְהַגִּבֶּנֶת שֶׁעַל גַּבּוֹ הִיא הַכֹּחַ שֶׁלּוֹ.

עִבּוּד לְפִי ״אֵיךְ קִבֵּל הַגָּמָל אֶת הַגִּבֶּנֶת״ (How the Camel Got His Hump) מִתּוֹךְ Just So Stories מֵאֵת רֶדְיַרְד קִיפְּלִינְג (אַנְגְּלִיָּה, 1902), עִבּוּד חָפְשִׁי.

לְאָן הַפְּעֻלָּה הוֹלֶכֶת?

בְּכָל מִשְׁפָּט יֵשׁ מִי שֶׁעוֹשֶׂה אֶת הַפְּעֻלָּה. לִפְעָמִים הַפְּעֻלָּה נִשְׁאֶרֶת אֵצֶל הָעוֹשֶׂה, וְלִפְעָמִים הִיא ״יוֹצֵאת״ מִמֶּנּוּ וְעוֹבֶרֶת אֶל מִישֶׁהוּ אוֹ אֶל מַשֶּׁהוּ אַחֵר.

פֹּעַל יוֹצֵא — הַפְּעֻלָּה עוֹבֶרֶת אֶל מִישֶׁהוּ אוֹ מַשֶּׁהוּ (מֻשָּׂא יָשִׁיר). אֶפְשָׁר לִשְׁאֹל אַחֲרֵי הַפֹּעַל: ״אֶת מִי?״ אוֹ ״אֶת מָה?״ וְיֵשׁ תְּשׁוּבָה.

פֹּעַל עוֹמֵד — הַפְּעֻלָּה נִשְׁאֶרֶת אֵצֶל הָעוֹשֶׂה. אִי אֶפְשָׁר לִשְׁאֹל ״אֶת מָה?״.

• מִלַּת יַחַס אֵינָהּ ״אֶת״: ״הָלַךְ אֶל הַנָּהָר״ וְ״יָשַׁב בַּמִּדְבָּר״ — הַמָּקוֹם אֵינוֹ מֻשָּׂא יָשִׁיר, וְהַפֹּעַל עוֹמֵד.

• כְּשֶׁהַמֻּשָּׂא לֹא מְיֻדָּע, ״אֶת״ נֶעֱלֶמֶת: ״לָעַס קוֹצִים״ — עֲדַיִן אֶפְשָׁר לִשְׁאֹל ״לָעַס מָה?״, וְלָכֵן הַפֹּעַל יוֹצֵא.

• פֹּעַל אֶחָד יָכוֹל לִהְיוֹת גַּם עוֹמֵד וְגַם יוֹצֵא: ״הַגָּמָל אָכַל״ (עוֹמֵד) לְעֻמַּת ״הַגָּמָל אָכַל קוֹצִים״ (יוֹצֵא).

      חִשְׁבוּ עַל חַיּוֹת בַּמִּדְבָּר אוֹ בַּחַוָּה. כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים: בִּשְׁלוֹשָׁה מֵהֶם פֹּעַל יוֹצֵא (עִם ״אֶת מָה?״), וּבִשְׁנַיִם פֹּעַל עוֹמֵד. דֻּגְמָה: ״הַגָּמָל נָשָׂא אֶת הַמַּשָּׂאוֹת״ (יוֹצֵא), ״הַשֶּׁמֶשׁ זָרְחָה״ (עוֹמֵד). אֶתְגָּר: הוֹסִיפוּ אַחֲרֵי כָּל מִשְׁפָּט בְּסוֹגְרַיִם ״יוֹצֵא״ אוֹ ״עוֹמֵד״.
