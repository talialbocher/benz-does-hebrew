📖 קְרִיאָה

יוֹם קַיִץ חַם וְיָבֵשׁ שׁוֹרֵר בַּשָּׂדֶה. הַשֶּׁמֶשׁ בּוֹעֶרֶת בַּשָּׁמַיִם, וְלֹא נָפְלָה טִפַּת גֶּשֶׁם כְּבָר שְׁבוּעוֹת רַבִּים. עוֹרֵב שָׁחֹר וְגָדוֹל טָס מֵעַל הָעֵצִים הַיְּבֵשִׁים, וּגְרוֹנוֹ יָבֵשׁ וְצוֹרֵב. הָעוֹרֵב צָמֵא מְאֹד, וְהוּא מְחַפֵּשׂ בְּכָל מָקוֹם מְעַט מַיִם לִשְׁתּוֹת.

הָעוֹרֵב טָס מֵעַל הַגִּנָּה, מֵעַל הַגָּדֵר, וּמֵעַל הַבַּיִת הַלָּבָן שֶׁבִּקְצֵה הַשָּׂדֶה. הוּא מַבִּיט לְמַטָּה בְּעֵינַיִם חַדּוֹת, וּמְחַפֵּשׂ בְּרֵכָה, נַחַל, אוֹ אֲפִלּוּ שְׁלוּלִית קְטַנָּה. אֲבָל בְּכָל מָקוֹם שֶׁהוּא בּוֹדֵק, הָאֲדָמָה יְבֵשָׁה וּסְדוּקָה. הָעוֹרֵב כִּמְעַט מְאַבֵּד אֶת תִּקְוָתוֹ.

לְפֶתַע מַבְחִין הָעוֹרֵב בְּמַשֶּׁהוּ מַבְרִיק לְיַד עֵץ זַיִת בּוֹדֵד. הוּא מְשַׁנֶּה כִּוּוּן וְטָס מַהֵר לְשָׁם. עַל הָאֲדָמָה עוֹמֵד כַּד חֶרֶס גָּדוֹל, וּבְתוֹכוֹ נִרְאִים מְעַט מַיִם צְלוּלִים - מַמָּשׁ בַּתַּחְתִּית. הָעוֹרֵב נוֹחֵת עַל שְׂפַת הַכַּד, וְלִבּוֹ הוֹלֵם בְּשִׂמְחָה.

אֲבָל כְּשֶׁהָעוֹרֵב מְנַסֶּה לְהַכְנִיס אֶת מַקּוֹרוֹ אֶל תּוֹךְ הַכַּד, הוּא מְגַלֶּה בְּעָיָה: הַמַּיִם עוֹמְדִים נָמוֹךְ מִדַּי, וּמַקּוֹרוֹ הַקָּצָר לֹא מַגִּיעַ אֲלֵיהֶם. הָעוֹרֵב דּוֹחֵף אֶת רֹאשׁוֹ עוֹד וָעוֹד לְתוֹךְ צַוַּאר הַכַּד הַצַּר, אֲבָל לַשָּׁוְא. הוּא מְנַסֶּה גַּם לְהַטּוֹת אֶת הַכַּד עִם כְּנָפָיו, אֲבָל הַכַּד כָּבֵד מִדַּי, וְלֹא זָז אֲפִלּוּ מְעַט.

הָעוֹרֵב יוֹשֵׁב עַל עֵץ הַזַּיִת הַסָּמוּךְ וְחוֹשֵׁב. הוּא לֹא מְוַתֵּר בְּקַלּוּת. "אֲנִי חָכָם וְסַבְלָנִי," הוּא אוֹמֵר לְעַצְמוֹ, "וּבְוַדַּאי אֶמְצָא דֶּרֶךְ לִשְׁתּוֹת מֵהַמַּיִם הָאֵלֶּה." הוּא סוֹקֵר בְּעֵינָיו אֶת הַסְּבִיבָה, מְחַפֵּשׂ רַעְיוֹן.

לְפֶתַע הוּא מַבְחִין בַּחֲלוּקֵי אֶבֶן קְטַנִּים הַמְּפֻזָּרִים עַל הָאֲדָמָה לְיַד הַכַּד. רַעְיוֹן מַבְרִיק עוֹלֶה בְּמֹחוֹ. הָעוֹרֵב לוֹקֵחַ בְּמַקּוֹרוֹ חָלוּק אֶחָד קָטָן, וּמַכְנִיס אוֹתוֹ בִּזְהִירוּת לְתוֹךְ הַכַּד. הוּא שׁוֹמֵעַ "פְּלוֹפּ" קָטָן, וְרוֹאֶה שֶׁהַמַּיִם עוֹלִים מְעַט - מְעַט מְאֹד, אֲבָל בְּכָל זֹאת עוֹלִים.

הָעוֹרֵב שָׂמֵחַ וְנִלְהָב. הוּא חוֹזֵר לָאֲדָמָה, לוֹקֵחַ חָלוּק נוֹסָף, וּמַשְׁלִיךְ אוֹתוֹ לְתוֹךְ הַכַּד. אַחַר כָּךְ עוֹד אֶחָד, וְעוֹד אֶחָד, וְעוֹד אֶחָד. הוּא עוֹבֵד בְּסַבְלָנוּת רַבָּה, בְּלִי לְהִתְעַיֵּף וּבְלִי לְהִתְיָאֵשׁ, וּמֵבִיא חָלוּק אַחֲרֵי חָלוּק אֶל תּוֹךְ הַכַּד. עִם כָּל חָלוּק שֶׁנִּכְנָס, הַמַּיִם עוֹלִים עוֹד קְצָת.

לְאַט לְאַט, אַחֲרֵי עֶשְׂרוֹת חֲלוּקֵי אֶבֶן, הַמַּיִם מַגִּיעִים סוֹף סוֹף עַד קָרוֹב לִשְׂפַת הַכַּד. הָעוֹרֵב מַכְנִיס אֶת מַקּוֹרוֹ פְּנִימָה, וְהַפַּעַם הוּא נוֹגֵעַ בַּמַּיִם הַקָּרִים! הוּא שׁוֹתֶה לְאַט וּבַהֲנָאָה גְּדוֹלָה, לוֹגְמָה אַחַר לוֹגְמָה, עַד שֶׁצִּמְאוֹנוֹ חוֹלֵף לְגַמְרֵי.

מֵאוֹתוֹ יוֹם וָהָלְאָה, כָּל הָעוֹרְבִים בָּאֵזוֹר מְסַפְּרִים אֶת סִפּוּרוֹ שֶׁל הָעוֹרֵב הֶחָכָם. הַסִּפּוּר מְלַמֵּד אוֹתָנוּ שֶׁלֹּא תָּמִיד צָרִיךְ כֹּחַ גָּדוֹל כְּדֵי לִפְתֹּר בְּעָיָה - לִפְעָמִים סַבְלָנוּת וּמַחְשָׁבָה חֲכָמָה עוֹשׂוֹת אֶת הָעֲבוֹדָה טוֹב יוֹתֵר מִכֹּחַ.

מְסֻפָּר עַל־פִּי מִשְׁלֵי אֵיזוֹפּוֹס (יָוָן הָעַתִּיקָה), "הָעוֹרֵב וְהַכַּד" (The Crow and the Pitcher) - מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ.

בְּשִׁיעוּר 59 לָמַדְנוּ לְהַבְחִין בֵּין שְׁלֹשָׁה זְמַנֵּי פֹּעַל: עָבָר, הֹוֶה וְעָתִיד. בְּשִׁיעוּר 60 לָמַדְנוּ שֶׁבֶּעָבָר יֵשׁ תֵּשַׁע צוּרוֹת שׁוֹנוֹת - אַחַת לְכָל גּוּף (אֲנִי, אַתָּה, אַתְּ, הוּא, הִיא, אֲנַחְנוּ, אַתֶּם, אַתֶּן, הֵם/הֵן). הַיּוֹם נְגַלֶּה שֶׁבַּהֹוֶה הַכְּלָלִים שׁוֹנִים לְגַמְרֵי!

בַּהֹוֶה: לֹא לְפִי גּוּף, אֶלָּא לְפִי מִין וּמִסְפָּר

בְּנִגּוּד לֶעָבָר, בַּהֹוֶה אֵין תֵּשַׁע צוּרוֹת - יֵשׁ רַק אַרְבַּע! הַפֹּעַל בַּהֹוֶה לֹא מִשְׁתַּנֶּה לְפִי מִי מְדַבֵּר (אֲנִי, אַתָּה אוֹ הוּא) - הוּא מִשְׁתַּנֶּה רַק לְפִי שְׁנֵי דְּבָרִים: מִינוֹ שֶׁל הַנּוֹשֵׂא (זָכָר אוֹ נְקֵבָה) וּמִסְפָּרוֹ (יָחִיד אוֹ רַבִּים). כָּךְ, אֲנִי, אַתָּה וְהוּא (כֻּלָּם זָכָר יָחִיד) יֹאמְרוּ בְּדִיּוּק אוֹתָהּ צוּרָה: "כּוֹתֵב".

טָעֻיּוֹת נְפוֹצוֹת לְהִזָּהֵר מֵהֶן

⚠️ טָעוּת 1 - מִין לֹא מַתְאִים: "הַיַּלְדָּה כּוֹתֵב שִׁיעוּרֵי בַּיִת" - טָעוּת! "הַיַּלְדָּה" הִיא נְקֵבָה, לָכֵן צָרִיךְ "כּוֹתֶבֶת".
⚠️ טָעוּת 2 - מִסְפָּר לֹא מַתְאִים: "הַיְּלָדִים הוֹלֵךְ לְבֵית סֵפֶר" - טָעוּת! "הַיְּלָדִים" הֵם רַבִּים, לָכֵן צָרִיךְ "הוֹלְכִים".
⚠️ טָעוּת 3 - בִּלְבּוּל עִם הֶעָבָר: אַל תְּנַסּוּ לִמְצֹא תֵּשַׁע צוּרוֹת שׁוֹנוֹת כְּמוֹ בֶּעָבָר (שִׁיעוּר 60) - בַּהֹוֶה יֵשׁ רַק אַרְבַּע, וְהֵן לֹא תְּלוּיוֹת בְּגוּף הַמְדַבֵּר בִּכְלָל.
הוּא כּוֹתֵב = זָכָר יָחִיד הִיא כּוֹתֶבֶת = נְקֵבָה יָחִיד הֵם כּוֹתְבִים = זָכָר רַבִּים הֵן כּוֹתְבוֹת = נְקֵבָה רַבִּים

✍️ מַטְלַת כְּתִיבָה

תָּאֲרוּ יוֹם בְּחַיֵּי מִשְׁפַּחַת צִפֳּרִים: אַבָּא צִפּוֹר, אִמָּא צִפּוֹר, וְאַרְבָּעָה אֶפְרוֹחִים - שְׁנֵי זְכָרִים וּשְׁתֵּי נְקֵבוֹת. בְּכָל מִשְׁפָּט הִשְׁתַּמְּשׁוּ בַּצּוּרַת הֹוֶה הַנְּכוֹנָה שֶׁל הַפֹּעַל, כָּךְ שֶׁתַּתְאִים לְמִין וּלְמִסְפָּר שֶׁל מִי שֶׁמְּבַצֵּעַ אֶת הַפְּעֻלָּה - לְמָשָׁל: "אַבָּא הָעוֹרֵב מְחַפֵּשׂ אֹכֶל בַּבֹּקֶר."
