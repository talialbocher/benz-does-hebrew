📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת מְאֹד חַי בְּיַפָּן כּוֹרֵת בַּמְבּוּק זָקֵן עִם אִשְׁתּוֹ. בְּנֵי הַזּוּג הָיוּ עֲנִיִּים וְלֹא הָיוּ לָהֶם יְלָדִים, אַךְ הֵם הָיוּ אֲנָשִׁים טוֹבִים וּמְרֻצִּים. בְּכָל בֹּקֶר יָצָא הַזָּקֵן אֶל הַיַּעַר וְכָרַת גִּזְעֵי בַּמְבּוּק, כְּדֵי לִקְלֹעַ מֵהֶם סַלִּים וּכְלֵי בַּיִת לִמְכִירָה.

יוֹם אֶחָד רָאָה הַזָּקֵן גֶּזַע בַּמְבּוּק מְיֻחָד, שֶׁזָּהַר בְּאוֹר עָדִין וְזָהֹב. בְּהִתְרַגְּשׁוּת רַבָּה בָּקַע אֶת הַגֶּזַע, וּלְתִמְהוֹנוֹ הַגָּדוֹל מָצָא בְּתוֹכוֹ יַלְדָּה זְעִירָה, לֹא גְּדוֹלָה מֵאֶצְבַּע, יוֹשֶׁבֶת וּמְחַיֶּכֶת אֵלָיו. "בְּוַדַּאי הַשָּׁמַיִם שָׁלְחוּ לִי אוֹתָךְ," אָמַר הַזָּקֵן בְּהִתְרַגְּשׁוּת, וְנָשָׂא אוֹתָהּ בִּזְהִירוּת הַבַּיְתָה בְּתוֹךְ כַּפּוֹת יָדָיו.

הַזָּקֵן וְאִשְׁתּוֹ אִמְּצוּ אֶת הַיַּלְדָּה וְגִדְּלוּ אוֹתָהּ בְּאַהֲבָה גְּדוֹלָה. מֵאוֹתוֹ יוֹם, בְּכָל פַּעַם שֶׁכָּרַת הַזָּקֵן גֶּזַע בַּמְבּוּק, מָצָא בְּתוֹכוֹ חֲתִיכַת זָהָב - וּבְנֵי הַזּוּג הֶעֱשִׁירוּ לְאַט־לְאַט. הַיַּלְדָּה עַצְמָהּ גָּדְלָה בִּמְהִירוּת מַפְלִיאָה: תּוֹךְ שְׁלוֹשָׁה חֳדָשִׁים בִּלְבַד הָפְכָה מִתִּינֹקֶת זְעִירָה לְאִשָּׁה צְעִירָה בַּעֲלַת יֹפִי מְסַנְוֵר. קָרְאוּ לָהּ קָגוּיָה־הִימֶה - "נְסִיכַת הַלַּיְלָה הַזּוֹהֵר."

שְׁמוּעוֹת עַל יָפְיָהּ שֶׁל קָגוּיָה הִתְפַּשְּׁטוּ בְּכָל הָאָרֶץ, וַחֲמִשָּׁה נְסִיכִים אֲצִילִים הִגִּיעוּ בְּזֶה אַחַר זֶה לְבַקֵּשׁ אֶת יָדָהּ. קָגוּיָה לֹא רָצְתָה לְהִנָּשֵׂא לְאַף אֶחָד מֵהֶם, וְלָכֵן נָתְנָה לְכָל אֶחָד מְשִׂימָה בִּלְתִּי אֶפְשָׁרִית: לְהָבִיא לָהּ אוֹצָר אַגָּדִי מִקָּצֶה הָעוֹלָם. אֶחָד הָיָה צָרִיךְ לְהָבִיא עָנָף שֶׁל אֶבֶן חֵן מֵהָר קָסוּם, וְאַחֵר - אֶבֶן יְקָרָה מִצַּוָּאר דְּרָקוֹן. כֻּלָּם נִכְשְׁלוּ, וְקָגוּיָה נִשְׁאֲרָה חָפְשִׁיָּה.

אֲפִלּוּ הַקֵּיסָר עַצְמוֹ שָׁמַע עַל יָפְיָהּ שֶׁל קָגוּיָה, וּבִקֵּר בְּבֵית הוֹרֶיהָ כְּדֵי לְבַקֵּשׁ אוֹתָהּ לְאִשָּׁה. גַּם לוֹ הֵשִׁיבָה קָגוּיָה בְּעֶדְנָה שֶׁהִיא אֵינָהּ יְכוֹלָה לְהִנָּשֵׂא לְאִישׁ - וְלֹא הִסְבִּירָה מַדּוּעַ. הַקֵּיסָר, שֶׁלֹּא וִתֵּר בְּקַלּוּת, הִמְשִׁיךְ לִשְׁלֹחַ לָהּ מִכְתָּבִים יְדִידוּתִיִּים בְּכָל עוֹנָה, וְהִיא הֵשִׁיבָה לוֹ בְּחֹם - אַךְ מֵעוֹלָם לֹא הִסְכִּימָה לְעָזֹב אֶת בֵּית הוֹרֶיהָ.

עִם הַשָּׁנִים שָׂם לֵב הַזָּקֵן שֶׁבְּכָל לֵיל יָרֵחַ מָלֵא יוֹשֶׁבֶת קָגוּיָה לְיַד הַחַלּוֹן וּבוֹכָה בְּשֶׁקֶט, מַבִּיטָה בַּיָּרֵחַ הָעָגֹל וְהַזּוֹהֵר. "מָה קוֹרֶה לָךְ, בִּתִּי?" שָׁאַל בִּדְאָגָה. לְבַסּוֹף גִּלְּתָה לוֹ קָגוּיָה סוֹד גָּדוֹל: הִיא אֵינָהּ בַּת אָדָם רְגִילָה, אֶלָּא נְסִיכָה מִן הַיָּרֵחַ, שֶׁנִּשְׁלְחָה לָעוֹלָם לִזְמַן מֻגְבָּל בִּלְבַד. בְּלֵיל הַיָּרֵחַ הַמָּלֵא הַקָּרוֹב יָבוֹאוּ בְּנֵי הַיָּרֵחַ לְקַחַת אוֹתָהּ הַבַּיְתָה, וְהִיא לֹא תּוּכַל לְהִשָּׁאֵר.

הַהוֹרִים נִשְׁבְּרוּ מִצַּעַר, וְהַקֵּיסָר, כְּשֶׁשָּׁמַע עַל כָּךְ, שָׁלַח חַיָּלִים רַבִּים לִשְׁמֹר עַל הַבַּיִת. אַךְ בְּלֵיל הַיָּרֵחַ הַמָּלֵא יָרַד מִן הַשָּׁמַיִם אוֹר מְסַנְוֵר, וּמֶרְכָּבָה זְהֻבָּה נָחֲתָה בֶּחָצֵר. הַחַיָּלִים נִסּוּ לְמָנֹעַ אֶת בּוֹאָם שֶׁל בְּנֵי הַיָּרֵחַ, אַךְ יְדֵיהֶם קָפְאוּ בָּאֲוִיר וְלֹא יָכְלוּ לְהָנִיעַ אֶצְבַּע.

לִפְנֵי שֶׁעָלְתָה עַל הַמֶּרְכָּבָה, כָּתְבָה קָגוּיָה מִכְתָּבִים חַמִּים לְהוֹרֶיהָ וְלַקֵּיסָר, וְהִשְׁאִירָה לַקֵּיסָר בַּקְבּוּק קָטָן שֶׁל סַם הָאַלְמָוֶת, כְּמַתָּנָה אַחֲרוֹנָה. אַחַר כָּךְ עָטְתָה גְּלִימַת נוֹצוֹת קְסוּמָה, וְאֶת כָּל צַעֲרָהּ עַל פְּרֵדָה שָׁכְחָה בְּבַת אַחַת - כִּי כָּךְ פּוֹעֶלֶת הַגְּלִימָה עַל בְּנֵי הַיָּרֵחַ. הִיא עָלְתָה עַל הַמֶּרְכָּבָה הַזְּהֻבָּה וְנֶעֶלְמָה בַּשָּׁמַיִם.

הַקֵּיסָר, שֶׁלֹּא רָצָה לָחְיוֹת לָנֶצַח בְּעוֹלָם שֶׁבּוֹ אֵין קָגוּיָה, צִוָּה עַל חַיָּלָיו לִשְׂרֹף אֶת סַם הָאַלְמָוֶת בְּרֹאשׁ הָהָר הַגָּבֹהַּ בְּיוֹתֵר בְּיַפָּן. עַד הַיּוֹם מְסַפְּרִים שֶׁמֵּאָז עוֹלֶה עָשָׁן מֵרֹאשׁ הָהָר הַזֶּה, שֶׁנִּקְרָא הַר פּוּגִ'י - וְזוֹ אַחַת הָאַגָּדוֹת הָעַתִּיקוֹת בְּיוֹתֵר בְּיַפָּן.

מְסֻפָּר עַל־פִּי "טָאקֶטוֹרִי מוֹנוֹגָטָרִי" (סִפּוּר כּוֹרֵת הַבַּמְבּוּק), אַגָּדָה יַפָּנִית עַתִּיקָה מֵהַמֵּאָה הַ־9 לִסְפִירָה, מְעֻבֶּדֶת.

⚠️ שִׂימוּ לֵב: "אַתָּה" (זָכָר) מְנֻקָּד בְּקָמָץ (תָּ) וְ"אַתְּ" (נְקֵבָה) מְנֻקָּד בִּשְׁוָא (תְּ) - הַהֶבְדֵּל קָטָן בַּכְּתָב אֲבָל חָשׁוּב בַּדִּבּוּר: "אַתָּה גָּדַלְתָּ" לְעֻמַּת "אַתְּ גָּדַלְתְּ". וְגוּף שְׁלִישִׁי יָחִיד זָכָר (הוּא) הוּא הַגּוּף הַיָּחִיד בֶּעָבָר שֶׁאֵין לוֹ שׁוּם סִיּוֹמֶת בִּכְלָל.
✏️ מִשְׂחַק דִּקְדּוּק — תַּפְזֹרֶת הַגּוּפִים
🔍 תַּפְזֹרֶת מִלִּים - כִּנּוּיֵי הַגּוּף
לְחַצְתֶּם עַל הָאוֹת הָרִאשׁוֹנָה וְעַל הָאַחֲרוֹנָה שֶׁל כָּל מִלָּה, כְּדֵי לִמְצֹא אֶת חֲמֵשֶׁת כִּנּוּיֵי הַגּוּף וּמִלַּת הַזְּמַן שֶׁמֻּסְתָּרִים בָּרֶשֶׁת
✍️ כְּתִיבָה
✍️ מַטְלַת כְּתִיבָה

כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים בִּזְמַן עָבָר עַל הַסִּפּוּר שֶׁל קָגוּיָה־הִימֶה, כָּל מִשְׁפָּט בְּגוּף אַחֵר, לְפִי הַהַנְחָיָה שֶׁלְּיַד כָּל שׁוּרָה - לְמָשָׁל: "הַזָּקֵן מָצָא יַלְדָּה בְּתוֹךְ הַבַּמְבּוּק."
