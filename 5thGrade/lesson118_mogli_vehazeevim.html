📖 קְרִיאָה

בְּלֵב הַגַּ׳נְגֶל הַהוֹדִי רָאֲתָה אִמָּא זְאֵבָה תִּינוֹק אֱנוֹשִׁי קָטָן לְיַד הַמְּעָרָה שֶׁלָּהּ. הַתִּינוֹק לֹא בָּכָה. הוּא צָחַק וְהוֹשִׁיט אֶת יָדָיו אֶל הַגּוּרִים. אִמָּא זְאֵבָה הִרְגִּישָׁה אַהֲבָה, וְאָמְרָה לְבַעֲלָהּ: ״אֲנִי רוֹצָה לְגַדֵּל אֶת הַיֶּלֶד הַזֶּה.״

בְּאוֹתוֹ לַיְלָה הִתְקָרֵב הַנָּמֵר הָרָעֵב אֶל הַמְּעָרָה וְשָׁאַג: ״תְּנוּ לִי אֶת הַיֶּלֶד!״ אִמָּא זְאֵבָה עָמְדָה בַּכְּנִיסָה וְעָנְתָה בְּקוֹל חָזָק: ״הַיֶּלֶד שֶׁלָּנוּ עַכְשָׁו. לֹא נִתֵּן לְךָ אוֹתוֹ.״ הַנָּמֵר גִּלְגֵּל עֵינַיִם וְהָלַךְ.

הַזְּאֵבִים קָרְאוּ לַיֶּלֶד מוֹגְלִי, וְהוּא גָּדַל בֵּינֵיהֶם. הוּא עָלָה עַל עֵצִים, שָׁתָה מִן הַנָּהָר וְרָץ עִם הַגּוּרִים. הַדֹּב בָּלוּ לִמֵּד אוֹתוֹ אֶת חֻקֵּי הַגַּ׳נְגֶל, וְהַפַּנְתֵּר בָּגִירָה הָיָה לוֹ חָבֵר וּמוֹרֶה.

שָׁנִים עָבְרוּ. הַנָּמֵר חָזַר וְסִפֵּר לַעֲדַת הַזְּאֵבִים שֶׁהַיֶּלֶד אֱנוֹשִׁי וְאֵינוֹ שַׁיָּךְ לַגַּ׳נְגֶל. כָּל הַזְּאֵבִים הִתְכַּנְּסוּ בַּמּוֹעָצָה. מוֹגְלִי עָלָה עַל הַסֶּלַע וְקָרָא בְּקוֹל: ״אֲנִי רָאִיתִי אֶת הָאֵשׁ שֶׁל בְּנֵי הָאָדָם!״ הוּא הֵרִים עָנָף בּוֹעֵר, וְהַנָּמֵר נָסוֹג וְנֶעֱלַם בֵּין הָעֵצִים.

אַחֲרֵי הַמּוֹעָצָה הֵבִין מוֹגְלִי שֶׁהוּא רוֹצֶה לָשׁוּב אֶל בְּנֵי הָאָדָם, אֲבָל לֹא לִשְׁכֹּחַ אֶת הַזְּאֵבִים. כְּשֶׁהָלַךְ, אִמָּא זְאֵבָה בָּכְתָה וְהַגּוּרִים בָּכוּ אִתָּהּ. מוֹגְלִי הִבְטִיחַ שֶׁיָּבוֹא לְבַקֵּר אוֹתָם בְּכָל פַּעַם שֶׁיִּתְגַּעְגַּע.

רַדְיַארְד קִיפְּלִינְג (אַנְגְּלִיָּה, 1894), מִתּוֹךְ ״סֵפֶר הַגַּ׳נְגֶל״, בְּעִבּוּד חָפְשִׁי וּמְקֻצָּר.

📚 דִּקְדּוּק — פְּעָלִים בְּגִזְרַת ל״ה

הַ־ה׳ שֶׁנֶּעֱלֶמֶת אוֹ נַעֲשֵׂית י׳

בַּסִּפּוּר פָּגַשְׁנוּ פְּעָלִים כְּמוֹ רָאָה, בָּכָה, עָלָה, שָׁתָה, עָנָה, רָצָה. בְּכֻלָּם הָאוֹת הָאַחֲרוֹנָה בַּשֹּׁרֶשׁ הִיא ה׳ (ר־א־ה, ב־כ־ה, ע־ל־ה). לָכֵן קוֹרְאִים לָהֶם פְּעָלִים בְּגִזְרַת ל״ה (״ל״ — הָאוֹת הָאַחֲרוֹנָה, ״ה״ — הָאוֹת שֶׁהִיא ה׳). בִּזְמַן עָבָר הַ־ה׳ הָאַחֲרוֹנָה מִשְׁתַּנָּה לְפִי הַסִּיּוֹמֶת.

🔑 הַכְּלָל: בְּעָבָר, בְּגוּף ״הוּא״ הַ־ה׳ נִשְׁאֶרֶת (רָאָה, בָּנָה). לִפְנֵי הַסִּיּוֹמוֹת ־תִּי, ־תָּ, ־תְּ, ־נוּ, ־תֶּם, ־תֶּן הִיא הוֹפֶכֶת לְ־י׳ (רָאִיתִי). בְּ״הִיא״ (־תָה) וּבְ״הֵם/הֵן״ (־וּ) הַ־ה׳ נֶעֱלֶמֶת (רָאֲתָה, רָאוּ). טִיפּ: שֵׁם הַפֹּעַל שֶׁל פְּעָלִים בְּגִזְרַת ל״ה מִסְתַּיֵּם כִּמְעַט תָּמִיד בְּ־וֹת: לִרְאוֹת, לִבְנוֹת, לִשְׁתּוֹת, לִבְכּוֹת, לַעֲלוֹת, לִרְצוֹת.

כְּלָלִים לִזְכֹּר

• בְּהֹוֶה: הוּא רוֹאֶה הִיא רוֹאָה הֵם רוֹאִים הֵן רוֹאוֹת — הַ־ה׳ שֶׁל הַשֹּׁרֶשׁ אֵינָהּ נִשְׁמַעַת.

• זְהִירוּת! פֹּעַל שֶׁנִּשְׁמַע כְּמוֹ ל״ה אֲבָל מִסְתַּיֵּם בָּ־א׳ הוּא בְּגִזְרָה אַחֶרֶת (ל״א): קָרָא — קָרָאתִי (הָ־א׳ נִשְׁמֶרֶת), אֲבָל בָּכָה — בָּכִיתִי (הָ־ה׳ נַעֲשֵׂית י׳).

סַפְּרוּ עַל מָה שֶׁקָּרָה לְמוֹגְלִי. כִּתְבוּ חָמֵשׁ שׁוּרוֹת, בְּכָל אַחַת פֹּעַל בְּגִזְרַת ל״ה בְּעָבָר, וְשִׂימוּ לֵב לַסִּיּוֹמֶת. לְמָשָׁל: ״מוֹגְלִי עָלָה עַל הַסֶּלַע וְהַנָּמֵר נָסוֹג.״
