📖 קְרִיאָה

פַּעַם הָיְתָה אִשָּׁה שֶׁרָצְתָה מְאֹד לְגַדֵּל יַלְדָּה קְטַנָּה מִשֶּׁלָּהּ, אֲבָל לֹא הָיוּ לָהּ יְלָדִים. יוֹם אֶחָד הָלְכָה אֶל מְכַשֵּׁפָה זְקֵנָה וְטוֹבַת לֵב שֶׁגָּרָה בִּקְצֵה הַכְּפָר, וּבִקְּשָׁה מִמֶּנָּה עֵצָה. הַמְּכַשֵּׁפָה נָתְנָה לָהּ גַּרְעִין שְׂעוֹרָה קָטָן וּמְיֻחָד, וְאָמְרָה לָהּ: "טִמְנִי אוֹתוֹ בֶּעָצִיץ, הַשְׁקִי אוֹתוֹ הֵיטֵב, וְחַכִּי לִרְאוֹת מָה יִצְמַח."

הָאִשָּׁה נָטְעָה אֶת הַגַּרְעִין בֶּעָצִיץ, וּכְבָר לְמָחֳרָת צָמַח מִמֶּנּוּ פֶּרַח צִבְעוֹנִי גָּדוֹל, שֶׁעֲלֵי הַכּוֹתֶרֶת שֶׁלּוֹ סְגוּרִים הֵיטֵב. כְּשֶׁנָּשְׁקָה הָאִשָּׁה עַל עֲלֵי הַכּוֹתֶרֶת, הֵם נִפְתְּחוּ לְאַט - וּבְתוֹךְ הַפֶּרַח יָשְׁבָה נַעֲרָה זְעִירָה, לֹא גְּדוֹלָה מֵאֶצְבַּע. הָאִשָּׁה הִתְרַגְּשָׁה כָּל כָּךְ, עַד שֶׁקָּרְאָה לַיַּלְדָּה הַקְּטַנְטַנָּה בְּשֵׁם שֶׁמַּתְאִים לָהּ בְּדִיּוּק: "אֶצְבְּעוֹנִית" - כִּי הָיְתָה קְטַנָּה כְּמוֹ אֶצְבַּע.

אֶצְבְּעוֹנִית יָשְׁנָה בְּעֶרֶשׂ קָטָן עָשׂוּי מִקְּלִפַּת אֱגוֹז, מְרֻפָּד בְּעָלִים סְגֻלִּים רַכִּים, וּמְכֻסָּה בְּעֶלֶה שֶׁל וֶרֶד אָדֹם. בַּיּוֹם הָיְתָה שָׁטָה בְּצַלַּחַת מַיִם, כְּשֶׁעֲלֵה כּוֹתֶרֶת לָבָן מְשַׁמֵּשׁ לָהּ סִירָה, וְזוּג שַׂעֲרוֹת סוּס מְשַׁמְּשׁוֹת מָשׁוֹט. הִיא שָׁרָה בְּקוֹל דַּקִּיק וְעָדִין, וְכָל מִי שֶׁשָּׁמַע אוֹתָהּ הִתְפַּלֵּא כַּמָּה יָפֶה קוֹלָהּ שֶׁל יְצוּרָה כֹּה קְטַנָּה.

לַיְלָה אֶחָד קָפְצָה פְּנִימָה דֶּרֶךְ הַחַלּוֹן הַפָּתוּחַ קַרְפָּדָה גְּדוֹלָה וּמְכֹעֶרֶת. הִיא רָאְתָה אֶת אֶצְבְּעוֹנִית הַיָּפָה יְשֵׁנָה בְּעֶרֶשׂ הָאֱגוֹז, וְחָשְׁבָה מִיָּד: "זוֹ תִּהְיֶה כַּלָּה מֻשְׁלֶמֶת לִבְנִי!" בְּלִי לְהָעִיר אוֹתָהּ, הֵרִימָה הַקַּרְפָּדָה אֶת הָעֶרֶשׂ כֻּלּוֹ וְקָפְצָה אִתּוֹ אֶל הַנָּהָר. שָׁם הִנִּיחָה אֶת אֶצְבְּעוֹנִית עַל עֲלֵה חֲבַצֶּלֶת רָחָב בְּאֶמְצַע הַזֶּרֶם הַזּוֹרֵם, כְּדֵי שֶׁלֹּא תּוּכַל לִבְרֹחַ.

כְּשֶׁהִתְעוֹרְרָה אֶצְבְּעוֹנִית וְרָאֲתָה שֶׁהִיא לְבַדָּהּ בְּלֵב הַנָּהָר, פָּרְצָה בִּבְכִי מַר. דָּגִים קְטַנִּים שֶׁשָּׂחוּ בַּמַּיִם רִחֲמוּ עָלֶיהָ, וְכִרְסְמוּ בִּזְרִיזוּת אֶת גִּבְעוֹל הֶעָלֶה עַד שֶׁנִּפְרַד מֵהַקַּרְקָעִית. פַּרְפַּר לָבָן שֶׁחָלַף בַּסְּבִיבָה קָשַׁר אֶת עַצְמוֹ בְּחוּט דַּק לֶעָלֶה הַצָּף, וְעָזַר לוֹ לָשׁוּט בְּמַהֲרָה בְּמוֹרַד הַנָּהָר, רָחוֹק רָחוֹק מֵהַקַּרְפָּדָה וּבְנָהּ.

חִפּוּשִׂית גְּדוֹלָה שֶׁחָלְפָה בָּאֲוִיר רָאֲתָה אֶת אֶצְבְּעוֹנִית, חָשְׁבָה שֶׁהִיא יָצוּר יָפֶה וּמְיֻחָד, וְחָטְפָה אוֹתָהּ אֶל צַמֶּרֶת עֵץ גָּבוֹהַּ כְּדֵי לְהַצִּיג אוֹתָהּ לִפְנֵי חֲבֵרוֹתֶיהָ הַחִפּוּשִׂיּוֹת. אַךְ הַחִפּוּשִׂיּוֹת הָאֲחֵרוֹת גִּחְכוּ: "אֵין לָהּ מְחוּשִׁים! יֵשׁ לָהּ רַק שְׁתֵּי רַגְלַיִם! מָה בְּדִיּוּק יָפֶה בָּהּ?" הַחִפּוּשִׂית הִתְבַּיְּשָׁה, הוֹרִידָה אֶת אֶצְבְּעוֹנִית עַל פֶּרַח חִנָּנִית בּוֹדֵד, וְעָזְבָה אוֹתָהּ שָׁם לְבַדָּהּ.

כָּל הַקַּיִץ חָיְתָה אֶצְבְּעוֹנִית לְבַדָּהּ בֵּין הַפְּרָחִים הַגְּדוֹלִים, שָׁתְתָה טַל בֹּקֶר וְטָעֲמָה דְּבַשׁ פְּרָחִים מָתוֹק, וְאָרְגָה לְעַצְמָהּ שִׂמְלָה קְטַנָּה מֵעֲלֵי עֵשֶׂב רַכִּים. כָּל עָלֶה נִרְאָה לָהּ עָנָק, וְכָל טִפַּת מַיִם עַל עָלֶה - כְּמוֹ אֲגַם שָׁלֵם. הִיא הִרְגִּישָׁה זְעִירָה בְּתוֹךְ עוֹלָם עֲנָקִי, אֲבָל גַּם חָפְשִׁיָּה וּשְׁלֵוָה, בֵּין הַדָּגִים, הַפַּרְפָּרִים וְעֵשֶׂב הַנָּהָר הָרָחָב.

אַךְ הַקַּיִץ חָלַף, וְאַחֲרָיו גַּם הַסְּתָיו - וְהַחֹרֶף הַקַּר הִגִּיעַ. עַכְשָׁו לֹא נוֹתְרוּ פְּרָחִים וְלֹא נוֹתַר אֹכֶל, וְהַשֶּׁלֶג הֵחֵל לְרַחֵף לְאַט. שִׂמְלָתָהּ הַדַּקָּה שֶׁל אֶצְבְּעוֹנִית לֹא הִסְפִּיקָה לְחַמֵּם אוֹתָהּ, וְהִיא רָעֲדָה מִקֹּר וּמֵרָעָב וְשׁוֹטְטָה בֵּין גִּבְעוֹלִים יְבֵשִׁים, לֹא יוֹדַעַת לְאָן לָלֶכֶת.

לְבַסּוֹף מָצְאָה פֶּתַח קָטָן שֶׁל מְחִלָּה חַמָּה בָּאֲדָמָה - בֵּיתוֹ שֶׁל עַכְבְּרוֹן שָׂדֶה טוֹב לֵב. הָעַכְבְּרוֹן רִחֵם עָלֶיהָ וְהִזְמִין אוֹתָהּ לְהִצְטָרֵף אֵלָיו לְכָל הַחֹרֶף, בִּתְנַאי שֶׁתְּסַדֵּר אֶת בֵּיתוֹ וּתְסַפֵּר לוֹ סִפּוּרִים בָּעֲרָבִים. אֶצְבְּעוֹנִית הִסְכִּימָה בְּשִׂמְחָה, וְסוֹף סוֹף הָיָה לָהּ מָקוֹם חָם וּבָטוּחַ.

שְׁכֵנוֹ הֶעָשִׁיר שֶׁל הָעַכְבְּרוֹן הָיָה חֲפַרְפֶּרֶת עִוֶּרֶת שֶׁגָּרָה תַּחַת הָאֲדָמָה, בְּבַיִת מָלֵא אוֹצָרוֹת אַךְ בְּלִי טִפַּת אוֹר שֶׁמֶשׁ. הַחֲפַרְפֶּרֶת שָׁמְעָה אֶת אֶצְבְּעוֹנִית שָׁרָה וְהִתְאַהֲבָה בְּקוֹלָהּ מִיָּד. הָעַכְבְּרוֹן, שֶׁרָצָה בְּטוֹבָתָהּ, הֶחְלִיט שֶׁעַל אֶצְבְּעוֹנִית לְהִנָּשֵׂא לַחֲפַרְפֶּרֶת הֶעָשִׁירָה - אַךְ אֶצְבְּעוֹנִית לֹא שָׂמְחָה כְּלָל: אֵיךְ תּוּכַל לִחְיוֹת כָּל חַיֶּיהָ מִתַּחַת לָאֲדָמָה, בְּלִי לִרְאוֹת שׁוּב שָׁמַיִם כְּחֻלִּים אוֹ פֶּרַח פּוֹרֵחַ?

בְּאַחַת הַמְּחִלּוֹת הָאֲרֻכּוֹת שֶׁחָפְרָה הַחֲפַרְפֶּרֶת מָצְאָה אֶצְבְּעוֹנִית סְנוּנִית קְפוּאָה, שׁוֹכֶבֶת בְּלִי נוֹעַ, כְּאִלּוּ מֵתָה. הִיא רִחֲמָה עָלֶיהָ, כִּסְּתָה אוֹתָהּ בְּעָלִים יְבֵשִׁים, וְשָׁבָה לְבַקֵּר אוֹתָהּ בְּסֵתֶר כָּל לַיְלָה. לְמַרְבֵּה הַשִּׂמְחָה, כְּשֶׁהִגִּיעַ הָאָבִיב הִתְעוֹרְרָה הַסְּנוּנִית לְחַיִּים, וְהוֹדְתָה לְאֶצְבְּעוֹנִית שֶׁהִצִּילָה אוֹתָהּ. אֲבָל אֶצְבְּעוֹנִית לֹא יָכְלָה לָעוּף אִתָּהּ - הִיא הִבְטִיחָה לָעַכְבְּרוֹן שֶׁתִּנָּשֵׂא לַחֲפַרְפֶּרֶת, וּמִלָּה הִיא מִלָּה.

בְּיוֹם הַחֲתֻנָּה, כְּשֶׁכָּל הַכֹּל הָיָה מוּכָן מִתַּחַת לָאֲדָמָה, יָצְאָה אֶצְבְּעוֹנִית לְרֶגַע הַחוּצָה כְּדֵי לְהִפָּרֵד מֵהַשֶּׁמֶשׁ בְּפַעַם הָאַחֲרוֹנָה בְּחַיֶּיהָ. אָז חָזְרָה הַסְּנוּנִית, רָאְתָה אֶת עֵינֶיהָ הָעֲצוּבוֹת, וְהֵבִינָה מִיָּד. "עֲלִי עַל גַּבִּי," קָרְאָה, "וְאֶקַּח אוֹתָךְ לְאֶרֶץ חַמָּה וְרַחוֹקָה, שָׁם תָּמִיד שֶׁמֶשׁ וּפְרָחִים." אֶצְבְּעוֹנִית לֹא הִסְּסָה אַף רֶגַע - הִיא עָלְתָה עַל גַּב הַסְּנוּנִית הַחַמָּה, וְיַחַד הֵן עָפוּ גָּבוֹהַּ, גָּבוֹהַּ מֵעַל הַעֲנָנִים.

הֵן נָחֲתוּ בְּאֶרֶץ שֶׁכֻּלָּהּ פְּרָחִים צִבְעוֹנִיִּים תַּחַת שֶׁמֶשׁ נִצְחִית. עַל פֶּרַח לָבָן וְעָדִין יָשַׁב נָסִיךְ זָעִיר, בְּדִיּוּק בְּגֹבַהּ שֶׁל אֶצְבְּעוֹנִית, עִם זוּג כְּנָפַיִם שְׁקוּפוֹת עַל גַּבּוֹ. הַנָּסִיךְ הִתְפַּעֵל: מֵעוֹלָם לֹא רָאָה נַעֲרָה כֹּה יָפָה - וּבְדִיּוּק בְּמִדָּתוֹ שֶׁלּוֹ! הוּא הָיָה מֶלֶךְ כָּל פְּרָחֵי הָאָרֶץ הַהִיא, וְשָׁאַל אוֹתָהּ אִם תַּסְכִּים לִהְיוֹת לוֹ לְמַלְכָּה.

כָּל הַפְּרָחִים בַּסְּבִיבָה נִפְתְּחוּ בְּבַת אַחַת, וּמִתּוֹכָם יָצְאוּ עוֹד יְצוּרִים זְעִירִים כְּמוֹתָהּ, כֻּלָּם שְׂמֵחִים בְּבוֹאָהּ. הֵם הֶעֱנִיקוּ לָהּ זוּג כְּנָפַיִם מִשֶּׁל עַצְמָהּ, כְּדֵי שֶׁתּוּכַל לָעוּף מִפֶּרַח לְפֶרַח כְּמוֹתָם, וְשֵׁם חָדָשׁ - "מַאיָה". אֲבָל אֶצְבְּעוֹנִית שָׁמְרָה בְּלִבָּהּ גַּם אֶת הַשֵּׁם הַיָּשָׁן: אַחֲרֵי כָּל כָּךְ הַרְבֵּה זְמַן שֶׁבּוֹ הָיְתָה הַקְּטַנָּה בְּיוֹתֵר בְּכָל מָקוֹם שֶׁהִגִּיעָה אֵלָיו, סוֹף סוֹף מָצְאָה מָקוֹם שֶׁבּוֹ הִיא בְּדִיּוּק בַּמִּדָּה הַנְּכוֹנָה.

מָה זֹאת מִלָּה מַקְטִינָה?

בְּעִבְרִית יֵשׁ דֶּרֶךְ מְיֻחֶדֶת לוֹמַר שֶׁמַּשֶּׁהוּ קָטָן, חָמוּד אוֹ צָעִיר - בְּלִי לְהוֹסִיף אֶת הַמִּלָּה "קָטָן" בִּכְלָל! מוֹסִיפִים לְשֵׁם הָעֶצֶם סִיּוֹמֶת מְיֻחֶדֶת, וְנוֹצֶרֶת "מִלָּה מַקְטִינָה" חֲדָשָׁה. לְמָשָׁל: "יֶלֶד" הוֹפֵךְ לְ"יַלְדּוֹן", וּ"סֵפֶר" הוֹפֵךְ לְ"סִפְרוֹן". שִׂימוּ לֵב: גַּם שֵׁם הַגִּבּוֹרָה שֶׁל הַסִּפּוּר, "אֶצְבְּעוֹנִית", בָּנוּי בְּדִיּוּק כָּךְ - מִן הַמִּלָּה "אֶצְבַּע" וְסִיּוֹמֶת מַקְטִינָה, כִּי הִיא הָיְתָה קְטַנָּה כְּמוֹ אֶצְבַּע!

הַתַּבְנִית הָרְגִילָה: סִיּוֹמֶת ־וֹן / ־וֹנָה / ־וֹנֶת
