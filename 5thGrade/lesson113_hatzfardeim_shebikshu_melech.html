📖 קְרִיאָה

בִּבְרֵכָה גְּדוֹלָה חָיוּ צְפַרְדְּעִים רַבּוֹת. הֵן קָפְצוּ מֵעָלֶה לְעָלֶה, שָׂחוּ בַּמַּיִם וְקִרְקְרוּ כָּל הַיּוֹם. אַךְ יוֹם אֶחָד אָמְרָה צְפַרְדֵּעַ זְקֵנָה: ״הַבְּרֵכָה שְׁקֵטָה מִדַּי, וַאֲנַחְנוּ בְּלִי מֶלֶךְ! בּוֹאוּ נְבַקֵּשׁ מֶלֶךְ מִזֶּאוּס.״

הַצְּפַרְדְּעִים הִתְאַסְּפוּ וְהִתְחַנְּנוּ לִפְנֵי זֶאוּס, מֶלֶךְ הָאֵלִים: ״תֵּן לָנוּ מֶלֶךְ, בְּבַקָּשָׁה!״ זֶאוּס צָחַק, וְהִשְׁלִיךְ אֶל הַמַּיִם גֶּזַע עֵץ גָּדוֹל. הַמַּיִם נִתְּזוּ בְּרַעַשׁ גָּדוֹל, וְהַצְּפַרְדְּעִים נִבְהֲלוּ וְהִסְתַּתְּרוּ בֵּין הַקָּנִים.

אַחֲרֵי רֶגַע יָצְאוּ הַצְּפַרְדְּעִים בִּזְהִירוּת. אַחַת מֵהֶן הֶעֱזָה וְנָגְעָה בַּגֶּזַע. ״הַמֶּלֶךְ הַזֶּה כָּבֵד וְשָׁקֵט! הוּא לֹא מְדַבֵּר, וְהוּא לֹא זָז״, קָרְאָה. מִיָּד קָפְצוּ כֻּלָּן עָלָיו, רָקְדוּ וְצָחֲקוּ.

אֲבָל בִּמְהֵרָה נִמְאַס לָהֶן. הֵן חָזְרוּ אֶל זֶאוּס וְהִתְלוֹנְנוּ: ״הַמֶּלֶךְ שֶׁלָּנוּ סְתָם גֶּזַע! אֲנַחְנוּ רוֹצוֹת מֶלֶךְ אֱמֶת, שֶׁיֹּאמַר לָנוּ מָה לַעֲשׂוֹת.״

זֶאוּס כָּעַס מְאוֹד וְשָׁלַח אֶל הַבְּרֵכָה חֲסִידָה גְּדוֹלָה. הַחֲסִידָה הָלְכָה בַּמַּיִם, פָּתְחָה אֶת מַקּוֹרָהּ וְטָרְפָה צְפַרְדְּעִים אַחַת אַחֲרֵי הַשְּׁנִיָּה. ״הַמֶּלֶךְ הֶחָדָשׁ רָעֵב וְחָזָק!״ צָעֲקוּ הַצְּפַרְדְּעִים הַמְּבֹהָלוֹת.

הֵן בָּכוּ וְהִתְחַנְּנוּ שׁוּב לִפְנֵי זֶאוּס, אֲבָל הוּא לֹא עָנָה. אָז הֵבִינוּ הַצְּפַרְדְּעִים שֶׁטָּעוּ: הַבְּרֵכָה הַשְּׁקֵטָה וְהַגֶּזַע הַשָּׁקֵט הָיוּ טוֹבִים לָהֶן הַרְבֵּה יוֹתֵר, אֲבָל כְּבָר הָיָה מְאֻחָר.

בְּמִשְׁפְּטִים הַמֻּדְגָּשִׁים שֶׁבְּתוֹךְ הַדִּבּוּר אֵין פֹּעַל — הֵם מִשְׁפָּטִים שֵׁמָנִיִּים. מְעֻבָּד לְפִי מָשָׁל מְפֻרְסָם שֶׁל אֵזוֹפּוֹס.

📚 דִּקְדּוּק — מִשְׁפָּט שֵׁמָנִי וּמִשְׁפָּט פֹּעֲלִי

מִשְׁפָּט שֵׁמָנִי וּמִשְׁפָּט פֹּעֲלִי

בְּכָל מִשְׁפָּט יֵשׁ נוֹשֵׂא (עַל מִי אוֹ עַל מָה מְדַבְּרִים) וְנָשׂוּא (מָה אוֹמְרִים עָלָיו). הַנָּשׂוּא קוֹבֵעַ אֶת סוּג הַמִּשְׁפָּט:

סוּג הַמִּשְׁפָּטהַנָּשׂוּאדֻּגְמָאוֹת
מִשְׁפָּט פֹּעֲלִיפֹּעַל — מַה הַנּוֹשֵׂא עוֹשֶׂההַצְּפַרְדֵּעַ קוֹפֶצֶת. הַחֲסִידָה טוֹרֶפֶת צְפַרְדְּעִים.
מִשְׁפָּט שֵׁמָנִישֵׁם עֶצֶם, שֵׁם תֹּאַר אוֹ צֵרוּף מִלַּת יַחַס — מַה הַנּוֹשֵׂא הוּא אוֹ אֵיפֹה הוּאהַגֶּזַע כָּבֵד. זֶאוּס מֶלֶךְ. הַצְּפַרְדְּעִים בַּבְּרֵכָה.

אֵיךְ מְזַהִים? מְחַפְּשִׂים אֶת הַנָּשׂוּא וְשׁוֹאֲלִים: ״הַאִם זֶה פֹּעַל?״

① אִם הַנָּשׂוּא הוּא פֹּעַל (קוֹפֶצֶת, אוֹכֵל, רָץ) — הַמִּשְׁפָּט פֹּעֲלִי.

② אִם אֵין פֹּעַל, וְהַנָּשׂוּא הוּא שֵׁם עֶצֶם (מֶלֶךְ), שֵׁם תֹּאַר (כָּבֵד) אוֹ צֵרוּף עִם מִלַּת יַחַס (בַּבְּרֵכָה) — הַמִּשְׁפָּט שֵׁמָנִי.

זְהִירוּת: מִשְׁפָּט שֵׁמָנִי מְתָאֵר גַּם הַרְגָּשָׁה אוֹ מַצָּב (״הַצְּפַרְדְּעִים עֲצוּבּוֹת״), אֲבָל הוּא עֲדַיִן שֵׁמָנִי, כִּי הַנָּשׂוּא שֵׁם תֹּאַר וְלֹא פֹּעַל. בַּשִּׁיעוּר הַזֶּה בּוֹדְקִים מִשְׁפָּטִים בִּזְמַן הוֹוֶה.
