📖 קְרִיאָה

לִפְנֵי אֲלָפֵי שָׁנִים, בָּאִי כְּרֵתִים שֶׁבְּיָם הַתִּיכוֹן, חַי אוֹמָן חָכָם וּמְחֻכָּם בְּשֵׁם דֵּדָלוֹס. הוּא יָדַע לִבְנוֹת כָּל דָּבָר - פְּסָלִים, מִגְדָּלִים, וַאֲפִלּוּ מְבוֹךְ עָנָק שֶׁל מַסְדְּרוֹנוֹת מִסְתּוֹרִיִּים בִּשְׁבִיל הַמֶּלֶךְ מִינוֹס. לְדֵדָלוֹס הָיָה בֵּן קָטָן וְאָהוּב, בְּשֵׁם אִיקָרוֹס.

יוֹם אֶחָד הִתְרַגֵּז הַמֶּלֶךְ מִינוֹס עַל דֵּדָלוֹס, וְצִוָּה לִנְעֹל אוֹתוֹ וְאֶת בְּנוֹ בְּתוֹךְ מִגְדָּל גָּבוֹהַ עַל שְׂפַת הַיָּם, שֶׁמִּמֶּנּוּ אֵין דֶּרֶךְ לְהִמָּלֵט. סְפִינוֹת הַמֶּלֶךְ שָׁמְרוּ עַל הַיָּם, וְחַיָּלָיו שָׁמְרוּ עַל כָּל דֶּרֶךְ בַּיַּבָּשָׁה - נִשְׁאַר רַק כִּוּוּן אֶחָד חָפְשִׁי: הַשָּׁמַיִם.

דֵּדָלוֹס הִתְבּוֹנֵן יוֹם אַחַר יוֹם בַּצִּפֳּרִים שֶׁעָפוּ חָפְשִׁי מֵעַל הַמִּגְדָּל, וְקִבֵּל אִידֵאָה נִפְלָאָה. בְּסַבְלָנוּת אָסַף נוֹצוֹת שֶׁנָּשְׁרוּ מֵהַצִּפֳּרִים, מִן הַקְּטַנָּה עַד הַגְּדוֹלָה, וְסִדֵּר אוֹתָן בְּשׁוּרוֹת מְדֻיָּקוֹת. בְּעֶזְרַת חוּטִים וְשַׁעֲוָה חַמָּה הִדְבִּיק אֶת הַנּוֹצוֹת זוֹ לְזוֹ, עַד שֶׁנּוֹצְרוּ שְׁתֵּי זוּגוֹת כְּנָפַיִם גְּדוֹלוֹת - זוּג אֶחָד לוֹ וְזוּג אֶחָד לְאִיקָרוֹס.

כְּשֶׁהַכְּנָפַיִם הָיוּ מוּכָנוֹת, קָשַׁר דֵּדָלוֹס אוֹתָן בִּרְצוּעוֹת עוֹר לִכְתֵפָיו וְלִכְתֵפֵי בְּנוֹ. הוּא הִזְהִיר אֶת אִיקָרוֹס בְּקוֹל רָצִינִי: "בְּנִי, טוּס בְּדֶּרֶךְ הָאֶמְצַע. אַל תָּעוּף נָמוּךְ מִדַּי מֵעַל הַיָּם, כִּי הַגַּלִּים יַרְטִיבוּ אֶת הַנּוֹצוֹת וְיַכְבִּידוּ עֲלֵיהֶן. וְאַל תָּעוּף גָּבוֹהַ מִדַּי לְעֵבֶר הַשֶּׁמֶשׁ, כִּי חֻמָּהּ יַמֵּס אֶת הַשַּׁעֲוָה וְהַכְּנָפַיִם יִתְפָּרְקוּ. הַבְטַח לִי!" אִיקָרוֹס הִבְטִיחַ בְּשִׂמְחָה, נִרְגָּשׁ מִדַּי כְּדֵי לְהַקְשִׁיב עַד הַסּוֹף.

הַשְּׁנַיִם קָפְצוּ מֵהַמִּגְדָּל, וּלְתִדְהֵמָתָם - הֵם עָפוּ! תּוֹשָׁבֵי הָאֵי לְמַטָּה הִבִּיטוּ מַעְלָה בְּפֶה פָּעוּר, בְּטוּחִים לְרֶגַע שֶׁהֵם רוֹאִים שְׁנֵי אֵלִים חוֹלְפִים בַּשָּׁמַיִם. דֵּדָלוֹס טָס לְאֹרֶךְ דֶּרֶךְ הָאֶמְצַע, לֹא נָמוּךְ וְלֹא גָּבוֹהַ, וּמֵאֲחוֹרָיו טָס אִיקָרוֹס.

הַתְּחוּשָׁה הָיְתָה מַסְחֶרֶרֶת מִדַּי בִּשְׁבִיל אִיקָרוֹס. הָרוּחַ לִטְּפָה אֶת פָּנָיו, הָעוֹלָם נִרְאָה קָטָן וְרָחוֹק, וְהוּא שָׁכַח לְגַמְרֵי אֶת אַזְהָרַת אָבִיו. פֶּתַע אַחַר פֶּתַע הוּא הֵחֵל לְהִתְרוֹמֵם גָּבוֹהַ יוֹתֵר, מְנַסֶּה לְהִתְקָרֵב אֶל הַשֶּׁמֶשׁ הַזּוֹהֶרֶת, שֶׁנִּרְאֲתָה לוֹ קְרוֹבָה כְּמוֹ חֲבֵרָה חֲדָשָׁה בַּשָּׁמַיִם.

לְאַט לְאַט הֵחֵלָּה הַשַּׁעֲוָה שֶׁחִבְּרָה בֵּין הַנּוֹצוֹת לְהֵימֵס בְּחֹם הַשֶּׁמֶשׁ. נוֹצָה אַחַת נָשְׁרָה, וְאַחֲרֶיהָ עוֹד אַחַת. אִיקָרוֹס הִבְחִין בַּפֶּלֶא הַזֶּה בְּרֶגַע שֶׁהָיָה כְּבָר מְאֻחָר מִדַּי - הַכְּנָפַיִם הִתְפָּרְקוּ לְאִטָּן, וְהוּא הֵחֵל לִצְנֹח לְעֵבֶר הַיָּם הָרָחוֹק.

"אַבָּא!" צָעַק אִיקָרוֹס בְּקוֹל שֶׁנִּשָּׂא בָּרוּחַ. דֵּדָלוֹס, שֶׁטָּס לְפָנָיו, הִסְתּוֹבֵב וְרָאָה רַק נוֹצוֹת מִתְעוֹפְפוֹת בַּמָּקוֹם שֶׁבּוֹ בְּנוֹ הָיָה אֲמוּר לִהְיוֹת. לִבּוֹ נִצְבַּט מִפַּחַד, וְהוּא צָנַח בִּמְהִירוּת לְעֵבֶר הַגַּלִּים.

לְמַזָּלוֹ שֶׁל אִיקָרוֹס, קְבוּצַת דַּיָּגִים שֶׁחָלְפָה בְּסִירָתָם בְּדִיּוּק בְּאוֹתוֹ מָקוֹם רָאֲתָה אֶת הַנְּפִילָה, וּמִהֲרָה לָשׁוּט אֶל הַמָּקוֹם שֶׁבּוֹ נָפַל. הֵם מָשׁוּ אוֹתוֹ מִן הַמַּיִם, מְבֹהָל וְעָיֵף אֲבָל חַי, וְהֵבִיאוּ אוֹתוֹ אֶל חוֹפוֹ שֶׁל אִי קָטָן קָרוֹב.

דֵּדָלוֹס, שֶׁחִפֵּשׂ בְּיֵאוּשׁ מִמַּעַל, גִּלָּה אֶת בְּנוֹ שׁוֹכֵב עַל הַחוֹל, רָטוּב וְרוֹעֵד אֲבָל בְּחַיִּים. הוּא נָפַל עַל בִּרְכָּיו וְחָבַק אוֹתוֹ בְּכֹל כֹּחוֹ, בֵּין דְּמָעוֹת שֶׁל פַּחַד וְשֶׁל שִׂמְחָה גַּם יַחַד.

כְּשֶׁאִיקָרוֹס הִתְאוֹשֵׁשׁ, אָמַר לְאָבִיו בְּקוֹל חַלָּשׁ: "עַכְשָׁו אֲנִי מֵבִין לָמָּה אָמַרְתָּ לִי לָטוּס בְּדֶרֶךְ הָאֶמְצַע. לֹא תָּמִיד הַגָּבוֹהַ בְּיוֹתֵר הוּא הַכִּי טוֹב." דֵּדָלוֹס חִבֵּק אוֹתוֹ שׁוּב וְהֵשִׁיב בְּחִיּוּךְ עָיֵף: "הַלֵּקַח הַזֶּה, בְּנִי, יִשָּׁאֵר אִתְּךָ הַרְבֵּה יוֹתֵר זְמָן מֵהַכְּנָפַיִם שֶׁאָבְדוּ לָנוּ."

מֵאָז מְסֻפֶּרֶת אַגָּדָה זוֹ בְּכָל הָעוֹלָם, וּמְלַמֶּדֶת שֶׁלִּפְעָמִים דַּוְקָא הָעֵצָה שֶׁמַּגְבִּילָה אֶת הַשִּׂמְחָה שֶׁלָּנוּ הִיא זוֹ שֶׁמְּגִנָּה עָלֵינוּ - וְשֶׁדֶּרֶךְ הָאֶמְצַע, לֹא גָּבוֹהַ מִדַּי וְלֹא נָמוּךְ מִדַּי, הִיא לְעִתִּים הַדֶּרֶךְ הַבְּטוּחָה בְּיוֹתֵר.

מְסֻפָּר עַל־פִּי מִיתוֹלוֹגְיָה יְוָנִית עַתִּיקָה, מִיתוֹס אִיקָרוֹס וְדֵדָלוֹס, מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ (עִם סוֹף מְרֻכָּךְ).

מָה זֶה מוּשָׂא?

בְּשִׁיעוּרִים 53 וְ-67 לָמַדְנוּ שֶׁלְּמִשְׁפָּט יֵשׁ נוֹשֵׂא (מִי עוֹשֶׂה אֶת הַפְּעֻלָּה) וְנָשׂוּא (הַפְּעֻלָּה עַצְמָהּ). לִפְעָמִים יֵשׁ בַּמִּשְׁפָּט גַּם מוּשָׂא: מִי אוֹ מָה שֶׁ"סוֹפֵג" אֶת הַפְּעֻלָּה שֶׁל הַנָּשׂוּא. יֵשׁ שְׁנֵי סוּגִים שֶׁל מוּשָׂא, וְהַסִּפּוּר עַל אִיקָרוֹס וְדֵדָלוֹס נָתַן לָנוּ דֻּגְמָאוֹת רַבּוֹת לִשְׁנֵיהֶם.

מוּשָׂא יָשִׁיר - מְחֻבָּר עִם "אֶת"

כְּשֶׁהַמּוּשָׂא הוּא שֵׁם עֶצֶם מְסֻיָּם, הוּא מִתְחַבֵּר לַנָּשׂוּא בְּלִי מִלַּת יַחַס - רַק בְּעֶזְרַת הַמִּלָּה "אֶת". כְּדֵי לְאַתֵּר מוּשָׂא יָשִׁיר, שׁוֹאֲלִים "אֶת מָה?" אוֹ "אֶת מִי?".

🔑 כְּלָל זָהָב: מוּשָׂא יָשִׁיר בָּא עִם "אֶת" (בְּלִי מִלַּת יַחַס נוֹסֶפֶת) - שׁוֹאֲלִים עָלָיו "אֶת מָה? אֶת מִי?". מוּשָׂא עָקִיף בָּא עִם מִלַּת יַחַס (לְ-, מִ-, עַל...) - שׁוֹאֲלִים עָלָיו "לְמִי? עַל מָה? מִמִּי?".

מִשְׁפָּט אֶחָד, שְׁנֵי מוּשָׂאִים

בְּמִשְׁפָּט אֶחָד יְכוֹלִים לְהוֹפִיע גַּם מוּשָׂא יָשִׁיר וְגַם מוּשָׂא עָקִיף בְּיַחַד: "דֵּדָלוֹס נָתַן אֶת הַכְּנָפַיִם לְאִיקָרוֹס." הַנָּשׂוּא הוּא "נָתַן", הַמּוּשָׂא הַיָּשִׁיר הוּא "אֶת הַכְּנָפַיִם" (מָה נָתַן?), וְהַמּוּשָׂא הֶעָקִיף הוּא "לְאִיקָרוֹס" (לְמִי נָתַן?).

💡 טִיפ: לֹא כָּל פֹּעַל דּוֹרֵשׁ מוּשָׂא. בַּמִּשְׁפָּט "אִיקָרוֹס צָנַח" אֵין מוּשָׂא בִּכְלָל - הַנָּשׂוּא "צָנַח" מַסְפִּיק לְעַצְמוֹ, בְּלִי "אֶת מָה" אוֹ "לְמָה".

כִּתְבוּ 5 מִשְׁפָּטִים מִשֶּׁלָּכֶם עַל אִיקָרוֹס וְדֵדָלוֹס, שֶׁבְּכָל אֶחָד מֵהֶם מוּשָׂא: בְּחֵלֶק מֵהַמִּשְׁפָּטִים הַשְׁתַּמְּשׁוּ בְּמוּשָׂא יָשִׁיר עִם "אֶת" (כְּמוֹ "דֵּדָלוֹס אָסַף אֶת הַנּוֹצוֹת"), וּבְחֵלֶק בְּמוּשָׂא עָקִיף עִם מִלַּת יַחַס (כְּמוֹ "אִיקָרוֹס הִקְשִׁיב לְאָבִיו"). וַדְּאוּ שֶׁמִּשְׁפָּט אֶחָד לְפָחוֹת כּוֹלֵל גַּם מוּשָׂא יָשִׁיר וְגַם מוּשָׂא עָקִיף בְּיַחַד.
