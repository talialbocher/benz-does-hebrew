📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת, בְּיָפָן, גָּרוּ זָקֵן וּזְקֵנָה בִּכְפָר קָטָן שֶׁלְּיַד נָהָר. לֹא הָיוּ לָהֶם יְלָדִים, וְכָל בּוֹקֶר הִתְפַּלְּלוּ שֶׁיִּוָּולֵד לָהֶם בֵּן.

יוֹם אֶחָד הָלְכָה הַזְּקֵנָה לְכַבֵּס בְּגָדִים בַּנָּהָר. פִּתְאוֹם הִיא רָאֲתָה אֲפַרְסֵק עֲנָק צָף עַל הַמַּיִם, גָּדוֹל בְּהַרְבֵּה מִכָּל אֲפַרְסֵק שֶׁרָאֲתָה בְּחַיֶּיהָ. בְּשִׂמְחָה רַבָּה הִיא שָׁלְתָה אוֹתוֹ אֶל הַחוֹף.

הָאֲפַרְסֵק הָעֲנָק הוּבָא הַבַּיְתָה בִּזְרוֹעוֹתֶיהָ, כְּדֵי שֶׁהַזָּקֵן יוּכַל לִטְעוֹם מִמֶּנּוּ יַחַד אִיתָּהּ.

כְּשֶׁהַזָּקֵן נִיסָּה לַחְתּוֹךְ אֶת הָאֲפַרְסֵק לִשְׁנַיִים, הוּא נִבְהַל מְאוֹד: מִתּוֹךְ הַפְּרִי בָּקַע קוֹל שֶׁל תִּינוֹק בּוֹכֶה! הַזָּקֵן וְהַזְּקֵנָה פָּתְחוּ אֶת הָאֲפַרְסֵק בִּזְהִירוּת, וְהַתִּינוֹק הוּצָא מִשָּׁם בְּשָׁלוֹם.

הַשְּׁנַיִים הָיוּ הֲמוּמִים וּשְׂמֵחִים גַּם יַחַד. הֵם קָרְאוּ לַתִּינוֹק "מוֹמוֹטָארוֹ" - "יֶלֶד הָאֲפַרְסֵק" בַּשָּׂפָה הַיַּפָּנִית - וְאִימְּצוּ אוֹתוֹ כִּבְנָם.

מוֹמוֹטָארוֹ הוּאֲכַל בָּאוֹכֶל הַטּוֹב בְּיוֹתֵר בַּכְּפָר, וְגָדַל מַהֵר יוֹתֵר מִכָּל יֶלֶד אַחֵר. תּוֹךְ שָׁנִים סְפוּרוֹת הוּא הָפַךְ לְנַעַר חָזָק וְחָכָם.

יוֹם אֶחָד סִיפְּרוּ תּוֹשָׁבֵי הַכְּפָר לְמוֹמוֹטָארוֹ שֶׁעַל אִי רָחוֹק בְּשֵׁם "אוֹנִיגָשִׁימָה" גָּרִים שֵׁדִים אַכְזָרִיִּים, שֶׁגּוֹנְבִים אוֹכֶל וְאוֹצָרוֹת מֵהַכְּפָרִים שֶׁלְּאוֹרֶךְ הַחוֹף. מוֹמוֹטָארוֹ הֶחְלִיט לָצֵאת בְּעַצְמוֹ לְהָבִיס אוֹתָם.

אִמּוֹ אָפְתָה לוֹ עוּגִיּוֹת דּוֹחַן קְטַנּוֹת וּמְתוּקוֹת לַדֶּרֶךְ, וּמוֹמוֹטָארוֹ הוּלְבַּשׁ בְּבִגְדֵי לוֹחֵם, עִם חֶרֶב בְּמוֹתְנָיו.

בַּדֶּרֶךְ אֶל הַחוֹף פָּגַשׁ מוֹמוֹטָארוֹ כֶּלֶב רָעֵב, קוֹף סַקְרָן וּפַסְיוֹנִית יָפָה. לְכָל אֶחָד מֵהֶם הוּא נָתַן עוּגִיַּית דּוֹחַן, וּבִתְמוּרָה שְׁלוֹשְׁתָּם הִצְטָרְפוּ אֵלָיו לַמַּסָּע.

מוֹמוֹטָארוֹ וַחֲבֵרָיו הַחֲדָשִׁים הוּשְׁבוּ בְּסִירָה קְטַנָּה, וְיַחַד חָתְרוּ בְּמֶרֶץ אֶל הָאִי הַמִּסְתּוֹרִי.

עַל הָאִי חִיכָּה קְרָב קָשֶׁה מוּל הַשֵּׁדִים. הַכֶּלֶב נָשַׁךְ, הַקּוֹף טִיפֵּס וְזָרַק אֲבָנִים, וְהַפַּסְיוֹנִית נִיקְּרָה מִלְמַעְלָה - וּמוֹמוֹטָארוֹ עַצְמוֹ נִלְחַם בְּאוֹמֶץ עִם חַרְבּוֹ. לְבַסּוֹף הַשֵּׁדִים נִכְנְעוּ וּבִיקְשׁוּ סְלִיחָה.

כָּל הָאוֹצָר שֶׁהַשֵּׁדִים גָּנְבוּ בְּמֶשֶׁךְ שָׁנִים הוּעֲלָה לַסִּירָה, וּבַדֶּרֶךְ חֲזָרָה לַכְּפָר הוּא הוּרַד בִּזְהִירוּת אֶל הַחוֹף. מוֹמוֹטָארוֹ חִילֵּק אֶת הָאוֹצָר בֵּין כָּל תּוֹשָׁבֵי הַכְּפָר, וְחָזַר הַבַּיְתָה כְּגִיבּוֹר.

עִיבּוּד לְסִיפּוּר הָעָם הַיַּפָּנִי עַל מוֹמוֹטָארוֹ, יֶלֶד הָאֲפַרְסֵק - מָסוֹרֶת עֲמָמִית יַפָּנִית

מכירים בניין חדש: הֻפְעַל

בשיעור 10 הכרנו את בניין הפעיל: בניין שבו מישהו עושה פעולה למישהו אחר - למשל, מישהו שמוציא, מביא, מאכיל או מלביש. בניין הֻפְעַל הוא הבניין הסביל המקביל לו: הוא מתאר שהפעולה קרתה למישהו, בלי לומר תמיד מי עשה אותה - למשל, תינוק שהוצא מהאפרסק (מישהו הוציא אותו), או אוצר שהועלה לסירה (מישהו העלה אותו).
