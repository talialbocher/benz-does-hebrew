📖 קְרִיאָה

בִּימֵי קֶדֶם, בִּכְפָר קָטָן בְּסִין, גָּר זָקֵן בֶּן תִּשְׁעִים בְּשֵׁם יוּ גּוֹנְג. מִמּוּל בֵּיתוֹ עָמְדוּ שְׁנֵי הָרִים עֲנָקִיִּים שֶׁחָסְמוּ אֶת הַדֶּרֶךְ. כְּדֵי לָצֵאת מֵהַכְּפָר, הָיוּ הַתּוֹשָׁבִים צְרִיכִים לָלֶכֶת יָמִים רַבִּים בִּשְׁבִיל מְתַפַּתֵּל סָבִיב הֶהָרִים.

יוֹם אֶחָד כִּנֵּס יוּ גּוֹנְג אֶת בָּנָיו וְאֶת נֶכְדָּיו וְאָמַר: "נַעֲבֹד כֻּלָּנוּ יַחַד כְּדֵי לִפְתֹּחַ אֶת הַדֶּרֶךְ! נִקַּח מַעְדֵּרִים וְסַלִּים כְּדֵי לַחְפֹּר אֶת הָאֲבָנִים וּלְהוֹבִיל אוֹתָן אֶל הַיָּם." כָּל בֹּקֶר קָמוּ עִם הַשַּׁחַר, כְּדֵי שֶׁיִּהְיֶה לָהֶם זְמַן לַעֲבֹד לִפְנֵי הַחֹם.

הַדֶּרֶךְ אֶל הַיָּם הָיְתָה אֲרֻכָּה כָּל כָּךְ, עַד שֶׁכָּל הָלוֹךְ וָשׁוֹב נִמְשַׁךְ שָׁנָה שְׁלֵמָה. זָקֵן חָכָם מִן הַכְּפָר הִתְקָרֵב וְלָעַג לוֹ: "אַתָּה בֶּן תִּשְׁעִים! אֵין לְךָ סִכּוּי לְהַזִּיז אֲפִלּוּ גִּבְעָה קְטַנָּה!"

יוּ גּוֹנְג חִיֵּךְ וְעָנָה: "כְּשֶׁאָמוּת, יַמְשִׁיכוּ בָּנַי. כְּשֶׁהֵם יָמוּתוּ, יַמְשִׁיכוּ נְכָדַי. הֶהָרִים לֹא גְּדֵלִים, אֲבָל הַמִּשְׁפָּחָה שֶׁלָּנוּ גְּדֵלָה. אֲנִי מַתְחִיל הַיּוֹם כְּדֵי שֶׁהַדּוֹרוֹת הַבָּאִים יִהְיוּ חָפְשִׁיִּים לָלֶכֶת בְּכָל מָקוֹם!"

רוּחוֹת הֶהָרִים שָׁמְעוּ אֶת הַדְּבָרִים וְנִבְהֲלוּ. הֵן עָלוּ לַשָּׁמַיִם וְסִפְּרוּ לְקֵיסַר הַשָּׁמַיִם עַל הָאִישׁ שֶׁלֹּא מְוַתֵּר. הַקֵּיסָר הִתְרַגֵּשׁ מֵהַהַתְמָדָה שֶׁלּוֹ. הוּא שָׁלַח שְׁתֵּי רוּחוֹת חֲזָקוֹת, כְּדֵי שֶׁיִּשְׂאוּ אֶת הֶהָרִים וְיַנִּיחוּ אוֹתָם רָחוֹק, אַחַת בַּמִּזְרָח וְאַחַת בַּדָּרוֹם.

מִמָּחֳרָת הָיְתָה הַדֶּרֶךְ פְּתוּחָה. יוּ גּוֹנְג לִמֵּד אֶת נֶכְדָּיו: "כְּדֵי לְהַגִּיעַ רָחוֹק, לֹא צָרִיךְ לָרוּץ. צָרִיךְ לְהַמְשִׁיךְ."

מְסֻפָּר עַל־פִּי סִפּוּר הָעָם הַסִּינִי הָעַתִּיק "יוּ גּוֹנְג מֵזִיז הָרִים" (愚公移山), הַמְּיֻחָס לַסֵּפֶר "לֶיְה־דְּזֶה" (סִין הָעַתִּיקָה), מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ.

כִּתְבוּ אַרְבָּעָה מִשְׁפְּטֵי תַּכְלִית עַל יוּ גּוֹנְג, הַמִּשְׁפָּחָה שֶׁלּוֹ אוֹ הַקֵּיסָר. בְּכָל מִשְׁפָּט שִׁלְבוּ "כְּדֵי" אוֹ "עַל מְנָת". לְדֻגְמָה: "הַנֶּכֶד מִלֵּא סַל כְּדֵי לְהוֹבִיל אֶת הָאֲבָנִים." אוֹ "הַקֵּיסָר שָׁלַח רוּחוֹת כְּדֵי שֶׁיִּפְתְּחוּ אֶת הַדֶּרֶךְ."
