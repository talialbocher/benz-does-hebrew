📚 סִיפּוּרִים בְּעִבְרִית — שִׁיעוּר 137

הָאִכָּר וְהַדֹּב

סִפּוּר עַם רוּסִי עַל צַמָּרוֹת, שָׁרָשִׁים וְאִכָּר פִּקֵּחַ
🎯 כִּיתָה ה׳  |  מִשְׁקַל קָטִיל
→ 133 → 134 → 135 → 136 שִׁיעוּר 137 ◀
📖 קְרִיאָה

בִּכְפָר קָטָן בְּרוּסְיָה, עַל שְׂפַת יַעַר עָבֹת, גָּר אִכָּר עָנִי וְחָרוּץ. בְּכָל אָבִיב יָצָא לַשָּׂדֶה עִם הַמַּחְרֵשָׁה שֶׁלּוֹ, כִּי הֶחָרִישׁ הוּא הָעֲבוֹדָה הָרִאשׁוֹנָה שֶׁל כָּל אִכָּר.

יוֹם אֶחָד, בְּאֶמְצַע הֶחָרִישׁ, יָצָא מִן הַיַּעַר דֹּב גָּדוֹל וְשָׂעִיר. "זֶה הַשָּׂדֶה שֶׁלִּי!" נָהַם הַדֹּב. "אִם תַּמְשִׁיךְ לַעֲבֹד בּוֹ, אֲנִי אֶטְרֹף אוֹתְךָ!"

הָאִכָּר נִבְהַל, אֲבָל חָשַׁב מַהֵר. "אַל תִּטְרֹף אוֹתִי, דֹּב יָקָר," אָמַר. "בּוֹא נִהְיֶה שֻׁתָּפִים. אֲנִי אֶחְרֹשׁ, אֶזְרַע וְאֶעֱבֹד, וְאַתָּה תְּקַבֵּל חֵצִי מִן הַיְּבוּל. מָה אַתָּה בּוֹחֵר: אֶת הַצַּמָּרוֹת — מָה שֶׁגָּדֵל מֵעַל הָאֲדָמָה — אוֹ אֶת הַשָּׁרָשִׁים שֶׁמִּתַּחַת לָהּ?"

הַדֹּב חָשַׁב רֶגַע וְאָמַר: "מָה שֶׁלְּמַעְלָה — שֶׁלִּי. אֶת הַשָּׁרָשִׁים הַמְּלֻכְלָכִים אַתָּה יָכוֹל לָקַחַת." "מְצֻיָּן," חִיֵּךְ הָאִכָּר. בְּאוֹתוֹ אָבִיב הוּא זָרַע בַּשָּׂדֶה לֶפֶת.

כָּל הַקַּיִץ עָבַד הָאִכָּר קָשֶׁה: הוּא הִשְׁקָה אֶת הַשָּׂדֶה, עָקַר עֲשָׂבִים וְשָׁמַר עַל הַצְּמָחִים. הַדֹּב רַק שָׁכַב בַּצֵּל וּפִהֵק. בַּסְּתָיו הִגִּיעַ זְמַן הָאָסִיף. הָאִכָּר שָׁלַף מִן הָאֲדָמָה שָׁרָשִׁים עָבִים, עֲגֻלִּים וּמְתוּקִים שֶׁל לֶפֶת, וְנָתַן לַדֹּב עֲרֵמָה גְּדוֹלָה שֶׁל עָלִים יְרֻקִּים.

הַדֹּב נָגַס בְּעָלֶה אֶחָד וְיָרַק אוֹתוֹ מִיָּד. "נָכוֹן שֶׁהֶעָלִים הָאֵלֶּה אֲכִילִים," נָהַם, "אֲבָל הֵם מָרִים וְקָשִׁים!" "אֲנִי שָׁמַרְתִּי עַל הַהֶסְכֵּם," אָמַר הָאִכָּר. "בָּחַרְתָּ בַּצַּמָּרוֹת, וְקִבַּלְתָּ אֶת הַצַּמָּרוֹת."

"בַּשָּׁנָה הַבָּאָה," אָמַר הַדֹּב בְּכַעַס, "אֲנִי אֶקַּח אֶת הַשָּׁרָשִׁים, וְאַתָּה תִּקַּח אֶת הַצַּמָּרוֹת!" "כִּרְצוֹנְךָ," עָנָה הָאִכָּר.

בָּאָבִיב הַבָּא חָרַשׁ הָאִכָּר שׁוּב אֶת הַשָּׂדֶה, אֲבָל הַפַּעַם הוּא זָרַע חִטָּה. הַשִּׁבֳּלִים צָמְחוּ גְּבוֹהוֹת וּזְהֻבּוֹת, וּבְסוֹף הַקַּיִץ הִגִּיעַ הַקָּצִיר. הָאִכָּר קָצַר אֶת הַשִּׁבֳּלִים בְּמַגָּל, אָסַף אֶת הַגַּרְעִינִים לְשַׂקִּים וְהִשְׁאִיר לַדֹּב אֶת הַשָּׁרָשִׁים.

הַדֹּב חָפַר וְחָפַר, אֲבָל מָצָא רַק שָׁרָשִׁים דַּקִּים וּקְנֵי קַשׁ יְבֵשִׁים וּשְׁבִירִים, שֶׁהִתְפּוֹרְרוּ בֵּין צִפָּרְנָיו. "שׁוּב רִמִּיתָ אוֹתִי!" שָׁאַג.

"בָּחַרְתָּ בְּלִי לִשְׁאֹל מָה אֲנִי זוֹרֵעַ," אָמַר הָאִכָּר בְּשֶׁקֶט. "מִי שֶׁלֹּא עוֹבֵד בַּשָּׂדֶה — לֹא יוֹדֵעַ מָה צוֹמֵחַ בּוֹ." הַדֹּב הִשְׁפִּיל אֶת רֹאשׁוֹ, וְהָאִכָּר רִחֵם עָלָיו. "בּוֹא נַעֲשֶׂה הֶסְכֵּם חָדָשׁ וְהוֹגֵן," הִצִּיעַ. "אַתָּה תִּשְׁמֹר עַל הַשָּׂדֶה מִפְּנֵי הַחֲזִירִים הַפְּרָאִיִּים, וַאֲנִי אֶתֵּן לְךָ מִכָּל יְבוּל: תַּפּוּחִים בִּזְמַן הַקָּטִיף, עֲנָבִים בִּזְמַן הַבָּצִיר וּדְבַשׁ מִן הַכַּוֶּרֶת."

מֵאָז הָיוּ הָאִכָּר וְהַדֹּב שֻׁתָּפִים אֲמִתִּיִּים. וְהַדֹּב לָמַד לָדַעַת: לִפְנֵי שֶׁבּוֹחֲרִים, כְּדַאי לְהָבִין עַל מָה מְדֻבָּר.

עַל פִּי סִפּוּר הָעַם הָרוּסִי "צַמָּרוֹת וְשָׁרָשִׁים" (Вершки и корешки), שֶׁמּוֹפִיעַ בָּאֹסֶף שֶׁל אָלֶכְּסַנְדֶּר אָפָנַסְיֶב (הַמֵּאָה ה־19) בַּשֵּׁם "הָאִכָּר וְהַדֹּב" (Мужик и медведь). סִפּוּר דּוֹמֶה מוֹפִיעַ גַּם אֵצֶל הָאַחִים גְּרִים: "הָאִכָּר וְהַשֵּׁד". מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ.

📚 דִּקְדּוּק — מִשְׁקַל קָטִיל

🌾 מִשְׁקַל קָטִיל — קָצִיר, בָּצִיר, שָׁבִיר

מִשְׁקָל הוּא "תַּבְנִית" שֶׁל תְּנוּעוֹת שֶׁאֶל תּוֹכָהּ "יוֹצְקִים" שֹׁרֶשׁ. בְּמִשְׁקַל קָטִיל יֵשׁ קָמָץ מִתַּחַת לָאוֹת הָרִאשׁוֹנָה שֶׁל הַשֹּׁרֶשׁ, וְחִירִיק מִתַּחַת לָאוֹת הַשְּׁנִיָּה, וְאַחֲרֶיהָ י׳: קָ ־ טִי ־ ל

מִשְׁפָּחָה 1 — עוֹנוֹת וַעֲבוֹדוֹת בַּחַקְלָאוּת. מִלִּים רַבּוֹת בְּמִשְׁקַל קָטִיל הֵן שְׁמוֹת שֶׁל עֲבוֹדָה בַּשָּׂדֶה וְשֶׁל הַזְּמַן שֶׁבּוֹ עוֹשִׂים אוֹתָהּ:

שֹׁרֶשׁמִלָּה בְּמִשְׁקַל קָטִילמָה זֶה?
ח־ר־שׁחָרִישׁהֲפִיכַת הָאֲדָמָה לִפְנֵי הַזְּרִיעָה
ק־צ־רקָצִירכְּרִיתַת הַחִטָּה וְהַשְּׂעוֹרָה
א־ס־ףאָסִיףאִסּוּף הַיְּבוּל בַּסְּתָיו (חַג הָאָסִיף = סֻכּוֹת)
ב־צ־רבָּצִירקְטִיף הָעֲנָבִים בַּכֶּרֶם
מ־ס־קמָסִיקקְטִיף הַזֵּיתִים
ג־ד־דגָּדִידקְטִיף הַתְּמָרִים
ק־ט־ףקָטִיףקְטִיף פֵּרוֹת מִן הָעֵצִים

מִשְׁפָּחָה 2 — תְּאָרִים שֶׁל "אֶפְשָׁר לְ...". מִשְׁקַל קָטִיל יוֹצֵר גַּם שְׁמוֹת תֹּאַר שֶׁאוֹמְרִים שֶׁאֶפְשָׁר לַעֲשׂוֹת מַשֶּׁהוּ לַדָּבָר:

שָׁבִיר = אֶפְשָׁר לִשְׁבֹּר אָכִיל = אֶפְשָׁר לֶאֱכֹל קָרִיא = אֶפְשָׁר לִקְרֹא שָׁמִיעַ = אֶפְשָׁר לִשְׁמֹעַ דָּלִיק = נִדְלָק בְּקַלּוּת

⚠️ הַקָּמָץ "בּוֹרֵחַ"! כְּשֶׁמּוֹסִיפִים סִיּוֹמֶת (נְקֵבָה אוֹ רַבִּים) אוֹ כְּשֶׁהַמִּלָּה בִּסְמִיכוּת, הַקָּמָץ הוֹפֵךְ לִשְׁוָא:

צוּרַת הַבָּסִיסצוּרָה עִם שְׁוָא
שָׁבִירכּוֹס שְׁבִירָה, כֵּלִים שְׁבִירִים
קָצִירקְצִיר הַחִטִּים (סְמִיכוּת)
קָרִיאכְּתִיבָה קְרִיאָה, אוֹתִיּוֹת קְרִיאוֹת

כְּשֶׁהָאוֹת הָרִאשׁוֹנָה גְּרוֹנִית (א, ה, ח, ע), הִיא מְקַבֶּלֶת חֲטַף פַּתָּח בִּמְקוֹם שְׁוָא: אָכִיל ← פֵּרוֹת אֲכִילִים חָרִישׁ ← חֲרִישׁ הַשָּׂדֶה

שִׂימוּ לֵב: בְּמִשְׁקַל קָטִיל יֵשׁ גַּם מִלִּים אֲחֵרוֹת, כְּמוֹ צָעִיר עָשִׁיר פָּקִיד שָׂעִיר — הַמִּשְׁקָל זֵהֶה, אֲבָל הֵן לֹא מִשְּׁתֵּי הַמִּשְׁפָּחוֹת. וּמִלָּה כְּמוֹ תַּלְמִיד אֵינָהּ בְּמִשְׁקַל קָטִיל בִּכְלָל: יֵשׁ בָּהּ ת׳ נוֹסֶפֶת בַּהַתְחָלָה (הַשֹּׁרֶשׁ ל־מ־ד).
