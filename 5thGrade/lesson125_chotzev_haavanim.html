📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת, לְמַרְגְּלוֹת הַר גָּבוֹהַּ, חַי חוֹצֵב אֲבָנִים עָנִי. בְּכָל בֹּקֶר, כְּשֶׁהַשֶּׁמֶשׁ עָלְתָה, הוּא כְּבָר עָמַד לְיַד הַסֶּלַע עִם הַפַּטִּישׁ וְהָאִזְמֵל שֶׁלּוֹ. הוּא חָצַב אֲבָנִים גְּדוֹלוֹת, וַאֲנָשִׁים בָּנוּ מֵהֶן בָּתִּים.

הוּא עָבַד בְּלִי הַפְסָקָה עַד שֶׁהַשֶּׁמֶשׁ שָׁקְעָה. בָּעֶרֶב חָזַר הַבַּיְתָה עָיֵף, אֲבָל שָׂמֵחַ בְּחֶלְקוֹ.

יוֹם אֶחָד הֵבִיא הַחוֹצֵב אֲבָנִים לְבֵיתוֹ שֶׁל סוֹחֵר עָשִׁיר. כַּאֲשֶׁר נִכְנַס פְּנִימָה, רָאָה מִטּוֹת רַכּוֹת, וִילוֹנוֹת שֶׁל מֶשִׁי וְשֻׁלְחָנוֹת מְלֵאִים כָּל טוּב. "הַלְוַאי שֶׁהָיִיתִי עָשִׁיר כָּמוֹהוּ!" נֶאֱנַח. "אָז לֹא הָיִיתִי צָרִיךְ לַעֲבֹד כָּל כָּךְ קָשֶׁה."

רוּחַ הָהָר שָׁמְעָה אֶת דְּבָרָיו. בִּזְמַן שֶׁיָּשַׁן בַּלַּיְלָה, הִיא הִגְשִׁימָה אֶת מִשְׁאַלְתּוֹ. כְּשֶׁהִתְעוֹרֵר, מָצָא אֶת עַצְמוֹ בְּבַיִת מְפֹאָר — וְהוּא הָיָה עָשִׁיר.

יָמִים אֲחָדִים הוּא נֶהֱנָה מֵעָשְׁרוֹ. אֲבָל אַחֲרֵי שֶׁרָאָה אֶת הַנָּסִיךְ עוֹבֵר בָּרְחוֹב — בְּכִרְכָּרָה מֻזְהֶבֶת, וּמְשָׁרְתִים מַחְזִיקִים מֵעָלָיו שִׁמְשִׁיָּה — הוּא קִנֵּא שׁוּב. "הַלְוַאי שֶׁהָיִיתִי נָסִיךְ!" קָרָא. וְהַמִּשְׁאָלָה הִתְגַּשְּׁמָה.

בַּקַּיִץ יָשַׁב הַנָּסִיךְ הֶחָדָשׁ בַּכִּרְכָּרָה. כְּשֶׁהַשֶּׁמֶשׁ לָהֲטָה עַל רֹאשׁוֹ, הַדֶּשֶׁא הִצְהִיב וּפָנָיו נָטְפוּ זֵעָה. אֲפִלּוּ הַשִּׁמְשִׁיָּה לֹא עָזְרָה. "הַשֶּׁמֶשׁ חֲזָקָה מִמֶּנִּי!" אָמַר. "הַלְוַאי שֶׁהָיִיתִי הַשֶּׁמֶשׁ!" וּמִיָּד הָפַךְ לַשֶּׁמֶשׁ.

עַכְשָׁו הוּא זָרַח בְּכָל כֹּחוֹ וְיִבֵּשׁ אֶת הַשָּׂדוֹת, עַד שֶׁעָנָן כָּבֵד כִּסָּה אוֹתוֹ, וְהָאוֹר שֶׁלּוֹ לֹא הִגִּיעַ עוֹד לָאָרֶץ. "הֶעָנָן חָזָק מִמֶּנִּי!" כָּעַס. "הַלְוַאי שֶׁהָיִיתִי עָנָן!"

כְּעָנָן, הוּא הִמְטִיר גְּשָׁמִים כְּבֵדִים. הַנְּהָרוֹת עָלוּ עַל גְּדוֹתֵיהֶם, וְהַשִּׁטָּפוֹן סָחַף עֵצִים וּגְדֵרוֹת. אֲבָל כְּשֶׁהַמַּיִם יָרְדוּ, הוּא רָאָה דָּבָר מְשֻׁנֶּה: הָהָר הַגָּדוֹל עָמַד בִּמְקוֹמוֹ וְלֹא זָז אֲפִלּוּ קְצָת. "הָהָר חָזָק מִכֻּלָּם! הַלְוַאי שֶׁהָיִיתִי הָהָר!"

כְּהַר, שׁוּם דָּבָר לֹא הִזִּיז אוֹתוֹ — לֹא חֹם הַשֶּׁמֶשׁ וְלֹא הַגֶּשֶׁם. אֲבָל בֹּקֶר אֶחָד שָׁמַע קוֹל מוּזָר לְמַרְגְּלוֹתָיו: טַק, טַק, טַק. לִפְנֵי שֶׁהֵבִין מָה קוֹרֶה, כְּבָר נִסְדַּק סֶלַע בְּצִדּוֹ. לְמַטָּה עָמַד אִישׁ קָטָן עִם פַּטִּישׁ וְאִזְמֵל — חוֹצֵב אֲבָנִים.

"אִישׁ קָטָן חָזָק יוֹתֵר מֵהָהָר הַגָּדוֹל?" תָּמַהּ. "הַלְוַאי שֶׁהָיִיתִי שׁוּב חוֹצֵב אֲבָנִים!" וְרוּחַ הָהָר הִגְשִׁימָה גַּם אֶת הַמִּשְׁאָלָה הָאַחֲרוֹנָה.

מֵאָז שֶׁחָזַר לַסֶּלַע, הוּא לֹא בִּקֵּשׁ עוֹד דָּבָר. הוּא חָצַב אֲבָנִים מִן הַבֹּקֶר וְעַד הָעֶרֶב, חָזַר הַבַּיְתָה עָיֵף וְחִיֵּךְ: עַכְשָׁו יָדַע שֶׁלְּכָל אֶחָד יֵשׁ כֹּחַ מִשֶּׁלּוֹ.

מָקוֹר: אַגָּדַת עַם מִמִּזְרַח אַסְיָה (מֻכֶּרֶת בְּיַפָּן וּבְסִין), שֶׁהִתְפַּרְסְמָה בְּאֵירוֹפָּה בִּזְכוּת הַסּוֹפֵר הַהוֹלַנְדִּי מוּלְטָטוּלִי (בַּמֵּאָה הַ־19). עִבּוּד חָפְשִׁי.

📚 דִּקְדּוּק — מִשְׁפַּט זְמַן

⏳ מִשְׁפַּט זְמַן — מִשְׁפָּט שֶׁעוֹנֶה עַל הַשְּׁאֵלָה "מָתַי?"

מִשְׁפַּט זְמַן הוּא מִשְׁפָּט מֻרְכָּב שֶׁיֵּשׁ בּוֹ שְׁנֵי חֲלָקִים: מִשְׁפָּט עִקָּרִי (מָה קָרָה), וּפְסוּקִית זְמַן שֶׁעוֹנָה עַל הַשְּׁאֵלָה מָתַי? פְּסוּקִית הַזְּמַן מַתְחִילָה בְּמִלַּת זְמַן (כְּמוֹ כְּשֶׁ־, כַּאֲשֶׁר, לִפְנֵי שֶׁ־), וְיֵשׁ בָּהּ פֹּעַל מִשֶּׁלָּהּ.

כְּשֶׁהִתְעוֹרֵר + מָצָא אֶת עַצְמוֹ בְּבַיִת מְפֹאָר

מָתַי מָצָא אֶת עַצְמוֹ בְּבַיִת מְפֹאָר? ← כְּשֶׁהִתְעוֹרֵר. זוֹ פְּסוּקִית הַזְּמַן.

מִלַּת הַזְּמַןמָה הִיא אוֹמֶרֶת?דֻּגְמָה מֵהַסִּפּוּר
כְּשֶׁ־ / כַּאֲשֶׁרבְּאוֹתוֹ רֶגַעכְּשֶׁהַמַּיִם יָרְדוּ, הוּא רָאָה דָּבָר מְשֻׁנֶּה.
בִּזְמַן שֶׁ־שְׁנֵי דְּבָרִים קוֹרִים יַחַדבִּזְמַן שֶׁיָּשַׁן, רוּחַ הָהָר הִגְשִׁימָה אֶת מִשְׁאַלְתּוֹ.
לִפְנֵי שֶׁ־הַמִּשְׁפָּט הָעִקָּרִי קָרָה קֹדֶםלִפְנֵי שֶׁהֵבִין מָה קוֹרֶה, כְּבָר נִסְדַּק סֶלַע.
אַחֲרֵי שֶׁ־הַמִּשְׁפָּט הָעִקָּרִי קָרָה אַחַר כָּךְאַחֲרֵי שֶׁרָאָה אֶת הַנָּסִיךְ, הוּא קִנֵּא שׁוּב.
עַד שֶׁ־נְקֻדַּת הַסִּיּוּםהוּא עָבַד עַד שֶׁהַשֶּׁמֶשׁ שָׁקְעָה.
מֵאָז שֶׁ־נְקֻדַּת הַהַתְחָלָה, וּמִשָּׁם וָהָלְאָהמֵאָז שֶׁחָזַר לַסֶּלַע, הוּא לֹא בִּקֵּשׁ עוֹד דָּבָר.
✏️ פְּסִיק: כְּשֶׁפְּסוּקִית הַזְּמַן בָּאָה בַּהַתְחָלָה — שָׂמִים פְּסִיק אַחֲרֶיהָ: כְּשֶׁהַשֶּׁמֶשׁ עָלְתָה, הוּא יָצָא לַעֲבֹד. כְּשֶׁהִיא בָּאָה בַּסּוֹף — בְּדֶרֶךְ כְּלָל אֵין פְּסִיק: הוּא עָבַד עַד שֶׁהַשֶּׁמֶשׁ שָׁקְעָה.
⚠️ מַלְכֹּדֶת 1 — לֹא כָּל כְּ־ הִיא זְמַן: "כְּעָנָן, הוּא הִמְטִיר גְּשָׁמִים" — כְּ + שֵׁם עֶצֶם פֵּרוּשׁוֹ "בְּתוֹר עָנָן", וְאֵין כָּאן פְּסוּקִית זְמַן. פְּסוּקִית זְמַן צְרִיכָה מִלַּת זְמַן וּפֹעַל: כְּשֶׁהָיָה עָנָן.
⚠️ מַלְכֹּדֶת 2 — צֵרוּף זְמַן בְּלִי פֹּעַל: "לִפְנֵי הַשְּׁקִיעָה הוּא נָח" — זֶה צֵרוּף זְמַן, אֲבָל לֹא פְּסוּקִית, כִּי אֵין בּוֹ פֹּעַל. "לִפְנֵי שֶׁהַשֶּׁמֶשׁ שָׁקְעָה, הוּא נָח" — זוֹ כְּבָר פְּסוּקִית זְמַן.
⚠️ מַלְכֹּדֶת 3 — זְמַן אוֹ סִבָּה? פְּסוּקִית שֶׁמַּתְחִילָה בְּ"כִּי" אוֹ "מִפְּנֵי שֶׁ־" עוֹנָה עַל לָמָּה? — זֶה מִשְׁפַּט סִבָּה (שִׁיעוּר 116). פְּסוּקִית זְמַן עוֹנָה עַל מָתַי?
✏️ מִשְׂחַק דִּקְדּוּק — יוֹם עֲבוֹדָה לְיַד הַסֶּלַע
🪨 יוֹם עֲבוֹדָה לְיַד הַסֶּלַע
עַל כָּל תְּשׁוּבָה נְכוֹנָה הַחוֹצֵב מוֹסִיף אֶבֶן לַקִּיר, וְהַשָּׁעָה מִתְקַדֶּמֶת — עַד הַשְּׁקִיעָה.

כְּתֹב חֲמִשָּׁה מִשְׁפְּטֵי זְמַן עַל הַיּוֹם שֶׁלְּךָ — בְּכָל מִשְׁפָּט מִלַּת זְמַן אַחֶרֶת. אַל תִּשְׁכַּח פְּסִיק אַחֲרֵי פְּסוּקִית הַזְּמַן כְּשֶׁהִיא בָּאָה בַּהַתְחָלָה. דֻּגְמָה: כְּשֶׁהַשָּׁעוֹן מְצַלְצֵל, אֲנִי קָם מִן הַמִּטָּה.
