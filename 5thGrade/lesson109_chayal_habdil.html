חַיַּל הַבְּדִיל הָאֵיתָן

סִפּוּר מֻכָּר מֵאֵת הַנְס כְּרִיסְטְיָאן אַנְדֶּרְסֶן, בְּעִבּוּד

📖 קְרִיאָה

בְּבַיִת אֶחָד גָּר יֶלֶד, וּבְיוֹם הֻלַּדְתּוֹ קִבֵּל קֻפְסָה עִם עֶשְׂרִים וַחֲמִשָּׁה חַיָּלֵי בְּדִיל. כֻּלָּם נוֹצְרוּ מִכַּף בְּדִיל יְשָׁנָה, וְלָכֵן הָיוּ דּוֹמִים זֶה לָזֶה: מַדִּים אֲדֻמִּים וּכְחֻלִּים, כּוֹבַע גָּבוֹהַּ וְרוֹבֶה עַל הַכָּתֵף.

רַק חַיָּל אֶחָד הָיָה שׁוֹנֶה. הוּא נוֹצַר אַחֲרוֹן, וְלֹא נִשְׁאַר מַסְפִּיק בְּדִיל בִּשְׁבִילוֹ, וְלָכֵן הָיְתָה לוֹ רֶגֶל אַחַת בִּלְבַד. אַף עַל פִּי כֵן עָמַד הַחַיָּל בְּגַאֲוָה וּבְיֹשֶׁר עַל רַגְלוֹ הָאַחַת, וְלֹא זָז.

עַל הַשֻּׁלְחָן עָמְדוּ עוֹד צַעֲצוּעִים רַבִּים. בֵּינֵיהֶם הָיָה אַרְמוֹן יָפֶה מִנְּיָר, וּלְיָדוֹ אֲגַם קָטָן מִמַּרְאָה. בְּפֶתַח הָאַרְמוֹן עָמְדָה רַקְדָּנִית עֲדִינָה מִנְּיָר, וְעַל שִׂמְלָתָהּ נִצְנֵץ כּוֹכָב קָטָן. הִיא פָּרְשָׂה אֶת זְרוֹעוֹתֶיהָ וְהֵרִימָה אֶת אַחַת מֵרַגְלֶיהָ כֹּה גָּבוֹהַּ, שֶׁהַחַיָּל לֹא רָאָה אוֹתָהּ וְחָשַׁב שֶׁגַּם לָהּ רֶגֶל אַחַת.

הַחַיָּל הִתְאַהֵב בָּרַקְדָּנִית. בַּלַּיְלָה, כְּשֶׁהַיֶּלֶד יָשַׁן, הַצַּעֲצוּעִים הִתְעוֹרְרוּ וְשִׂחֲקוּ. רַק הַחַיָּל וְהָרַקְדָּנִית לֹא זָזוּ מִמְּקוֹמָם, וְהוּא לֹא הוֹרִיד מִמֶּנָּה אֶת הָעֵינַיִם. אֲבָל מִתּוֹךְ קֻפְסַת טַבָּק קְטַנָּה קָפַץ שְׁדוֹן שָׁחוֹר וְלָחַשׁ בְּכַעַס: ״חַיָּל בְּדִיל, אַל תִּסְתַּכֵּל עַל הָרַקְדָּנִית!״

בַּבֹּקֶר הִנִּיחַ הַיֶּלֶד אֶת הַחַיָּל עַל אֶדֶן הַחַלּוֹן. פִּתְאֹם נִשְׁבָה רוּחַ חֲזָקָה, הַחַלּוֹן נִפְתַּח, וְהַחַיָּל נָפַל מִן הַקּוֹמָה הַשְּׁלִישִׁית אֶל הָרְחוֹב! הוּא נָחַת עַל הָרֹאשׁ וְנִתְקַע בֵּין אַבְנֵי הַמִּדְרָכָה. הַיֶּלֶד וְהַמְּנַקָּה יָרְדוּ לְחַפֵּשׂ אוֹתוֹ, אֲבָל לֹא מָצְאוּ.

אַחַר כָּךְ יָרַד גֶּשֶׁם עָז, וְהָרְחוֹב הִתְמַלֵּא מַיִם. שְׁנֵי נְעָרִים עָבְרוּ שָׁם. הֵם מָצְאוּ אֶת הַחַיָּל, עָשׂוּ סִירָה מֵעִתּוֹן, הִנִּיחוּ אוֹתוֹ בְּתוֹכָהּ וְהֵשִׁיטוּ אוֹתָהּ בַּתְּעָלָה. הַסִּירָה נָסְעָה מַהֵר, וְהַחַיָּל עָמַד עַל רַגְלוֹ הָאַחַת וּפָנָיו לְפָנִים.

פִּתְאֹם הוֹפִיעָה חֻלְדַּת מַיִם גְּדוֹלָה וְצָעֲקָה: ״דַּרְכּוֹן! תֵּן לִי דַּרְכּוֹן!״ הַחַיָּל שָׁתַק וְהֶחֱזִיק אֶת הָרוֹבֶה חָזָק. הַחֻלְדָּה רָדְפָה אַחֲרָיו, אֲבָל הַזֶּרֶם הָיָה חָזָק מִמֶּנָּה.

הַתְּעָלָה נִכְנְסָה לְצִנּוֹר אָרֹךְ וְחָשׁוּךְ, וְשָׁם נִשְׁמַע שָׁאוֹן עָצוּם. הַסִּירָה הִגִּיעָה אֶל מַפָּל, הִסְתּוֹבְבָה שָׁלוֹשׁ פְּעָמִים וְהִתְמַלְּאָה מַיִם. הָעִתּוֹן הִתְרַכֵּךְ וְנִקְרַע, אֲבָל הַחַיָּל חָשַׁב עַל הָרַקְדָּנִית, וַאֲפִלּוּ בַּחֲשֵׁכָה לֹא הִרְפָּה מֵהָרוֹבֶה.

בַּסּוֹף נִבְלַע הַחַיָּל בְּפִי דָּג גָּדוֹל. בְּתוֹךְ הַבֶּטֶן הָיָה צַר, אֲבָל הוּא שָׁכַב יָשָׁר, עִם הָרוֹבֶה עַל הַכָּתֵף. אַחֲרֵי זְמַן רַב נִתְפַּס הַדָּג, נִמְכַּר בַּשּׁוּק וְהוּבָא אֶל מִטְבַּח בֵּיתוֹ שֶׁל הַיֶּלֶד. הַטַּבָּחִית פָּתְחָה אֶת בֶּטֶן הַדָּג וְהִתְפַּלְּאָה: ״זֶה הַחַיָּל שֶׁאָבַד לַיֶּלֶד!״

הַיֶּלֶד שָׂמַח מְאֹד, וְהַחַיָּל חָזַר אֶל הַשֻּׁלְחָן. הוּא רָאָה שׁוּב אֶת הָאַרְמוֹן וְאֶת הָרַקְדָּנִית, שֶׁעוֹד עָמְדָה עַל רֶגֶל אַחַת. אֲבָל פִּתְאֹם, בְּלִי סִבָּה בְּרוּרָה, זָרַק הַיֶּלֶד אֶת הַחַיָּל אֶל הַתַּנּוּר. הַחַיָּל הִרְגִּישׁ אֶת הַחֹם הַגָּדוֹל, אַךְ לֹא זָז: הוּא הִמְשִׁיךְ לַעֲמֹד יָשָׁר וְלִהְיוֹת נֶאֱמָן.

אָז נִפְתַּח הַחַלּוֹן, וְרוּחַ נָשְׂאָה אֶת הָרַקְדָּנִית וְהִטִּיסָה אוֹתָהּ אֶל הַתַּנּוּר. הִיא עָפָה אֶל הַחַיָּל, וְהָאֵשׁ אָחֲזָה בָּהּ. הַחַיָּל הִתְחִיל לְהִתְמוֹסֵס, וְכָל מַה שֶּׁנִּשְׁאַר מִמֶּנּוּ הָיָה לֵב קָטָן מִבְּדִיל. כְּשֶׁהַמְּנַקָּה פִּנְּתָה בַּבֹּקֶר אֶת הָאֵפֶר, הִיא מָצְאָה אֶת הַלֵּב, וּלְצִדּוֹ אֶת הַכּוֹכָב שֶׁל הָרַקְדָּנִית, שָׁחוֹר כְּמוֹ פֶּחָם.

הַסִּפּוּר מְבֻסָּס עַל ״חַיַּל הַבְּדִיל הָאֵיתָן״ מֵאֵת הַנְס כְּרִיסְטְיָאן אַנְדֶּרְסֶן (דֶּנְמַרְק, 1838), בְּעִבּוּד וְקִצּוּר.

אוֹת אַחַת, כַּמָּה צוּרוֹת

וָו הַחִבּוּר מְחַבֶּרֶת בֵּין מִלִּים וּבֵין מִשְׁפָּטִים, כְּמוֹ ״חַיָּל וְרַקְדָּנִית״. בְּדֶרֶךְ כְּלָל הִיא מְנֻקֶּדֶת בִּשְׁוָא (וְ), אֲבָל לִפְעָמִים הִיא מְשַׁנָּה צוּרָה כְּדֵי לְהַתְאִים לַמִּלָּה שֶׁאַחֲרֶיהָ. בַּסִּפּוּר פָּגַשְׁנוּ אֶת כָּל הַצּוּרוֹת:

הַזָּוִית הֶחָדָשָׁה בַּחֲזָרָה: לִפְנֵי חֲטַף ״וְ״ אֵינָהּ נִשְׁאֶרֶת שְׁוָא! כְּשֶׁהַמִּלָּה הַבָּאָה מַתְחִילָה בַּחֲטַף־פַּתַּח (אֲ, חֲ, עֲ) הַוָּו נִהְיֵית וַ: ״וַחֲמִשָּׁה״, ״וַאֲפִלּוּ״. כְּשֶׁהִיא מַתְחִילָה בַּחֲטַף־סֶגּוֹל (אֱ) הַוָּו נִהְיֵית וֶ: ״וֶאֱמֶת״.

שִׂימוּ לֵב: ״וְהַחַיָּל״ ו״וְאֶת״ נִשְׁאָרוֹת ״וְ״, כִּי הֵא הַיְּדִיעָה וְאָלֶף עִם סֶגּוֹל אֵינָן שְׁוָא וְאֵינָן חֲטַף. וְהַוָּו תָּמִיד נִכְתֶּבֶת צְמוּדָה לַמִּלָּה.

      בַּלַּיְלָה הַצַּעֲצוּעִים שֶׁל הַיֶּלֶד מִתְעוֹרְרִים. כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים עַל מָה שֶׁקּוֹרֶה בְּחֶדֶר הַצַּעֲצוּעִים. בְּכָל מִשְׁפָּט שַׁלְּבוּ שְׁתֵּי מִלִּים שֶׁמְּחֻבָּרוֹת בְּוָו הַחִבּוּר, בַּצּוּרָה שֶׁמְּבֻקֶּשֶׁת בַּשּׁוּרָה, וְשִׂימוּ לֵב לְנִקּוּד הַוָּו. דֻּגְמָה: ״הַדֻּבִּי וְהַבֻּבָּה יוֹשְׁבִים עַל הַשֻּׁלְחָן.״
