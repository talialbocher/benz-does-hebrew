📖 קְרִיאָה

פַּעַם, בְּיַעַר גָּדוֹל וְסָבוּךְ, גָּרוּ אָח וְאָחוֹת בְּשֵׁם הַנְזֶל וְגַרְטֶל עִם אֲבִיהֶם וְעִם אִשְׁתּוֹ הַחוֹרֶגֶת. בַּבַּיִת שָׂרַר רָעָב גָּדוֹל, וְלֹא הָיָה מַסְפִּיק אוֹכֶל לְכוּלָּם.

יוֹם אֶחָד הֶחְלִיטוּ הָאָב וְאִשְׁתּוֹ לָקַחַת אֶת הַיְלָדִים עָמוֹק אֶל תּוֹךְ הַיַּעַר, וּלְהַשְׁאִיר אוֹתָם שָׁם לְבַד. הַנְזֶל שָׁמַע אֶת הַשִּׂיחָה בַּלַּיְלָה, וְלִפְנֵי שֶׁיָּצְאוּ לַדֶּרֶךְ הוּא אָסַף בַּחֲשַׁאי צְרוֹרוֹת קְטַנִּים שֶׁל אֲבָנִים לְבָנוֹת.

בַּדֶּרֶךְ אֶל תּוֹךְ הַיַּעַר, הַנְזֶל פִּיזֵּר אֶת הָאֲבָנִים הַלְּבָנוֹת מֵאֲחוֹרָיו, אַחַת אַחַת, בְּלִי שֶׁאִישׁ יָשִׂים לֵב. כְּשֶׁהָאָב וְאִשְׁתּוֹ נֶעֶלְמוּ בֵּין הָעֵצִים, פָּחֲדוּ הַנְזֶל וְגַרְטֶל מְאוֹד - אֲבָל כְּשֶׁעָלָה הַיָּרֵחַ, הָאֲבָנִים הַלְּבָנוֹת נָצְצוּ בָּאוֹר, וְהֶרְאוּ אֶת הַדֶּרֶךְ חֲזָרָה הַבַּיְתָה.

אַחֲרֵי כַּמָּה יָמִים נִיסּוּ הָאָב וְאִשְׁתּוֹ שׁוּב לָקַחַת אֶת הַיְלָדִים לַיַּעַר. הַפַּעַם לֹא הִסְפִּיק הַנְזֶל לֶאֱסוֹף אֲבָנִים, וּבִמְקוֹמָן הוּא פִּיזֵּר פֵּירוּרֵי לֶחֶם לְאוֹרֶךְ הַדֶּרֶךְ. אֲבָל צִיפּוֹרֵי הַיַּעַר אָכְלוּ אֶת כָּל הַפֵּירוּרִים, וְהַיְלָדִים נִשְׁאֲרוּ לְבַד, אֲבוּדִים בֵּין הָעֵצִים.

אַחֲרֵי שֶׁהָלְכוּ זְמַן רַב, הַנְזֶל וְגַרְטֶל הִגִּיעוּ לְבַיִת מוּזָר וּמְיוּחָד. קִירוֹת הַבַּיִת הָיוּ עֲשׂוּיִים מֵעוּגָה מְתוּקָה, הַגַּג כּוּסָּה בְּשׁוֹקוֹלָד, וְהַשּׁוּלְחָן בִּפְנִים סוּדַּר בְּעוּגוֹת וּבְמַמְתַּקִּים צִבְעוֹנִיִּים. הַיְלָדִים הָרְעֵבִים הִתְחִילוּ מִיָּד לֶאֱכוֹל חֲתִיכוֹת מֵהַבַּיִת.

פִּתְאוֹם נִפְתְּחָה הַדֶּלֶת, וְיָצְאָה מִתּוֹכָהּ אִישָּׁה זְקֵנָה. הִיא חִייְכָה בְּנוֹעַם וְהִזְמִינָה אֶת הַיְלָדִים פְּנִימָה. אֲבָל הָאִישָּׁה הַזֹּאת הָיְיתָה לְמַעֲשֶׂה מְכַשֵּׁפָה מְרוּשַּׁעַת, שֶׁהַבַּיִת שֶׁלָּהּ קוּשַּׁט בְּמַמְתַּקִּים כְּדֵי לְפַתּוֹת יְלָדִים רְעֵבִים אֶל תּוֹךְ הַמַּלְכּוֹדֶת שֶׁלָּהּ.

הַמְכַשֵּׁפָה סָגְרָה אֶת הַנְזֶל בִּכְלוּב קָטָן בֶּחָצֵר, כְּדֵי לְפַטֵּם אוֹתוֹ. בְּכָל יוֹם הִיא בִּיקְשָׁה מִגַּרְטֶל לְהָכִין לוֹ אוֹכֶל, וּבָשָׂר שָׁמֵן בּוּשַּׁל עַל הָאֵשׁ כְּדֵי שֶׁהַנְזֶל יִהְיֶה שָׁמֵן וּמוּכָן לַאֲרוּחָה שֶׁל הַמְכַשֵּׁפָה. בְּכָל עֶרֶב הַכְּלוּב נוּקָּה עַל יְדֵי גַּרְטֶל, כְּדֵי שֶׁהַמְכַשֵּׁפָה לֹא תִּכְעַס עָלֶיהָ.

הַמְכַשֵּׁפָה לֹא רָאֲתָה טוֹב, וְלָכֵן כָּל בּוֹקֶר הִיא בִּיקְשָׁה מֵהַנְזֶל לְהוֹשִׁיט אֶצְבַּע דֶּרֶךְ סוֹרְגֵי הַכְּלוּב, כְּדֵי לִבְדּוֹק אִם הוּא כְּבָר שָׁמֵן מַסְפִּיק. הַנְזֶל, שֶׁהָיָה חָכָם, הוֹשִׁיט בִּמְקוֹם זֹאת עֶצֶם קְטַנָּה שֶׁמָּצָא בַּכְּלוּב - וְכָךְ הַמְכַשֵּׁפָה חָשְׁבָה שֶׁהוּא עֲדַיִין רָזֶה מִדַּי, וְלֹא מִיהֲרָה לֶאֱכוֹל אוֹתוֹ.

יוֹם אֶחָד רָצְתָה הַמְכַשֵּׁפָה לְהַדְלִיק אֶת הַתַּנּוּר כְּדֵי לֶאֱפוֹת אֶת הַנְזֶל, וּבִיקְשָׁה מִגַּרְטֶל לְהִיכָּנֵס פְּנִימָה וְלִבְדּוֹק אִם הָאֵשׁ כְּבָר חַמָּה מַסְפִּיק. גַּרְטֶל הֵבִינָה מִיָּד אֶת הַכַּוָּונָה הָרָעָה, וְהֶעֱמִידָה פָּנִים שֶׁאֵינָהּ מְבִינָה אֵיךְ בְּדִיּוּק צָרִיךְ לְהִיכָּנֵס לַתַּנּוּר. כְּשֶׁהַמְכַשֵּׁפָה עַצְמָהּ הִתְכּוֹפְפָה כְּדֵי לְהַרְאוֹת לָהּ אֵיךְ, גַּרְטֶל דָּחֲפָה אוֹתָהּ בְּחוֹזְקָה לְתוֹךְ הַתַּנּוּר, וְסָגְרָה מַהֵר אֶת הַדֶּלֶת.

גַּרְטֶל מִיהֲרָה לְשַׁחְרֵר אֶת הַנְזֶל מֵהַכְּלוּב. הוּא שׁוּחְרַר מִיָּד, וּשְׁנֵי הַיְלָדִים חִיבְּקוּ זֶה אֶת זֶה בְּשִׂמְחָה גְּדוֹלָה. הֵם מָצְאוּ בְּבֵית הַמְכַשֵּׁפָה תֵּיבָה מְלֵאָה זָהָב וַאֲבָנִים יְקָרוֹת, וּמִילְּאוּ בָּהֶן אֶת הַכִּיסִים.

הַנְזֶל וְגַרְטֶל הָלְכוּ בַּדֶּרֶךְ חֲזָרָה הַבַּיְתָה, וּכְשֶׁהִגִּיעוּ, אֲבִיהֶם שָׂמַח מְאוֹד לִרְאוֹתָם - הוּא הִתְחָרֵט עַל מָה שֶׁקָּרָה, וְאִשְׁתּוֹ הָרָעָה כְּבָר לֹא הָיְיתָה בַּבַּיִת. מֵאוֹתוֹ יוֹם הִתְגּוֹרְרוּ הַנְזֶל, גַּרְטֶל וַאֲבִיהֶם בְּיַחַד, וְלֹא חָסַר לָהֶם עוֹד דָּבָר.

עִיבּוּד לְסִיפּוּר עַם מוּכָּר מֵאֵת הָאַחִים גְּרִים (גֶּרְמַנְיָה) - נַחֲלַת הַכְּלָל

מכירים בניין חדש: פֻּעַל

בשיעורים הקודמים (10, 12, 13, 15) הכרנו את הבניינים הפעיל, פיעל, נפעל והתפעל. היום נכיר בניין נוסף: בניין פֻּעַל. זהו הבניין הסביל (הפסיבי) של בניין פיעל, בדיוק כמו שבניין נפעל הוא לפעמים הבניין הסביל של בניין קל.

בבניין פיעל מישהו עושה פעולה לדבר מה - למשל, המכשפה קישטה את הבית בממתקים. בבניין פֻּעַל מדגישים את הדבר שקרה לו, בלי לומר תמיד מי עשה את הפעולה - הבית קושט בממתקים.
