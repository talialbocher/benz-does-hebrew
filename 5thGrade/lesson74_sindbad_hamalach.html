📖 קְרִיאָה

בְּעִיר בַּגְדָּאד הָעַתִּיקָה חַי פַּעַם סוֹחֵר עָשִׁיר בְּשֵׁם סִינְדְּבָּד. הָיוּ לוֹ בַּיִת גָּדוֹל, שָׂדוֹת פּוֹרִים וַחֲבֵרִים רַבִּים, אַךְ לִבּוֹ לֹא יָדַע מְנוּחָה. בְּכָל לַיְלָה יָשַׁב עַל גַּג בֵּיתוֹ, הִבִּיט אֶל הַיָּם הָרָחוֹק, וְחָלַם עַל אֲרָצוֹת שֶׁמֵּעוֹלָם לֹא רָאָה. יוֹם אֶחָד הֶחְלִיט: הוּא יִמְכֹּר חֵלֶק מֵרְכוּשׁוֹ, יִקְנֶה סְחוֹרָה, וְיֵצֵא לְהַפְלָגָה רִאשׁוֹנָה בְּחַיָּיו.

סִינְדְּבָּד עָלָה עַל אֳנִיָּה גְּדוֹלָה יַחַד עִם קְבוּצַת סוֹחֲרִים אֲחֵרִים. הָרוּחַ מִלְּאָה אֶת הַמִּפְרָשִׂים, וְהָאֳנִיָּה שָׁטָה בֵּין אִיִּים קְטַנִּים וּנְמֵלִים רְחוֹקִים. סִינְדְּבָּד נֶהֱנָה מְאֹד מֵהַנּוֹף וּמֵהַסִּפּוּרִים שֶׁסִּפְּרוּ לוֹ הַמַּלָּחִים עַל אֲרָצוֹת קְסוּמוֹת. הוּא יָשַׁב עִמָּם עַל סִפּוּן הָאֳנִיָּה שָׁעוֹת אֲרֻכּוֹת, וְשָׂמַח שֶׁיָּצָא סוֹף־סוֹף מִבֵּיתוֹ.

יוֹם אֶחָד, בְּלֵב יָם רָחוֹק, הִבְחִין רַב הַחוֹבֵל בְּאִי יָרֹק קָטָן, שָׁטוּחַ וְשָׁקֵט. הוּא הֵנִיחַ אֶת הָאֳנִיָּה לַעֲגֹן לְיָדוֹ, וְהַמַּלָּחִים יָרְדוּ אֶל הָאִי עִם סִינְדְּבָּד כְּדֵי לָנוּחַ וּלְבַשֵּׁל אֲרוּחָה חַמָּה. הֵם הִדְלִיקוּ מְדוּרָה גְּדוֹלָה מַמָּשׁ עַל גַּבּוֹ שֶׁל הָאִי - וְאָז, לְפֶתַע, הֵחֵל הַקַּרְקַע לִרְעֹד תַּחְתֵּיהֶם.

הָאִי לֹא הָיָה אִי כְּלָל! זֶה הָיָה לִוְיָתָן עֲנָק, שֶׁיָּשֵׁן עַל פְּנֵי הַיָּם שָׁנִים רַבּוֹת כָּל־כָּךְ, עַד שֶׁצְּמָחִים גָּדְלוּ עַל גַּבּוֹ. חֹם הָאֵשׁ הֵעִיר אוֹתוֹ מִשְּׁנָתוֹ, וְהוּא צָלַל בְּבַת אַחַת אֶל תּוֹךְ הַמַּיִם הָעֲמֻקִּים. הַמַּלָּחִים שֶׁהִסְפִּיקוּ לָרוּץ בַּחֲזָרָה לָאֳנִיָּה נִמְלְטוּ בְּפַחַד, אַךְ סִינְדְּבָּד נוֹתַר לְבַדּוֹ בַּמַּיִם, אוֹחֵז בְּקֶרֶשׁ עֵץ שֶׁצָּף לְיָדוֹ.

שָׁעוֹת רַבּוֹת שָׂחָה סִינְדְּבָּד עִם הַקֶּרֶשׁ, עַד שֶׁגַּלִּים חֲזָקִים סָחֲפוּ אוֹתוֹ אֶל חוֹף אִי אַחֵר, אֲמִתִּי הַפַּעַם. עָיֵף וְרָעֵב, טִפֵּס סִינְדְּבָּד עַל הַחוֹל, וְנִרְדַּם מִיָּד. כְּשֶׁהִתְעוֹרֵר, רָאָה לְיָדוֹ רוֹעֶה צָעִיר שֶׁשָּׁמַר עַל סוּסִים יָפִים. הָרוֹעֶה נִבְהַל לִרְאוֹת אָדָם זָר עַל הַחוֹף, אַךְ כְּשֶׁשָּׁמַע אֶת סִפּוּרוֹ שֶׁל סִינְדְּבָּד, רִחֵם עָלָיו וְהֵבִיא אוֹתוֹ עִמּוֹ אֶל אַרְמוֹנוֹ שֶׁל הַמֶּלֶךְ מִהְרְגָ'אן, שֶׁלּוֹ שַׁיָּכִים הַסּוּסִים.

הַמֶּלֶךְ מִהְרְגָ'אן קִבֵּל אֶת סִינְדְּבָּד בְּלֵב פָּתוּחַ, הִלְבִּישׁ אוֹתוֹ בְּבִגְדֵי כָּבוֹד וְהוֹשִׁיב אוֹתוֹ לְיָדוֹ בַּסְּעֻדּוֹת. סִינְדְּבָּד סִפֵּר לוֹ עַל בַּגְדָּאד וְעַל הָאֳנִיָּה שֶׁאִבֵּד, וְהַמֶּלֶךְ הֶעֱנִיק לוֹ תַּפְקִיד חָשׁוּב בְּמֶמְשַׁלְתּוֹ. חֹדֶשׁ אַחַר חֹדֶשׁ עָבַר, וְהָאַחְרָיוּת שֶׁהֻטְּלָה עַל סִינְדְּבָּד גָּדְלָה - אַךְ הוּא מֵעוֹלָם לֹא שָׁכַח אֶת הַיָּם וְאֶת הַחֲלוֹם לַחֲזֹר הַבַּיְתָה.

יוֹם אֶחָד עָמַד סִינְדְּבָּד עַל שְׂפַת הַנָּמֵל, וְלִבּוֹ כִּמְעַט פָּסַק מִלִּפְעֹם: הָאֳנִיָּה שֶׁעָגְנָה שָׁם נִרְאֲתָה בְּדִיּוּק כְּמוֹ הָאֳנִיָּה שֶׁלּוֹ! הוּא רָץ אֵלֶיהָ וְדִבֵּר עִם רַב הַחוֹבֵל, שֶׁסִּפֵּר לוֹ שֶׁכֻּלָּם חָשְׁבוּ שֶׁסִּינְדְּבָּד טָבַע בַּיָּם, וְלָכֵן שָׁמְרוּ עֲבוּרוֹ בֶּאֱמוּנָה אֶת כָּל הַסְּחוֹרָה שֶׁהִשְׁאִיר עַל הַסִּפּוּן.

בְּשִׂמְחָה רַבָּה מָכַר סִינְדְּבָּד אֶת סְחוֹרָתוֹ בְּרֶוַח נָאֶה, וְנִפְרַד מֵהַמֶּלֶךְ וּמֵחֲבֵרָיו הַחֲדָשִׁים בְּחֹם רַב. הַמֶּלֶךְ נָתַן לוֹ מַתָּנוֹת יְקָרוֹת לְדֶרֶךְ, וְאָמַר לוֹ: "הַשָּׁלוֹם יִהְיֶה עִמְּךָ, יְדִידִי, וּתְפִלָּתִי עָלֶיךָ בְּכָל דֶּרֶךְ שֶׁתֵּלֵךְ." סִינְדְּבָּד חָזַר לְבַגְדָּאד עָשִׁיר בְּזָהָב וּבְחָכְמָה, וְסִפֵּר לְכָל מִי שֶׁפָּגַשׁ אֶת סִפּוּר הַהַפְלָגָה הָרִאשׁוֹנָה שֶׁלּוֹ - זוֹ שֶׁלִּמְּדָה אוֹתוֹ שֶׁגַּם בָּרֶגַע הַקָּשֶׁה בְּיוֹתֵר, כְּדַאי לִשְׁמֹר עַל אֹמֶץ וְתִקְוָה.

מְסֻפָּר עַל־פִּי "מַסָּעוֹת סִינְדְּבָּד הַמַּלָּח", מִתּוֹךְ אוֹסֵף "אֶלֶף לַיְלָה וָלַיְלָה" (עוֹלָם עֲרָבִי), מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ.

כִּנּוּיֵי גוּף חֲבוּרִים עִם מִלַּת הַיַּחַס "עִם"

בְּשִׁיעוּר 54 לָמַדְנוּ אֵיךְ מִלַּת הַיַּחַס "לְ-" מִצְטָרֶפֶת לְכִנּוּיֵי גוּף (לִי, לְךָ, לוֹ...), וּבְשִׁיעוּר 57 לָמַדְנוּ אֶת כִּנּוּיֵי הַגּוּף שֶׁל "אֶת" (אוֹתִי, אוֹתְךָ, אוֹתוֹ...). הַיּוֹם נִלְמַד עוֹד שְׁתֵּי מִלּוֹת יַחַס חֲשׁוּבוֹת שֶׁמִּצְטָרְפוֹת לְכִנּוּיֵי גוּף בְּאוֹתָהּ הַדֶּרֶךְ: "עִם" (מַשְׁמָעוּתָהּ - יַחַד עִם מִישֶׁהוּ) וְ"עַל" (מַשְׁמָעוּתָהּ - מֵעַל, אוֹ אַחְרָיוּת/נוֹשֵׂא).

כִּתְבוּ 5 מִשְׁפָּטִים, שֶׁבְּכָל אֶחָד מֵהֶם כִּנּוּי גוּף חָבוּר אֶחָד לְפָחוֹת שֶׁל "עִם" אוֹ שֶׁל "עַל". לְדֻגְמָה: "סִינְדְּבָּד רָצָה שֶׁחֲבֵרוֹ יַפְלִיג עִמּוֹ אֶל הָאִי הֶחָדָשׁ."
