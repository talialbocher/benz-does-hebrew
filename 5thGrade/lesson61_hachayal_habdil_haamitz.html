📚 סִיפּוּרִים בְּעִבְרִית — שִׁיעוּר 61 (שִׁיעוּר חֲזָרָה)

הַחַיָּל הַבְּדִיל הָאַמִּיץ

חַיָּלוֹן בְּדִיל בַּעַל רֶגֶל אַחַת מִתְאַהֵב בְּרַקְדָנִית נְיָר - וְשִׁיעוּר חֲזָרָה עַל מִלִּים מַקְטִינוֹת, עִם תּוֹסֶפֶת מַפְתִּיעָה עַל גְּוָנֵי צֶבַע

📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת חַי בְּעִיר קְטַנָּה יֶלֶד בְּשֵׁם אוּרִי. לְיוֹם הֻלַּדְתּוֹ קִבֵּל אוּרִי קֻפְסָה גְּדוֹלָה, וּבְתוֹכָהּ עֶשְׂרִים וַחֲמִשָּׁה חַיָּלוֹנִים קְטַנִּים, עֲשׂוּיִים מִבְּדִיל נוֹצֵץ. כָּל הַחַיָּלוֹנִים הָיוּ זֶהִים לְגַמְרֵי - חוּץ מֵאֶחָד. הַבְּדִיל שֶׁמִּמֶּנּוּ נוֹצַר הַחַיָּלוֹן הָאַחֲרוֹן לֹא הִסְפִּיק, וְלָכֵן יָצְקוּ לוֹ רַק רֶגֶל אַחַת בִּלְבַד. אַף אֶחָד לֹא חָשַׁב שֶׁהוּא פָּחוֹת שָׁוֶה בִּגְלַל זֶה - וּבְוַדַּאי לֹא הוּא עַצְמוֹ.

אוּרִי הֶעֱמִיד אֶת כָּל הַחַיָּלוֹנִים בְּשׁוּרָה עַל שֻׁלְחַן הַמִּשְׂחָקִים שֶׁלּוֹ, לְיַד אַרְמוֹן קָטָן עָשׂוּי נְיָר צִבְעוֹנִי. בְּפֶתַח הָאַרְמוֹן עָמְדָה רַקְדָנִית נְיָר עֲדִינָה, לְבוּשָׁה בְּשִׂמְלָה אֲדַמְדַּמָּה שֶׁדָּהֲתָה מְעַט מֵרֹב שָׁנִים. הָרַקְדָנִית הָיְתָה מְקֻפֶּלֶת בְּזָוִית מְיֻחֶדֶת, כָּךְ שֶׁנִּרְאֲתָה עוֹמֶדֶת עַל רֶגֶל אַחַת בִּלְבַד, וְרַגְלָהּ הַשְּׁנִיָּה הָיְתָה מוּרֶמֶת גָּבוֹהַּ מֵאֲחוֹרֶיהָ. הַחַיָּלוֹן בַּעַל רֶגֶל אַחַת הִבִּיט בָּהּ מֵהַצַּד הָאַחֵר שֶׁל הַשֻּׁלְחָן וְחָשַׁב לְעַצְמוֹ: "בְּוַדַּאי גַּם לָהּ יֵשׁ רַק רֶגֶל אַחַת, בְּדִיּוּק כָּמוֹנִי." מֵאוֹתוֹ רֶגַע הֵבִיא הַחַיָּלוֹן אֶת כָּל לִבּוֹ אֵלֶיהָ.

בַּלַּיְלָה, כְּשֶׁכָּל בְּנֵי הַבַּיִת יָשְׁנוּ, קָפַץ מִתּוֹךְ קֻפְסַת טַבָּק יְשָׁנָה שֶׁעָמְדָה עַל הַשֻּׁלְחָן שֵׁדוֹן קָטָן וְעַז מַבָּט. הַשֵּׁדוֹן רָאָה אֵיךְ הַחַיָּלוֹן מַבִּיט בָּרַקְדָנִית בְּלִי הֶרֶף, וְכָעַס. "לָמָּה אַתָּה מַבִּיט בְּמִי שֶׁלֹּא שַׁיֶּכֶת לְךָ?" קָרָא בְּזַעַף. אֲבָל הַחַיָּלוֹן לֹא עָנָה לוֹ מִלָּה. הוּא לֹא יָכוֹל הָיָה לָזוּז מֵהַשֻּׁלְחָן בְּעַצְמוֹ, וְלָכֵן פָּשׁוּט הִמְשִׁיךְ לְהַבִּיט קָדִימָה בְּשֶׁקֶט וּבְנֶאֱמָנוּת, כְּפִי שֶׁחַיָּל בְּדִיל אַמִּיץ אָמוּר לַעֲשׂוֹת.

לְמָחֳרָת בַּבֹּקֶר הִנִּיחַ אוּרִי אֶת הַחַיָּלוֹן עַל אֶדֶן הַחַלּוֹן. פִּתְאוֹם נָשְׁבָה רוּחַ חֲזָקָה, הַחַלּוֹן נִפְתַּח לִרְוָחָה, וְהַחַיָּלוֹן נָפַל מִשְּׁלוֹשָׁה קוֹמוֹת אֶל הָרְחוֹב. הוּא נָחַת בֵּין אַבְנֵי הַמִּדְרָכָה, רֹאשׁוֹ לְמַטָּה וְרַגְלוֹ הָאַחַת לְמַעְלָה, וְנִשְׁאַר שָׁם בְּלִי לְהַשְׁמִיעַ קוֹל תְּלוּנָה. כְּשֶׁהֵחֵל לָרֶדֶת גֶּשֶׁם חָזָק, מָצְאוּ אוֹתוֹ שְׁנֵי יְלָדִים שֶׁשִּׂחֲקוּ בָּרְחוֹב, הִנִּיחוּ אוֹתוֹ בְּתוֹךְ סִירוֹנֶת נְיָר קְטַנָּה, וְשָׁלְחוּ אוֹתָהּ לָשׁוּט בְּתַעֲלַת הַמַּיִם שֶׁזָּרְמָה בְּצַד הָרְחוֹב.

הַסִּירוֹנֶת שָׁטָה מַהֵר וּמַהֵר, עַד שֶׁנֶּעֶלְמָה בְּתוֹךְ תְּעָלָה חֲשׁוּכָה שֶׁל מֵי גְּשָׁמִים. הַנְּיָר הַדַּק הֻרְטַב, הַסִּירוֹנֶת הִתְמַלְּאָה מַיִם, וּלְבַסּוֹף שָׁקְעָה כֻּלָּהּ. הַחַיָּלוֹן שָׁקַע יַחַד אִתָּהּ, וּבְאוֹתוֹ רֶגַע מַמָּשׁ שָׂחָה לְעֶבְרוֹ דָּג גָּדוֹל וְבָלַע אוֹתוֹ בְּבַת אַחַת. בְּתוֹךְ בֶּטֶן הַדָּג הָיָה חָשׁוּךְ אֲפִלּוּ יוֹתֵר מִבְּתוֹךְ הַתְּעָלָה, אֲבָל הַחַיָּלוֹן לֹא אִבֵּד אֶת קוֹר רוּחוֹ אֲפִלּוּ לְרֶגַע.

לְמַזָּלוֹ שֶׁל הַחַיָּלוֹן, אוֹתוֹ דָּג נִתְפַּס בִּידֵי דַּיָּג, נִמְכַּר בַּשּׁוּק, וְהִגִּיעַ אֶל מִטְבַּח בֵּיתוֹ שֶׁל אוּרִי - בְּדִיּוּק אוֹתוֹ בַּיִת שֶׁמִּמֶּנּוּ יָצָא לְמַסָּעוֹ! כְּשֶׁפָּתְחָה הַטַּבָּחִית אֶת בֶּטֶן הַדָּג לְהָכִין אוֹתוֹ לַאֲרוּחָה, נִדְהֲמָה לִמְצֹא בְּתוֹכוֹ חַיָּלוֹן בְּדִיל, שָׁלֵם וְנָקִי כְּמוֹ בַּיּוֹם שֶׁבּוֹ נוֹצַר. הִיא הֵרִימָה אוֹתוֹ בִּזְהִירוּת וְהֵשִׁיבָה אוֹתוֹ אֶל אוֹתוֹ הַשֻּׁלְחָן שֶׁמִּמֶּנּוּ נָפַל - וְשָׁם, מוּלוֹ מַמָּשׁ, שׁוּב עָמְדָה הָרַקְדָנִית הַקְּטַנָּה בְּאוֹתָהּ שִׂמְלָה אֲדַמְדַּמָּה, כְּאִלּוּ לֹא עָבַר יוֹם אֶחָד.

אֲבָל בְּאוֹתוֹ עֶרֶב, כְּשֶׁשִּׂחֲקוּ יַלְדֵי הַבַּיִת לְיַד הָאָח הַבּוֹעֶרֶת, נָשְׁבָה שׁוּב רוּחַ פִּתְאוֹמִית דֶּרֶךְ הַדֶּלֶת הַפְּתוּחָה. הָרוּחַ הֵזִיזָה אֶת הַחַיָּלוֹן מִקְּצֵה הַשֻּׁלְחָן, וְהוּא הֶחְלִיק וְנָפַל הַיְשֵׁר לְתוֹךְ הָאֵשׁ הַלּוֹהֶטֶת. רֶגַע אַחַר כָּךְ נָשְׁבָה הָרוּחַ שׁוּב, וְהֵזִיזָה גַּם אֶת הָרַקְדָנִית הַקַּלָּה מִשְּׂפַת הָאַרְמוֹן - וְהִיא צָנְחָה לְתוֹךְ הָאֵשׁ יָשָׁר אֶל צִדּוֹ שֶׁל הַחַיָּלוֹן. הַחַיָּלוֹן הִבִּיט בָּהּ בְּעֵינַיִם דּוֹלְקוֹת, מַרְגִּישׁ שֶׁגּוּפוֹ הֵחֵל לְהִמַּס לְאַט לְאַט בְּחֹם הָאֵשׁ - אֲבָל גַּם עַכְשָׁו לֹא הֵסִיר אֶת מַבָּטוֹ מִמֶּנָּה אֲפִלּוּ לְשֶׁנִיָּה.

לְמָחֳרָת בַּבֹּקֶר, כְּשֶׁפִּנְּתָה הַטַּבָּחִית אֶת אֵפֶר הָאָח, מָצְאָה בֵּין הַגֶּחָלִים הַקָּרוֹת חֲתִיכַת בְּדִיל קְטַנָּה שֶׁנָמְסָה וְהִתְקַשְּׁתָה מֵחָדָשׁ בְּצוּרָה מֻזָּרָה - בְּדִיּוּק בְּצוּרַת לֵב קָטָן. לְיָדָהּ מָצְאָה גַּם כּוֹכָבִית זְעִירָה שֶׁחָרְכָה קְצָת בָּאֵשׁ וְנֶהְפְּכָה לִשְׁחַרְחֹרֶת - כָּל מַה שֶּׁנּוֹתַר מִשִּׂמְלַת הָרַקְדָנִית.

אוּרִי מָצָא אֶת שְׁנֵי הַשְּׂרִידִים בֵּין הָאֵפֶר, וְהֵבִין מִיָּד שֶׁמַּשֶּׁהוּ מְיֻחָד קָרָה כָּאן. הוּא נִקָּה בִּזְהִירוּת אֶת לֵב הַבְּדִיל הַקָּטָן, שָׂם אוֹתוֹ בְּתוֹךְ קֻפְסָה עֲדִינָה יַחַד עִם הַכּוֹכָבִית הַשְּׂרוּפָה, וְשָׁמַר אֶת שְׁנֵיהֶם כְּמַזְכֶּרֶת יְקָרָה עַד יוֹם בָּגְרוּתוֹ. מִדֵּי פַּעַם הָיָה מוֹצִיא אֶת הַקֻּפְסָה וּמְסַפֵּר לְכָל מִי שֶׁשָּׁאַל עַל חַיָּלוֹן בְּדִיל קָטָן אֶחָד, שֶׁנִּשְׁאַר נֶאֱמָן וְאַמִּיץ עַד הָרֶגַע הָאַחֲרוֹן מַמָּשׁ.

מְסֻפָּר עַל־פִּי "הַחַיָּל הַבְּדִיל הָאַמִּיץ" (The Steadfast Tin Soldier) מֵאֵת הַנְס כְּרִיסְטִיאַן אַנְדֶּרְסֶן (דֶּנְמַרְק), 1838 - מְעֻבָּד וּמְרֻכָּךְ בְּסוֹפוֹ.

בְּשִׁיעוּר 45 לָמַדְנוּ שֶׁבְּעִבְרִית אֶפְשָׁר לְהַגִּיד שֶׁמַּשֶּׁהוּ קָטָן בִּשְׁתֵּי דְּרָכִים: (1) הוֹסָפַת סִיּוֹמֶת ־וֹן/־וֹנָה/־וֹנֶת בְּסוֹף הַמִּלָּה (יֶלֶד ← יַלְדּוֹן, חַיָּל ← חַיָּלוֹן), אוֹ (2) כֶּפֶל שְׁתֵּי הָאוֹתִיּוֹת הָאַחֲרוֹנוֹת בַּשֹּׁרֶשׁ (כֶּלֶב ← כְּלַבְלַב, חָתוּל ← חֲתַלְתּוּל). הַיּוֹם נְגַלֶּה שֶׁלַּתַּבְנִית הַשְּׁנִיָּה - כֶּפֶל הָאוֹתִיּוֹת - יֵשׁ עוֹד שִׁמּוּשׁ מַפְתִּיעַ, שֶׁלֹּא קָשׁוּר בִּכְלָל לְגֹדֶל!

הַמְצִיאוּ דְּמוּת שֶׁל צַעֲצוּעַ קָטָן מִשֶּׁלָּכֶם, עִם שֵׁם בְּמִלָּה מַקְטִינָה (כְּמוֹ "דֻּבּוֹן" אוֹ "חֲתַלְתּוּל"), וְתָאֲרוּ אֶת צֶבַע הַצַּעֲצוּעַ בְּעֶזְרַת גָּוֶן מֻחְלָשׁ (כְּמוֹ "יְרַקְרַק" אוֹ "כְּחַלְחַל"). כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים לְפִי הַהַנְחָיָה שֶׁלְּיַד כָּל שׁוּרָה - לְמָשָׁל: "קָרָאתִי לַצַּעֲצוּעַ שֶׁלִּי 'כְּלַבְלַב הַזָּהָב', וְהַפַּרְוָה שֶׁלּוֹ הָיְתָה צְהַבְהַבָּה וְרַכָּה."
