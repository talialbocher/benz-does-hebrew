📖 קְרִיאָה

בְּעֵמֶק יָרֹק אֶחָד רָעוּ שְׁלוֹשָׁה שׁוֹרִים: שׁוֹר לָבָן, שׁוֹר שָׁחֹר וְשׁוֹר חוּם. הֵם הָיוּ חֲבֵרִים טוֹבִים, וְתָמִיד אָכְלוּ, שָׁתוּ וְנָחוּ יַחַד.

בַּיַּעַר הַסָּמוּךְ גָּר אֲרִי רָעֵב שֶׁהִתְגַּנֵּב אֶל הָעֵמֶק כָּל בֹּקֶר. הוּא רָצָה לֶאֱכֹל שׁוֹר, אֲבָל פָּחַד לְהִתְקָרֵב. ״שְׁלוֹשָׁה שׁוֹרִים עוֹמְדִים יַחַד, וְהַקַּרְנַיִם הַחַדּוֹת שֶׁלָּהֶם מְסֻכָּנוֹת,״ חָשַׁב הָאֲרִי. ״אִם אֶתְקֹף אֶחָד מֵהֶם, הַשְּׁנַיִם הָאֲחֵרִים יָבוֹאוּ לַעֲזֹר לוֹ.״

הָאֲרִי הֶעָרוּם חָשַׁב וְחָשַׁב, עַד שֶׁהָיְתָה לוֹ תּוֹכְנִית. הוּא לָחַשׁ לַשּׁוֹר הַלָּבָן: ״שָׁמַעְתִּי שֶׁהַשּׁוֹר הַשָּׁחֹר מְדַבֵּר עָלֶיךָ דְּבָרִים רָעִים.״ אַחַר כָּךְ לָחַשׁ לַשּׁוֹר הַשָּׁחֹר מִלִּים דּוֹמוֹת עַל הַשּׁוֹר הַחוּם. הַשּׁוֹרִים שֶׁהָיוּ חֲבֵרִים טוֹבִים הִפְסִיקוּ לְדַבֵּר זֶה עִם זֶה, וְכָל אֶחָד עָמַד לְבַד, רָחוֹק מִן הַשְּׁנַיִם הָאֲחֵרִים.

בַּיּוֹם שֶׁבּוֹ הִתְקִיף הָאֲרִי, הַשּׁוֹר הַלָּבָן עָמַד לְבַד בְּקָצֵה הָעֵמֶק. הָאֲרִי קָפַץ עָלָיו, וְהַשּׁוֹר צָעַק: ״עֶזְרָה!״ הַשּׁוֹר הַשָּׁחֹר וְהַשּׁוֹר הַחוּם שָׁמְעוּ אֶת הַצְּעָקָה, אֲבָל לֹא זָזוּ, כִּי הֵם עֲדַיִן הָיוּ כּוֹעֲסִים.

רַק כְּשֶׁהַשּׁוֹר הַלָּבָן כִּמְעַט נָפַל, הִבְחִינוּ הַשְּׁנַיִם הָאֲחֵרִים בַּטָּעוּת הַנּוֹרָאָה שֶׁעָשׂוּ. הֵם רָצוּ אֵלָיו בְּכָל כֹּחָם, וּשְׁלוֹשְׁתָּם עָמְדוּ יַחַד, כָּתֵף אֶל כָּתֵף. הָאֲרִי רָאָה שְׁלוֹשָׁה שׁוֹרִים עוֹמְדִים יַחַד, וּבָרַח חֲזָרָה אֶל הַיַּעַר. מֵאָז, בְּכָל פַּעַם שֶׁמִּישֶׁהוּ לָחַשׁ לָהֶם דְּבָרִים רָעִים עַל חָבֵר, הֵם שָׁאֲלוּ אֶת הֶחָבֵר עַצְמוֹ.

אִיזוֹפּוֹס (יָוָן הָעַתִּיקָה), ״הָאֲרִי וּשְׁלוֹשֶׁת הַשּׁוֹרִים״, בְּעִבּוּד חָפְשִׁי.

• הַפְּסוּקִית בָּאָה מִיָּד אַחֲרֵי שֵׁם הָעֶצֶם שֶׁהִיא מְתָאֶרֶת: ״הָאֲרִי שֶׁגָּר בַּיַּעַר הִפְחִיד אֶת הַשּׁוֹר״ (וְלֹא ״הָאֲרִי הִפְחִיד אֶת הַשּׁוֹר בַּיַּעַר שֶׁגָּר״).

• לֹא כָּל ״שֶׁ־״ פּוֹתֵחַ פְּסוּקִית זִיקָה! אַחֲרֵי פְּעָלִים כְּמוֹ אָמַר, שָׁמַע, חָשַׁב ״שֶׁ־״ מַתְחִיל אֶת מַה שֶּׁנֶּאֱמַר אוֹ נִחְשַׁב, וְלֹא מְתָאֵר שֵׁם עֶצֶם: שָׁמַעְתִּי שֶׁהוּא כּוֹעֵס — זוֹ אֵינָהּ פְּסוּקִית זִיקָה.

🔑 הַכְּלָל: פְּסוּקִית זִיקָה פּוֹתַחַת בְּ״שֶׁ־״, בָּאָה מִיָּד אַחֲרֵי שֵׁם עֶצֶם וּמוֹסִיפָה עָלָיו מֵידָע. ״שֶׁ־״ יָכוֹל לְהַחֲלִיף נוֹשֵׂא אוֹ מוּשָׂא. מִבְחָן: אִם אֶפְשָׁר לִשְׁאֹל ״אֵיזֶה שֵׁם עֶצֶם?״ — זוֹ פְּסוּקִית זִיקָה.
