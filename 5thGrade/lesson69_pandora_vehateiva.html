📖 קְרִיאָה

לִפְנֵי אֲלָפֵי שָׁנִים, בִּימֵי יָוָן הָעַתִּיקָה, שָׁלְטוּ בָּעוֹלָם אֵלִים רַבִּים מֵרֹאשׁ הַר אוֹלִימְפּוֹס. פְּרוֹמֶתֶאוּס, אֵל פִּקֵּחַ שֶׁאָהַב אֶת בְּנֵי הָאָדָם, גָּנַב פַּעַם אֵשׁ מִן הַשָּׁמַיִם וְנָתַן אוֹתָהּ בְּמַתָּנָה לִבְנֵי הָאָדָם עַל הָאֲדָמָה. זֶאוּס, רֹאשׁ הָאֵלִים, כָּעַס מְאֹד עַל הַגְּנֵבָה, וְהֶחְלִיט לְהַעֲנִישׁ אֶת בְּנֵי הָאָדָם בְּדַרְכּוֹ הַמְּיֻחֶדֶת.

זֶאוּס פָּנָה אֶל הָאֵלִים הָאֻמָּנִים שֶׁבָּהָר וּבִקֵּשׁ מֵהֶם לִיצֹר אִשָּׁה רִאשׁוֹנָה, יָפָה וּמְיֻחֶדֶת, מֵעָפָר וּמַיִם. כָּל אֵל נָתַן לָהּ מַתָּנָה מִשֶּׁלּוֹ - יֹפִי, קֶסֶם בַּדִּבּוּר, סַקְרָנוּת בִּלְתִּי נִגְמֶרֶת. בִּזְכוּת הַמַּתָּנוֹת הָרַבּוֹת שֶׁקִּבְּלָה, קָרְאוּ לָהּ הָאֵלִים בַּשֵּׁם פַּנְדּוֹרָה - שֶׁפֵּרוּשׁוֹ, בְּיָוָנִית עַתִּיקָה, "כָּל הַמַּתָּנוֹת".

לִפְנֵי שֶׁשָּׁלַח אֶת פַּנְדּוֹרָה אֶל עוֹלָם בְּנֵי הָאָדָם, נָתַן לָהּ זֶאוּס תֵּבָה קְטַנָּה וּסְגוּרָה הֵיטֵב, וְאָמַר בְּקוֹל חָמוּר: "קְחִי אֶת הַתֵּבָה הַזֹּאת אִתָּךְ - אֲבָל אַל תִּפְתְּחִי אוֹתָהּ לְעוֹלָם, יְהִי מַה יִּהְיֶה!" פַּנְדּוֹרָה לֹא יָדְעָה מָה יֵשׁ בִּפְנִים, אַךְ הִבְטִיחָה בְּרָצוֹן: "אֲנִי לֹא אֶפְתַּח אוֹתָהּ, אֲנִי מַבְטִיחָה."

בָּאָרֶץ חַי אָז אָחִיו שֶׁל פְּרוֹמֶתֶאוּס, אֶפִּימֶתֶאוּס, שֶׁהָיָה טוֹב לֵב אַךְ קְצָת פָּזִיז. פְּרוֹמֶתֶאוּס הִזְהִיר אוֹתוֹ פַּעַם אַחַר פַּעַם: "אַל תְּקַבֵּל שׁוּם מַתָּנָה מִזֶּאוּס, אָחִי - הוּא עֲדַיִן כּוֹעֵס עָלֵינוּ עַל גְּנֵבַת הָאֵשׁ!" אֲבָל כְּשֶׁרָאָה אֶפִּימֶתֶאוּס אֶת פַּנְדּוֹרָה, הוּא הִתְאַהֵב בְּיָפְיָהּ בְּמַבָּט רִאשׁוֹן - וְשָׁכַח לְגַמְרֵי אֶת אַזְהָרַת אָחִיו. הוּא נָשָׂא אוֹתָהּ לְאִשָּׁה, וְהֵם חָיוּ יַחְדָּיו בְּבַיִת קָטָן וְנָעִים.

פַּנְדּוֹרָה הֵנִיחָה אֶת הַתֵּבָה עַל מַדָּף גָּבֹהַּ בְּפִנַּת הַבַּיִת, וְלֹא נָגְעָה בָּהּ יוֹם אוֹ יוֹמַיִם. אֲבָל בְּכָל פַּעַם שֶׁעָבְרָה לְיָדָהּ, הֶעֱיפָה מַבָּט קָצָר וְתָהֲתָה: מָה בֶּאֱמֶת יֵשׁ שָׁם בִּפְנִים? הַסַּקְרָנוּת שֶׁקִּבְּלָה בְּמַתָּנָה מֵהָאֵלִים הֵחֵלָּה לְכַרְסֵם בְּתוֹכָהּ, יוֹם אַחַר יוֹם.

אֶפִּימֶתֶאוּס, שֶׁרָאָה אוֹתָהּ מַבִּיטָה בַּתֵּבָה שׁוּב וָשׁוּב, אָמַר לָהּ בְּדַאֲגָה: "אַל תַּעֲשִׂי שׁוּם דָּבָר פָּזִיז, פַּנְדּוֹרָה. זֶאוּס בִּקֵּשׁ שֶׁלֹּא נִפְתַּח אוֹתָהּ - וַאֲנַחְנוּ לֹא נָבִין לָמָּה, אֲבָל צָרִיךְ לִשְׁמֹעַ." פַּנְדּוֹרָה חִיְּכָה וְעָנְתָה: "אַל תִּדְאַג, אֲנִי לֹא אֶפְתַּח אוֹתָהּ. הֵן הִבְטַחְתִּי." אֲבָל בְּלִבָּהּ פְּנִימָה, הַסַּקְרָנוּת רַק גָּדְלָה וְגָדְלָה.

לַיְלָה אֶחָד, כְּשֶׁאֶפִּימֶתֶאוּס יָשַׁן, לֹא יָכְלָה פַּנְדּוֹרָה לְהִתְאַפֵּק יוֹתֵר. הִיא קָרְבָה בְּשֶׁקֶט אֶל הַתֵּבָה, הֵרִימָה אֶת הַמִּכְסֶה רַק סֶנְטִימֶטֶר אֶחָד - "רַק לְהָצִיץ, בְּלִי לְהוֹצִיא כְּלוּם," חָשְׁבָה לְעַצְמָהּ. אֲבָל בְּאוֹתוֹ רֶגַע פָּרְצוּ מִתּוֹךְ הַתֵּבָה עֲשָׁנִים כֵּהִים, וּבְתוֹכָם כָּל הַצָּרוֹת שֶׁל הָעוֹלָם: מַחֲלָה, עֶצֶב, קִנְאָה, רָעָב וְרִיב. הֵם עָפוּ הַחוּצָה בִּמְהִירוּת וְהִתְפַּזְּרוּ בְּכָל כִּוּוּן.

פַּנְדּוֹרָה נִבְהֲלָה וְטָרְקָה אֶת הַמִּכְסֶה בְּחָזְקָה, אֲבָל כְּבָר הָיָה מְאֻחָר מִדַּי - כִּמְעַט כָּל מַה שֶּׁהָיָה בַּתֵּבָה כְּבָר בָּרַח הַחוּצָה. הִיא יָשְׁבָה עַל הָרִצְפָּה וּבָכְתָה: "מָה עָשִׂיתִי? אָמַרְתִּי לְעַצְמִי שֶׁלֹּא אֶפְתַּח אוֹתָהּ..." לְפֶתַע שָׁמְעָה קוֹל עָדִין בּוֹקֵעַ מִתּוֹךְ הַתֵּבָה: "אַל תִּבְכִּי, פַּנְדּוֹרָה. פִּתְחִי גַּם אוֹתִי - וְלֹא תִּתְחָרְטִי."

בְּיָד רוֹעֶדֶת הֵרִימָה פַּנְדּוֹרָה שׁוּב אֶת הַמִּכְסֶה, וְהַפַּעַם יָצְאָה מִן הַתֵּבָה יְצוּרָה זְעִירָה וּמְאִירָה בְּאוֹר רַךְ - זוֹ הָיְתָה הַתִּקְוָה. הַתִּקְוָה עָפָה הַחוּצָה אַחֲרֵי כָּל שְׁאָר הַדְּבָרִים הָרָעִים, וְטָסָה בְּעוֹלָם כֻּלּוֹ - אֶל כָּל אָדָם שֶׁפָּגַשׁ צָרָה אוֹ קֹשִׁי, כְּדֵי לְהַזְכִּיר לוֹ שֶׁתָּמִיד יֵשׁ סִבָּה לְהַאֲמִין שֶׁמָּחָר יִהְיֶה טוֹב יוֹתֵר.

כְּשֶׁחָזַר אֶפִּימֶתֶאוּס וְשָׁמַע מָה קָרָה, הוּא לֹא כָּעַס עַל פַּנְדּוֹרָה. הוּא אָחַז בְּיָדָהּ וְאָמַר: "אֶת הָרָעוֹת כְּבָר אִי אֶפְשָׁר לְהַחֲזִיר לַתֵּבָה - אֲבָל הַתִּקְוָה תָּמִיד תִּהְיֶה אִתָּנוּ, בְּכָל בַּיִת וּבְכָל לֵב. מֵעַתָּה לֹא נִפְחַד מִן הַקֹּשִׁי, כִּי נֵדַע שֶׁהוּא לֹא יִשָּׁאֵר לְעוֹלָם." וְכָךְ, מֵאָז וְעַד הַיּוֹם, כְּשֶׁאָדָם עוֹבֵר יְמֵי צָרָה, אוֹמְרִים שֶׁ"תִּקְוָתָהּ שֶׁל פַּנְדּוֹרָה" עֲדַיִן טָסָה בָּעוֹלָם, וּמַזְכִּירָה לוֹ: אַל תְּאַבֵּד אֶת הַתִּקְוָה.

מְסֻפָּר עַל־פִּי מִיתוֹס "תֵּבַת פַּנְדּוֹרָה" (Pandora's Box), מִן הַמִּיתוֹלוֹגְיָה הַיְּוָנִית הָעַתִּיקָה, מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ.

📚 דִּקְדּוּק — שְׁלִילַת צִוּוּי וְעָתִיד

אֵיךְ אוֹמְרִים "אַל תַּעֲשֶׂה!" בְּעִבְרִית?

כְּשֶׁרוֹצִים לְבַקֵּשׁ מִמִּישֶׁהוּ שֶׁלֹּא יַעֲשֶׂה מַשֶּׁהוּ עַכְשָׁו - פְּקֻדָּה אוֹ אַזְהָרָה יְשִׁירָה - מִשְׁתַּמְּשִׁים בְּמִלַּת הַשְּׁלִילָה "אַל", וְאַחֲרֶיהָ צוּרַת עָתִיד שֶׁל הַפֹּעַל (וְלֹא צוּרַת צִוּוּי!). זֶאוּס לֹא אָמַר "אַל פִּתְחִי" - הוּא אָמַר: "אַל תִּפְתְּחִי", כִּי "תִּפְתְּחִי" הִיא צוּרַת הֶעָתִיד שֶׁל "אַתְּ" בַּפֹּעַל "לִפְתֹּחַ".

🔑 כְּלָל זָהָב: שְׁלִילַת צִוּוּי תָּמִיד נִבְנֵית מִצּוּרַת הֶעָתִיד שֶׁל הַפֹּעַל, וּלְעוֹלָם לֹא מִצּוּרַת הַצִּוּוּי עַצְמָהּ. "אַל" מִצְטָרֵף אֶל צוּרַת הֶעָתִיד - בְּדִיּוּק כְּמוֹ שֶׁ"לְכִי" הוֹפֶכֶת לְ"אַל תֵּלְכִי", לֹא לְ"אַל לְכִי".

"אַל" לְעֻמַּת "לֹא" + עָתִיד

יֵשׁ הֶבְדֵּל חָשׁוּב בֵּין שְׁתֵּי צוּרוֹת הַשְּׁלִילָה: "אַל" מְשַׁמֵּשׁ בְּדֶרֶךְ כְּלָל בִּפְנִיָּה יְשִׁירָה לְגוּף שֵׁנִי (אַתָּה/אַתְּ/אַתֶּם/אַתֶּן), כְּפִקֻּדָּה אוֹ אַזְהָרָה דְּחוּפָה בָּרֶגַע הַזֶּה. "לֹא" + עָתִיד יָכוֹל לְהוֹפִיעַ עִם כָּל גוּף (אֲנִי, אַתָּה, הוּא, אֲנַחְנוּ, הֵם...), וּמְתָאֵר סְתָם הַצְהָרָה אוֹ הַחְלָטָה עַל מַשֶּׁהוּ שֶׁיִּקְרֶה אוֹ לֹא יִקְרֶה בֶּעָתִיד - לֹא בְּהֶכְרֵחַ פְּקֻדָּה.

כִּתְבוּ 5 מִשְׁפָּטִים מִשֶּׁלָּכֶם: בְּחֵלֶק מֵהֶם שַׁלְּבוּ "אַל" + פֹּעַל בֶּעָתִיד (פְּקֻדָּה אוֹ אַזְהָרָה עַכְשָׁוִית), וּבְחֵלֶק שַׁלְּבוּ "לֹא" + פֹּעַל בֶּעָתִיד (הַצְהָרָה אוֹ הַבְטָחָה כְּלָלִית). לְדֻגְמָה: "אַל תִּשְׁכַּח לְהָבִיא אֶת הַתִּיק מָחָר בַּבֹּקֶר" (פְּקֻדָּה עַכְשָׁוִית), אוֹ "אֲנִי לֹא אֶשְׁכַּח אֶת הַיּוֹם הַזֶּה לְעוֹלָם" (הַצְהָרָה עַל הֶעָתִיד).
