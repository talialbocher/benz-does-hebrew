הַקּוֹסֵם מֵאֶרֶץ עוֹץ

סוּפָה חֲזָקָה שׁוֹלַחַת אֶת דּוֹרוֹתִי וְאֶת כַּלְבָּהּ טוֹטוֹ אֶל אֶרֶץ קְסוּמָה, שָׁם הֵם פּוֹגְשִׁים חֲבֵרִים מְיֻחָדִים בַּדֶּרֶךְ אֶל הַקּוֹסֵם - וְשִׁיעוּר חָדָשׁ עַל לְשׁוֹן הַצִּוּוּי בְּכָל הַגּוּפִים

📖 קְרִיאָה

הָיְתָה הָיְתָה יַלְדָּה בְּשֵׁם דּוֹרוֹתִי, שֶׁגָּרָה בְּחַוָּה קְטַנָּה בְּמִישׁוֹרֵי קֶנְזָס עִם דּוֹדָהּ הֶנְרִי, דּוֹדָתָהּ אֶם וְכַלְבָּהּ הַנֶּאֱמָן טוֹטוֹ. יוֹם אֶחָד הִגִּיעָה אֶל הַחַוָּה סוּפָה עֲצוּמָה שֶׁהֵרִימָה אֶת כָּל הַבַּיִת הָעֵצִי בָּאֲוִיר, וְטָסָה אִתּוֹ - וְעִם דּוֹרוֹתִי וְטוֹטוֹ בִּפְנִים - רָחוֹק רָחוֹק מֵעַל הֶעָבִים.

הַבַּיִת נָחַת בְּבַת אַחַת בְּאֶרֶץ צִבְעוֹנִית וּמוּזָרָה בְּשֵׁם עוֹץ, וְנָחַת מַמָּשׁ עַל בֵּיתָהּ שֶׁל מְכַשֵּׁפָה רָעָה שֶׁמֵּתָה בּוֹ בָּרֶגַע. תּוֹשָׁבֵי הַמָּקוֹם, אֲנָשִׁים קְטַנִּים שֶׁנִּקְרְאוּ "בְּנֵי מַאנְצְ'קִין", יָצְאוּ לְקַדֵּם אֶת פְּנֵי דּוֹרוֹתִי בְּשִׂמְחָה רַבָּה, כִּי סוֹף־סוֹף נִגְאֲלוּ מִשִּׁלְטוֹנָהּ הָרַע. מְכַשֵּׁפָה טוֹבָה נָתְנָה לְדוֹרוֹתי אֶת נַעֲלֵי הַכֶּסֶף הַקְּסוּמוֹת שֶׁל הַמְּכַשֵּׁפָה הָרָעָה, וְאָמְרָה לָהּ: "לְכִי בַּדֶּרֶךְ הַמְּרֻצֶּפֶת בִּלְבֵנִים צְהֻבּוֹת, וְהַגִּיעִי אֶל עִיר הַזָּהָב הַגְּדוֹלָה - שָׁם גָּר קוֹסֵם רַב־כֹּחַ שֶׁיּוּכַל לַעֲזֹר לָךְ לַחְזֹר הַבַּיְתָה."

דּוֹרוֹתִי וְטוֹטוֹ יָצְאוּ לַדֶּרֶךְ הָאֲרֻכָּה. בַּדֶּרֶךְ פָּגְשׁוּ דַּחְלִיל תָּלוּי עַל מוֹט בְּתוֹךְ שְׂדֵה תִּירָס, שֶׁהִתְלוֹנֵן שֶׁאֵין לוֹ שֵׂכֶל וּמֹחַ אֲמִתִּי - רַק קַשׁ בְּתוֹךְ רֹאשׁוֹ. "בּוֹא אִתָּנוּ אֶל הַקּוֹסֵם," הִצִּיעָה דּוֹרוֹתִי בְּחִיּוּךְ, "אוּלַי הוּא יִתֵּן לְךָ שֵׂכֶל!" הַדַּחְלִיל שָׂמַח מְאֹד וְהִצְטָרֵף מִיָּד אֶל הַחֲבוּרָה.

בְּהֶמְשֵׁךְ הַדֶּרֶךְ מָצְאוּ אִישׁ פַּח קָפוּא בְּתוֹךְ הַיַּעַר, שֶׁכְּבָר זְמַן רַב לֹא זָז מִמְּקוֹמוֹ. "עִזְרוּ לִי, בְּבַקָּשָׁה!" קָרָא אִישׁ הַפַּח בְּקוֹל חָלוּשׁ. הַחֲבֵרִים עָזְרוּ לוֹ לָזוּז שׁוּב, וְהוּא סִפֵּר לָהֶם שֶׁכָּל מִשְׁאֲלוֹתוֹ הִיא לֵב אֲמִתִּי שֶׁיּוּכַל לֶאֱהֹב בּוֹ. "בּוֹאוּ אִתָּנוּ אֶל הַקּוֹסֵם!" קָרְאוּ דּוֹרוֹתִי וְהַדַּחְלִיל בְּיַחַד, וְאִישׁ הַפַּח הִצְטָרֵף בְּשִׂמְחָה.

מִיָּד אַחַר כָּךְ הוֹפִיעַ מִבֵּין הָעֵצִים אַרְיֵה עָנָק שֶׁשָּׁאַג בְּקוֹל רָם - אַךְ כְּשֶׁטוֹטוֹ נָבַח לְעֶבְרוֹ, בָּרַח הָאַרְיֵה בְּפַחַד. הִתְבָּרֵר שֶׁלַּמְרוֹת גָּדְלוֹ הוּא פַּחְדָּן גָּדוֹל, שֶׁמְּבַקֵּשׁ רַק דָּבָר אֶחָד: אֹמֶץ לֵב. "בּוֹא גַּם אַתָּה אִתָּנוּ," אָמְרוּ לוֹ הַחֲבֵרִים, "אוּלַי הַקּוֹסֵם יוּכַל לַעֲזֹר גַּם לְךָ." וְכָךְ יָצְאוּ יַחַד בַּדֶּרֶךְ הַצְּהֻבָּה - דּוֹרוֹתִי, טוֹטוֹ, הַדַּחְלִיל, אִישׁ הַפַּח וְהָאַרְיֵה.

כְּשֶׁהִגִּיעוּ סוֹף־סוֹף אֶל שַׁעֲרֵי עִיר הַזָּהָב, אָמַר לָהֶם הַשּׁוֹמֵר: "חַכּוּ כָּאן!" וְנֶעֱלַם לְרֶגַע. כְּשֶׁחָזַר, נָתַן לְכֻלָּם מִשְׁקָפַיִם יְרֻקִּים "כְּדֵי שֶׁלֹּא יְסֻנְוְרוּ מֵאוֹר הַזָּהָב", וְהוֹבִיל אוֹתָם אֶל אוּלַם הַכֵּס. שָׁם, מִתּוֹךְ עָשָׁן וְקוֹל רוֹעֵם, אָמַר הַקּוֹסֵם הַגָּדוֹל: "הָבִיאוּ לִי אֶת מַטְאֲטֵא הַמְּכַשֵּׁפָה הָרָעָה שֶׁל הַמַּעֲרָב - וְרַק אָז אֶעֱנֶה לְכָל בַּקָּשׁוֹתֵיכֶם!"

הַחֲבוּרָה יָצְאָה אֶל טִירַת הַמְּכַשֵּׁפָה הָרָעָה שֶׁל הַמַּעֲרָב. הַמְּכַשֵּׁפָה רָאֲתָה אוֹתָם מִתְקָרְבִים וְצָעֲקָה לִזְאֵבֶיהָ: "תִּפְשׂוּ אוֹתָם מִיָּד!" אַךְ אִישׁ הַפַּח וְהָאַרְיֵה הִתְגּוֹנְנוּ בְּאֹמֶץ, וְהִצִּילוּ אֶת חַבְרֵיהֶם. לְבַסּוֹף, כְּשֶׁהַמְּכַשֵּׁפָה נִסְּתָה לִתְפֹּס אֶת דּוֹרוֹתִי, שָׁפְכָה עָלֶיהָ דּוֹרוֹתִי בְּטָעוּת דְּלִי מַיִם - וְהַמְּכַשֵּׁפָה הַנּוֹרָאָה נָמַסָּה בְּמָקוֹם, טִפָּה אַחַר טִפָּה, עַד שֶׁלֹּא נִשְׁאַר מִמֶּנָּה דָּבָר.

בַּחֲזָרָה בָּעִיר, הֶעֱנִיק הַקּוֹסֵם לְכָל אֶחָד מֵהֶם אֶת מַה שֶּׁבִּקֵּשׁ: לַדַּחְלִיל - תְּעוּדָה מְפֹאֶרֶת שֶׁתּוֹכִיחַ שֶׁהוּא חָכָם; לְאִישׁ הַפַּח - שָׁעוֹן בְּצוּרַת לֵב, לְאוֹת שֶׁיֵּשׁ לוֹ עַכְשָׁו לֵב אוֹהֵב; וְלָאַרְיֵה - מַדַלְיָה נוֹצֶצֶת לְאֹמֶץ. רַק אָז הִתְגַּלָּה הַסּוֹד: הַ"קּוֹסֵם הַגָּדוֹל" הָיָה בֶּאֱמֶת אִישׁ קָטָן וְרָגִיל שֶׁהִסְתַּתֵּר מֵאֲחוֹרֵי וִילוֹן וְהִפְעִיל מְכוֹנוֹת רוֹעֲשׁוֹת.

הַקּוֹסֵם נִסָּה לָקַחַת אֶת דּוֹרוֹתִי הַבַּיְתָה בְּכַדּוּר פּוֹרֵחַ, אֲבָל הַכַּדּוּר הִמְרִיא בְּטָעוּת בְּלִי שֶׁהִסְפִּיקָה לַעֲלוֹת אֵלָיו. אָז הוֹפִיעָה מְכַשֵּׁפָה טוֹבָה נוֹסֶפֶת בְּשֵׁם גְּלִינְדָה, וְגִלְּתָה לְדוֹרוֹתי שֶׁלְּנַעֲלֵי הַכֶּסֶף שֶׁלָּהּ יֵשׁ כֹּחַ קֶסֶם מִשֶּׁלָּהֶן. "עִצְמִי אֶת עֵינַיִךְ," אָמְרָה גְּלִינְדָה בְּעֶדְנָה, "הַקִּישִׁי בַּעֲקֵבַיִךְ שָׁלֹשׁ פְּעָמִים, וְאִמְרִי: 'אֵין מָקוֹם כְּמוֹ הַבַּיִת!'"

דּוֹרוֹתִי עָשְׂתָה בְּדִיּוּק כְּפִי שֶׁנֶּאֱמַר לָהּ - וּכְשֶׁפָּקְחָה אֶת עֵינֶיהָ, מָצְאָה אֶת עַצְמָהּ שׁוּב בַּחַוָּה בְּקֶנְזָס, מְחַבֶּקֶת אֶת דּוֹדָתָהּ אֶם. "אֵין מָקוֹם כְּמוֹ הַבַּיִת," לָחֲשָׁה בְּאֹשֶׁר, וְיָדְעָה שֶׁלְּעוֹלָם לֹא תִּשְׁכַּח אֶת חֲבֵרֶיהָ הַנִּפְלָאִים מֵאֶרֶץ עוֹץ.

מְסֻפָּר עַל־פִּי "הַקּוֹסֵם מֵאֶרֶץ עוֹץ" (The Wonderful Wizard of Oz) מֵאֵת ל. פְרַנְק בָּאוּם (אַרְצוֹת הַבְּרִית, 1900) - יְצִירָה בְּנַחֲלַת הַכְּלָל, מְעֻבֶּדֶת וּמְסֻפֶּרֶת מֵחָדָשׁ.

📚 דִּקְדּוּק — לְשׁוֹן צִוּוּי, כָּל הַגּוּפִים

מָתַי מִשְׁתַּמְּשִׁים בִּלְשׁוֹן צִוּוּי?

לְשׁוֹן צִוּוּי מְשַׁמֶּשֶׁת רַק כְּשֶׁפּוֹנִים בְּמִשְׁפָּט יָשָׁר אֶל מִישֶׁהוּ וּמְבַקְּשִׁים מִמֶּנּוּ לַעֲשׂוֹת מַשֶּׁהוּ - כְּלוֹמַר, רַק בְּגוּף שֵׁנִי: אַתָּה, אַתְּ, אַתֶּם וְאַתֶּן. אֵין צוּרַת צִוּוּי לְ"אֲנִי", "הוּא", "הִיא", "אֲנַחְנוּ" אוֹ "הֵם" - לְגוּפִים אֵלֶּה פּוֹנִים תָּמִיד בִּלְשׁוֹן עָתִיד רָגִילָה.

אֵיךְ בּוֹנִים אֶת הַצִּוּוּי?

הַדֶּרֶךְ הַקַּלָּה בְּיוֹתֵר לִבְנוֹת צוּרַת צִוּוּי הִיא לָקַחַת אֶת צוּרַת הֶעָתִיד שֶׁל אוֹתוֹ גּוּף - וּלְהַשְׁמִיט (לְהוֹרִיד) אֶת אוֹת הַתָּי"ו שֶׁבִּתְחִלַּת הַמִּלָּה.

🔑 אַתָּה תֵּלֵךְ ← לֵךְ. פָּשׁוּט מוֹרִידִים אֶת הַ"תּ" מֵהַתְחָלַת צוּרַת הֶעָתִיד, וּמְקַבְּלִים אֶת צוּרַת הַצִּוּוּי.

שִׂימוּ לֵב: כְּמוֹ בְּתוֹאֲרֵי הַשֵּׁם וּבְפָעֳלֵי הֶעָבָר/הֹוֶה/עָתִיד שֶׁלָּמַדְנוּ, גַּם בַּצִּוּוּי יֵשׁ הֶבְדֵּל בֵּין זָכָר לְנֵקֵבָה בְּיָחִיד (לֵךְ מוּל לְכִי), אֲבָל בָּרַבִּים בְּדֶרֶךְ כְּלָל מִשְׁתַּמְּשִׁים בְּאוֹתָהּ צוּרָה גַּם לְזָכָר וְגַם לְנֵקֵבָה (לְכוּ).
לְכִי בַּדֶּרֶךְ! בּוֹאוּ אִתָּנוּ! עִזְרוּ לִי! חַכּוּ כָּאן! אִמְרִי אֶת זֶה!
