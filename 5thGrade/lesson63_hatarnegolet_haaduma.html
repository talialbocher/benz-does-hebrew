📖 קְרִיאָה

בַּחֲצַר שֶׁל מֶשֶׁק קָטָן גָּרָה תַּרְנְגֹלֶת אֲדֻמָּה קְטַנָּה. לְיָדָהּ גָּרִים שְׁלוֹשָׁה חֲבֵרִים: כֶּלֶב עָצֵל, חָתוּל עָצֵל וְאַוָּז עַצְלָנִית. שְׁלֹשְׁתָּם אוֹהֲבִים לָנוּחַ בַּצֵּל כָּל הַיּוֹם, לְשַׂחֵק וּלְנַמְנֵם - אֲבָל לֹא אוֹהֲבִים לַעֲבֹד בִּכְלָל.

יוֹם אֶחָד מוֹצֵאת הַתַּרְנְגֹלֶת הָאֲדֻמָּה כַּמָּה גַּרְעִינֵי חִטָּה מְפֻזָּרִים בֶּחָצֵר. הִיא חוֹשֶׁבֶת רֶגַע וְאוֹמֶרֶת בְּקוֹל: "אִם נִזְרַע אֶת הַגַּרְעִינִים הָאֵלֶּה, נוּכַל לְגַדֵּל חִטָּה וְלֶאֱפוֹת מִמֶּנָּה לֶחֶם טָעִים! מִי יַעֲזֹר לִי לִזְרֹעַ אֶת הַחִטָּה?"

"לֹא אֲנִי," אוֹמֵר הַכֶּלֶב וּמַמְשִׁיךְ לִישֹׁן. "לֹא אֲנִי," אוֹמֵר הֶחָתוּל וּמְלַטֵּף אֶת פַּרְוָתוֹ. "לֹא אֲנִי," אוֹמֶרֶת הָאַוָּז וְקוֹפֶצֶת לַבְּרֵכָה. "טוֹב מְאֹד," אוֹמֶרֶת הַתַּרְנְגֹלֶת הָאֲדֻמָּה, "אֲנִי אֶזְרַע אֶת הַחִטָּה בְּעַצְמִי!" וְכָךְ הִיא עוֹשָׂה.

עָבְרוּ שָׁבוּעוֹת, וְהַחִטָּה צָמְחָה גְּבוֹהָה וּזְהֻבָּה בַּשָּׂדֶה. הִגִּיעַ הַזְּמַן לִקְצֹר אוֹתָהּ. הַתַּרְנְגֹלֶת הָאֲדֻמָּה קוֹרֵאת שׁוּב לַחֲבֵרֶיהָ: "מִי יַעֲזֹר לִי לִקְצֹר אֶת הַחִטָּה?" וְשׁוּב שׁוֹמַעַת שְׁלוֹשָׁה "לֹא אֲנִי" - וְשׁוּב אוֹמֶרֶת: "אָז אֲנִי אֶקְצֹר אוֹתָהּ בְּעַצְמִי!"

אַחֲרֵי הַקָּצִיר, יֵשׁ לִטְחֹן אֶת גַּרְעִינֵי הַחִטָּה בְּטַחֲנָה וְלַהֲפֹךְ אוֹתָם לְקֶמַח. הַתַּרְנְגֹלֶת שׁוֹאֶלֶת: "מִי יַעֲזֹר לִי לָשֵׂאת אֶת הַשַּׂקִּים אֶל הַטַּחֲנָה וְלִטְחֹן אֶת הַחִטָּה?" שׁוּב שְׁלֹשְׁתָּם עוֹנִים "לֹא אֲנִי" - וְשׁוּב הַתַּרְנְגֹלֶת אוֹמֶרֶת: "אָז אֲנִי אֶטְחַן אוֹתָהּ בְּעַצְמִי!"

עִם הַקֶּמַח הַטָּרִי, הַתַּרְנְגֹלֶת רוֹצָה לֶאֱפוֹת לֶחֶם. הִיא שׁוֹאֶלֶת בְּפַעַם הָרְבִיעִית: "מִי יַעֲזֹר לִי לָלוּשׁ אֶת הַבָּצֵק וְלֶאֱפוֹת אֶת הַלֶּחֶם?" וְשׁוּב - "לֹא אֲנִי", "לֹא אֲנִי", "לֹא אֲנִי". "בְּסֵדֶר גָּמוּר," עוֹנָה הַתַּרְנְגֹלֶת בְּלִי לְהִתְרַגֵּז, "אֲנִי אֹפֶה אֶת הַלֶּחֶם בְּעַצְמִי!"

רֵיחַ נִפְלָא שֶׁל לֶחֶם טָרִי מִתְפַּשֵּׁט בְּכָל הֶחָצֵר. הַכֶּלֶב מֵרִים אֶת רֹאשׁוֹ, הֶחָתוּל פּוֹקֵחַ אֶת עֵינָיו, וְהָאַוָּז מַפְסִיקָה לִשְׂחוֹת. שְׁלֹשְׁתָּם רָצִים בִּמְהִירוּת אֶל הַתַּרְנְגֹלֶת, וּבִטְנָם מְקַרְקֶרֶת בְּרַעַב.

הַתַּרְנְגֹלֶת מוֹצִיאָה מֵהַתַּנּוּר כִּכַּר לֶחֶם חַמָּה וּזְהֻבָּה, וְשׁוֹאֶלֶת: "מִי יַעֲזֹר לִי לֶאֱכֹל אֶת הַלֶּחֶם?" הַפַּעַם, כֻּלָּם עוֹנִים בְּיַחַד וּבְהִתְלַהֲבוּת: "אֲנִי!" קוֹרֵא הַכֶּלֶב. "אֲנִי!" קוֹרֵא הֶחָתוּל. "אֲנִי!" קוֹרֵאת הָאַוָּז.

אֲבָל הַתַּרְנְגֹלֶת הָאֲדֻמָּה מְנַעֶרֶת בְּרֹאשָׁהּ. "לֹא," הִיא אוֹמֶרֶת בְּשֶׁקֶט. "אַתֶּם לֹא עֲזַרְתֶּם לִי לִזְרֹעַ, לֹא עֲזַרְתֶּם לִקְצֹר, לֹא עֲזַרְתֶּם לִטְחֹן וְלֹא עֲזַרְתֶּם לֶאֱפוֹת. לָכֵן - אֲנִי וְאֶפְרוֹחַי נֹאכַל אֶת הַלֶּחֶם בְּעַצְמֵנוּ." הִיא קוֹרֵאת לְאֶפְרוֹחֶיהָ הַקְּטַנִּים, וְיַחַד הֵם נֶהֱנִים מֵאֲרוּחָה חַמָּה וְטוֹבָה.

מֵאוֹתוֹ יוֹם וָהָלְאָה, לָמְדוּ הַכֶּלֶב, הֶחָתוּל וְהָאַוָּז שֶׁמִּי שֶׁלֹּא עוֹזֵר בָּעֲבוֹדָה - לֹא יִזְכֶּה גַּם בַּתּוֹצָאָה. הַסִּפּוּר מְלַמֵּד אוֹתָנוּ שֶׁמִּי שֶׁיַּעֲבֹד קָשֶׁה, הוּא זֶה שֶׁיֵּהָנֶה מִפְּרִי עֲבוֹדָתוֹ.

מְסֻפָּר עַל־פִּי "הַתַּרְנְגֹלֶת הָאֲדֻמָּה הַקְּטַנָּה" (The Little Red Hen) - סִפּוּר עַם אַנְגְּלִי-אֲמֶרִיקָנִי מָסוֹרְתִּי, שֶׁהוֹפִיעַ לָרִאשׁוֹנָה בִּדְפוּס בִּשְׁנַת 1874 - מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ.

📚 דִּקְדּוּק — הַטָּיַת הַפֹּעַל בֶּעָתִיד

חֲזָרָה קְצָרָה: מָה כְּבָר לָמַדְנוּ עַל הַפֹּעַל?

בְּשִׁיעוּר 59 לָמַדְנוּ לְהַבְחִין בֵּין שְׁלֹשָׁה זְמַנֵּי פֹּעַל: עָבָר, הֹוֶה וְעָתִיד. בְּשִׁיעוּר 60 לָמַדְנוּ שֶׁבֶּעָבָר יֵשׁ תֵּשַׁע צוּרוֹת - אַחַת לְכָל גּוּף. בְּשִׁיעוּר 62 גִּלִּינוּ שֶׁבַּהֹוֶה יֵשׁ רַק אַרְבַּע צוּרוֹת, לְפִי מִין וּמִסְפָּר בִּלְבַד. הַיּוֹם נַגִּיעַ לַזְּמַן הָאַחֲרוֹן - עָתִיד - וְנִלְמַד שֶׁגַּם בּוֹ, כְּמוֹ בֶּעָבָר, יֵשׁ צוּרָה שׁוֹנָה לְכָל גּוּף.

עָתִיד: אַרְבַּע אוֹתִיּוֹת-מַפְתֵּחַ בְּתִחְלַת הַפֹּעַל

בִּזְמַן עָתִיד, כָּל צוּרַת פֹּעַל מַתְחִילָה בְּאַחַת מֵאַרְבַּע אוֹתִיּוֹת: א (אֲנִי), ת (אַתָּה / אַתְּ / הִיא / אַתֶּם / אַתֶּן), י (הוּא / הֵם / הֵן), אוֹ נ (אֲנַחְנוּ). הִנֵּה הַטַּבְלָה הַמְּלֵאָה שֶׁל הַפֹּעַל "כָּתַב" (בִּנְיָן קַל) בֶּעָתִיד:

⭐ עֻבְדָּה מְעַנְיֶנֶת: "אַתָּה תִּכְתֹּב" וְ"הִיא תִּכְתֹּב" - בִּזְמַן עָתִיד, זוֹ בְּדִיּוּק אוֹתָהּ צוּרָה! רַק הַמִּלָּה שֶׁלִּפְנֵי הַפֹּעַל (אַתָּה / הִיא) מְגַלָּה לָנוּ מִי בֶּאֱמֶת מְדַבֵּר אוֹ מְבַצֵּעַ אֶת הַפְּעֻלָּה.

⚠️ טָעוּת נְפוֹצָה: "אֲנִי כָּתַב" - טָעוּת! בֶּעָתִיד חַיֶּבֶת לְהוֹפִיעַ אוֹת הַמַּפְתֵּחַ בְּתִחְלַת הַמִּלָּה - "אֲנִי אֶכְתֹּב". אַל תִּשְׁכְּחוּ אֶת הָאוֹת אֲפִלּוּ שֶׁהַשֹּׁרֶשׁ (כ-ת-ב) נִשְׁאָר אוֹתוֹ שֹׁרֶשׁ.
אֲנִי אֶקְצֹר אַתָּה תִּטְחַן הִיא תֹּאפֶה אֲנַחְנוּ נַעֲזֹר הֵם יַעַזְרוּ
