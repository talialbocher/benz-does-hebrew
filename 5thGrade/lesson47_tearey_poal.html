📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת חַי בִּכְפָר קָטָן אִכָּר עָנִי עִם אִשְׁתּוֹ. לָאִכָּר לֹא הָיוּ שָׂדוֹת גְּדוֹלִים וְלֹא הַרְבֵּה כֶּסֶף, אֲבָל הָיְתָה לוֹ חֶלְקַת אֲדָמָה קְטַנָּה וְלוּל צָנוּעַ עִם כַּמָּה תַּרְנְגֹלוֹת וְאַוָּזָה אַחַת לְבָנָה. כָּל בֹּקֶר קָם הָאִכָּר מֻקְדָּם וְעָבַד בְּשַׁקְדָנוּת בַּשָּׂדֶה, וְאִשְׁתּוֹ טִפְּלָה בַּעֲדֶר הָעוֹפוֹת בִּמְסִירוּת.

יוֹם אֶחָד, כְּשֶׁנִּכְנְסָה הָאִשָּׁה אֶל הַלּוּל לֶאֱסֹף בֵּיצִים, הִיא קָפְאָה בִּמְקוֹמָהּ. בְּתוֹךְ הַקֵּן שֶׁל הָאַוָּזָה נִצְנְצָה בֵּיצָה מוּזָרָה - כְּבֵדָה, קָשָׁה וּצְהֻבָּה כְּמוֹ שֶׁמֶשׁ. הִיא קָרְאָה לְבַעֲלָהּ בְּקוֹל נִרְגָּשׁ, וּשְׁנֵיהֶם בָּדְקוּ אֶת הַבֵּיצָה בִּזְהִירוּת רַבָּה. זֶה הָיָה זָהָב טָהוֹר!

מֵאוֹתוֹ יוֹם, הֵטִילָה הָאַוָּזָה בְּכָל בֹּקֶר בֵּיצַת זָהָב אַחַת - לֹא יוֹתֵר וְלֹא פָּחוֹת. הָאִכָּר וְאִשְׁתּוֹ לָקְחוּ כָּל בֵּיצָה בְּשֶׁקֶט אֶל הָעִיר וּמָכְרוּ אוֹתָהּ בְּמְחִיר גָּבוֹהַּ. לְאַט לְאַט הֵם נַעֲשׂוּ עֲשִׁירִים יוֹתֵר וְיוֹתֵר, קָנוּ בְּגָדִים חֲדָשִׁים, שֻׁלְחָן נָאֶה וְאַף פָּרָה שְׁמֵנָה.

אֲבָל הָאִכָּר, שֶׁהָיָה רָגִיל לְכֶסֶף מוּעָט, הֵחֵל לְהִתְקַנֵּא בְּעֹשֶׁר שֶׁמִּתְקַדֵּם כָּל כָּךְ לְאַט. "בֵּיצָה אַחַת בַּיּוֹם - זֶה מְעַט מִדַּי!" הוּא הִתְלוֹנֵן בְּכַעַס לְאִשְׁתּוֹ. "וַדַּאי יֵשׁ בְּתוֹךְ הָאַוָּזָה עוֹד וְעוֹד בֵּיצֵי זָהָב! אִם רַק אֶפְתַּח אוֹתָהּ בְּבַת אַחַת, אֶהְיֶה הֶעָשִׁיר בְּיוֹתֵר בַּכְּפָר כֻּלּוֹ!"

אִשְׁתּוֹ נִסְּתָה לְהַרְגִּיעַ אוֹתוֹ בְּסַבְלָנוּת: "אַל תְּמַהֵר," הִיא לָחֲשָׁה, "הָאַוָּזָה נוֹתֶנֶת לָנוּ מַתָּנָה בְּכָל בֹּקֶר. עָדִיף לְחַכּוֹת בְּסַבְלָנוּת וְלֹא לְקַלְקֵל אֶת הַמַּזָּל שֶׁלָּנוּ." אֲבָל הָאִכָּר כְּבָר לֹא הִקְשִׁיב. הַחֶמְדָּנוּת גָּבְרָה עָלָיו.

לְמָחֳרָת בַּבֹּקֶר, בְּעוֹד הָאַוָּזָה מְנַמְנֶמֶת בְּשֶׁקֶט בַּקֵּן, תָּפַס הָאִכָּר סַכִּין חַדָּה. בִּפְזִיזוּת גְּדוֹלָה, בְּלִי לַחְשֹׁב פַּעֲמַיִם, הוּא חָתַךְ אֶת בֶּטֶן הָאַוָּזָה בְּבַת אַחַת.

אֲבָל בְּתוֹךְ הַבֶּטֶן לֹא הָיָה שׁוּם דָּבָר מְיֻחָד - רַק בָּשָׂר וְאֵבָרִים רְגִילִים, בְּדִיּוּק כְּמוֹ בְּכָל אַוָּזָה אַחֶרֶת בָּעוֹלָם. לֹא הָיְתָה שָׁם אֲפִלּוּ בֵּיצַת זָהָב אַחַת קְטַנָּה.

הָאִכָּר יָשַׁב עַל הָאֲדָמָה וּבָכָה בְּמָרָה. הָאַוָּזָה שֶׁהֵטִילָה לוֹ בֵּיצַת זָהָב בְּכָל בֹּקֶר - מֵתָה. עַכְשָׁו לֹא יִהְיֶה עוֹד זָהָב, לֹא מָחָר וְלֹא לְעוֹלָם. מִתּוֹךְ רְצוֹנוֹ לְהִתְעַשֵּׁר מַהֵר, אִבֵּד הָאִכָּר אֶת כָּל מָה שֶׁהָיָה יָכוֹל לְקַבֵּל לְאַט וּבְבִטָּחוֹן.

הָאִכָּר וְאִשְׁתּוֹ חָזְרוּ לִחְיוֹת בְּעֹנִי, בְּדִיּוּק כְּמוֹ לִפְנֵי שֶׁמָּצְאוּ אֶת הָאוֹצָר. מֵאָז וְעַד הַיּוֹם מְסַפְּרִים אֶת הַסִּפּוּר הַזֶּה לִילָדִים, כְּדֵי לְלַמֵּד שֶׁמִּי שֶׁרוֹצֶה הַכֹּל בְּבַת אַחַת, בְּלִי סַבְלָנוּת - לְעִתִּים קְרוֹבוֹת מְאַבֵּד גַּם אֶת מָה שֶׁכְּבָר הָיָה לוֹ.

מְסֻפָּר עַל־פִּי מָשָׁל מֵאֵת אֵיזוֹפּוֹס (יָוָן הָעַתִּיקָה), "הָאַוָּזָה שֶׁהֵטִילָה בֵּיצֵי זָהָב" - מְעֻבָּד
