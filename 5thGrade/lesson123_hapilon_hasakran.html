📖 קְרִיאָה

בְּאַפְרִיקָה, לְיַד נָהָר רָחָב, גָּר פִּילוֹן קָטָן. הָיָה לוֹ חֹטֶם קָצָר וּמְעֻגָּל כְּמוֹ מַגָּף, וּבְלִבּוֹ הָיְתָה סַקְרָנוּת גְּדוֹלָה. הוּא שָׁאַל שְׁאֵלוֹת בְּלִי סוֹף: ״לָמָּה לַזֶּבְּרָה יֵשׁ פַּסִּים?״ ״מַדּוּעַ הַנְּשָׁרִים עָפִים גָּבוֹהַּ?״ ״מָה אוֹכֵל הַתַּנִּין לַאֲרוּחַת עֶרֶב?״

הַדּוֹדִים הַגְּדוֹלִים שֶׁלּוֹ נִרְגְּזוּ מִן הַשְּׁאֵלוֹת וְאָמְרוּ: ״שְׁתֹק כְּבָר!״ אֲבָל הַפִּילוֹן לֹא הִפְסִיק לִשְׁאֹל. כְּשֶׁשָּׁאַל עַל הַתַּנִּין, כֻּלָּם הִשְׁתַּתְּקוּ בְּפַחַד. אָז לָחֲשָׁה לוֹ צִפּוֹר חֲכָמָה: ״לֵךְ אֶל גְּדַת הַנָּהָר וּשְׁאַל אֶת הַתַּנִּין בְּעַצְמְךָ.״

הַפִּילוֹן אָזַר אֹמֶץ וְיָצָא לַדֶּרֶךְ. עַל סֶלַע גָּדוֹל שָׁכַב פִּיתוֹן אָרֹךְ. ״הֲרָאִיתָ אֶת הַתַּנִּין?״ שָׁאַל הַפִּילוֹן. ״אֲנִי מַכִּיר אוֹתוֹ הֵיטֵב״, אָמַר הַפִּיתוֹן. ״הִזָּהֵר, יֶלֶד! הוּא חָכָם, וְיֵשׁ לוֹ שִׁנַּיִם חַדּוֹת.״

בַּנָּהָר שָׂחָה תַּנִּין אָפֹר וְחִיֵּךְ חִיּוּךְ רָחָב. ״בּוֹא קָרוֹב, יֶלֶד״, אָמַר. ״הַיּוֹם אֶאֱכַל פִּילוֹן לַאֲרוּחַת עֶרֶב.״ הַפִּילוֹן הִתְקָרֵב, וְהַתַּנִּין תָּפַס אֶת הַחֹטֶם שֶׁלּוֹ בַּשִּׁנַּיִם וְהִתְחִיל לִמְשֹׁךְ. ״אוֹי!״ צָעַק הַפִּילוֹן, וְהַכְּאֵב הָיָה חָזָק מְאֹד.

הַפִּיתוֹן שָׁמַע אֶת הַצְּעָקָה וְהִגִּיעַ בִּמְהִירוּת. הוּא כָּרַךְ אֶת עַצְמוֹ סְבִיב הַפִּילוֹן וְהִתְחִיל לִמְשֹׁךְ בַּכִּוּוּן הַשֵּׁנִי. הַתַּנִּין מָשַׁךְ, הַפִּיתוֹן מָשַׁךְ, וְהַחֹטֶם הִתְמַתֵּחַ וְהִתְמַתֵּחַ, עַד שֶׁהַתַּנִּין הִרְפָּה אוֹתוֹ וְנָפַל לַמַּיִם בִּפְלוּץ. עַכְשָׁו הָיָה הַחֹטֶם אָרֹךְ וְגָּמִישׁ, כְּמוֹ חֵדֶק!

הַפִּילוֹן יָשַׁב עַל הַגָּדָה וְקֵרֵר אֶת הַחֵדֶק הַכּוֹאֵב בַּמַּיִם. בְּסַבְלָנוּת הוּא נִסָּה לְהָזִיז אוֹתוֹ, וְגִלָּה שֶׁהוּא יָכוֹל לִקְטֹף בּוֹ פֵּרוֹת וּלְהַתִּיז מַיִם עַל הַגַּב. כְּשֶׁחָזַר הַבַּיְתָה, הַדּוֹדִים הִתְפַּלְאוּ, וְגַם הֵם הִסְתַּקְרְנוּ. מֵאָז יֵשׁ לְכָל הַפִּילִים חֵדֶק, וְהֵם שׁוֹאֲלִים בְּלִי פַּחַד, כִּי הַחָכְמָה מַתְחִילָה בְּסַקְרָנוּת.

רֶדְיַארְד קִיפְּלִינְג (אַנְגְּלִיָּה, 1902), ״הַפִּילוֹן הַסַּקְרָן״ (מִתּוֹךְ הַקֶּבֶץ Just So Stories), בְּעִבּוּד חָפְשִׁי עִם סוֹף מְרֻכָּךְ.

אֶפְשָׁר לָגַעַת בּוֹ אוֹ לֹא?

בְּשִׁיעוּר 88 לָמַדְנוּ שֶׁיֵּשׁ שְׁנֵי סוּגִים שֶׁל שֵׁמוֹת עֶצֶם. שֵׁם עֶצֶם מוּחָשִׁי הוּא דָּבָר שֶׁאֶפְשָׁר לִרְאוֹת, לָגַעַת, לִשְׁמֹעַ, לְהָרִיחַ אוֹ לִטְעֹם: חֹטֶם, נָהָר, תַּנִּין, סֶלַע, קוֹל. שֵׁם עֶצֶם מֻפְשָׁט הוּא רֶגֶשׁ, תְּכוּנָה אוֹ רַעֲיוֹן, וְאֶפְשָׁר רַק לְהַרְגִּישׁ אוֹ לַחְשֹׁב עָלָיו: סַקְרָנוּת, אֹמֶץ, פַּחַד, כְּאֵב, חָכְמָה, סַבְלָנוּת.

אֵיךְ יוֹצְרִים שֵׁם עֶצֶם מֻפְשָׁט?

אֶפְשָׁר לִבְנוֹת שֵׁם עֶצֶם מֻפְשָׁט מִשֵּׁם תֹּאַר: סַקְרָן ← סַקְרָנוּת סַבְלָן ← סַבְלָנוּת חָכָם ← חָכְמָה אַמִּיץ ← אֹמֶץ שָׂמֵחַ ← שִׂמְחָה עָצוּב ← עֶצֶב. הַסִּיּוֹמֶת הַנְּפוֹצָה הִיא ־וּת, אֲבָל לִפְעָמִים הַמִּלָּה מִשְׁתַּנָּה יוֹתֵר.

יוֹצְאִים מִן הַכְּלָל

• לֹא כָּל מִלָּה בְּ־וּת הִיא מֻפְשֶׁטֶת: חֲנוּת הִיא מָקוֹם שֶׁאֶפְשָׁר לְהִכָּנֵס אֵלָיו, וְלָכֵן הִיא מוּחָשִׁית.

• קוֹל וְרֵיחַ הֵם מוּחָשִׁיִּים, כִּי אֶפְשָׁר לִשְׁמֹעַ אוֹתָם וּלְהָרִיחַ אוֹתָם, אַף עַל פִּי שֶׁאֵין לָגַעַת בָּהֶם.
