📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת, בִּכְפַר דַּיָּגִים קָטָן עַל חוֹף הַיָּם בְּיַפָּן, חַי דַּיָּג צָעִיר וְטוֹב לֵב בְּשֵׁם אוּרָשִׁימָה טָארוֹ. כָּל בֹּקֶר יָצָא אוּרָשִׁימָה בְּסִירָתוֹ אֶל הַיָּם הַכָּחֹל, וְדָג בְּסַבְלָנוּת עַד שֶׁהָיָה בּוֹ דַּי דָּגִים לְפַרְנֵס אֶת אִמּוֹ הַזְּקֵנָה.

יוֹם אֶחָד, בְּדַרְכּוֹ הַבַּיְתָה, רָאָה אוּרָשִׁימָה קְבוּצַת יְלָדִים עַל הַחוֹף מְצִיקִים לְצָב קָטָן - זוֹרְקִים בּוֹ אֲבָנִים וְצוֹחֲקִים. הַדַּיָּג הַצָּעִיר לֹא יָכוֹל הָיָה לְהַמְשִׁיךְ בְּדַרְכּוֹ בְּלִי לַעֲזֹר. הוּא בִּקֵּשׁ מֵהַיְּלָדִים לְהַנִּיחַ לַצָּב, קָנָה אוֹתוֹ מֵהֶם בְּמַטְבְּעוֹת אֲחָדִים, וְהֵשִׁיב אוֹתוֹ בִּזְהִירוּת אֶל הַמַּיִם.

כַּמָּה יָמִים אַחַר כָּךְ, כְּשֶׁדָּג אוּרָשִׁימָה כְּרָגִיל מֵעַל סִירָתוֹ, צָף לְיָדוֹ צָב עֲנָק וּפָנָה אֵלָיו בְּקוֹל אָדִיב: "מֶלֶךְ הַדְּרָקוֹנִים שֶׁל הַיָּם שׁוֹלֵחַ לְךָ תּוֹדָה עַל שֶׁהִצַּלְתָּ אֶת נִכְדָּתוֹ, וּמְבַקֵּשׁ לְהַזְמִינְךָ אֶל אַרְמוֹנוֹ שֶׁמִּתַּחַת לַיָּם." אוּרָשִׁימָה, סַקְרָן וְנִרְגָּשׁ, טִפֵּס עַל גַּב הַצָּב, וְיַחַד צָלְלוּ עָמֹק־עָמֹק אֶל תּוֹךְ הַמַּיִם הַכְּחֻלִּים.

עָמֹק־עָמֹק מִתַּחַת לַגַּלִּים הִתְגַּלָּה אַרְמוֹן מַפְלִיא, בָּנוּי אַלְמֻגִּים וּפְנִינִים נוֹצְצוֹת. שָׁם קִדְּמָה אֶת פְּנֵי אוּרָשִׁימָה נְסִיכָה יָפָה בְּשֵׁם אוֹטוֹהִימֶה, וְעָרְכָה לִכְבוֹדוֹ מְסִבָּה גְּדוֹלָה: דָּגִים צִבְעוֹנִיִּים רָקְדוּ בֵּין הָאַלְמֻגִּים, וְשֻׁלְחָנוֹת עֲמוּסִים מַעֲדַנֵּי יָם. הַזְּמַן שָׁם עָבַר בְּלִי הַרְגֵּשׁ - נִדְמָה הָיָה שֶׁחָלְפוּ רַק שְׁלוֹשָׁה יָמִים נְעִימִים.

אֲבָל אוּרָשִׁימָה הִתְגַּעְגֵּעַ לְאִמּוֹ, וְלִפְנֵי שֶׁהֶחְלִיט לַחֲזֹר הַבַּיְתָה, יָשַׁב וְכָתַב לָהּ מִכְתָּב:

אַרְמוֹן הַדְּרָקוֹן שֶׁבְּתוֹךְ הַיָּם, יוֹם שְׁלִישִׁי אִמָּא יְקָרָה, אֲנִי כּוֹתֵב לָךְ כְּדֵי לְסַפֵּר עַל הַהַרְפַּתְקָה הַמֻּפְלָאָה שֶׁעָבַרְתִּי. קֹדֶם כֹּל הִצַּלְתִּי צָב קָטָן שֶׁיְּלָדִים הִצִּיקוּ לוֹ עַל הַחוֹף. אַחַר כָּךְ שָׁב הַצָּב הַגָּדוֹל וְלָקַח אוֹתִי עַל גַּבּוֹ אֶל אַרְמוֹן פֶּלֶא מִתַּחַת לַמַּיִם. בֵּינְתַיִם אֲנִי מִתְאָרֵחַ כָּאן אֵצֶל נְסִיכָה טוֹבַת לֵב, אֲבָל לְבַסּוֹף אֲנִי מִתְגַּעְגֵּעַ אֵלַיִךְ וְרוֹצֶה לַחֲזֹר הַבַּיְתָה בְּקָרוֹב. בְּאַהֲבָה רַבָּה, בִּנֵךְ, אוּרָשִׁימָה

כְּשֶׁשָּׁמְעָה אוֹטוֹהִימֶה שֶׁאוּרָשִׁימָה רוֹצֶה לָשׁוּב אֶל הַיַּבָּשָׁה, נֶעֱצְבָה, אַךְ לֹא עָצְרָה אוֹתוֹ. הִיא נָתְנָה לוֹ תֵּבָה קְטַנָּה וּמְקֻשֶּׁטֶת, וְהִזְהִירָה אוֹתוֹ בְּקוֹל רְצִינִי: "קַח אֶת הַתֵּבָה הַזֹּאת אִתְּךָ, אַךְ לְעוֹלָם אַל תִּפְתַּח אוֹתָהּ - וְאִם תִּפְתַּח, לֹא תּוּכַל לָשׁוּב לְכָאן לְעוֹלָם."

הַצָּב הֵשִׁיב אֶת אוּרָשִׁימָה אֶל הַחוֹף שֶׁמִּמֶּנּוּ יָצָא. אַךְ מַשֶּׁהוּ הָיָה מוּזָר: הַבַּיִת שֶׁלּוֹ נֶעֱלַם, וּבִמְקוֹמוֹ עָמְדוּ בָּתִּים זָרִים שֶׁמֵּעוֹלָם לֹא רָאָה. הוּא שָׁאַל זָקֵן עוֹבֵר אֹרַח עַל מִשְׁפַּחְתּוֹ, וְהַזָּקֵן עָנָה בְּתִמָּהוֹן: "אוּרָשִׁימָה טָארוֹ? זֶה שֵׁם שֶׁשָּׁמַעְתִּי רַק בְּסִפּוּרִים יְשָׁנִים שֶׁל סַבָּא שֶׁלִּי - הוּא נֶעֱלַם בַּיָּם לִפְנֵי מֵאוֹת שָׁנִים!"

אוּרָשִׁימָה עָמַד בְּהֶלֶם וּמְבֻלְבָּל. מַה שֶּׁהִרְגִּישׁ כִּשְׁלוֹשָׁה יָמִים בְּאַרְמוֹן הַדְּרָקוֹן הָיָה, לְמַעֲשֶׂה, מֵאוֹת שָׁנִים בָּעוֹלָם שֶׁלְּמַעְלָה. בְּצַעַר גָּדוֹל וּבְלִי לְהַרְהֵר, שָׁכַח אֶת אַזְהֲרַת הַנְּסִיכָה וּפָתַח אֶת הַתֵּבָה. עָנָן לָבָן שֶׁל עָשָׁן קַל עָלָה מִמֶּנָּה, וּשְׂעָרוֹ שֶׁל אוּרָשִׁימָה הִלְבִּין בְּבַת אַחַת.

אַךְ אוּרָשִׁימָה לֹא נִשְׁאַר לְבַדּוֹ. תּוֹשָׁבֵי הַכְּפָר הֶחָדָשׁ, שֶׁשָּׁמְעוּ אֶת סִפּוּרוֹ הַמֻּפְלָא, קִבְּלוּ אוֹתוֹ בְּחֶמְלָה וְנָתְנוּ לוֹ בַּיִת קָטָן לְיַד הַחוֹף. שָׁנִים רַבּוֹת יָשַׁב שָׁם הַזָּקֵן הֶחָדָשׁ, וְסִפֵּר לְיַלְדֵי הַכְּפָר עַל אַרְמוֹן הַפְּלָאִים שֶׁמִּתַּחַת לַיָּם, וְעַל הַצָּב הַקָּטָן שֶׁגְּמוּלוֹ הָיָה גָּדוֹל מֵהַכֹּל. הַיְּלָדִים הָיוּ מַקְשִׁיבִים בְּפֶה פָּתוּחַ, וְלוֹמְדִים מִמֶּנּוּ שֶׁתָּמִיד כְּדַאי לַעֲזֹר לְמִי שֶׁחַלָּשׁ מִמְּךָ - וְשֶׁחָשׁוּב לְהַקְשִׁיב לְאַזְהָרוֹת, גַּם כְּשֶׁקָּשֶׁה לַעֲמֹד בַּפִּתּוּי.

מְסֻפָּר עַל־פִּי הָאַגָּדָה הַיַּפָּנִית הָעַתִּיקָה עַל אוּרָשִׁימָה טָארוֹ, הַמֻּכֶּרֶת בְּיַפָּן כְּבָר מֵהַמֵּאָה הַשְּׁמִינִית לִסְפִירָה, מְעֻבֶּדֶת וּמְסֻפֶּרֶת מֵחָדָשׁ עִם סוֹף מְרֻכָּךְ.
