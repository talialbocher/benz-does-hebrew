📖 קְרִיאָה

הָיָה פַּעַם אַרְיֵה גָּדוֹל וְחָזָק, מֶלֶךְ הַיַּעַר. יוֹם אֶחָד יָצָא הָאַרְיֵה לָצוּד עִם שְׁנֵי חֲבֵרִים: חֲמוֹר חָזָק וְשׁוּעָל עָרוּם. "הַיּוֹם נָצוּד יַחַד," אָמַר הָאַרְיֵה, "וְנַחֲלֹק אֶת הַצַּיִד בֵּינֵינוּ."

כָּל הַבֹּקֶר רָצוּ שְׁלָשְׁתָּם בֵּין הָעֵצִים, עַד שֶׁלָכְדוּ אַיָּל שָׁמֵן וּגְדוֹל. הָאַרְיֵה הִצְבִּיעַ עַל הָאַיָּל וְאָמַר לַחֲמוֹר: "הָאַיָּל הַזֶּה שֶׁלָּנוּ! חַלֵּק אוֹתוֹ בֵּינֵינוּ, וְחַלֵּק בְּצֶדֶק."

הַחֲמוֹר חָשַׁב רֶגַע וְחִלֵּק אֶת הָאַיָּל לִשְׁלוֹשָׁה חֲלָקִים שָׁוִים. "זֶה חֵלֶק לָאַרְיֵה, זֶה חֵלֶק לִי וְזֶה חֵלֶק לַשּׁוּעָל," אָמַר בְּגַאֲוָה. הָאַרְיֵה כָּעַס מְאֹד. "הַחֲלָקִים הָאֵלֶּה שָׁוִים? אֲנִי הַמֶּלֶךְ!" שָׁאַג, וְהַחֲמוֹר בָּרַח בְּפַחַד אֶל תּוֹךְ הַיַּעַר וְלֹא חָזַר.

הָאַרְיֵה הִסְתַּכֵּל עַל הַשּׁוּעָל. "עַכְשָׁו אַתָּה," אָמַר, "חַלֵּק אֶת הַצַּיִד, אֲבָל תִּהְיֶה זָהִיר!" הַשּׁוּעָל הִצִּיץ אֶל הָעֵצִים הָהֵם, הָרְחוֹקִים, שֶׁשָּׁם נֶעֱלַם הַחֲמוֹר, וְהֵבִין שֶׁעָלָיו לִהְיוֹת זָהִיר.

הוּא הִגִּישׁ לָאַרְיֵה כִּמְעַט אֶת כָּל הָאַיָּל וְשָׁמַר לְעַצְמוֹ חֵלֶק קָטָן מְאֹד. "הַחֵלֶק הַזֶּה הוּא שֶׁלְךָּ, הַמֶּלֶךְ," אָמַר, "וְהַחֵלֶק הַקָּטָן הַזֶּה מַסְפִּיק לִי."

הָאַרְיֵה חִיֵּךְ בְּשִׂמְחָה. "זֹאת חֲלוּקָה מְצֻיֶּנֶת! מִי לִמֵּד אוֹתְךָ לַחֲלֹק כָּךְ?" "הַחֲמוֹר שֶׁבָּרַח," עָנָה הַשּׁוּעָל. "לָמַדְתִּי מִמֶּנּוּ מָה קוֹרֶה לְמִי שֶׁלֹּא חוֹשֵׁב עַל הַמֶּלֶךְ."

מֵהַיּוֹם הַהוּא חָלְקוּ הָאַרְיֵה וְהַשּׁוּעָל אֶת הַצַּיִד בְּשָׁלוֹם. וְהַשּׁוּעָל זָכַר אֶת הַלֶּקַח: חָכָם הוּא מִי שֶׁלּוֹמֵד מִטָּעֻיּוֹת שֶׁל אֲחֵרִים.

הַמָּשָׁל מְבֻסָּס עַל "חֵלֶק הָאַרְיֵה" מֵאֵת אֵיזוֹפּוֹס (יָוָן הָעַתִּיקָה), מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ.

📚 דִּקְדּוּק — כִּנּוּיֵי רֶמֶז (חֲזָרָה)

מָה מַצְבִּיעִים וְעַל מִי?

כְּשֶׁהָאַרְיֵה הִצְבִּיעַ עַל הָאַיָּל, הוּא אָמַר "הָאַיָּל הַזֶּה". כִּנּוּיֵי רֶמֶז הֵם מִלִּים שֶׁמַּצְבִּיעוֹת עַל מִישֶׁהוּ אוֹ עַל מַשֶּׁהוּ: קָרוֹב אֵלֵינוּ אוֹ רָחוֹק מִמֶּנּוּ. הֵם גַּם מִתְאִימִים לְשֵׁם הָעֶצֶם בְּמִין וּבְמִסְפָּר.

🔑 כְּלָל: לִפְנֵי שֵׁם עֶצֶם מֻגְדָּר (עִם ה׳ הַיְדִיעָה) כִּנּוּי הָרֶמֶז בָּא אַחֲרָיו, וְהַקָּרוֹב מְקַבֵּל ה׳: "הָאַיָּל הַזֶּה", "הַחֲלוּקָה הַזֹּאת", "הַחֲלָקִים הָאֵלֶּה". כְּשֶׁאֵין שֵׁם עֶצֶם אַחֲרָיו, כִּנּוּי הָרֶמֶז עוֹמֵד לְבַד: "זֶה חֵלֶק לִי", "זֹאת חֲלוּקָה מְצֻיֶּנֶת".
💡 שִׂימוּ לֵב לַמִּכְשׁוֹלִים: שֵׁמוֹת עֶצֶם בְּזָכָר שֶׁרַבִּים שֶׁלָּהֶם נִגְמָר בְּ־וֹת (מָקוֹם ← מְקוֹמוֹת, שֻׁלְחָן ← שֻׁלְחָנוֹת) נִשְׁאָרִים זָכָר, וְלָכֵן: "הַמְּקוֹמוֹת הָהֵם" (לֹא "הָהֵן"). וּבְקָרוֹב אֵין הֶבְדֵּל בֵּין זָכָר לִנְקֵבָה בְּרַבִּים: "הָאֵלֶּה".
