לִפְנֵי שָׁנִים רַבּוֹת, בָּאִיִּים הַיָּפִים שֶׁל הָאוֹקְיָנוֹס הַשָּׁקֵט, חַיּוּ בְּנֵי אָדָם שֶׁהִתְלוֹנְנוּ יוֹם אַחַר יוֹם עַל אוֹתָהּ בְּעָיָה בְּדִיּוּק: הַשֶּׁמֶשׁ הָיְתָה עוֹלָה בַּבֹּקֶר וְשׁוֹקַעַת כִּמְעַט מִיָּד, כְּאִלּוּ מִהֲרָה לָנוּחַ. הַיּוֹם הָיָה קָצָר מִדַּי כְּדֵי לְדַיֵּג בַּיָּם, לְעַבֵּד אֶת הָאֲדָמָה אוֹ לְיַבֵּשׁ דָּגִים בָּרוּחַ הַחַמָּה, וְהַלַּיְלָה הָאָרֹךְ הִשְׁתָּרֵר עַל הָאִי שָׁעוֹת רַבּוֹת מִדַּי.

בְּאֶחָד הַכְּפָרִים חַי נַעַר קָטָן וְעַקְשָׁן בְּשֵׁם מָאוּי, שֶׁהָיָה, כָּךְ סִפְּרוּ, חֲצִי-אֵל - חֲצִי אָדָם וַחֲצִי מִישֶׁהוּ בַּעַל כֹּחַ עַל-טִבְעִי. יְלָדִים אֲחֵרִים בַּכְּפָר חָלְקוּ בְּדֵעוֹתֵיהֶם עָלָיו: יְלָדוֹת מְבֻגָּרוֹת יוֹתֵר טָעֲנוּ שֶׁמָּאוּי הוּא הֶחָכָם וְהָאַמִּיץ בְּיוֹתֵר מִבֵּין כָּל יַלְדֵי הָאִי, וְאִלּוּ קְשִׁישִׁים מְסֻיָּמִים בַּכְּפָר הִתְלוֹנְנוּ שֶׁהוּא רַק יֶלֶד קַנְטְרָן שֶׁתָּמִיד מְחַפֵּשׂ צָרוֹת. אֵלֶּה הָיוּ, כַּמּוּבָן, רַק דֵּעוֹת - וְלֹא כֻּלָּם הִסְכִּימוּ בֵּינֵיהֶם.

יוֹם אֶחָד חָזַר מָאוּי הַבַּיְתָה עִם סַל דָּגִים גָּדוֹל, וְגִלָּה שֶׁאִמּוֹ, הִינָה, לֹא הִסְפִּיקָה לְיַבֵּשׁ אֲפִלּוּ דָּג אֶחָד לִפְנֵי שֶׁהַחֲשֵׁכָה יָרְדָה עַל הָאִי. "מִישֶׁהוּ חַיָּב לְלַמֵּד אֶת הַשֶּׁמֶשׁ הַזֹּאת דֶּרֶךְ אֶרֶץ!" קָרָא מָאוּי בְּכַעַס. הִינָה חִיְּכָה וְגִלְּתָה לוֹ סוֹד יָשָׁן: כְּדֵי לִלְכֹּד יְצוּר חָזָק כָּל כָּךְ, צָרִיךְ חֶבֶל חָזָק עוֹד יוֹתֵר - אֶחָד אָרוּג מִסִּיבֵי עֵץ הַקּוֹקוֹס וּמִשְּׂעַר רֹאשָׁהּ שֶׁל אֲחוֹתוֹ הַגְּדוֹלָה שֶׁל מָאוּי.

מָאוּי וְאֶחָיו עָבְדוּ יַחַד שָׁעוֹת רַבּוֹת וְאָרְגוּ שְׁמוֹנָה חֲבָלִים חֲזָקִים, כָּל אֶחָד עִם מַלְכֹּדֶת קְשׁוּרָה בִּקְצֵהוּ. לְאַחַר מִכֵּן טִפְּסוּ כֻּלָּם בַּלַּיְלָה אֶל פִּי הַר הַגַּעַשׁ הַגָּבוֹהַּ בָּאִי, שָׁם, כָּךְ סִפְּרוּ הַזְּקֵנִים, יוֹצֵאת הַשֶּׁמֶשׁ כָּל בֹּקֶר מִתּוֹךְ הָאֲדָמָה. מָאוּי פָּרַשׂ אֶת חֲבָלָיו סָבִיב לִשְׂפַת הַמַּכְתֵּשׁ וְחִכָּה בְּשֶׁקֶט לְיַד הָאֹפֶק, בְּעוֹד אֶחָיו מִסְתַּתְּרִים מֵאֲחוֹרֵי סְלָעִים.

עִם עֲלוֹת הַשַּׁחַר הֵחֵלָּה הַשֶּׁמֶשׁ לְהָצִיץ מֵעַל שְׂפַת הַמַּכְתֵּשׁ - וְקֶרֶן רִאשׁוֹנָה, אַחַר כָּךְ שְׁנִיָּה, יָצְאוּ מִמֶּנָּה כְּמוֹ רַגְלַיִם אֲרֻכּוֹת שֶׁל אוֹר. בְּרֶגַע שֶׁקֶּרֶן שֶׁל הַשֶּׁמֶשׁ יָצְאָה דַּיָּהּ הַחוּצָה, קָפַץ מָאוּי מִמַּחֲבוֹאוֹ וְתָפַס אוֹתָהּ בְּלוּלָאַת הַחֶבֶל. קֶרֶן אַחַר קֶרֶן נִלְכְּדָה, עַד שֶׁכָּל הַשֶּׁמֶשׁ הָיְתָה קְשׁוּרָה חָזָק אֶל סַלְעֵי הָהָר וְלֹא יָכְלָה לָזוּז מִמְּקוֹמָהּ.

הַשֶּׁמֶשׁ צָעֲקָה בְּכֹאֵב וּבְפַחַד, כִּי הַחֲבָלִים חָנְקוּ אֶת קַרְנֶיהָ הַלּוֹהֲטוֹת. "שַׁחְרְרוּ אוֹתִי!" הִתְחַנְּנָה. מָאוּי עָמַד מוּלָהּ בְּלִי לָזוּז וְאָמַר בְּקוֹל תַּקִּיף: "רַק אִם תַּבְטִיחִי לָלֶכֶת לְאַט יוֹתֵר בַּשָּׁמַיִם, כְּדֵי שֶׁלִּבְנֵי הָאָדָם יִהְיֶה מַסְפִּיק אוֹר לָעֲבוֹדָה וְלַחַיִּים." הַשֶּׁמֶשׁ, חֲלוּשָׁה מִדַּי כְּדֵי לְהִתְנַגֵּד, הִבְטִיחָה לוֹ בְּקוֹל חַלָּשׁ: "אֲנִי מַבְטִיחָה. מֵעַתָּה אֵלֵךְ לְאַט יוֹתֵר, יוֹם אַחַר יוֹם."

רַק אָז שִׁחְרֵר מָאוּי אֶת הַחֲבָלִים, קֶרֶן אַחַר קֶרֶן, וְהַשֶּׁמֶשׁ הִמְשִׁיכָה בְּדַרְכָּהּ בַּשָּׁמַיִם - הַפַּעַם בְּקֶצֶב אִטִּי וְשָׁקֵט הַרְבֵּה יוֹתֵר. מֵאוֹתוֹ יוֹם וָאֵילָךְ הָיָה הַיּוֹם אָרֹךְ דַּיּוֹ כְּדֵי לְדַיֵּג, לְעַבֵּד אֶת הָאֲדָמָה, לְבַשֵּׁל וּלְיַבֵּשׁ אֶת הַדָּגִים בַּשֶּׁמֶשׁ הַחַמָּה - וְעַל כָּךְ יָכְלוּ תּוֹשְׁבֵי הָאִי לְהוֹדוֹת עַד הַיּוֹם.

כְּשֶׁחָזַר מָאוּי לַכְּפָר עִם אֶחָיו, כֻּלָּם כְּבָר יָדְעוּ עַל הַמַּעֲשֶׂה. יַלְדֵי הַכְּפָר הֵרִיעוּ לוֹ וְקָרְאוּ לוֹ הַגִּבּוֹר הַגָּדוֹל בְּיוֹתֵר שֶׁחַי אֵי פַּעַם עַל הָאִי, וְאִלּוּ הַקְּשִׁישִׁים שֶׁתָּמִיד הִתְלוֹנְנוּ עָלָיו הוֹדוּ, הַפַּעַם בְּחִיּוּךְ, שֶׁאוּלַי טָעוּ לְגַבָּיו. אֲבָל הָעֻבְדָּה שֶׁכֻּלָּם יָכְלוּ לִרְאוֹת בְּעֵינֵיהֶם, בְּלִי שׁוּם וִכּוּחַ, הָיְתָה פְּשׁוּטָה: הַיּוֹם, מֵאוֹתוֹ בֹּקֶר וָהָלְאָה, הָיָה אָרֹךְ יוֹתֵר.

מְסֻפָּר עַל־פִּי אַגָּדָה פּוֹלִינֵזִית עַתִּיקָה (מֵהַמָּסֹרֶת הַהָאוַאִית וְהַמָּאוֹרִית) עַל מָאוּי חֲצִי-הָאֵל וּלְכִידַת הַשֶּׁמֶשׁ, מְעֻבֶּדֶת וּמְסֻפֶּרֶת מֵחָדָשׁ.

      ✍️ כְּתִיבָה
      ✍️ מַטְלַת כְּתִיבָה

      כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים: שְׁלֹשָׁה מִשְׁפְּטֵי עֻבְדָּה (מָה שֶׁבֶּאֱמֶת קָרָה בַּסִּפּוּר, אֶפְשָׁר לְהוֹכִיחַ) וּשְׁנֵי מִשְׁפְּטֵי דֵּעָה מִשֶּׁלָּכֶם (מָה שֶׁאַתֶּם חוֹשְׁבִים אוֹ מַרְגִּישִׁים עַל מָאוּי אוֹ עַל הַסִּפּוּר). לְדֻגְמָה, מִשְׁפַּט עֻבְדָּה: "מָאוּי לָכַד אֶת הַשֶּׁמֶשׁ בְּעֶזְרַת חֲבָלִים." מִשְׁפַּט דֵּעָה: "לְדַעְתִּי, הִינָה הָיְתָה הַדְּמוּת הֲכִי חֲכָמָה בַּסִּפּוּר."
