📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת חַי בְּעִיר לְיַד הַיָּם דַּיָּג עָנִי וְזָקֵן, שֶׁפַּרְנָסָתוֹ הָיְתָה תְּלוּיָה בַּיָּם. בְּכָל בֹּקֶר יָצָא אֶל הַחוֹף וְהֵטִיל אֶת רִשְׁתּוֹ אַרְבַּע פְּעָמִים בִּלְבַד, לֹא יוֹתֵר - כָּךְ נָהַג מֵאָז יְמֵי נְעוּרָיו, וְהָיָה מְרֻצֶּה מִמָּה שֶׁהַיָּם נָתַן לוֹ.

בְּאוֹתוֹ יוֹם הֵטִיל אֶת הָרֶשֶׁת פַּעַם רִאשׁוֹנָה, וְכַאֲשֶׁר מָשַׁךְ אוֹתָהּ אֶל הַחוֹף מָצָא בְּתוֹכָהּ רַק חֲמוֹר מֵת וְכָבֵד. בַּפַּעַם הַשְּׁנִיָּה מָשַׁךְ כַּד חֶרֶס שָׁבוּר, מָלֵא חוֹל וּצְדָפִים. בַּפַּעַם הַשְּׁלִישִׁית לֹא הֶעֱלָה מְאוּמָה מִלְּבַד אֲבָנִים וְעֵשֶׂב יָם רָטֹב. הַדַּיָּג נֶאֱנַח: "נוֹתְרָה לִי רַק פַּעַם אַחַת לְהַטִּיל אֶת הָרֶשֶׁת הַיּוֹם - הַלְוַאי שֶׁתָּבִיא לִי פַּרְנָסָה."

בַּפַּעַם הָרְבִיעִית הִרְגִּישׁ הַדַּיָּג שֶׁמַּשֶּׁהוּ כָּבֵד מְאֹד נִתְפַּס בָּרֶשֶׁת. בְּקֹשִׁי רַב מָשַׁךְ וּמָשַׁךְ, עַד שֶׁעָלָה מִן הַמַּיִם כַּד נְחֹשֶׁת יָשָׁן וְכָבֵד, סָתוּם בִּפְקַק עוֹפֶרֶת שֶׁעָלָיו הָיָה טָבוּעַ חוֹתָם מוּזָר. "אוּלַי יֵשׁ בְּתוֹכוֹ מַשֶּׁהוּ יָקָר," חָשַׁב הַדַּיָּג בְּשִׂמְחָה, וּמִהֵר לְשַׁבֵּר אֶת פְּקַק הָעוֹפֶרֶת בְּסַכִּינוֹ.

מִיָּד יָצָא מִתּוֹךְ הַכַּד עַמּוּד עָשָׁן עָבֶה, שֶׁהָלַךְ וְהִתְרוֹמֵם לַשָּׁמַיִם עַד שֶׁכִּסָּה אֶת הַשֶּׁמֶשׁ. לְאַט לְאַט הִתְעַבָּה הֶעָשָׁן וְקִבֵּל צוּרָה שֶׁל עֲנָק אַדִּיר, גָּבֹהַּ מֵעֵץ הַתָּמָר הַגָּבוֹהַּ בְּיוֹתֵר בָּעִיר. הַדַּיָּג נִרְתַּע לְאָחוֹר בְּפַחַד, וְהֶעָנָק - שֶׁהָיָה, כַּמּוּבָן, גִּ'ינִי רַב־כֹּחַ - הִבִּיט בּוֹ בְּזַעַם וְקָרָא: "הִתְכּוֹנֵן לָמוּת, בֶּן־אָדָם!"

הַדַּיָּג, אַף שֶׁרָעַד מִפַּחַד, אָסַף אֶת כָּל אֹמֶץ לִבּוֹ וְשָׁאַל: "מַדּוּעַ תַּהֲרֹג אוֹתִי? הֲרֵי שִׁחְרַרְתִּי אוֹתְךָ מִן הַכַּד!" הַגִּ'ינִי הֵשִׁיב בְּקוֹל רוֹעֵם: "לִפְנֵי מֵאוֹת שָׁנִים סֵרַבְתִּי לְצַיֵּת לַמֶּלֶךְ שְׁלֹמֹה הֶחָכָם, וְהוּא כָּלָא אוֹתִי בַּכַּד הַזֶּה וְהִשְׁלִיךְ אוֹתִי לַיָּם. בְּמֵאָה הַשָּׁנִים הָרִאשׁוֹנוֹת נִשְׁבַּעְתִּי לְהַעֲשִׁיר אֶת מִי שֶׁיְּשַׁחְרֵר אוֹתִי. אֲבָל הַמֵּאוֹת חָלְפוּ וְאִישׁ לֹא בָּא, וְעִם הַזְּמַן הַכַּעַס גָּבַר עָלַי - עַד שֶׁנִּשְׁבַּעְתִּי שֶׁבִּמְקוֹם לְהַעֲשִׁיר אֶת מְשַׁחְרְרִי, אֶהֱרֹג אוֹתוֹ בִּמְקוֹם זֹאת."

הַדַּיָּג הֵבִין שֶׁכֹּחַ לֹא יוֹעִיל לוֹ מוּל עֲנָק כָּזֶה, וְנֶאֱחַז בְּעָרְמָה בִּמְקוֹם זֶה. הוּא הֵרִים גַּבָּה בְּסָפֵק וְאָמַר: "אֲנִי מִתְקַשֶּׁה לְהַאֲמִין. אַתָּה עֲנָק כֹּה גָּדוֹל - אֵיךְ יָכֹלְתָּ בִּכְלָל לְהִכָּנֵס לְכַד קָטָן שֶׁכָּזֶה? אֲנִי בָּטוּחַ שֶׁזֶּה בִּלְתִּי אֶפְשָׁרִי." הַגִּ'ינִי, שֶׁנֶּעֱלַב מִסָּפֵק זֶה בְּכֹחוֹ, קָרָא: "אֲנִי אוֹכִיחַ לְךָ!" וְהִתְחִיל שׁוּב לְהִתְמַסֵּס לְעָשָׁן, וְשָׁב וְנִכְנַס פְּנִימָה אֶל תּוֹךְ הַכַּד הַצַּר.

בְּרֶגַע שֶׁהָאַחֲרוֹן מִן הֶעָשָׁן נֶעֱלַם בְּתוֹךְ הַכַּד, חָטַף הַדַּיָּג בִּמְהִירוּת אֶת פְּקַק הָעוֹפֶרֶת וְסָתַם בּוֹ אֶת פִּי הַכַּד בְּחֹזֶק. "עַכְשָׁו אַתָּה כָּלוּא שׁוּב," אָמַר הַדַּיָּג בְּקוֹל תַּקִּיף, "וְאֶזְרֹק אוֹתְךָ בַּחֲזָרָה לְמַצוּלוֹת הַיָּם, כְּדֵי שֶׁתֵּדַע אֵיךְ מְשַׁלְּמִים רָעָה תַּחַת טוֹבָה." מִתּוֹךְ הַכַּד נִשְׁמַע קוֹלוֹ הֶעָמוּם וְהַנִּבְהָל שֶׁל הַגִּ'ינִי, מִתְחַנֵּן לִסְלִיחָה.

הַגִּ'ינִי הֵבִין שֶׁטָּעָה, וּמִתּוֹךְ הַכַּד נִשְׁבַּע שְׁבוּעָה חֲדָשָׁה וַאֲמִתִּית: אִם יְשַׁחְרֵר אוֹתוֹ הַדַּיָּג, לֹא רַק שֶׁלֹּא יִפְגַּע בּוֹ - אֶלָּא יוֹבִיל אוֹתוֹ אֶל שָׂדֶה נִסְתָּר שֶׁל אֲבָנִים יְקָרוֹת, שֶׁדַּי בָּהֶן כְּדֵי לְפַרְנֵס אֶת כָּל תּוֹשְׁבֵי הַכְּפָר עַד סוֹף יְמֵיהֶם. הַדַּיָּג, שֶׁכְּבָר לָמַד לְהַבְחִין בֵּין דְּבָרִים שֶׁאֶפְשָׁר לִבְדֹּק לְבֵין הַבְטָחוֹת בִּלְבַד, בִּקֵּשׁ הוֹכָחָה קְטַנָּה קֹדֶם - וְהַגִּ'ינִי, בְּלִי לְהִתְעַכֵּב, הוֹצִיא מִתּוֹךְ הַכַּד חֲתִיכַת זָהָב טְהוֹרָה כְּאוֹת רְצִינוּת. רַק אָז שִׁחְרֵר הַדַּיָּג אֶת פְּקַק הָעוֹפֶרֶת, וּשְׁנֵיהֶם יָצְאוּ יַחַד אֶל שְׂדֵה הָאֲבָנִים הַיְּקָרוֹת.

מֵאוֹתוֹ יוֹם וָהָלְאָה חָזַר הַדַּיָּג לְכַפְרוֹ עָשִׁיר מְאֹד - אַךְ לֹא שָׁכַח אֶת יְמֵי הָעֹנִי, וְחִלֵּק חֵלֶק גָּדוֹל מֵאוֹצָרוֹ בֵּין שְׁכֵנָיו הָעֲנִיִּים. וְעַד הַיּוֹם מְסַפְּרִים בְּאוֹתָהּ עִיר עַל הַדַּיָּג הֶחָכָם שֶׁנִּצַּח עֲנָק בְּעָרְמָה, וְלֹא בְּכֹחַ.

מְסֻפָּר עַל־פִּי "הַדַּיָּג וְהַגִּ'ינִי" מִתּוֹךְ "אֶלֶף לַיְלָה וָלַיְלָה" (עוֹלָם עֲרָבִי), מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ עִם סוֹף מְרֻכָּךְ.

חֲוַת דֵּעָה מְנֻמֶּקֶת

בַּשִׁיעוּר הַקּוֹדֵם לָמַדְנוּ לְהַבְחִין בֵּין עֻבְדָּה לְדֵעָה. עַכְשָׁו נִלְמַד לִכְתֹּב חֲוַת דֵּעָה מְנֻמֶּקֶת - לֹא רַק לוֹמַר מָה אֲנַחְנוּ חוֹשְׁבִים, אֶלָּא גַּם לְהַסְבִּיר לָמָּה. חֲוַת דֵּעָה מְנֻמֶּקֶת טוֹבָה בְּנוּיָה מִשְּׁלוֹשָׁה חֲלָקִים: מִשְׁפַּט עֶמְדָּה בָּרוּר, לְפָחוֹת שְׁנֵי נִמּוּקִים שֶׁמַּסְבִּירִים אוֹתָהּ, וּמִשְׁפַּט סִכּוּם שֶׁחוֹזֵר וּמְחַזֵּק אֶת הָעֶמְדָּה.

כִּתְבוּ חֲוַת דֵּעָה מְנֻמֶּקֶת בִּשְׁאֵלָה: הַאִם לְדַעְתְּכֶם הַדַּיָּג נָהַג בְּחָכְמָה כְּשֶׁרִמָּה אֶת הַגִּ'ינִי בַּחֲזָרָה לְתוֹךְ הַכַּד? כִּתְבוּ אַרְבָּעָה מִשְׁפָּטִים: (1) מִשְׁפַּט עֶמְדָּה בָּרוּר, (2) נִמּוּק רִאשׁוֹן עִם מִלַּת נִמּוּק (כִּי / מִכֵּיוָן שֶׁ- / מִשּׁוּם שֶׁ-) וְעֻבְדָּה מִן הַסִּפּוּר, (3) נִמּוּק שֵׁנִי בְּאוֹתָהּ הַצּוּרָה, (4) מִשְׁפַּט סִכּוּם שֶׁמְּחַזֵּק אֶת הָעֶמְדָּה. לְדֻגְמָה, מִשְׁפַּט עֶמְדָּה: "לְדַעְתִּי הַדַּיָּג נָהַג בְּחָכְמָה."
