הָאַרְנָב וְהַקִּפּוֹד

מַעֲשִׂיָּה שֶׁל הָאַחִים גְּרִים עַל מֵרוֹץ מְשֻׁנֶּה בִּשְׂדֵה הַלְּפָתוֹת

📖 קְרִיאָה

לְיַד שְׂדֵה לְפָתוֹת גָּדוֹל גָּר פַּעַם קִפּוֹד קָטָן עִם אִשְׁתּוֹ וִילָדָיו. בְּבֹקֶר אֶחָד שֶׁל סוֹף הַקַּיִץ הוּא קָם מֻקְדָּם, עָמַד בְּפֶתַח הַבַּיִת וְשָׁר לְעַצְמוֹ שִׁיר קָטָן. ״אֵלֵךְ לִרְאוֹת מָה שְׁלוֹם הַלְּפָתוֹת,״ אָמַר לְאִשְׁתּוֹ, וְיָצָא לַדֶּרֶךְ.

בַּשָּׂדֶה פָּגַשׁ אֶת הָאַרְנָב, שֶׁגַּם הוּא יָצָא לְטַיֵּל. הַקִּפּוֹד אָמַר לוֹ בְּנִימוּס: ״בֹּקֶר טוֹב!״ אֲבָל הָאַרְנָב הַגַּאַוְתָן לֹא עָנָה לוֹ. הוּא הִבִּיט בָּרַגְלַיִם הַקְּצָרוֹת וְהָעֲקֻמּוֹת שֶׁל הַקִּפּוֹד וְצָחַק: ״מָה אַתָּה עוֹשֶׂה כָּאן כָּל כָּךְ מֻקְדָּם? עִם רַגְלַיִם כָּאֵלֶּה אַתָּה בֶּטַח לֹא רָץ לְשׁוּם מָקוֹם!״

הַקִּפּוֹד נֶעֱלַב מְאוֹד, כִּי הוּא אָהַב אֶת רַגְלָיו בְּדִיּוּק כְּמוֹ שֶׁהֵן. ״אַתָּה חוֹשֵׁב שֶׁאַתָּה מָהִיר מִמֶּנִּי?״ שָׁאַל. ״אֲנִי מוּכָן לְהִתְעָרֵב אִתְּךָ שֶׁאֲנִי אָרוּץ מַהֵר מִמְּךָ!״ הָאַרְנָב צָחַק עַד שֶׁכִּמְעַט נָפַל. ״אַתָּה? בְּסֵדֶר גָּמוּר! נִתְעָרֵב עַל מַטְבֵּעַ זָהָב. בּוֹא נַתְחִיל מִיָּד!״

״לֹא עַכְשָׁו,״ עָנָה הַקִּפּוֹד בְּשַׁלְוָה. ״עוֹד לֹא אָכַלְתִּי אֲרוּחַת בֹּקֶר. אֲנִי אָשׁוּב לְכָאן בְּעוֹד חֲצִי שָׁעָה.״ אֲבָל זוֹ לֹא הָיְתָה הַסִּבָּה הָאֲמִתִּית: בְּרֹאשׁוֹ כְּבָר הִתְחִיל לְהִתְבַּשֵּׁל רַעְיוֹן.

כְּשֶׁבָּא הַבַּיְתָה, קָרָא לְאִשְׁתּוֹ: ״קוּמִי מַהֵר וּבוֹאִי אִתִּי לַשָּׂדֶה!״ הוּא סִפֵּר לָהּ עַל הַהִתְעָרְבוּת, וְהִיא נִבְהֲלָה: ״הִשְׁתַּגַּעְתָּ? אַל תָּרוּץ נֶגֶד אַרְנָב!״ ״אַל תִּדְאֲגִי,״ עָנָה הַקִּפּוֹד, ״יֵשׁ לִי תָּכְנִית.״

בַּשָּׂדֶה הָיוּ שְׁנֵי תְּלָמִים אֲרֻכִּים, זֶה לְיַד זֶה. ״אַתְּ תַּעַמְדִי כָּאן, בַּקָּצֶה הַתַּחְתּוֹן שֶׁל הַתֶּלֶם,״ הִסְבִּיר הַקִּפּוֹד. ״כְּשֶׁהָאַרְנָב יַגִּיעַ, קוּמִי וְקִרְאִי לוֹ: ׳אֲנִי כְּבָר כָּאן!׳ וְעַד אָז אַל תָּזוּזִי מִמְּקוֹמֵךְ.״ אִשְׁתּוֹ הִתְחַבְּאָה בַּתֶּלֶם וְנָחָה שָׁם בְּשֶׁקֶט.

הַקִּפּוֹד שָׁב אֶל הָאַרְנָב וְעָמַד בְּרֹאשׁ הַתֶּלֶם הַשֵּׁנִי. ״אַחַת, שְׁתַּיִם, שָׁלוֹשׁ!״ קָרָא הָאַרְנָב, וְטָס קָדִימָה כְּמוֹ רוּחַ סְעָרָה. הַקִּפּוֹד רָץ רַק שְׁלוֹשָׁה צְעָדִים, וְאָז הִתְכּוֹפֵף וְנִשְׁאַר בַּתֶּלֶם בְּשֶׁקֶט.

כְּשֶׁהָאַרְנָב הִגִּיעַ בִּנְשִׁימָה כְּבֵדָה לְסוֹף הַשָּׂדֶה, קָמָה אֵשֶׁת הַקִּפּוֹד מִן הַתֶּלֶם וְקָרְאָה: ״אֲנִי כְּבָר כָּאן!״ הָאַרְנָב נִדְהַם. הוּא לֹא יָדַע שֶׁזּוֹ אִשְׁתּוֹ שֶׁל הַקִּפּוֹד, כִּי קִפּוֹד וְקִפּוֹדָה נִרְאִים כִּמְעַט אוֹתוֹ דָּבָר. ״עוֹד פַּעַם!״ צָעַק. ״נָרוּץ בַּחֲזָרָה!״

וְכָךְ רָץ הָאַרְנָב הָלוֹךְ וָשׁוֹב, הָלוֹךְ וָשׁוֹב. בְּכָל פַּעַם שֶׁהִגִּיעַ לְקָצֶה, קָם קִפּוֹד מִן הַתֶּלֶם וְקָרָא: ״אֲנִי כְּבָר כָּאן!״ שִׁבְעִים וְשָׁלוֹשׁ פְּעָמִים חָצָה הָאַרְנָב אֶת הַשָּׂדֶה. בַּפַּעַם הַשִּׁבְעִים וְאַרְבַּע הוּא כְּבָר לֹא הִצְלִיחַ לְהַגִּיעַ לַסּוֹף: הוּא נָפַל בְּאֶמְצַע הַשָּׂדֶה, עָיֵף וּמֻתָּשׁ, וְלֹא זָז.

הַקִּפּוֹד וְאִשְׁתּוֹ לָקְחוּ אֶת מַטְבֵּעַ הַזָּהָב וְשָׁבוּ הַבַּיְתָה מְרֻצִּים, וְשָׁרוּ כָּל הַדֶּרֶךְ. וּמֵאוֹתוֹ יוֹם, אַף אַרְנָב לֹא צָחַק עוֹד עַל הָרַגְלַיִם שֶׁל קִפּוֹד.

בְּשִׁיעוּר 121 לָמַדְנוּ שֶׁבְּגִזְרַת ע״ו הָאוֹת הָאֶמְצָעִית שֶׁל הַשֹּׁרֶשׁ (ו׳ אוֹ י׳) נֶעֱלֶמֶת בֶּעָבָר: קָם, רַצְתִּי, שָׁרוּ. הַיּוֹם נִרְאֶה מָה קוֹרֶה בֶּעָתִיד וּבַצִּוּוּי — שָׁם הִיא חוֹזֶרֶת, בְּדִיּוּק כְּמוֹ בְּשֵׁם הַפֹּעַל.

⚠️ מַלְכֹּדֶת: קוּמוּ! (צִוּוּי, עִם ו׳) לְעֻמַּת הֵם קָמוּ (עָבָר, בְּלִי ו׳). וְגַם: רוּצוּ! לְעֻמַּת הֵם רָצוּ.
🚫 צִוּוּי שְׁלִילִי = אַל + עָתִיד: אַל תָּרוּץ! אַל תָּזוּזִי! — וְלֹא ״אַל רוּץ״.
⭐ יוֹצֵא דֹּפֶן — בָּא: בֶּעָתִיד וּבַצִּוּוּי יֵשׁ חוֹלָם וְלֹא שׁוּרוּק: אָבוֹא יָבוֹא בּוֹא! וְשִׂימוּ לֵב: אָבוֹא (אֲנִי אַגִּיעַ) שׁוֹנֶה מִאָבִיא (אֲנִי אֶקַּח אִתִּי).

כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים עַל הַסִּפּוּר אוֹ עַל עַצְמְכֶם, וּבְכָל מִשְׁפָּט הִשְׁתַּמְּשׁוּ בְּפֹעַל מִגִּזְרַת ע״ו לְפִי הַהוֹרָאָה. שִׂימוּ לֵב: בֶּעָתִיד וּבַצִּוּוּי הָאוֹת הָאֶמְצָעִית חוֹזֶרֶת! דֻּגְמָה: מָחָר אָקוּם מֻקְדָּם וְאָרוּץ לַגַּן.
