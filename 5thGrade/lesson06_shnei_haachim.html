📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת חָיוּ בֶּהָרִים שֶׁל יְרוּשָׁלַיִם שְׁנֵי אַחִים, שֶׁיָּרְשׁוּ יַחַד שָׂדֶה שֶׁל חִטִּים מֵאֲבִיהֶם.

הָאָח הַבְּכוֹר הָיָה נָשׂוּי וְאָב לִילָדִים רַבִּים, וְהָאָח הַצָּעִיר הָיָה רַוָּק וְגָר לְבַדּוֹ.

בְּכָל שָׁנָה, בִּזְמַן הַקָּצִיר, חָלְקוּ הָאַחִים אֶת הַיְּבוּל בֵּינֵיהֶם בְּמִדָּה שָׁוָה - עֲרֵמַת עֳמָרִים לְכָל אֶחָד.

לַיְלָה אֶחָד לֹא הָיָה הָאָח הַבְּכוֹר יָכוֹל לִישֹׁן, וְחָשַׁב לְעַצְמוֹ: "לְאָחִי אֵין יְלָדִים שֶׁיִּדְאֲגוּ לוֹ בַּזִּקְנָה, וְלִי יֵשׁ הַרְבֵּה. אֶקַּח כַּמָּה עֳמָרִים מֵהָעֲרֵמָה שֶׁלִּי, וְאוֹסִיף לוֹ בְּסֵתֶר."

הוּא קָם בַּלַּיְלָה וְעָשָׂה כֵּן, בְּלִי שֶׁאִישׁ יֵדַע.

בְּאוֹתוֹ לַיְלָה מַמָּשׁ, גַּם הָאָח הַצָּעִיר לֹא הָיָה יָכוֹל לִישֹׁן, וְחָשַׁב לְעַצְמוֹ: "לְאָחִי הַבְּכוֹר יֵשׁ יְלָדִים רַבִּים לְפַרְנֵס, וְלִי אֵין. אֶקַּח כַּמָּה עֳמָרִים מֵהָעֲרֵמָה שֶׁלִּי, וְאוֹסִיף לוֹ בְּסֵתֶר."

גַּם הוּא קָם וְעָשָׂה כֵּן, בְּצִנְעָה גְּמוּרָה.

בַּבֹּקֶר הִתְפַּלְּאוּ שְׁנֵיהֶם לִרְאוֹת שֶׁהָעֲרֵמוֹת נִשְׁאֲרוּ בְּאוֹתוֹ הַגֹּדֶל, לַמְרוֹת שֶׁכָּל אֶחָד לָקַח מֵהֶן בַּלַּיְלָה.

כָּךְ קָרָה שׁוּב וָשׁוּב, לַיְלָה אַחַר לַיְלָה - עַד שֶׁלַּיְלָה אֶחָד נִפְגְּשׁוּ הָאַחִים בְּאֶמְצַע הַדֶּרֶךְ, כָּל אֶחָד נוֹשֵׂא עֳמָרִים אֶל עֵבֶר אָחִיו.

אָז הֵבִינוּ הַשְּׁנַיִם מָה קָרָה כָּל הַלֵּילוֹת הָאֵלֶּה, וְחִבְּקוּ זֶה אֶת זֶה בְּאַהֲבָה גְּדוֹלָה.

עַל־פִּי הָאַגָּדָה, בְּדִיּוּק בַּמָּקוֹם שֶׁבּוֹ נִפְגְּשׁוּ שְׁנֵי הָאַחִים - נִבְנָה מְאֻחָר יוֹתֵר בֵּית הַמִּקְדָּשׁ בִּירוּשָׁלַיִם.

הַלֶּקַח מֵהָאַגָּדָה: נְתִינָה בְּצִנְעָה, בְּלִי לְחַכּוֹת לְתוֹדָה - הִיא אַחַת הַדְּרָכִים הַיָּפוֹת בְּיוֹתֵר לְהַרְאוֹת אַהֲבָה.

אַגָּדַת עַם יְהוּדִית עַתִּיקָה עַל בִּנְיַן בֵּית הַמִּקְדָּשׁ בִּירוּשָׁלַיִם, מְעֻבֶּדֶת - נַחֲלַת הַכְּלָל

מהי מילה לועזית?

מילים לועזיות הן מילים שהגיעו לעברית משפות אחרות - יוונית, לטינית, ערבית, אנגלית, צרפתית, רוסית ועוד. חלק מהמילים הלועזיות משמשות אותנו כבר מאות שנים ואין להן תחליף, אבל לכמה מהן יש מילה עברית מקבילה, שהומצאה או חודשה במיוחד כדי להחליף את המילה הלועזית.
