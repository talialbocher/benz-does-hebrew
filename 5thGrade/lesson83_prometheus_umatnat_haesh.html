פְּרוֹמֶתֵאוּס וּמַתְּנַת הָאֵשׁ

📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת מְאֹד, כְּשֶׁעוֹלַם הָאֵלִים וְעוֹלַם בְּנֵי הָאָדָם הָיוּ קְרוֹבִים זֶה לָזֶה, יָשְׁבוּ הָאֵלִים בְּהֵיכְלֵיהֶם עַל הַר הָאוֹלִימְפּוֹס, וְאִלּוּ בְּנֵי הָאָדָם חָיוּ עַל פְּנֵי הָאֲדָמָה בְּלִי שׁוּם נוֹחוּת. לֹא הָיְתָה לָהֶם אֵשׁ - לֹא לְבִשּׁוּל, לֹא לְחִמּוּם בַּלֵּילוֹת הַקָּרִים, וְלֹא לַעֲבוֹדַת מַתֶּכֶת.

בֵּין הַטִּיטָנִים - בְּנֵי הַדּוֹר הַקָּדוּם שֶׁקָּדַם לְאֵלֵי הָאוֹלִימְפּוֹס - הָיָה פְּרוֹמֶתֵאוּס, טִיטָן חָכָם וְאוֹהֵב אָדָם, שֶׁרָאָה אֶת סִבְלָם שֶׁל בְּנֵי הָאָדָם וְרִחֵם עֲלֵיהֶם מְאֹד.

פְּרוֹמֶתֵאוּס פָּנָה אֶל זֵאוּס, מֶלֶךְ הָאֵלִים, וּבִקֵּשׁ מִמֶּנּוּ אֵשׁ עֲבוּר בְּנֵי הָאָדָם. אַךְ זֵאוּס סֵרֵב בְּתֹקֶף: "אִם נִתֵּן לָהֶם אֵשׁ, יִהְיוּ חֲזָקִים מִדַּי וְעַצְמָאִיִּים מִדַּי - הֵם עֲלוּלִים אֲפִלּוּ לְהִתְחָרוֹת בָּנוּ, הָאֵלִים."

פְּרוֹמֶתֵאוּס לֹא וִתֵּר. לַיְלָה אֶחָד הִתְגַּנֵּב אֶל הָאָח הַבּוֹעֶרֶת בְּהֵיכַל הָאֵלִים, לָקַח קָנֶה חָלוּל וְסָתוּם, וְהֶחְבִּיא בְּתוֹכוֹ גַּחֶלֶת לוֹחֶשֶׁת. בְּמִסְתָּרִים יָרַד מִן הָהָר אֶל בְּנֵי הָאָדָם, וּמָסַר לָהֶם אֶת הַשַּׁלְהֶבֶת הַיְּקָרָה.

מֵאוֹתוֹ רֶגַע הִשְׁתַּנּוּ חַיֵּי בְּנֵי הָאָדָם. הֵם לָמְדוּ לְבַשֵּׁל אֹכֶל חַם, לְחַמֵּם אֶת בָּתֵּיהֶם בַּלֵּילוֹת הַקָּרִים, וּלְהַתִּיךְ מַתֶּכֶת וּלְיַצֵּר כֵּלִים וּכְלֵי עֲבוֹדָה. הָאֵשׁ הָיְתָה חֲשׁוּבָה כׇּל כָּךְ, עַד שֶׁבְּנֵי הָאָדָם הֵחֵלּוּ לְהִתְקַדֵּם וְלִבְנוֹת עָרִים וּכְפָרִים.

אַךְ מֵעַל הָהָר הִבְחִין זֵאוּס בְּעַמּוּדֵי הֶעָשָׁן הָעוֹלִים מִן הַכְּפָרִים, וְהֵבִין מִיָּד מָה קָרָה. הוּא כָּעַס כׇּל כָּךְ, עַד שֶׁגָּזַר עַל פְּרוֹמֶתֵאוּס עֹנֶשׁ נוֹרָא בְּיוֹתֵר.

זֵאוּס צִוָּה לִכְבֹּל אֶת פְּרוֹמֶתֵאוּס בְּשַׁלְשְׁלָאוֹת כְּבֵדוֹת אֶל צוּק גָּבֹהַּ בְּהָרֵי הַקַּוְקָז. בְּכָל יוֹם הָיָה נֶשֶׁר עֲנָק מַגִּיעַ וּמְנַקֵּר בִּכְבֵדוֹ, וּבְכָל לַיְלָה הָיָה הַכָּבֵד צוֹמֵחַ מֵחָדָשׁ - כָּךְ שֶׁהָעֹנֶשׁ הָיָה חוֹזֵר עַל עַצְמוֹ לְלֹא סוֹף.

פְּרוֹמֶתֵאוּס סָבַל כׇּל כָּךְ שָׁנִים רַבּוֹת, עַד שֶׁכִּמְעַט כָּל הַיְּצוּרִים בַּיְּקוּם שָׁכְחוּ מִמֶּנּוּ - אַךְ בְּנֵי הָאָדָם, שֶׁנֶּהֱנוּ מִן הָאֵשׁ שֶׁהֶעֱנִיק לָהֶם, הִמְשִׁיכוּ לְסַפֵּר אֶת סִפּוּרוֹ וּלְכַנּוֹתוֹ גִּבּוֹר וִידִיד הָאֱנוֹשׁוּת.

יוֹם אֶחָד עָבַר בַּמָּקוֹם הַגִּבּוֹר הֶרְקוּלֵס, בְּנוֹ שֶׁל זֵאוּס, בְּמַסָּע אָרֹךְ שֶׁל גְּבוּרָה. כְּשֶׁרָאָה אֶת סִבְלוֹ הַנּוֹרָא שֶׁל פְּרוֹמֶתֵאוּס, הִתְמַלֵּא רַחֲמִים רַבִּים, וּבִקֵּשׁ מִזֵּאוּס רְשׁוּת לְשַׁחְרְרוֹ. הֶרְקוּלֵס הָיָה חָזָק כׇּל כָּךְ, עַד שֶׁהִצְלִיחַ לִירוֹת בְּקֶשֶׁת בַּנֶּשֶׁר וְלִשְׁבֹּר אֶת הַשַּׁלְשְׁלָאוֹת בְּיָדָיו.

כָּךְ, אַחֲרֵי שָׁנִים אֲרֻכּוֹת שֶׁל סֵבֶל, שֻׁחְרַר פְּרוֹמֶתֵאוּס לַחֹפֶשׁ. וְעַד הַיּוֹם מְסַפְּרִים אֶת סִפּוּרוֹ כְּדֵי לִזְכֹּר: מִי שֶׁמַּעֲנִיק מַתָּנָה גְּדוֹלָה לַזּוּלַת, אֲפִלּוּ בִּמְחִיר יָקָר, זוֹכֶה בַּסּוֹף בְּהַכָּרַת תּוֹדָה שֶׁאֵינָהּ נִשְׁכַּחַת.

מְסֻפָּר עַל־פִּי הַמִּיתוֹלוֹגְיָה הַיְּוָנִית הָעַתִּיקָה, מִיתוֹס פְּרוֹמֶתֵאוּס וּגְנֵבַת הָאֵשׁ, מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ עִם סוֹף מְרֻכָּךְ (הַתְמַקְּדוּת בְּשִׁחְרוּרוֹ בִּידֵי הֶרְקוּלֵס, בְּלִי תֵּאוּר מְפֹרָט וּמַמֻּשָּׁךְ שֶׁל סִבְלוֹ).

📚 דִּקְדּוּק — מִשְׁפַּט תּוֹצָאָה

מִשְׁפַּט תּוֹצָאָה — "כׇּל כָּךְ... עַד שֶׁ..."

בְּשִׁיעוּרִים קוֹדְמִים לָמַדְנוּ עַל מִשְׁפַּט סִבָּה וְתוֹצָאָה עִם מִלִּים כְּמוֹ "כִּי" אוֹ "בִּגְלַל שֶׁ-" (שִׁיעוּר 3, 11) - מִלָּה אַחַת מַסְבִּירָה אֶת הַסִּבָּה לְמַשֶּׁהוּ שֶׁקָּרָה. עַכְשָׁו נַכִּיר סוּג נוֹסָף שֶׁל מִשְׁפָּט: מִשְׁפַּט תּוֹצָאָה. בְּמִשְׁפָּט כָּזֶה, תֵּאוּר שֶׁל עֹצֶם גָּבוֹהַּ (עִם הַמִּלִּים "כׇּל כָּךְ") גּוֹרֵם לְתוֹצָאָה קִיצוֹנִית אוֹ מַפְתִּיעָה, שֶׁמּוֹפִיעָה אַחֲרֵי הַמִּלָּה "עַד שֶׁ-".

🔑 כְּלָל: הַמִּלָּה "כׇּל כָּךְ" בָּאָה לְיַד תֹּאַר אוֹ פֹּעַל שֶׁמְּתָאֵר עֹצֶם (חָזָק, כָּעַס, אָהַב, סָבַל...), וְ"עַד שֶׁ-" בָּא לִפְנֵי הַתּוֹצָאָה שֶׁקָּרְתָה בִּגְלַל אוֹתוֹ עֹצֶם. שְׁתֵּי הַמִּלִּים עוֹבְדוֹת תָּמִיד יַחַד כְּזוּג - אִי אֶפְשָׁר לְהִשְׁתַּמֵּשׁ בְּ"עַד שֶׁ-" בְּמִשְׁפַּט תּוֹצָאָה בְּלִי "כׇּל כָּךְ" (אוֹ מִלַּת עֹצֶם דּוֹמָה) קֹדֶם לָכֵן.
💡 שִׂימוּ לֵב: מִשְׁפַּט תּוֹצָאָה שׁוֹנֶה מִמִּשְׁפַּט תְּנַאי (שִׁיעוּר 39, "אִם... אָז..."). מִשְׁפַּט תְּנַאי מְתָאֵר אֶפְשָׁרוּת שֶׁעֲדַיִן לֹא קָרְתָה בֶּאֱמֶת, וּמִשְׁפַּט תּוֹצָאָה מְתָאֵר מָה שֶׁבֶּאֱמֶת קָרָה בִּגְלַל עֹצֶם גָּבוֹהַּ.

כִּתְבוּ שְׁנֵי מִשְׁפְּטֵי תּוֹצָאָה בְּתַבְנִית "כׇּל כָּךְ... עַד שֶׁ..." - אֶחָד עַל אֵרוּעַ מִתּוֹךְ הַסִּפּוּר עַל פְּרוֹמֶתֵאוּס, וְאֶחָד עַל אֵרוּעַ מֵהַחַיִּים שֶׁלָּכֶם. לְכָל מִשְׁפָּט כִּתְבוּ תְּחִלָּה אֶת הָעֹצֶם (עִם "כׇּל כָּךְ") וְאַחַר כָּךְ אֶת הַתּוֹצָאָה (עִם "עַד שֶׁ-"). לְדֻגְמָה: עֹצֶם - "הָרוּחַ נָשְׁבָה כׇּל כָּךְ חָזָק"; תּוֹצָאָה - "עַד שֶׁכָּל הֶעָלִים עָפוּ מִן הָעֵצִים".
