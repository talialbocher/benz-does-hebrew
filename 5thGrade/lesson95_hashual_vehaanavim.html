📖 קְרִיאָה

בְּיוֹם קַיִץ חַם הָלַךְ שׁוּעָל רָעֵב בְּכֶרֶם שֶׁעַל רֶגֶל הָהָר. כְּבָר יוֹמַיִם הוּא לֹא אָכַל דָּבָר, וְהַבֶּטֶן שֶׁלּוֹ קִרְקְרָה בְּקוֹל.

פִּתְאֹם הִבְחִין הַשּׁוּעָל בְּאֶשְׁכּוֹל עֲנָבִים גָּדוֹל. הָעֲנָבִים הָיוּ בְּשֵׁלִים וּמַבְרִיקִים, אֲבָל הֵם הָיוּ תְּלוּיִים גָּבוֹהַּ מֵעַל לְרֹאשׁוֹ, עַל גֶּפֶן עַתִּיקָה. הַשּׁוּעָל לָקַק אֶת שְׂפָתָיו וְאָמַר: "אֵיזֶה אֹכֶל מְצֻיָּן! כָּל כָּךְ מְתוּקִים! אֲנִי חַיָּב לְהַשִּׂיג אוֹתָם."

הוּא קָפַץ וְנָשַׁךְ אֶת הָאֲוִיר, אֲבָל לֹא הִגִּיעַ. הוּא קָפַץ שֵׁנִית, וְשׁוּב לֹא הִגִּיעַ. בַּפַּעַם הַשְּׁלִישִׁית קָפַץ בְּכָל הַכֹּחַ וְנָפַל עַל גַּבּוֹ בִּכְאֵב.

אַחַר כָּךְ עָלָה הַשּׁוּעָל עַל גֶּזַע הַגֶּפֶן, אֲבָל הַגֶּזַע הָיָה חָלָק, וְהוּא גָּלַשׁ מַטָּה. עָלֶה יָבֵשׁ נָשַׁר מֵלְמַעְלָה וְנָחַת עַל אַפּוֹ.

הַשּׁוּעָל יָשַׁב עַל הַקַּרְקַע וְהֵרִים אֶת הָרֹאשׁ. הָעֲנָבִים הָיוּ מַמָּשׁ שָׁם, אֲבָל רְחוֹקִים מִמֶּנּוּ.

צִפּוֹר קְטַנָּה, שֶׁיָּשְׁבָה עַל עָנָף סָמוּךְ, שָׂמָה לֵב לְכָל מַה שֶׁקָּרָה וְחִיְּכָה. הַשּׁוּעָל הִתְבַּיֵּשׁ. הוּא מָשַׁךְ בִּכְתֵפָיו וְאָמַר בְּקוֹל רָם: "בֶּאֱמֶת! הָעֲנָבִים הָאֵלֶּה בֶּטַח חֲמוּצִים, וַאֲנִי בִּכְלָל לֹא רוֹצֶה אוֹתָם!"

וְהוּא הָלַךְ לְבֵיתוֹ בְּפָנִים חֲמוּצִים, וְהַבֶּטֶן שֶׁלּוֹ עֲדַיִן קִרְקְרָה.

הַסִּפּוּר מְבֻסָּס עַל הַמָּשָׁל הַמְּפֻרְסָם "הַשּׁוּעָל וְהָעֲנָבִים" מֵאֵת אִיזוֹפּוֹס (יָוָן הָעַתִּיקָה), מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ.

📚 דִּקְדּוּק — מִלִּים רַבּוֹת־מַשְׁמָעוּת

מִלָּה אַחַת, כַּמָּה פֵּרוּשִׁים

בְּעִבְרִית יֵשׁ הַרְבֵּה מִלִּים רַבּוֹת־מַשְׁמָעוּת: מִלָּה אַחַת שֶׁכּוֹתְבִים בְּדִיּוּק אוֹתָהּ, וְיֵשׁ לָהּ שְׁנֵי פֵּרוּשִׁים אוֹ יוֹתֵר. הַפֵּרוּשׁ הַנָּכוֹן תָּלוּי בְּהֶקְשֵׁר, כְּלוֹמַר בִּשְׁאָר הַמִּשְׁפָּט.

שִׂימוּ לֵב: לִפְעָמִים הַכְּתִיב זֵהֶה אֲבָל הַנִּקּוּד וְהַהֲגָיָה שׁוֹנִים: עָלָה (פֹּעַל) וְעָלֶה (שֵׁם עֶצֶם). בְּמִלּוֹן, מִלִּים כָּאֵלֶּה מוֹפִיעוֹת לִפְעָמִים בְּעֶרְכִּים נִפְרָדִים.

🔎 אֵיךְ מַכְרִיעִים? קוֹרְאִים אֶת כָּל הַמִּשְׁפָּט, בּוֹדְקִים עִם אֵילוּ מִלִּים הַמִּלָּה מְחֻבֶּרֶת, וּמַחֲלִיפִים אוֹתָהּ בְּפֵרוּשׁ. אִם הַמִּשְׁפָּט נִשְׁמָע הִגָּיוֹנִי — זֶה הַפֵּרוּשׁ הַנָּכוֹן.
💡 לֹא כָּל פֵּרוּשׁ מְתָאִים לְכָל מִשְׁפָּט. בַּצֵּרוּף "שֵׁן שׁוּם" הַמִּלָּה "שֵׁן" אֵינָהּ הַשֵּׁן שֶׁבַּפֶּה, וּבַצֵּרוּף "בַּיִת בְּשִׁיר" הַמִּלָּה "בַּיִת" אֵינָהּ מָקוֹם מְגוּרִים.
