פַּעֲמוֹן הַצֶּדֶק

אַגָּדָה אִיטַלְקִית עַל פַּעֲמוֹן, עֲנַף גֶּפֶן וְסוּס זָקֵן שֶׁבִּקֵּשׁ צֶדֶק

📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת, בָּעִיר הַקְּטַנָּה אַטְרִי שֶׁבְּאִיטַלְיָה, מָלַךְ מֶלֶךְ שֶׁאָהַב צֶדֶק יוֹתֵר מִכֹּל. יוֹם אֶחָד צִוָּה לִתְלוֹת פַּעֲמוֹן גָּדוֹל מִנְּחֹשֶׁת בְּמִגְדָּל קָטָן בְּכִכַּר הָעִיר. מֵהַפַּעֲמוֹן הִשְׁתַּלְשֵׁל חֶבֶל אָרֹךְ, שֶׁהִגִּיעַ כִּמְעַט עַד הָאֲדָמָה.

"מֵעַכְשָׁו," הִכְרִיז הַמֶּלֶךְ, "כָּל מִי שֶׁנַּעֲשָׂה לוֹ עָוֶל — עָנִי אוֹ עָשִׁיר, צָעִיר אוֹ זָקֵן — יִמְשֹׁךְ בַּחֶבֶל. הַשּׁוֹפְטִים יִשְׁמְעוּ אֶת הַפַּעֲמוֹן, יָבוֹאוּ מִיָּד וְיִשְׁפְּטוּ בְּצֶדֶק." הַמֶּלֶךְ בִּקֵּשׁ מִסּוֹפֵר הָעִיר לַחֲרֹט אֶת הַצַּו עַל אֶבֶן לְיַד הַמִּגְדָּל. אֲבָל הָאֶבֶן הָיְתָה קְטַנָּה, וְהַסּוֹפֵר הֵבִין שֶׁכָּל הַמִּלִּים לֹא יִכָּנְסוּ. מָה עָשָׂה? הוּא קִצֵּר אוֹתָן בְּעֶזְרַת רָאשֵׁי תֵּבוֹת. עַל הָאֶבֶן נֶחְרַט:

כָּל אָדָם שֶׁנַּעֲשָׂה לוֹ עָוֶל יִמְשֹׁךְ בַּחֶבֶל, וּבֵיהַ״ד יִשְׁפֹּט בְּצֶדֶק.
ע״פ צַו הַמֶּלֶךְ

אַנְשֵׁי הָעִיר לָמְדוּ מַהֵר לִקְרֹא אֶת הַכְּתֹבֶת: "בֵּיהַ״ד" הוּא בֵּית הַדִּין, וְ"ע״פ" פֵּרוּשׁוֹ עַל פִּי. שָׁנִים רַבּוֹת צִלְצֵל הַפַּעֲמוֹן: עָנִי שֶׁנִּגְנַב מִמֶּנּוּ כֶּבֶשׂ, אַלְמָנָה שֶׁשְּׁכֵנָהּ הֵזִיז אֶת הַגָּדֵר שֶׁלָּהּ, נַעַר שֶׁלֹּא קִבֵּל אֶת שְׂכָרוֹ — כֻּלָּם מָשְׁכוּ בַּחֶבֶל, וְהַשּׁוֹפְטִים בָּאוּ וְשָׁפְטוּ.

אֲבָל עִם הַזְּמַן הַחֶבֶל נִשְׁחַק מֵרֹב הַיָּדַיִם שֶׁמָּשְׁכוּ בּוֹ, עַד שֶׁנִּשְׁאַר מִמֶּנּוּ רַק קֶטַע קָצָר. בָּעִיר לֹא נִמְצָא חֶבֶל אָרֹךְ מַסְפִּיק, וְאֶחָד הַתּוֹשָׁבִים אָמַר: "עַד שֶׁיַּגִּיעַ חֶבֶל חָדָשׁ, נִקְשֹׁר כָּאן עָנָף שֶׁל גֶּפֶן." הוּא קָשַׁר לַפַּעֲמוֹן עָנָף אָרֹךְ וְגָמִישׁ, מָלֵא עָלִים יְרֻקִּים.

בְּאוֹתָהּ עִיר גָּר אַבִּיר זָקֵן. פַּעַם הָיָה אַמִּיץ וְרָכַב עַל סוּסוֹ הַנֶּאֱמָן לַקְּרָבוֹת, אֲבָל בְּזִקְנָתוֹ אָהַב רַק דָּבָר אֶחָד: זָהָב. הוּא מָכַר אֶת כְּלֵי הַנֶּשֶׁק, אֶת הַכְּלָבִים וְאֶת הַשָּׂדוֹת, וְסָפַר אֶת מַטְבְּעוֹתָיו כָּל לַיְלָה.

"לָמָּה לְהַאֲכִיל סוּס זָקֵן, שֶׁכְּבָר אִי אֶפְשָׁר לִרְכֹּב עָלָיו?" אָמַר לְעַצְמוֹ. הוּא גֵּרֵשׁ אֶת הַסּוּס מִן הָאֻרְוָה, וְהַסּוּס הַמִּסְכֵּן נָדַד בָּרְחוֹבוֹת, רָעֵב וְצָמֵא, וְלִחֵךְ עֲשָׂבִים יְבֵשִׁים לְיַד הַגְּדֵרוֹת.

יוֹם אֶחָד, בִּשְׁעַת צָהֳרַיִם חַמָּה, הִגִּיעַ הַסּוּס לַכִּכָּר. הוּא רָאָה אֶת עֲלֵי הַגֶּפֶן הַיְּרֻקִּים, מָתַח אֶת צַוָּארוֹ וְהִתְחִיל לִלְעֹס. וּבְכָל פַּעַם שֶׁמָּשַׁךְ עָלֶה — "דִּין־דּוֹן! דִּין־דּוֹן!" — צִלְצֵל הַפַּעֲמוֹן בְּכָל הָעִיר.

הַשּׁוֹפְטִים מִהֲרוּ לַכִּכָּר. בְּרֹאשָׁם הָלַךְ יוֹ״ר בֵּית הַדִּין — יוֹשֵׁב הָרֹאשׁ, שׁוֹפֵט זָקֵן וְחָכָם. הֵם צִפּוּ לִמְצֹא אָדָם שֶׁנִּפְגַּע, וּמָצְאוּ סוּס כָּל כָּךְ רָזֶה, שֶׁאֶפְשָׁר הָיָה לִסְפֹּר אֶת צַלְעוֹתָיו. "הַסּוּס הַזֶּה בָּא לְבַקֵּשׁ צֶדֶק," אָמַר הַיּוֹ״ר. "מִי בְּעָלָיו?"

הָאַבִּיר הוּבָא לַכִּכָּר. "הַסּוּס הַזֶּה שֵׁרֵת אוֹתְךָ בְּנֶאֱמָנוּת כָּל חַיָּיו," אָמַר הַשּׁוֹפֵט. "הוּא נָשָׂא אוֹתְךָ בַּקְּרָבוֹת וְהִצִּיל אֶת חַיֶּיךָ. וְעַכְשָׁו, כְּשֶׁהִזְדַּקֵּן, אַתָּה מַשְׁאִיר אוֹתוֹ רָעֵב?" הָאַבִּיר הִשְׁפִּיל אֶת עֵינָיו וְלֹא עָנָה.

בְּאוֹתוֹ עֶרֶב כָּתַב סוֹפֵר הָעִיר בְּסֵפֶר בֵּית הַדִּין, בְּעַמּ׳ 112:

יוֹם ג׳. הַתּוֹבֵעַ: סוּס זָקֵן וְרָעֵב. הַנִּתְבָּע: הָאַבִּיר, בְּעָלָיו.
פְּסַק הַדִּין: חֲצִי מֵהַזָּהָב שֶׁל הָאַבִּיר יְשַׁמֵּשׁ לְמָזוֹן, לְמַיִם, לְאֻרְוָה חַמָּה וְכוּ׳ — עַד סוֹף יְמֵי הַסּוּס.
נִכְתַּב ע״י סוֹפֵר בֵּיהַ״ד.

כְּשֶׁשָּׁמַע הַמֶּלֶךְ מָה קָרָה, חִיֵּךְ וְאָמַר: "הַפַּעֲמוֹן שֶׁלִּי מְצַלְצֵל גַּם בִּשְׁבִיל מִי שֶׁאֵינוֹ יָכוֹל לְדַבֵּר. זֶהוּ הַצֶּדֶק הָאֲמִתִּי." וּמֵאָז, כָּךְ מְסַפְּרִים, חַי הַסּוּס הַזָּקֵן בְּאֻרְוָה חַמָּה וְאָכַל שִׁבֹּלֶת שׁוּעָל כָּל בֹּקֶר. וְאַף אֶחָד בָּעִיר לֹא רָצָה לְהַחְלִיף אֶת עֲנַף הַגֶּפֶן בְּחֶבֶל חָדָשׁ.

מָקוֹר: אַגָּדָה אִיטַלְקִית עַתִּיקָה מֵהָעִיר אַטְרִי, שֶׁהִתְפַּרְסְמָה בָּעוֹלָם בִּזְכוּת הַשִּׁיר "The Bell of Atri" שֶׁל הַמְּשׁוֹרֵר הָאֲמֶרִיקָאִי הֶנְרִי לוֹנְגְפֶלוֹ (1863), בְּעִבּוּד חָפְשִׁי. הַמִּלִּים הַמְּסֻמָּנוֹת הֵן רָאשֵׁי תֵּבוֹת וְקִצּוּרִים.

📚 דִּקְדּוּק — רָאשֵׁי תֵּבוֹת וְקִצּוּרִים

מָה פֵּרוּשׁ "בֵּיהַ״ד"? רָאשֵׁי תֵּבוֹת וְקִצּוּרִים

לִפְעָמִים אֵין מָקוֹם (כְּמוֹ עַל הָאֶבֶן הַקְּטַנָּה בַּסִּפּוּר) אוֹ אֵין זְמַן לִכְתֹּב צֵרוּף שָׁלֵם — וְאָז מְקַצְּרִים. בָּעִבְרִית יֵשׁ שְׁתֵּי דְּרָכִים עִקָּרִיּוֹת לְקַצֵּר:

סוּגמָה לוֹקְחִיםאֵיזֶה סִימָןדֻּגְמָאוֹת
רָאשֵׁי תֵּבוֹתאוֹת מִכָּל מִלָּה בַּצֵּרוּףגֵּרְשַׁיִם ״ לִפְנֵי הָאוֹת הָאַחֲרוֹנָהע״פ = עַל פִּי · ע״י = עַל יְדֵי · ת״א = תֵּל אָבִיב
קִצּוּרחֵלֶק מִמִּלָּה אַחַתגֵּרֵשׁ ׳ בַּסּוֹףעַמּ׳ = עַמּוּד · מס׳ = מִסְפָּר · וְכוּ׳ = וְכוּלֵי (וְעוֹד דְּבָרִים כָּאֵלֶּה)

אֵיךְ קוֹרְאִים? אֶת רֹב רָאשֵׁי הַתֵּבוֹת קוֹרְאִים בְּמִלּוּאָם: כּוֹתְבִים ע״י וְקוֹרְאִים "עַל יְדֵי", כּוֹתְבִים יוֹ״ר וְקוֹרְאִים "יוֹשֵׁב רֹאשׁ", כּוֹתְבִים ת״א וְקוֹרְאִים "תֵּל אָבִיב". אֲבָל יֵשׁ רָאשֵׁי תֵּבוֹת שֶׁהָפְכוּ לְמִלָּה שֶׁקּוֹרְאִים כְּמוֹ שֶׁהִיא:

רָאשֵׁי תֵּבוֹתהַצֵּרוּף הַמָּלֵאאֵיךְ קוֹרְאִים
בֵּיהַ״דבֵּית הַדִּיןבְּמִלּוּאָם: "בֵּית הַדִּין"
ע״פעַל פִּיבְּמִלּוּאָם: "עַל פִּי"
צָהַ״לצְבָא הֲגַנָּה לְיִשְׂרָאֵלכְּמִלָּה: "צָהַל"
דּוּ״חַדִּין וְחֶשְׁבּוֹןכְּמִלָּה: "דּוּחַ"
מַנְכָּ״למְנַהֵל כְּלָלִיכְּמִלָּה: "מַנְכָּל"

שִׂימוּ לֵב לְה׳ הַיְּדִיעָה: בְּ"בֵּית הַדִּין" לוֹקְחִים גַּם אֶת הַ־ה׳ — וְלָכֵן כּוֹתְבִים בֵּיהַ״ד, וְכָךְ גַּם בֵּית הַסֵּפֶר ← בֵּיהַ״ס.

גֵּרֵשׁ אַחֲרֵי אוֹת אַחַת יָכוֹל לְסַמֵּן גַּם מִסְפָּר: "יוֹם ג׳" בְּסֵפֶר בֵּית הַדִּין הוּא הַיּוֹם הַשְּׁלִישִׁי בַּשָּׁבוּעַ, וְ"כִּיתָה ה׳" הִיא הַכִּיתָה הַחֲמִישִׁית.

שְׁמוֹנֶה קְעָרוֹת רֵיקוֹת בָּאֻרְוָה הַחֲדָשָׁה
✍️ כְּתִיבָה
✍️ מַטְלַת כְּתִיבָה

כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים. בְּכָל מִשְׁפָּט הִשְׁתַּמְּשׁוּ בְּרָאשֵׁי תֵּבוֹת אוֹ בְּקִצּוּר אֶחָד, וְכִתְבוּ אַחֲרָיו בְּסוֹגְרַיִם אֶת הַצֵּרוּף הַמָּלֵא. שִׂימוּ לֵב: גֵּרְשַׁיִם לִפְנֵי הָאוֹת הָאַחֲרוֹנָה, גֵּרֵשׁ בְּסוֹף קִצּוּר. דֻּגְמָה: הַסּוּס צִלְצֵל בַּפַּעֲמוֹן, וְהַשּׁוֹפְטִים שֶׁל בֵּיהַ״ד (בֵּית הַדִּין) מִהֲרוּ לַכִּכָּר.
