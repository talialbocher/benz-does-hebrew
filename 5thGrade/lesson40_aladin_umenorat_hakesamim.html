📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת, בְּעִיר רְחוֹקָה וַעֲתִיקָה, חַי נַעַר עָנִי בְּשֵׁם אָלָאדִּין. אָבִיו הָאֻמָּן מֵת כְּשֶׁהָיָה יֶלֶד קָטָן, וְאִמּוֹ הִתְפַּרְנְסָה בְּקֹשִׁי מִתְּפִירָה. אָלָאדִּין בִּילָה אֶת רֹב יָמָיו בְּמִשְׂחָקִים בִּרְחוֹבוֹת הַשּׁוּק, וְלֹא חָשַׁב הַרְבֵּה עַל הֶעָתִיד.

יוֹם אֶחָד נִיגַּשׁ אֶל אָלָאדִּין זָר מְבֻגָּר, לָבוּשׁ בְּגָדִים יְקָרִים, וְטָעַן שֶׁהוּא דּוֹדוֹ הָאָבוּד. "אֲנִי אֲחִי אָבִיךָ הַמָּנוֹחַ," אָמַר הָאִישׁ בְּחִיּוּךְ רָחָב, "וְחִיפַּשְׂתִּי אוֹתְךָ שָׁנִים רַבּוֹת!" הָאִישׁ (שֶׁלְּמַעֲשֶׂה לֹא הָיָה קָרוֹב מִשְׁפָּחָה כְּלָל, אֶלָּא קוֹסֵם עַרְמוּמִי שֶׁחִיפֵּשׂ נַעַר תָּמִים שֶׁיַּעֲזֹר לוֹ) הִבְטִיחַ לְאָלָאדִּין בְּגָדִים חֲדָשִׁים וְחַיִּים טוֹבִים יוֹתֵר.

הַקּוֹסֵם לָקַח אֶת אָלָאדִּין לְטִיּוּל אָרֹךְ אֶל מִחוּץ לָעִיר, עַד שֶׁהִגִּיעוּ לָהָר צָחִיחַ. הוּא הִצִּית אֵשׁ קְטַנָּה וְאָמַר מִלּוֹת קֶסֶם, וְהָאֲדָמָה נִפְעֲרָה וְגִילְּתָה כְּנִיסָה לְמְעָרָה חֲשׁוּכָה. "רֵד לַמְּעָרָה," אָמַר הַקּוֹסֵם לְאָלָאדִּין, "וְתָבִיא לִי מִשָּׁם רַק מְנוֹרָה יְשָׁנָה שֶׁתִּמְצָא שָׁם. אַל תִּיגַּע בְּשׁוּם דָּבָר אַחֵר, וְלֹא מְשַׁנֶּה כַּמָּה זָהָב תִּרְאֶה סְבִיבְךָ!"

אָלָאדִּין יָרַד לַמְּעָרָה בִּזְהִירוּת וּמָצָא אוֹתָהּ מְלֵאָה אֲבָנִים יְקָרוֹת וְזָהָב נוֹצֵץ, אֲבָל הוּא נִזְכַּר בְּבַקָּשַׁת הַקּוֹסֵם וְלֹא נָגַע בָּהֶם. בְּמֶרְכַּז הַמְּעָרָה מָצָא מְנוֹרָה יְשָׁנָה וּמֻכְתֶּמֶת, וְלִידָהּ טַבַּעַת פְּשׁוּטָה שֶׁהִנִּיחַ בְּכִיסוֹ בְּלִי לַחְשֹׁב פַּעֲמַיִם. כְּשֶׁחָזַר לְפֶתַח הַמְּעָרָה עִם הַמְּנוֹרָה, בִּיקֵּשׁ הַקּוֹסֵם שֶׁיּוֹשִׁיט לוֹ אוֹתָהּ מִיָּד - אֲבָל אָלָאדִּין סֵירֵב, וּבִיקֵּשׁ קֹדֶם לָצֵאת בְּבִטְחָה.

הַקּוֹסֵם הִתְרַגֵּז מְאֹד, מִלְמֵל מִלּוֹת קֶסֶם זוֹעֲפוֹת, וְסָגַר אֶת פֶּתַח הַמְּעָרָה בְּאֶבֶן כְּבֵדָה, עִם אָלָאדִּין לָכוּד בִּפְנִים בָּאֲפֵלָה. בְּיֵאוּשׁוֹ, אָלָאדִּין שִׁפְשֵׁף בְּלִי כַּוָּנָה אֶת הַטַּבַּעַת שֶׁבְּיָדוֹ - וּלְפֶתַע עָלָה עֲנַן עָשָׁן, וּמִתּוֹכוֹ הוֹפִיעַ יְצוּר עֲנָק. "אֲנִי עֶבֶד הַטַּבַּעַת," הִכְרִיז הַיְּצוּר בְּקוֹל רוֹעֵם, "מָה תִּרְצֶה מִמֶּנִּי!"

אָלָאדִּין בִּיקֵּשׁ לַחֲזֹר הַבַּיְתָה בְּשָׁלוֹם, וְהַיְּצוּר הֶעֱבִיר אוֹתוֹ מִיָּד אֶל בֵּית אִמּוֹ, כְּשֶׁהַמְּנוֹרָה עֲדַיִן בְּיָדוֹ. כְּשֶׁהָאֵם נִיסְּתָה לְנַקּוֹת אֶת הַמְּנוֹרָה הַמֻּכְתֶּמֶת כְּדֵי לִמְכֹּר אוֹתָהּ, שִׁפְשְׁפָה אוֹתָהּ בְּמַטְלִית - וַעֲנַן עָשָׁן נוֹסָף עָלָה מִמֶּנָּה, גָּדוֹל הַרְבֵּה יוֹתֵר מֵהָרִאשׁוֹן. "אֲנִי עֶבֶד הַמְּנוֹרָה," אָמַר הָעֲנָק הֶחָדָשׁ, "כָּל מַה שֶּׁתְּבַקְשִׁי - אֶעֱשֶׂה בִּשְׁבִילֵךְ."

מֵאוֹתוֹ הָרֶגַע הִשְׁתַּנּוּ הַחַיִּים שֶׁל אָלָאדִּין וְאִמּוֹ. הַגֵּ'נִי הֵבִיא לָהֶם אֲרוּחוֹת עֲשִׁירוֹת, בְּגָדִים מְפוֹאָרִים וּבַסּוֹף אֲפִלּוּ אַרְמוֹן מְפוֹאָר, שֶׁקָּם בְּלֵיל אֶחָד לְיַד אַרְמוֹן הַמֶּלֶךְ. אָלָאדִּין נָשָׂא לְאִשָּׁה אֶת בַּת הַמֶּלֶךְ, וְהַשְּׁנַיִם חָיוּ בְּאֹשֶׁר בָּאַרְמוֹן הֶחָדָשׁ שֶׁלָּהֶם.

אֲבָל הַקּוֹסֵם הָעַרְמוּמִי לֹא שָׁכַח אֶת הַמְּנוֹרָה שֶׁאָבְדָה לוֹ, וְגִילָּה שֶׁאָלָאדִּין עוֹדֶנּוּ חַי וּמַצְלִיחַ. הוּא הִתְחַפֵּשׂ לְרוֹכֵל זָקֵן וְהִסְתּוֹבֵב בָּרְחוֹבוֹת וְקָרָא בְּקוֹל: "מְנוֹרוֹת יְשָׁנוֹת בִּתְמוּרָה לַחֲדָשׁוֹת! מִי מוּכָן לְהַחְלִיף?" בַּת הַמֶּלֶךְ, שֶׁלֹּא יָדְעָה דָּבָר עַל כּוֹחָהּ הַסּוֹדִי שֶׁל הַמְּנוֹרָה הַיְּשָׁנָה, נָתְנָה לָרוֹכֵל אֶת מְנוֹרַת אָלָאדִּין בְּלִי לַחְשֹׁב פַּעֲמַיִם.

בְּרֶגַע שֶׁהַמְּנוֹרָה הִגִּיעָה לִידֵי הַקּוֹסֵם, הוּא שִׁפְשֵׁף אוֹתָהּ וּפָקַד עַל עֶבֶד הַמְּנוֹרָה לְהַעֲבִיר אֶת הָאַרְמוֹן כֻּלּוֹ, עִם בַּת הַמֶּלֶךְ בְּתוֹכוֹ, לִמְדִינָה רְחוֹקָה. כְּשֶׁאָלָאדִּין חָזַר וְגִילָּה שֶׁהָאַרְמוֹן וְאִשְׁתּוֹ נֶעֶלְמוּ, הוּא הֵבִין מִיָּד מִי עוֹמֵד מֵאֲחוֹרֵי כָּךְ.

אָלָאדִּין שִׁפְשֵׁף אֶת הַטַּבַּעַת שֶׁעֲדַיִן נִשְׁאֲרָה אֶצְלוֹ, וּבִיקֵּשׁ מֵעֶבֶד הַטַּבַּעַת לְהָבִיא אוֹתוֹ אֶל אִשְׁתּוֹ. יַחַד תִּכְנְנוּ הַשְּׁנַיִם תָּכְנִית: בַּת הַמֶּלֶךְ הִרְדִּימָה אֶת הַקּוֹסֵם בְּסַם, וְאָלָאדִּין תָּפַס בַּחֲזָרָה אֶת הַמְּנוֹרָה. הוּא שִׁפְשֵׁף אוֹתָהּ מִיָּד וּבִיקֵּשׁ מֵעֶבֶד הַמְּנוֹרָה לְהַחְזִיר אֶת הָאַרְמוֹן לִמְקוֹמוֹ וּלְהַעֲנִישׁ אֶת הַקּוֹסֵם עַל מַעֲשָׂיו. מֵאָז חָיוּ אָלָאדִּין וּבַת הַמֶּלֶךְ בְּאֹשֶׁר וּבְשַׁלְוָה, וְזָכְרוּ תָּמִיד לִשְׁמֹר הֵיטֵב עַל מַה שֶּׁיָּקָר לָהֶם.

תזכורת: מירכאות (" ") לציטוט מדויק

למדנו בשיעור 23 שמירכאות מסמנות מילים מדויקות שמישהו אמר, מילה במילה. לפני הציטוט מופיעות בדרך כלל נקודתיים (:). לדוגמה: היצור הכריז: "אני עבד הטבעת, מה תרצה ממני!" שימו לב: כיוון שסימן הקריאה הוא חלק ממה שהיצור באמת צעק, הוא נשאר בתוך המירכאות, לפני הסימן הסוגר.

תזכורת: סוגריים ( ) למידע נוסף

סוגריים מוסיפים למשפט הסבר או מידע נוסף, שאינו הכרחי - אפשר להוריד אותו והמשפט העיקרי עדיין יובן. לדוגמה: האיש (שלמעשה לא היה קרוב משפחה כלל) הבטיח לאלאדין חיים טובים.

כלל חדש: שורה חדשה לכל דובר בדו-שיח

כשכותבים שיחה בין שתי דמויות או יותר, נהוג להתחיל שורה או פסקה חדשה בכל פעם שדמות אחרת מתחילה לדבר. כך הקורא יודע בקלות מי אומר מה, בלי להתבלבל.
