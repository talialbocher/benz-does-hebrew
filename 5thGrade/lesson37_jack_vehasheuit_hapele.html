📖 קְרִיאָה

💡 לְחַץ עַל הַמִּלִּים הַמֻּדְגָּשׁוֹת כְּדֵי לִרְאוֹת מָה הֵן אוֹמְרוֹת
פֶּרֶק א׳ · הַהַחְלָפָה הַמּוּזָרָה

פַּעַם, בִּכְפָר קָטָן, גָּרוּ גֵּ'ק וְאִמּוֹ בְּבַיִת עָנִי, וּפָרָה אַחַת הָיְתָה כָּל מַה שֶּׁנִּשְׁאַר לָהֶם בָּעוֹלָם.

יוֹם אֶחָד בִּיקְּשָׁה הָאֵם מִגֵּ'ק לִמְכֹּר אֶת הַפָּרָה בַּשּׁוּק וּלְהָבִיא הַבַּיְתָה כֶּסֶף לְאֹכֶל.

בַּדֶּרֶךְ פָּגַשׁ גֵּ'ק זָקֵן מוּזָר, שֶׁהִצִּיעַ לוֹ לְהַחְלִיף אֶת הַפָּרָה בִּשְׂקִית קְטַנָּה שֶׁל שְׁעוּעִית קְסוּמָה. "זוֹהִי שְׁעוּעִית פֶּלֶא," לָחַשׁ הַזָּקֵן, "הִיא תִּצְמַח עַד הַשָּׁמַיִם עוֹד לִפְנֵי הַבֹּקֶר."

גֵּ'ק, סַקְרָן וְתָמִים, הִסְכִּים מִיָּד לְהַחְלָפָה וְחָזַר הַבַּיְתָה בְּגַאֲוָה - עַד שֶׁאִמּוֹ רָאֲתָה אֶת הַשְּׁעוּעִית וְכָעֲסָה מְאֹד. הִיא זָרְקָה אֶת הַגַּרְגִּירִים מִבְּעַד לַחַלּוֹן, וְשָׁלְחָה אֶת גֵּ'ק לִישֹׁן בְּלִי אֲרוּחַת עֶרֶב.

פֶּרֶק ב׳ · הַגִּבְעוֹל הָעֲנָק

לְמָחֳרָת בַּבֹּקֶר הִתְעוֹרֵר גֵּ'ק וְגִילָּה שֶׁגִּבְעוֹל עֲנָק צָמַח בֶּחָצֵר בְּמַהֲלַךְ הַלַּיְלָה, גָּבוֹהַּ כָּל כָּךְ שֶׁרֹאשׁוֹ נֶעֱלַם בֵּין הֶעָנָנִים.

בְּלִי לַחְשֹׁב פַּעֲמַיִם, טִיפֵּס גֵּ'ק עַל הַגִּבְעוֹל שָׁעָה אַחַר שָׁעָה, עַד שֶׁהִגִּיעַ אֶל פִּסְגָּה מוּזָרָה - עוֹלָם חָדָשׁ שֶׁהִתְפָּרֵשׂ מֵעַל הֶעָנָנִים.

שָׁם, בִּקְצֵה שְׁבִיל אֶבֶן, עָמְדָה טִירָה עֲנָקִית. גֵּ'ק נִכְנַס בְּשֶׁקֶט, וּלְפֶתַע שָׁמַע קוֹל רַעַם: "אֲנִי מֵרִיחַ דַּם שֶׁל בֶּן אָדָם!" זֶה הָיָה קוֹלוֹ שֶׁל עֲנָק שֶׁגָּר בַּטִּירָה עִם אִשְׁתּוֹ.

אֵשֶׁת הָעֲנָק, שֶׁרִיחֲמָה עַל הַיֶּלֶד הָרָעֵב, הֶחְבִּיאָה אֶת גֵּ'ק בְּתַנּוּר גָּדוֹל, וּמִשָּׁם הוּא רָאָה אֶת הָעֲנָק סוֹפֵר אֶת אוֹצַר הַזָּהָב שֶׁלּוֹ לִפְנֵי שֶׁנִּרְדַּם.

פֶּרֶק ג׳ · הָאוֹצָר

גֵּ'ק הִמְתִּין בְּסַבְלָנוּת, חָטַף שְׂקִית אַחַת מְלֵאָה מַטְבְּעוֹת זָהָב, וְגָלַשׁ בִּזְהִירוּת בְּמוֹרַד הַגִּבְעוֹל חֲזָרָה הַבַּיְתָה.

בַּפַּעַם הַשְּׁנִיָּה טִיפֵּס גֵּ'ק שׁוּב, וְגִילָּה בַּטִּירָה תַּרְנְגֹלֶת שֶׁמַּטִּילָה בֵּיצַת זָהָב בְּכָל בֹּקֶר. הוּא תָּפַס אוֹתָהּ בִּזְרִיזוּת וּבָרַח שׁוּב בְּמוֹרַד הַגִּבְעוֹל.

פֶּרֶק ד׳ · הַנֵּבֶל וְהַבְּרִיחָה

בַּפַּעַם הַשְּׁלִישִׁית לָקַח גֵּ'ק אֶת הַפְּרִיט הַיָּקָר בְּיוֹתֵר בַּטִּירָה - נֵבֶל קָסוּם שֶׁמְּנַגֵּן לְבַדּוֹ בְּלִי יַד אָדָם - אַךְ הַפַּעַם הִתְעוֹרֵר הָעֲנָק וְרָדַף אַחֲרָיו בְּמוֹרַד הַגִּבְעוֹל.

גֵּ'ק הִגִּיעַ רִאשׁוֹן לַקַּרְקַע, תָּפַס גַּרְזֶן וְכָרַת אֶת הַגִּבְעוֹל בְּכָל כֹּחוֹ. הָעֲנָק נָפַל אַרְצָה עִם קוֹל רַעַשׁ אַדִּיר, וְגֵ'ק וְאִמּוֹ חָיוּ מְאֻשָּׁרִים וַעֲשִׁירִים עַד סוֹף יְמֵיהֶם.

סִפּוּר עַם מֻכָּר מֵאַנְגְּלִיָּה - "גֵּ'ק וּשְׁעוּעִית הַפֶּלֶא" - מְעֻבָּד

📚 דִּקְדּוּקמִלִּים לוֹעֲזִיּוֹת וּמַקְבִּילוֹתֵיהֶן הָעִבְרִיּוֹת (חֲזָרָה)

תזכורת: מהי מילה לועזית?

מילה לועזית היא מילה שהגיעה לעברית משפה אחרת - יוונית, לטינית, ערבית, אנגלית, צרפתית, רוסית ועוד (למדנו לראשונה בשיעור 6). לחלק מהמילים הלועזיות יש היום מילה עברית תקנית שמחליפה אותן, ולחלקן אין תחליף מקובל, והן פשוט הפכו לחלק מהעברית המדוברת.
