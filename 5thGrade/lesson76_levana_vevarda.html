📖 קְרִיאָה

בְּבַיִת קָטָן בִּקְצֵה הַיַּעַר גָּרוּ אֵם אַלְמָנָה וּשְׁתֵּי בְּנוֹתֶיהָ, לְבָנָה וְוַרְדָּה. לִפְנֵי הַבַּיִת צָמְחוּ שְׁנֵי שִׂיחֵי וֶרֶד: אֶחָד עִם וְרָדִים לְבָנִים, וְהַשֵּׁנִי עִם וְרָדִים אֲדֻמִּים. לְבָנָה, שֶׁהָיְתָה שְׁקֵטָה וּמְתוּנָה, אָהֲבָה לָשֶׁבֶת לְיַד שִׂיחַ הַוְּרָדִים הַלְּבָנִים; וַרְדָּה, שֶׁהָיְתָה עַלִּיזָה וְאָהֲבָה לָרוּץ בַּיַּעַר, יָשְׁבָה תָּמִיד לְיַד שִׂיחַ הַוְּרָדִים הָאֲדֻמִּים. שְׁתֵּיהֶן אָהֲבוּ זוֹ אֶת זוֹ מְאֹד, וְכִמְעַט לֹא רָבוּ אַף פַּעַם.

בְּעֶרֶב חֹרֶף אֶחָד, כְּשֶׁהָרוּחַ יִלְּלָה בַּחוּץ וְהַשֶּׁלֶג יָרַד בְּכֹבֶד, נִשְׁמְעָה נְקִישָׁה בַּדֶּלֶת. וַרְדָּה פָּתְחָה, וּלְמוּלָהּ עָמַד דֹּב עֲנָקִי, שָׁחֹר וּמְכֻסֶּה שֶׁלֶג. הִיא צָעֲקָה בְּפַחַד וְנָסוֹגָה אֲחוֹרַנִּית, אַךְ הַדֹּב אָמַר בְּקוֹל רַךְ: "אַל תִּפְחֲדוּ מִמֶּנִּי. אֲנִי רַק רוֹצֶה לְהִתְחַמֵּם מְעַט לְיַד הָאֵשׁ שֶׁלָּכֶן, וְלֹא אֶפְגַּע בְּאִישׁ." לַמְרוֹת שֶׁפָּחֲדוּ בַּהַתְחָלָה, הָאֵם וְהַבָּנוֹת הִזְמִינוּ אוֹתוֹ פְּנִימָה - כִּי הֵבִינוּ מִיָּד שֶׁקּוֹלוֹ שֶׁל הַדֹּב עָדִין וְלֹא מְאַיֵּם.

הַדֹּב יָשַׁב לְיַד הָאֵח כָּל הָעֶרֶב, וּלְבָנָה וְוַרְדָּה הֵסִירוּ מֵעַל גַּבּוֹ אֶת שְׁאֵרִית הַשֶּׁלֶג בְּמַקֵּל עֵץ, בְּעֵת שֶׁהוּא נֶהֱנֶה וְנוֹחֵר בְּשֶׁקֶט. מֵאוֹתוֹ עֶרֶב וָהָלְאָה חָזַר הַדֹּב כִּמְעַט בְּכָל עֶרֶב חֹרֶף, וְהַמִּשְׁפָּחָה חִכְּתָה לוֹ בְּשִׂמְחָה. הַבָּנוֹת שִׂחֲקוּ עִמּוֹ וְהִתְגַּלְגְּלוּ עַל גַּבּוֹ הָרָחָב, וְאַף לֹא פָּחֲדוּ מִמֶּנּוּ יוֹתֵר. בְּנוֹסָף, לִמֵּד אוֹתָן הַדֹּב סִפּוּרִים עַל הַיַּעַר הָרָחוֹק, וְהַלֵּילוֹת הַחֲשׁוּכִים חָלְפוּ מַהֵר יוֹתֵר בִּזְכוּתוֹ.

כְּשֶׁהִגִּיעַ הָאָבִיב וְהַשֶּׁלֶג נָמַס, אָמַר הַדֹּב לַמִּשְׁפָּחָה שֶׁעָלָיו לַעֲזֹב לַיַּעַר. "בְּעוֹנַת הַקֶּרַח הָיָה קַל לִי לְהִשָּׁאֵר אֶצְלְכֶן," הִסְבִּיר, "כִּי הָאֲדָמָה קְפוּאָה וְאִישׁ לֹא יָכוֹל לַחְפֹּר בָּהּ. אֲבָל עַכְשָׁו, כְּשֶׁהָאֲדָמָה רַכָּה, גַּמָּדִים רָעִים עָלוּלִים לִגְנֹב אֶת הָאוֹצָר שֶׁאֲנִי שׁוֹמֵר עָלָיו." בְּצֵאתוֹ מִבַּעַד לַדֶּלֶת נִתְפְּסָה פִּסַּת פַּרְוָתוֹ בְּמַסְגֶּרֶת הַדֶּלֶת, וּלְבָנָה הָיְתָה בְּטוּחָה שֶׁרָאֲתָה נִצְנוּץ שֶׁל זָהָב מִתַּחַת לַפַּרְוָה הַשְּׁחוֹרָה. אַךְ הַדֹּב מִהֵר וְנֶעֱלַם בֵּין הָעֵצִים, וְהִיא לֹא הִסְפִּיקָה לִשְׁאֹל.

בְּקַיִץ יָצְאוּ לְבָנָה וְוַרְדָּה לֶאֱסֹף עֵצִים לְמַדּוּרָה. בַּדֶּרֶךְ מָצְאוּ עֵץ כָּרוּת שֶׁבְּתוֹךְ סִדְקוֹ הָיָה תָּקוּעַ גַּמָּד קָטָן: זְקָנוֹ הַלָּבָן הָאָרֹךְ נִלְכַּד בֵּין שְׁנֵי חֶלְקֵי הַגֶּזַע, וְהוּא קָפַץ מִצַּד לְצַד כְּמוֹ כֶּלֶב קָשׁוּר בְּחֶבֶל. "מָה אַתֶּן עוֹמְדוֹת שָׁם?!" צָעַק הַגַּמָּד בְּזַעַם. "בּוֹאוּ עִזְרוּ לִי מִיָּד, בִּמְקוֹם לְהִסְתַּכֵּל!" וַרְדָּה מִהֲרָה לְהָבִיא מִסְפָּרַיִם קְטַנִּים, וּבִזְהִירוּת גָּזְרָה אֶת קְצֵה הַזָּקָן הַתָּקוּעַ, וְשִׁחְרְרָה אֶת הַגַּמָּד. לַמְרוֹת שֶׁהֵן הִצִּילוּ אוֹתוֹ, הַגַּמָּד לֹא אָמַר מִלַּת תּוֹדָה אַחַת: הוּא חָטַף שַׂק כֶּסֶף שֶׁהִסְתַּתֵּר בֵּין שָׁרְשֵׁי הָעֵץ, נָזַף בָּהֶן עַל שֶׁקָּצְצוּ לוֹ מִזְּקָנוֹ הַיָּפֶה, וְנֶעֱלַם בַּסְּבַךְ בְּלִי לְהַבִּיט אֲחוֹרָה.

שָׁבוּעַ אַחַר כָּךְ יָשְׁבוּ הָאֲחָיוֹת עַל גְּדַת נָחָל, וְרָאוּ אֶת אוֹתוֹ הַגַּמָּד שׁוּב - הַפַּעַם זְקָנוֹ הִסְתַּבֵּךְ בְּחוּט דַּיִג, וְדָג גָּדוֹל שֶׁנִּתְפַּס בַּקָּרֶס מָשַׁךְ אוֹתוֹ בְּחָזְקָה לְעֵבֶר הַמַּיִם. "תַּעֲשׂוּ מַשֶּׁהוּ, אֲוִילוֹת שֶׁכָּמוֹתָן!" צָעַק. שׁוּב גָּזְרָה וַרְדָּה בִּזְהִירוּת אֶת קְצֵה הַזָּקָן, וְהַגַּמָּד נִשְׁמַט חָפְשִׁי מֵהַחוּט. אַף עַל פִּי שֶׁהַצָּלַת חַיָּיו הָיְתָה בִּזְכוּתָן מַמָּשׁ, הוֹצִיא הַגַּמָּד מִתּוֹךְ כִּיסוֹ שַׂק קָטָן שֶׁל פְּנִינִים, מִלְמֵל שֶׁקִּלְקְלוּ לוֹ סוֹפִית אֶת הַזָּקָן הַמְּפֹאָר, וְרָץ מֵהֶן בְּלִי לְהוֹדוֹת.

בְּפַעַם הַשְּׁלִישִׁית מָצְאוּ הָאֲחָיוֹת אֶת הַגַּמָּד כְּשֶׁנֶשֶׁר עָנָק תָּפַס אוֹתוֹ בְּצִפָּרְנָיו וְנִסָּה לַהֲרִימוֹ אֶל הַשָּׁמַיִם. הַבָּנוֹת תָּפְסוּ אֶת רַגְלָיו וּמָשְׁכוּ בְּכָל כֹּחָן, עַד שֶׁהַנֶּשֶׁר וִתֵּר וְעָף בְּלִי טַרְפּוֹ. הַנֶּשֶׁר רַק בִּצֵּעַ אֶת טִבְעוֹ כְּצִפּוֹר טוֹרֶפֶת. הַגַּמָּד, לְעֻמַּת זֹאת, הָיָה מְסֻכָּן מִסִּבָּה אַחֶרֶת לְגַמְרֵי: גַּם כְּשֶׁהִצִּילוּ אֶת חַיָּיו מַמָּשׁ, הוּא לֹא טָרַח לְהַגִּיד תּוֹדָה - רַק חָטַף שַׂק שֶׁל אֲבָנִים יְקָרוֹת שֶׁנָּפַל מֵעֻקְפּוֹ, נָזַף בָּהֶן עַל שֶׁקָּרְעוּ אֶת בִּגְדוֹ, וְנֶעֱלַם כְּדַרְכּוֹ. פַּעַם אַחַר פַּעַם הוֹכִיחַ הַגַּמָּד שֶׁהוּא כָּפוּי טוֹבָה בְּמֵלוֹא מוּבַן הַמִּלָּה.

פַּעַם רְבִיעִית, בְּסוֹף הַקַּיִץ, יָצְאוּ הָאֲחָיוֹת לָעִיר לִקְנוֹת חוּטִים וּמַחֲטִים, וּבַדֶּרֶךְ בַּיַּעַר מָצְאוּ אֶת הַגַּמָּד יוֹשֵׁב לְיַד עֲרֵמַת אוֹצָר נוֹצֵץ שֶׁפָּרַשׂ עַל הָאָרֶץ: אֲבָנִים יְקָרוֹת, זָהָב וְכֶסֶף שֶׁצָּבַר לְאֹרֶךְ הַשָּׁנִים. הוּא לֹא שָׂם לֵב לִשְׁאָגָה שֶׁהִתְקָרְבָה מִבֵּין הָעֵצִים - עַד שֶׁדֹּב שָׁחֹר קָפַץ מֵאֲחוֹרָיו. "רַחֵם עָלַי!" צָעַק הַגַּמָּד בְּאֵימָה, "קַח אֶת כָּל הָאוֹצָר, רַק אַל תִּפְגַּע בִּי!" אֲבָל הַדֹּב לֹא רָצָה בָּאוֹצָר: בְּמַכָּה אַחַת שֶׁל כַּפּוֹ הוּא הִפְחִיד אֶת הַגַּמָּד כָּל כָּךְ, שֶׁזֶּה בָּרַח בִּצְעָקָה אֶל תּוֹךְ הַסְּבַךְ - וְלֹא נִרְאָה עוֹד בַּיַּעַר הַהוּא לְעוֹלָם.

בְּאוֹתוֹ רֶגַע מַמָּשׁ נָשְׁרָה פַּרְוַת הַדֹּב מֵעָלָיו כְּמוֹ מְעִיל יָשָׁן, וְלִפְנֵי הָאֲחָיוֹת הַנִּדְהָמוֹת עָמַד נָסִיךְ צָעִיר, לָבוּשׁ בֶּגֶד זָהָב. "הַגַּמָּד הָרַע כִּשֵּׁף אוֹתִי לִהְיוֹת דֹּב," הִסְבִּיר הַנָּסִיךְ, "וְגָנַב מֵאָחִי אֶת מַמְלַכְתּוֹ וְאֶת אוֹצָרוֹ - הָאוֹצָר עַצְמוֹ שֶׁפָּרוּשׂ כָּאן עַל הָאֲדָמָה. רַק כְּשֶׁהַגַּמָּד יִבְרַח מִפַּחַד מֵהַיַּעַר הַזֶּה לְעוֹלָמִים, יִשָּׁבֵר הַקֶּסֶם - וְזֶה בְּדִיּוּק מָה שֶׁקָּרָה הָרֶגַע, בִּזְכוּתְכֶן." לַמְרוֹת שֶׁהַנָּסִיךְ הָיָה זָר לָהֶן, לְבָנָה וְוַרְדָּה זָכְרוּ מִיָּד אֶת קוֹלוֹ הָרַךְ שֶׁל הַדֹּב, וְיָדְעוּ שֶׁזֶּה אוֹתוֹ חָבֵר בְּדִיּוּק.

הַנָּסִיךְ קָרָא לְאָחִיו, שֶׁחָזַר מִיָּד לְאַרְמוֹן מַמְלַכְתּוֹ הַגְּזוּלָה בְּעֶזְרַת הָאוֹצָר שֶׁהֻחְזַר. בְּנוֹסָף לְהַחְזָרַת הַמַּמְלָכָה, הֵבִיא הַנָּסִיךְ אֶת הָאֵם וְאֶת שְׁתֵּי הַבָּנוֹת לָגוּר בְּקִרְבַת הָאַרְמוֹן. לְבָנָה נִשְׂאָה לַנָּסִיךְ שֶׁהָיָה פַּעַם הַדֹּב, וְוַרְדָּה נִשְׂאָה לְאָחִיו. אֶת שְׁנֵי שִׂיחֵי הַוְּרָדִים - הַלָּבָן וְהָאָדֹם - שָׁתְלוּ מֵחָדָשׁ בְּגַן הָאַרְמוֹן, וְהֵם פָּרְחוּ שָׁם שָׁנָה אַחַר שָׁנָה, בְּדִיּוּק כְּמוֹ שֶׁפָּרְחוּ לִפְנֵי הַבַּיִת הַקָּטָן בִּקְצֵה הַיַּעַר.

מְסֻפָּר עַל־פִּי "שֶׁלֶג-לָבָן וְוֶרֶד-אָדֹם" (Schneeweißchen und Rosenrot), אַגָּדַת עַם מוּכֶּרֶת שֶׁל הָאַחִים גְּרִים (גֶּרְמַנְיָה, פֻּרְסְמָה לָרִאשׁוֹנָה בְּ־1837), מְעֻבֶּדֶת וּמְסֻפֶּרֶת מֵחָדָשׁ (בָּעִבְרִית נִקְרְאוּ הָאֲחָיוֹת לְבָנָה וְוַרְדָּה; בַּגִּרְסָה הַמְּקוֹרִית הַגַּמָּד נֶהֱרָג בְּסוֹף הַסִּפּוּר בִּידֵי הַדֹּב, וְכָאן רֻכַּךְ הַסּוֹף - הַגַּמָּד רַק בּוֹרֵחַ בְּפַחַד וְלֹא נִרְאֶה עוֹד).

כְּבָר הִכַּרְנוּ מִלּוֹת קִשּׁוּר פְּשׁוּטוֹת כְּמוֹ "אֲבָל" וְ"כִּי". הַיּוֹם נַכִּיר שְׁתֵּי מִלִּים חֲזָקוֹת יוֹתֵר שֶׁמַּבִּיעוֹת נִגּוּד בְּתוֹךְ מִשְׁפָּט אֶחָד: "לַמְרוֹת שֶׁ-" וְ"אַף עַל פִּי שֶׁ-" (בְּעֶרֶךְ אוֹתָהּ מַשְׁמָעוּת). שְׁתֵּיהֶן חַיָּבוֹת לְהוֹפִיעַ עִם מִלַּת הַזִּיקָה "שֶׁ-" מִיָּד אַחֲרֵיהֶן (כְּמוֹ שֶׁלָּמַדְנוּ עַל "שֶׁ-" בְּשִׁיעוּר 72), וְהֵן מְצַיְּנוֹת שֶׁמַּשֶּׁהוּ קָרָה הֵפֶךְ מִמַּה שֶּׁהָיִינוּ מְצַפִּים לוֹ.
