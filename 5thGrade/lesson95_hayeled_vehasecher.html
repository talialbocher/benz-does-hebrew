📖 קְרִיאָה

הָיָה פַּעַם יֶלֶד הוֹלַנְדִּי בֶּן שְׁמוֹנֶה בְּשֵׁם פֶּטֶר. הוּא גָּר בִּכְפָר קָטָן עַל שְׂפַת הַיָּם, בְּאֶרֶץ שֶׁחֲלָקִים גְּדוֹלִים מִמֶּנָּה נְמוּכִים יוֹתֵר מִן הַיָּם. מָה מֵגֵן עַל הַכְּפָר מִפְּנֵי הַגַּלִּים? סֶכֶר אָרֹךְ וְחָזָק, שֶׁנִּבְנָה מֵאֲבָנִים וּמֵעָפָר.

כָּל תּוֹשְׁבֵי הַכְּפָר יָדְעוּ שֶׁהַסֶּכֶר חָשׁוּב מְאֹד. בְּכָל בֹּקֶר עָבְרוּ שׁוֹמְרִים לְאָרְכּוֹ וּבָדְקוּ אוֹתוֹ. אִם מָצְאוּ סֶדֶק קָטָן, הֵם תִּקְּנוּ אוֹתוֹ מִיָּד, כִּי יָדְעוּ שֶׁסֶּדֶק קָטָן עָלוּל לְהִתְפַּתֵּחַ לְהַצָּפָה גְּדוֹלָה. אַבָּא שֶׁל פֶּטֶר הָיָה אֶחָד מֵהַשּׁוֹמְרִים.

בְּאֶחָד מֵעַרְבֵי הַסְּתָו שָׁלְחָה אִמָּא אֶת פֶּטֶר לְבַקֵּר זָקֵן חוֹלֶה שֶׁגָּר מֵעֵבֶר לַסֶּכֶר, וְנָתְנָה לוֹ סַל עוּגִיּוֹת. פֶּטֶר הָלַךְ בְּשִׂמְחָה וְשָׁהָה אֵצֶל הַזָּקֵן זְמַן רַב. כְּשֶׁיָּצָא לַחֲזֹר הַבַּיְתָה, כְּבָר יָרַד הַלַּיְלָה.

בַּדֶּרֶךְ חָזְרָה שָׁמַע פֶּטֶר קוֹל מוּזָר: טִפְטוּף מַיִם. הוּא הִתְקָרֵב וְהִבִּיט בַּסֶּכֶר, וְרָאָה נֶקֶב זָעִיר, וּמִמֶּנּוּ זָרַם פֶּלֶג דַּק שֶׁל מַיִם.

פֶּטֶר הֵבִין מִיָּד מָה יִקְרֶה: הַמַּיִם יַרְחִיבוּ אֶת הַנֶּקֶב מִשְּׁנִיָּה לִשְׁנִיָּה, וּבַלַּיְלָה יִשְׁטְפוּ אֶת הַכְּפָר כֻּלּוֹ. הוּא חָשַׁב לָרוּץ וְלִקְרֹא לַשּׁוֹמְרִים, אֲבָל פָּחַד שֶׁעַד שֶׁיַּחֲזֹר, הַנֶּקֶב כְּבָר יִגְדַּל. לָכֵן עָשָׂה דָּבָר אֶחָד: הוּא תָּקַע אֶת אֶצְבָּעוֹ בַּנֶּקֶב, וְהַמַּיִם נֶעֶצְרוּ.

הַלַּיְלָה הָיָה אָרֹךְ וְקַר. הָרוּחַ נָשְׁבָה, הַגֶּשֶׁם יָרַד, וְאֶצְבְּעוֹתָיו שֶׁל פֶּטֶר קָפְאוּ מִקֹּר. הוּא צָעַק לְעֶזְרָה, אֲבָל אִישׁ לֹא שָׁמַע אוֹתוֹ. פֶּטֶר חָשַׁב עַל אִמָּא וְאַבָּא, וְעַל כָּל יַלְדֵי הַכְּפָר שֶׁיְּשֵׁנִים בְּמִטּוֹתֵיהֶם, וְלֹא זָז.

בַּבֹּקֶר עָבַר שָׁם אִישׁ זָקֵן וְשָׁמַע אֶת קוֹלוֹ שֶׁל פֶּטֶר. הוּא רָץ לִקְרֹא לְתוֹשְׁבֵי הַכְּפָר, וְיַחַד תִּקְּנוּ אֶת הַנֶּקֶב. כָּל הַתּוֹשָׁבִים הוֹדוּ לְפֶטֶר וְקָרְאוּ לוֹ גִּבּוֹר. אַבָּא חִבֵּק אוֹתוֹ בְּחֹם, וּפֶטֶר חִיֵּךְ חִיּוּךְ עָיֵף. מֵאָז זָכְרוּ כֻּלָּם שֶׁגַּם יֶלֶד קָטָן יָכוֹל לְהַצִּיל כְּפָר שָׁלֵם.

הַסִּפּוּר מְבֻסָּס עַל הַסִּפּוּר עַל הַיֶּלֶד וְהַסֶּכֶר מִתּוֹךְ הַסֵּפֶר "הַנְס בְּרִינְקֶר, אוֹ מַחֲלִיקֵי הַכֶּסֶף" מֵאֵת מֶרִי מֵייפְּס דּוֹדְג' (אַרְצוֹת הַבְּרִית, 1865), מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ.

📚 דִּקְדּוּק — פִּסְקָה

מִשְׁפָּטִים שֶׁעוֹבְדִים יַחַד: מָה זֹאת פִּסְקָה?

פִּסְקָה הִיא קְבוּצָה שֶׁל מִשְׁפָּטִים הָעוֹסְקִים בְּרַעְיוֹן אֶחָד. כְּשֶׁעוֹבְרִים לְרַעְיוֹן חָדָשׁ, מַתְחִילִים פִּסְקָה חֲדָשָׁה בְּשׁוּרָה חֲדָשָׁה. בַּסִּפּוּר שֶׁקְּרָאתֶם יֵשׁ שֶׁבַע פִּסְקָאוֹת, וּבְכָל אַחַת מֵהֶן רַעְיוֹן אַחֵר.

פִּסְקָה טוֹבָה בְּנוּיָה מִשְּׁלוֹשָׁה חֲלָקִים:

הַחֵלֶקמָה תַּפְקִידוֹ?דֻּגְמָה
מִשְׁפָּט פְּתִיחָהאוֹמֵר עַל מָה הַפִּסְקָההַסֶּכֶר חָשׁוּב מְאֹד לַכְּפָר.
מִשְׁפָּטֵי פְּרָטִיםנוֹתְנִים דֻּגְמָאוֹת, סִבּוֹת וּפְרָטִיםהוּא עוֹצֵר אֶת מֵי הַיָּם. שׁוֹמְרִים בּוֹדְקִים אוֹתוֹ בְּכָל בֹּקֶר.
מִשְׁפָּט סִיכּוּםסוֹגֵר אֶת הַפִּסְקָה וְחוֹזֵר עַל הָרַעְיוֹןבְּלִי הַסֶּכֶר, הַמַּיִם הָיוּ שׁוֹטְפִים אֶת הַכְּפָר.
🔑 כְּלָל: כָּל הַמִּשְׁפָּטִים בַּפִּסְקָה עוֹסְקִים בְּאוֹתוֹ רַעְיוֹן. מִשְׁפָּט שֶׁאֵינוֹ קָשׁוּר לָרַעְיוֹן — לֹא שַׁיָּךְ לַפִּסְקָה, וְעָדִיף לְהוֹצִיא אוֹתוֹ אוֹ לִכְתֹּב אוֹתוֹ בְּפִסְקָה אַחֶרֶת.
💡 שִׂימוּ לֵב: מִשְׁפָּט פְּתִיחָה בַּפִּסְקָה אֵינוֹ "נוֹשֵׂא" שֶׁל מִשְׁפָּט (שִׁיעוּר 53). כָּאן "פְּתִיחָה" פֵּרוּשָׁהּ הַמִּשְׁפָּט הָרִאשׁוֹן שֶׁמַּצִּיג אֶת הָרַעְיוֹן. וְלֹא תָּמִיד חַיָּבִים מִשְׁפָּט סִיכּוּם — בְּפִסְקָה קְצָרָה מְאֹד אֶפְשָׁר לְוַתֵּר עָלָיו.

מָתַי מַתְחִילִים פִּסְקָה חֲדָשָׁה? כְּשֶׁמִּשְׁתַּנִּים הָרַעְיוֹן, הַזְּמַן, הַמָּקוֹם אוֹ הַדְּמוּת שֶׁעוֹסְקִים בָּהּ. בַּסִּפּוּר: "בַּבֹּקֶר עָבַר שָׁם אִישׁ זָקֵן..." — הַזְּמַן הִשְׁתַּנָּה, לָכֵן זוֹ פִּסְקָה חֲדָשָׁה.

כִּתְבוּ פִּסְקָה קְצָרָה עַל מָקוֹם שֶׁאַתֶּם אוֹהֲבִים. כִּתְבוּ מִשְׁפָּט פְּתִיחָה, שְׁלוֹשָׁה מִשְׁפָּטֵי פְּרָטִים וּמִשְׁפָּט סִיכּוּם. לְדֻגְמָה: הַמִּטְבָּח שֶׁלָּנוּ הוּא הַמָּקוֹם הָאָהוּב עָלַי בַּבַּיִת.
