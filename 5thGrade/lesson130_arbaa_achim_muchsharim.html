אַרְבָּעָה אַחִים מֻכְשָׁרִים

מַעֲשִׂיָּה שֶׁל הָאַחִים גְּרִים עַל אַרְבַּע אֻמָּנֻיּוֹת, חָמֵשׁ בֵּיצִים וּדְרָקוֹן אֶחָד

📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת חַי בִּכְפָר קָטָן נַגָּר עָנִי, וְהָיוּ לוֹ אַרְבָּעָה בָּנִים. יוֹם אֶחָד קָרָא לָהֶם וְאָמַר: "בָּנַי הַיְּקָרִים, אֵין לִי כֶּסֶף לָתֵת לָכֶם. צְאוּ אֶל הָעוֹלָם, לִמְדוּ כָּל אֶחָד אֻמָּנוּת טוֹבָה, וְשׁוּבוּ אֵלַי בְּעוֹד אַרְבַּע שָׁנִים."

הָאַחִים יָצְאוּ לַדֶּרֶךְ וְהִגִּיעוּ אֶל פָּרָשַׁת דְּרָכִים. "כָּאן נִפָּרֵד," אָמַר הַבְּכוֹר, "וְכָל אֶחָד יֵלֵךְ בְּדֶרֶךְ אַחֶרֶת." הֵם הִתְחַבְּקוּ, וְכָל אֶחָד פָּנָה לְכִוּוּן אַחֵר.

הָאָח הָרִאשׁוֹן פָּגַשׁ אִישׁ זָרִיז שֶׁהָיָה פַּעַם גַּנָּב מְפֻרְסָם. "אֲנִי לֹא רוֹצֶה לִהְיוֹת גַּנָּב!" אָמַר הָאָח. הָאִישׁ צָחַק: "אֲלַמֵּד אוֹתְךָ אֶת זְרִיזוּת הַיָּדַיִם, וְאַתָּה תִּשְׁתַּמֵּשׁ בָּהּ רַק לְמַטָּרוֹת טוֹבוֹת." הָאָח הִסְכִּים וְנִשְׁאַר אֶצְלוֹ.

הָאָח הַשֵּׁנִי לָמַד אֵצֶל חוֹקֵר כּוֹכָבִים. בְּסוֹף הַלִּמּוּדִים קִבֵּל מִמֶּנּוּ מִשְׁקֶפֶת קְסוּמָה, שֶׁבְּעֶזְרָתָהּ אֶפְשָׁר לִרְאוֹת כָּל מַה שֶּׁקּוֹרֶה בַּשָּׁמַיִם וּבָאָרֶץ.

הָאָח הַשְּׁלִישִׁי לָמַד אֵצֶל צַיָּד בַּיַּעַר, וְקִבֵּל מִמֶּנּוּ קֶשֶׁת שֶׁהַחִצִּים שֶׁלָּהּ פּוֹגְעִים תָּמִיד בַּמַּטָּרָה. הָאָח הָרְבִיעִי לָמַד אֵצֶל חַיָּט, וְקִבֵּל מַחַט שֶׁתּוֹפֶרֶת כָּל דָּבָר — בַּד, עֵץ וַאֲפִלּוּ אֶבֶן — בְּלִי שֶׁיִּשָּׁאֵר סִימָן.

אַחֲרֵי אַרְבַּע שָׁנִים חָזְרוּ הָאַחִים הַבַּיְתָה. הָאָב רָצָה לִבְדֹּק אֶת הָאֻמָּנֻיּוֹת שֶׁלָּהֶם. הוּא הֶרְאָה לָהֶם עֵץ גָּבוֹהַּ וְשָׁאַל אֶת הַבֵּן הַשֵּׁנִי: "מָה אַתָּה רוֹאֶה בְּרֹאשׁ הָעֵץ?" הַבֵּן הִבִּיט בַּמִּשְׁקֶפֶת וְעָנָה: "קֵן, וּבוֹ צִפּוֹר שֶׁיּוֹשֶׁבֶת עַל חָמֵשׁ בֵּיצִים." הָאָח הָרִאשׁוֹן טִפֵּס בְּשֶׁקֶט וְהוֹצִיא אֶת הַבֵּיצִים מִתַּחַת לַצִּפּוֹר, וְהִיא אֲפִלּוּ לֹא הִרְגִּישָׁה.

הָאָב הִנִּיחַ אֶת חָמֵשׁ הַבֵּיצִים עַל הַשֻּׁלְחָן. הַצַּיָּד יָרָה חֵץ אֶחָד, וְהַחֵץ חָצָה אֶת כָּל הַבֵּיצִים לִשְׁנַיִם. מִיָּד תָּפַר הַחַיָּט אֶת הַקְּלִפּוֹת, וְהָאָח הָרִאשׁוֹן הֶחְזִיר אוֹתָן לַקֵּן בְּלִי שֶׁהַצִּפּוֹר הִרְגִּישָׁה. כַּעֲבֹר כַּמָּה יָמִים בָּקְעוּ הָאֶפְרוֹחִים, וּלְכָל אֶחָד מֵהֶם הָיָה פַּס אָדֹם דַּק עַל הַצַּוָּאר — בְּדִיּוּק בַּמָּקוֹם שֶׁבּוֹ עָבְרָה הַמַּחַט.

זְמַן קָצָר אַחַר כָּךְ הִגִּיעָה לַכְּפָר בְּשׂוֹרָה רָעָה: דְּרָקוֹן חָטַף אֶת בַּת הַמֶּלֶךְ! הַמֶּלֶךְ הִכְרִיז שֶׁמִּי שֶׁיַּצִּיל אוֹתָהּ יְקַבֵּל פְּרָס גָּדוֹל. הָאָח הַשֵּׁנִי הִבִּיט בַּמִּשְׁקֶפֶת וְרָאָה אֶת הַנְּסִיכָה רְחוֹקָה מְאוֹד, עַל סֶלַע בְּאֶמְצַע הַיָּם, וְהַדְּרָקוֹן שׁוֹמֵר עָלֶיהָ.

הַמֶּלֶךְ נָתַן לָאַחִים סְפִינָה, וְהֵם הִפְלִיגוּ אֶל הַסֶּלַע עִם מַלָּח זָקֵן שֶׁהִכִּיר אֶת הַיָּם. כְּשֶׁהִגִּיעוּ, הַדְּרָקוֹן יָשַׁן, וְרֹאשׁוֹ הַכָּבֵד הָיָה מֻנָּח עַל בִּרְכֵּי הַנְּסִיכָה. הָאָח הָרִאשׁוֹן הִתְגַּנֵּב בְּשֶׁקֶט, הוֹצִיא אוֹתָהּ מִתַּחַת לָרֹאשׁ הַגָּדוֹל, וְכֻלָּם מִהֲרוּ בַּחֲזָרָה אֶל הַסְּפִינָה.

אֲבָל הַדְּרָקוֹן הִתְעוֹרֵר וְעָף אַחֲרֵיהֶם בִּשְׁאָגָה. הַצַּיָּד מָתַח אֶת הַקֶּשֶׁת וְיָרָה, וְהַחֵץ פָּגַע בִּכְנַף הַדְּרָקוֹן. הַדְּרָקוֹן נָפַל אֶל הַיָּם, וּבִנְפִילָתוֹ פָּגַע בַּסְּפִינָה וְשָׁבַר אוֹתָהּ. אַחַר כָּךְ שָׂחָה הַדְּרָקוֹן הַרְחֵק. הַחַיָּט שָׁלַף אֶת הַמַּחַט וְתָפַר מַהֵר אֶת הַקְּרָשִׁים, וְהַסְּפִינָה הִמְשִׁיכָה לְשׁוֹט בְּשָׁלוֹם אֶל הַחוֹף.

בָּאַרְמוֹן הֵכִין טַבָּח הַמֶּלֶךְ סְעֻדָּה גְּדוֹלָה. אֲבָל אָז הִתְחִילוּ הָאַחִים לְהִתְוַכֵּחַ. "בִּזְכוּתִי מָצָאנוּ אוֹתָהּ!" אָמַר הַשֵּׁנִי. "בִּזְכוּתִי הִיא יָצְאָה מִן הַסֶּלַע!" אָמַר הָרִאשׁוֹן. "וּבִזְכוּתִי הַדְּרָקוֹן נָפַל!" אָמַר הַצַּיָּד. "וּבִזְכוּתִי לֹא טָבַעְנוּ!" אָמַר הַחַיָּט.

הַמֶּלֶךְ חִיֵּךְ וְאָמַר: "אַף אֶחָד מִכֶּם לֹא הָיָה מַצְלִיחַ לְבַדּוֹ. רַק אַרְבַּעְתְּכֶם יַחַד הִצַּלְתֶּם אֶת בִּתִּי." וְהוּא נָתַן לְכָל אֶחָד מֵהָאַחִים פְּרָס שָׁוֶה. הָאַחִים חָזְרוּ אֶל אֲבִיהֶם הַנַּגָּר, וּמֵאָז לֹא הָיוּ עוֹד עֲנִיִּים — וְגַם לֹא הִתְוַכְּחוּ עוֹד מִי מֵהֶם חָשׁוּב יוֹתֵר.

מָקוֹר: "אַרְבָּעָה אַחִים מֻכְשָׁרִים" (Die vier kunstreichen Brüder), מַעֲשִׂיָּה שֶׁל הָאַחִים גְּרִים (גֶּרְמַנְיָה, הַמֵּאָה ה־19), בְּעִבּוּד חָפְשִׁי וּמְקֻצָּר. הַמִּלִּים הַמְּסֻמָּנוֹת הֵן שֵׁמוֹת בְּמִשְׁקַל קַטָּל.

נַגָּר, חַיָּט, צַיָּד — הַתַּבְנִית שֶׁל בַּעֲלֵי הַמִּקְצוֹעַ

מִשְׁקָל הוּא תַּבְנִית שֶׁל תְּנוּעוֹת, שֶׁאֶל תּוֹכָהּ "יוֹצְקִים" אֶת אוֹתִיּוֹת הַשֹּׁרֶשׁ. בְּמִשְׁקַל קַטָּל יֵשׁ פַּתָּח תַּחַת הָאוֹת הָרִאשׁוֹנָה שֶׁל הַשֹּׁרֶשׁ, וּבָאוֹת הַשְּׁנִיָּה יֵשׁ דָּגֵשׁ וְקָמָץ. כָּךְ: קַ־טָּ־ל ← נַ־גָּ־ר.

שֵׁם בְּמִשְׁקַל קַטָּל מְתָאֵר אָדָם שֶׁעוֹשֶׂה אֶת אוֹתָהּ פְּעֻלָּה שׁוּב וָשׁוּב — בְּדֶרֶךְ כְּלָל כְּמִקְצוֹעַ. כְּדֵי לְהָבִין אֶת הַמִּלָּה, מְחַפְּשִׂים אֶת הַשֹּׁרֶשׁ וְשׁוֹאֲלִים: מָה הָאָדָם הַזֶּה עוֹשֶׂה?

נְקֵבָה וְרַבִּים: בִּנְקֵבָה מוֹסִיפִים ־ִית (טַבָּחִית, נַגָּרִית) אוֹ ־ֶת (חַיֶּטֶת, צַיֶּרֶת). בָּרַבִּים: נַגָּרִים, חַיָּטִים · נַגָּרִיּוֹת, חַיָּטוֹת.

אוֹת גְּרוֹנִית בָּאֶמְצַע: הָאוֹתִיּוֹת א, ה, ח, ע, ר לֹא מְקַבְּלוֹת דָּגֵשׁ. לָכֵן כְּשֶׁהֵן הָאוֹת הַשְּׁנִיָּה בַּשֹּׁרֶשׁ, הַדָּגֵשׁ נֶעֱלָם: נַהָג (נ־ה־ג) — בְּלִי דָּגֵשׁ בְּה׳. לִפְנֵי ר׳ הַפַּתָּח אַף הוֹפֵךְ לְקָמָץ: פָּרָשׁ (פ־ר־ש).

לֹא כָּל בַּעַל מִקְצוֹעַ הוּא בְּמִשְׁקַל קַטָּל: רוֹפֵא, שׁוֹטֵר וּמוֹרֶה בְּנוּיִים כְּמוֹ פֹּעַל בַּהֹוֶה; עִתּוֹנַאי בָּנוּי עִם הַסִּיּוֹמֶת ־ַאי. וְגַם לֹא כָּל מִלָּה עִם דָּגֵשׁ הִיא קַטָּל: צִיּוּר הוּא דָּבָר שֶׁעָשׂוּ, וְצַיָּר הוּא הָאָדָם שֶׁמְּצַיֵּר.

גַּם תְּכוּנָה אוֹ הֶרְגֵּל: לִפְעָמִים קַטָּל מְתָאֵר אָדָם שֶׁעוֹשֶׂה מַשֶּׁהוּ שׁוּב וָשׁוּב, גַּם אִם זֶה לֹא מִקְצוֹעַ טוֹב — כְּמוֹ גַּנָּב (ג־נ־ב).

מִצְאוּ שִׁבְעָה שֵׁמוֹת בְּמִשְׁקַל קַטָּל (הַמִּלִּים כְּתוּבוֹת בְּלִי נִקּוּד). לַחֲצוּ עַל הָאוֹת הָרִאשׁוֹנָה וְאַחַר כָּךְ עַל הָאוֹת הָאַחֲרוֹנָה שֶׁל הַמִּלָּה. עַל כָּל מִלָּה — אֶפְרוֹחַ חָדָשׁ בּוֹקֵעַ!

דַּמְיְנוּ שֶׁלַּנַּגָּר הָיָה בֵּן חֲמִישִׁי. אֵצֶל אֵיזֶה בַּעַל מִקְצוֹעַ הוּא לָמַד, וְאֵיךְ עָזַר לְאֶחָיו? כִּתְבוּ חָמֵשׁ שׁוּרוֹת, וּבְכָל שׁוּרָה הַשְׁתַּמְּשׁוּ לְפָחוֹת בְּשֵׁם אֶחָד בְּמִשְׁקַל קַטָּל. לְמָשָׁל: "הָאָח הַחֲמִישִׁי לָמַד אֵצֶל צַיָּר, וְהַצִּיּוּרִים שֶׁלּוֹ נִרְאוּ כְּמוֹ אֲמִיתִיִּים."
