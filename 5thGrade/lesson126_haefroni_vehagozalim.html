📖 קְרִיאָה

בְּאֶמְצַע שְׂדֵה חִטָּה גָּדוֹל, בֵּין הַשִּׁבֳּלִים, גָּרָה אִמָּא עֶפְרוֹנִי עִם שְׁלוֹשָׁה גּוֹזָלִים קְטַנִּים. הַקֵּן שֶׁלָּהֶם הָיָה מֻסְתָּר עַל הָאֲדָמָה, וּמֵעָלָיו הָלְכָה הַחִטָּה וְגָבְהָה מִיּוֹם לְיוֹם. "הַחִטָּה צוֹמַחַת מַהֵר," אָמְרָה הָאֵם, "וּכְשֶׁהִיא תִּהְיֶה זְהֻבָּה, יַגִּיעַ זְמַן הַקָּצִיר."

בְּכָל בֹּקֶר יָצְאָה הָאֵם לְחַפֵּשׂ אֹכֶל. לִפְנֵי שֶׁעָפָה, הָיְתָה אוֹמֶרֶת לַגּוֹזָלִים: "הַקְשִׁיבוּ הֵיטֵב לְכָל מִלָּה שֶׁאוֹמֵר הָאִכָּר בַּעַל הַשָּׂדֶה, וּבָעֶרֶב סַפְּרוּ לִי הַכֹּל."

יוֹם אֶחָד הִגִּיעַ הָאִכָּר לַשָּׂדֶה עִם בְּנוֹ. הוּא מִשֵּׁשׁ אֶת הַשִּׁבֳּלִים וְאָמַר: "הַחִטָּה כְּבָר בְּשֵׁלָה. הַיּוֹם אֲנִי שׁוֹלֵחַ אוֹתְךָ אֶל הַשְּׁכֵנִים, וּנְבַקֵּשׁ מֵהֶם לָבוֹא מָחָר לַעֲזֹר לָנוּ בַּקָּצִיר."

בָּעֶרֶב, כְּשֶׁחָזְרָה הָאֵם, צִיְּצוּ הַגּוֹזָלִים בְּבֶהָלָה: "אִמָּא, עָלֵינוּ לִבְרֹחַ מִיָּד! הָאִכָּר שׁוֹלֵחַ אֶת בְּנוֹ לִקְרֹא לַשְּׁכֵנִים!" אֲבָל הָאֵם רַק חִיְּכָה. "אִם הָאִכָּר בּוֹטֵחַ בַּשְּׁכֵנִים," אָמְרָה, "אֵין סִבָּה לְפַחֵד עֲדַיִן. הַלַּיְלָה נָנוּחַ בְּשֶׁקֶט."

לְמָחֳרָת חִכּוּ הָאִכָּר וּבְנוֹ בַּשָּׂדֶה מִן הַבֹּקֶר וְעַד הַצָּהֳרַיִם, אֲבָל אַף שָׁכֵן לֹא בָּא. "הַשְּׁכֵנִים עֲסוּקִים בַּשָּׂדוֹת שֶׁלָּהֶם," נֶאֱנַח הָאִכָּר. "עוֹד הַיּוֹם אֶשְׁלַח הוֹדָעָה לַקְּרוֹבִים שֶׁלָּנוּ — לַדּוֹדִים וְלִבְנֵי הַדּוֹדִים. בַּמִּשְׁפָּחָה אֲנִי בּוֹטֵחַ."

בָּעֶרֶב סִפְּרוּ הַגּוֹזָלִים לָאֵם מָה שָׁמְעוּ, וְשׁוּב הִתְחַנְּנוּ: "אִמָּא, בּוֹאִי נִבְרַח!" אֲבָל הָאֵם לֹא נִבְהֲלָה. "אֲנִי יוֹדַעַת מָה יִקְרֶה," אָמְרָה. "גַּם לַקְּרוֹבִים יֵשׁ שָׂדוֹת וַעֲבוֹדָה מִשֶּׁלָּהֶם. מִי שֶׁמְּחַכֶּה לַאֲחֵרִים — מְחַכֶּה הַרְבֵּה זְמַן. נִשָּׁאֵר עוֹד לַיְלָה."

וּבֶאֱמֶת, גַּם בַּיּוֹם הַשְּׁלִישִׁי לֹא בָּא אִישׁ. הָאִכָּר הִבִּיט בַּשִּׁבֳּלִים הַיְּבֵשׁוֹת וְאָמַר לִבְנוֹ: "דַּי לְחַכּוֹת. מָחָר עִם שַׁחַר אֶפְתַּח אֶת הַמַּחְסָן, נִקַּח שְׁנֵי מַגָּלִים, וַאֲנִי וְאַתָּה נִקְצֹר אֶת הַשָּׂדֶה בְּעַצְמֵנוּ."

בָּעֶרֶב, כְּשֶׁסִּפְּרוּ הַגּוֹזָלִים לָאֵם אֶת הַדְּבָרִים הָאֵלֶּה, הִיא אָמְרָה מִיָּד: "עַכְשָׁו הִגִּיעַ הַזְּמַן לִבְרֹחַ! כָּל עוֹד הָאִכָּר חִכָּה לַאֲחֵרִים, לֹא הָיְתָה סַכָּנָה. אֲבָל מִי שֶׁבּוֹטֵחַ רַק בְּעַצְמוֹ — בֶּאֱמֶת עוֹשֶׂה אֶת הָעֲבוֹדָה."

בְּאוֹתוֹ לַיְלָה, לְאוֹר הַיָּרֵחַ, הוֹבִילָה הָאֵם אֶת הַגּוֹזָלִים אֶל קֵן חָדָשׁ בִּקְצֵה הַשָּׂדֶה, מִתַּחַת לְשִׂיחַ קוֹצָנִי. הַגּוֹזָלִים כְּבָר לָמְדוּ לְעוֹפֵף מְעַט, וְהֵם קָפְצוּ וְרִפְרְפוּ אַחֲרֶיהָ.

בַּבֹּקֶר, כְּשֶׁהָאִכָּר וּבְנוֹ קָצְרוּ אֶת הַחִטָּה, הִבִּיטוּ בָּהֶם הַגּוֹזָלִים מִן הַקֵּן הֶחָדָשׁ — בְּטוּחִים וּשְׂמֵחִים. "אִמָּא," לָחַשׁ הַקָּטָן מִכֻּלָּם, "אֵיךְ יָדַעְתְּ?" "פָּשׁוּט מְאוֹד," עָנְתָה הָאֵם. "מִי שֶׁסּוֹמֵךְ עַל אֲחֵרִים — מְחַכֶּה. מִי שֶׁעוֹשֶׂה בְּעַצְמוֹ — מַצְלִיחַ."

מָקוֹר: מָשָׁל מְיֻחָס לְאִיזוֹפּוֹס (יָוָן הָעַתִּיקָה), "הָעֶפְרוֹנִי וְגוֹזָלָיו", בְּעִבּוּד חָפְשִׁי.

📚 דִּקְדּוּק — פַּתָּח גְּנוּבָה וּפְעָלִים שֶׁסּוֹפָם ח׳ אוֹ ע׳

פַּתָּח גְּנוּבָה — הַפַּתָּח שֶׁ"גּוֹנֵב" אֶת הַמָּקוֹם

בַּסִּפּוּר הוֹפִיעוּ הַרְבֵּה פְּעָלִים שֶׁהָאוֹת הָאַחֲרוֹנָה בַּשֹּׁרֶשׁ שֶׁלָּהֶם הִיא ח׳ אוֹ ע׳: שׁ־ל־ח, ב־ר־ח, ב־ט־ח, פ־ת־ח, שׁ־מ־ע, י־ד־ע. הָאוֹתִיּוֹת ח׳ וְע׳ הֵן אוֹתִיּוֹת גְּרוֹנִיּוֹת (הֵן נֶהֱגוֹת מֵהַגָּרוֹן), וְהֵן "אוֹהֲבוֹת" אֶת הַתְּנוּעָה אַ — פַּתָּח.

1. פַּתָּח גְּנוּבָה. כְּשֶׁמִּלָּה נִגְמֶרֶת בְּ־ח׳ אוֹ בְּ־ע׳, וְלִפְנֵיהֶן יֵשׁ תְּנוּעָה אַחֶרֶת (חוֹלָם, שׁוּרוּק, חִירִיק אוֹ צֵירֵי), מוֹסִיפִים פַּתָּח מִתַּחַת לָאוֹת הָאַחֲרוֹנָה. אֶת הַפַּתָּח הַזֶּה קוֹרְאִים לִפְנֵי הָאוֹת — כְּאִלּוּ הוּא "גָּנַב" אֶת מְקוֹמָהּ. לָכֵן קוֹרְאִים לוֹ "פַּתָּח גְּנוּבָה": רוּחַ נִשְׁמַע "רוּ־אַח", וְשׁוֹמֵעַ נִשְׁמַע "שׁוֹ־מֵ־אַע".

רוּחַיָרֵחַשִׂיחַתַּפּוּחַשׁוֹלֵחַיוֹדֵעַמַצְלִיחַ

כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים עַל הַסִּפּוּר אוֹ עַל עַצְמְכֶם, וּבְכָל מִשְׁפָּט הִשְׁתַּמְּשׁוּ בַּפֹּעַל שֶׁמּוֹפִיעַ בַּשּׁוּרָה, בַּצּוּרָה הַנְּכוֹנָה. דֻּגְמָה: "בַּבֹּקֶר אֲנִי שׁוֹמֵעַ אֶת הַצִּפּוֹרִים מִן הַחַלּוֹן."
