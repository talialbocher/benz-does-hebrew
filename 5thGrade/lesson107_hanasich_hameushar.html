הַנָּסִיךְ הַמְּאֻשָּׁר

סִפּוּר מֻכָּר מֵאֵת אוֹסְקָר וַיְילְד, בְּעִבּוּד

📖 קְרִיאָה

בְּעִיר אַחַת, עַל עַמּוּד גָּבוֹהַּ, עָמַד פֶּסֶל שֶׁל הַנָּסִיךְ הַמְּאֻשָּׁר. גּוּפוֹ הָיָה מְצֻפֶּה בְּעָלֵי זָהָב דַּקִּים, בִּמְקוֹם עֵינָיו נָצְצוּ שְׁנֵי סַפִּירִים כְּחֻלִּים, וְעַל חַרְבּוֹ זָרַח אֹדֶם אָדֹם. תּוֹשְׁבֵי הָעִיר הָיוּ עוֹצְרִים לְיָדוֹ וְאוֹמְרִים: ״כַּמָּה הוּא יָפֶה!״

בְּלַיְלָה אֶחָד, בְּסוֹף הַקַּיִץ, נָחֲתָה סְנוּנִית קְטַנָּה בֵּין רַגְלָיו שֶׁל הַפֶּסֶל. כָּל חֲבֵרוֹתֶיהָ כְּבָר עָפוּ לְמִצְרַיִם, וְהִיא נִשְׁאֲרָה מֵאָחוֹר. כְּשֶׁנִּרְדְּמָה, יָרְדוּ עָלֶיהָ טִפּוֹת מַיִם גְּדוֹלוֹת. הִיא שָׁאֲלָה אֶת עַצְמָהּ: ״אֵיךְ יוֹרֵד גֶּשֶׁם, אִם הַשָּׁמַיִם צְלוּלִים?״ אָז הִסְתַּכְּלָה לְמַעְלָה וְרָאֲתָה שֶׁהַפֶּסֶל בּוֹכֶה.

״מַדּוּעַ אַתָּה בּוֹכֶה?״ שָׁאֲלָה הַסְּנוּנִית. הַנָּסִיךְ עָנָה: ״כְּשֶׁהָיִיתִי חַי, לֹא יָדַעְתִּי מָה זֶה עֶצֶב. הָיִיתִי מְשַׂחֵק כָּל הַיּוֹם בְּגַן הָאַרְמוֹן, וְהַחוֹמָה הַגְּבוֹהָה הָיְתָה מַסְתִּירָה מִמֶּנִּי אֶת הָעִיר. עַכְשָׁו אֲנִי עוֹמֵד לְמַעְלָה וְרוֹאֶה אֶת כָּל הַצָּרוֹת שֶׁלָּהּ.

שָׁם, בְּבַיִת קָטָן, גָּרָה תּוֹפֶרֶת עֲנִיָּה. בְּנָהּ הַחוֹלֶה הָיָה בּוֹכֶה כָּל לַיְלָה וּמְבַקֵּשׁ תַּפּוּזִים, וְהִיא לֹא הָיְתָה יְכוֹלָה לִקְנוֹת לוֹ אֲפִלּוּ אֶחָד. סְנוּנִית, קְחִי אֶת הָאֹדֶם מֵחַרְבִּי וְהָבִיאִי לָהּ!״

הַסְּנוּנִית הָיְתָה עֲיֵפָה וְקַר לָהּ, אֲבָל הִיא הִסְכִּימָה. הִיא לָקְחָה אֶת הָאֹדֶם וְעָפָה אֶל הַבַּיִת הַקָּטָן. שָׁם הִנִּיחָה אֶת הָאֹדֶם עַל הַשֻּׁלְחָן וְהֵנִיפָה כְּנָפַיִם מֵעַל מִצְחוֹ שֶׁל הַיֶּלֶד, עַד שֶׁנִּרְדַּם.

מִלַּיְלָה לְלַיְלָה בִּקֵּשׁ הַנָּסִיךְ מֵהַסְּנוּנִית עוֹד וָעוֹד. אֶת הַסַּפִּיר הָרִאשׁוֹן נָתַן לְסוֹפֵר צָעִיר שֶׁהָיָה יוֹשֵׁב בַּעֲלִיָּה קָרָה וְכוֹתֵב מַחֲזֶה, כִּי לֹא הָיָה לוֹ מָה לֶאֱכֹל. וְאֶת הַסַּפִּיר הַשֵּׁנִי נָתַן לְיַלְדָּה קְטַנָּה שֶׁהָיְתָה מוֹכֶרֶת גִּפְרוּרִים בָּרְחוֹב.

עַכְשָׁו הַנָּסִיךְ כְּבָר לֹא רָאָה, כִּי נָתַן אֶת שְׁתֵּי עֵינָיו. הַסְּנוּנִית אָמְרָה: ״אֲנִי כְּבָר לֹא אֶעֱזֹב אוֹתְךָ!״ וְנִשְׁאֲרָה. בַּיּוֹם הִיא הָיְתָה יוֹשֶׁבֶת עַל כְּתֵפוֹ וּמְסַפֶּרֶת לוֹ עַל מִצְרַיִם, וּבַלַּיְלָה הִיא הָיְתָה עָפָה בֵּין הַבָּתִּים וּמְדַוַּחַת לוֹ מָה הִיא רוֹאָה.

הַחֹרֶף בָּא, וְהָעִיר הָיְתָה מִתְכַּסָּה בְּשֶׁלֶג. הַסְּנוּנִית הָיְתָה קוֹפֵאת מִקֹּר יוֹתֵר מִיּוֹם לְיוֹם. בַּסּוֹף הִיא נָשְׁקָה אֶת הַנָּסִיךְ וְנָפְלָה לְרַגְלָיו, וְלִבּוֹ הֶעָשׂוּי מֵעוֹפֶרֶת נִסְדַּק.

בַּבֹּקֶר רָאָה רֹאשׁ הָעִיר אֶת הַפֶּסֶל, בְּלִי זָהָב וּבְלִי אֲבָנִים, וְצִוָּה: ״לְהַמִּיס אוֹתוֹ!״ אֲבָל הַלֵּב הַנִּסְדָּק לֹא הִתְמוֹסֵס בַּכּוּר, וְהִשְׁלִיכוּ אוֹתוֹ אֶל הָאַשְׁפָּה, לְיַד הַסְּנוּנִית.

אָמַר אֱלֹהִים לְאֶחָד מֵהַמַּלְאָכִים: ״הָבֵא לִי אֶת שְׁנֵי הַדְּבָרִים הַיְּקָרִים בְּיוֹתֵר בָּעִיר.״ הַמַּלְאָךְ הֵבִיא אֶת הַלֵּב וְאֶת הַסְּנוּנִית, וְשָׁם בַּגָּן הַשְּׁמֵימִי הֵם נִשְׁאֲרוּ יַחַד לָנֶצַח.

אוֹסְקָר וַיְילְד (סוֹפֵר אִירִי, 1888), ״הַנָּסִיךְ הַמְּאֻשָּׁר״, בְּעִבּוּד חָפְשִׁי וּמְקֻצָּר.

📚 דִּקְדּוּק — הָיָה + הוֹוֶה

אֵיךְ מְסַפְּרִים עַל מַה שֶּׁקָּרָה שׁוּב וָשׁוּב?

בַּסִּפּוּר מָצָאנוּ מִשְׁפָּטִים כָּאֵלֶּה: ״תּוֹשְׁבֵי הָעִיר הָיוּ עוֹצְרִים לְיָדוֹ״. אֵין כָּאן פְּעֻלָּה אַחַת שֶׁקָּרְתָה פַּעַם אַחַת (עָצְרוּ), אֶלָּא דָּבָר שֶׁקָּרָה שׁוּב וָשׁוּב, אוֹ שֶׁנִּמְשַׁךְ זְמַן רַב. בּוֹנִים אֶת הַצּוּרָה הַזֹּאת מִשְּׁנֵי חֲלָקִים: הַפֹּעַל הָיָה (בָּעָבָר) וְאַחֲרָיו פֹּעַל בְּהוֹוֶה.

כְּלָלִים לִזְכֹּר

• שְׁנֵי הַחֲלָקִים מִתְאִימִים לַנּוֹשֵׂא בְּמִין וּבְמִסְפָּר: הַסְּנוּנִית הָיְתָה מְסַפֶּרֶת הַסְּנוּנִיּוֹת הָיוּ מְסַפְּרוֹת לֹא אוֹמְרִים ״הַיְּלָדִים הָיָה מְשַׂחֲקִים״.

• מִלִּים שֶׁרוֹמְזוֹת עַל הֶרְגֵּל: כָּל לַיְלָה תָּמִיד בְּדֶרֶךְ כְּלָל לִפְעָמִים מִיּוֹם לְיוֹם

• לְעֻמַּת זֹאת, פְּעֻלָּה שֶׁקָּרְתָה פַּעַם אַחַת כּוֹתְבִים בְּעָבָר רָגִיל: ״בְּלַיְלָה אֶחָד נָחֲתָה סְנוּנִית״.

• זְהִירוּת: ״הָיָה״ עִם שֵׁם תֹּאַר אוֹ שֵׁם עֶצֶם אֵינוֹ הֶרְגֵּל, אֶלָּא מַצָּב בָּעָבָר: ״הוּא הָיָה עָצוּב״. הֶרְגֵּל מִתְקַבֵּל רַק כְּשֶׁאַחֲרֵי ״הָיָה״ בָּא פֹּעַל בְּהוֹוֶה (עוֹצֵר, מְסַפֵּר, בּוֹכֶה).

סַפְּרוּ עַל מַה שֶּׁהֱיִיתֶם עוֹשִׂים כְּשֶׁהֱיִיתֶם קְטַנִּים. כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים עִם ״הָיָה + הוֹוֶה״, לְפִי הַהוֹרָאוֹת. לְמָשָׁל: ״הָיִיתִי אוֹהֵב לְשַׂחֵק בַּחוֹל כָּל יוֹם.״
