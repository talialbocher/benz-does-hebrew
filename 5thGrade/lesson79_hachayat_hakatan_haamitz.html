📖 קְרִיאָה

בְּעִיר קְטַנָּה בְּגֶרְמַנְיָה, לִפְנֵי שָׁנִים רַבּוֹת, יָשַׁב חַיָּט קָטָן וְזָרִיז לְיַד חַלּוֹן בֵּיתוֹ, וְתָפַר בְּגֵאוּת חֻלְצָה נָאָה. יוֹם קַיִץ חַם אֶחָד מָרַח הַחַיָּט רִבָּה מְתוּקָה עַל פְּרוּסַת לֶחֶם, וְהִנִּיחַ אוֹתָהּ עַל הַשֻּׁלְחָן.

רֵיחַ הָרִבָּה מָשַׁךְ אֶל הַחֶדֶר נְחִיל שֶׁל שִׁבְעָה זְבוּבִים רְעֵבִים, שֶׁיָּשְׁבוּ לָהֶם בְּשַׁלְוָה עַל פְּרוּסַת הַלֶּחֶם. הַחַיָּט הַכּוֹעֵס תָּפַס פִּסַּת בַּד, הֵנִיף אוֹתָהּ בְּמַכָּה אַחַת חֲזָקָה - וְהִנֵּה, כָּל שִׁבְעַת הַזְּבוּבִים שָׁכְבוּ מֵתִים עַל הַשֻּׁלְחָן. "אֵיזֶה גִּבּוֹר אֲנִי!" קָרָא הַחַיָּט בְּגַאֲוָה. "הָעִיר כֻּלָּהּ חַיֶּבֶת לָדַעַת עַל כָּךְ!"

בְּהִתְלַהֲבוּת רַבָּה תָּפַר הַחַיָּט חֲגוֹרָה רְחָבָה, וְרָקַם עָלֶיהָ בְּחוּטֵי זָהָב אֶת הַמִּלִּים: "שִׁבְעָה בְּמַכָּה אַחַת". הוּא חָגַר אוֹתָהּ בְּגַאֲוָה, אָרַז חֲתִיכַת גְּבִינָה וְצִפּוֹר קְטַנָּה שֶׁתָּפַס, וְיָצָא מִן הָעִיר אֶל הָעוֹלָם הָרָחָב, כְּדֵי לְחַפֵּשׂ הַרְפַּתְקָאוֹת.

בְּלֵב יַעַר עָבֹת פָּגַשׁ הַחַיָּט עֲנָק עֲצוּם מִמַּדִּים. הָעֲנָק קָרָא אֶת הַכָּתוּב עַל הַחֲגוֹרָה וְשָׁאַל בְּלַעַג: "שִׁבְעָה בְּמַכָּה אַחַת? אַתָּה בֶּאֱמֶת חָזָק מִמֶּנִּי?" הַחַיָּט לֹא נִבְהַל. "בְּוַדַּאי," עָנָה בְּשַׁלְוָה, "וְאִם לֹא תַּאֲמִין, בּוֹא נִבְחַן."

הָעֲנָק לָקַח אֶבֶן בְּיָדוֹ וְלָחַץ עָלֶיהָ בְּכָל כֹּחוֹ, עַד שֶׁטִּפּוֹת מַיִם נִסְחֲטוּ מִמֶּנָּה. הַחַיָּט הֶעָרוּם הוֹצִיא בַּחֲשַׁאי אֶת חֲתִיכַת הַגְּבִינָה שֶׁטָּמַן בְּכִיסוֹ, לָחַץ עָלֶיהָ - וְנוֹזֵל לָבָן נָטַף מִבֵּין אֶצְבְּעוֹתָיו. "רְאֵה," אָמַר, "אֲנִי סוֹחֵט אֶת הָאֶבֶן חָזָק מִמְּךָ!" הָעֲנָק פָּעַר אֶת עֵינָיו בְּתִמָּהוֹן.

אַחַר כָּךְ הֵרִים הָעֲנָק אֶבֶן כְּבֵדָה וְזָרַק אוֹתָהּ רָחוֹק מְאֹד אֶל תּוֹךְ הַיַּעַר. "עַכְשָׁו תּוֹרְךָ," אָמַר בְּגַאֲוָה. הַחַיָּט הוֹצִיא מִכִּיסוֹ אֶת הַצִּפּוֹר הַקְּטַנָּה שֶׁתָּפַס בַּדֶּרֶךְ, וּבִמְקוֹם לִזְרֹק אֶבֶן, שִׁחְרֵר אוֹתָהּ לָעוּף. "רָאִיתָ?" אָמַר הַחַיָּט. "הָאֶבֶן שֶׁלִּי כָּל כָּךְ קַלָּה - הִיא עֲדַיִן טָסָה, וְלֹא נוֹחֶתֶת בִּכְלָל!" הָעֲנָק הִבִּיט בַּשָּׁמַיִם בְּפֶה פָּעוּר, מְשֻׁכְנָע כָּעֵת שֶׁהַחַיָּט חָזָק מִכָּל עֲנָק שֶׁפָּגַשׁ בְּחַיָּיו.

בְּכָבוֹד רַב הֵבִיא הָעֲנָק אֶת הַחַיָּט אֶל מְעָרָה עֲנָקִית, שָׁם יָשְׁנוּ עוֹד שְׁנֵי עֲנָקִים חֲבֵרָיו. הַחַיָּט הַפִּקֵּחַ לֹא שָׁכַב בַּמִּטָּה הָעֲנָקִית שֶׁהוּכְנָה לוֹ, אֶלָּא הִתְכַּרְבֵּל בְּפִנָּה חֲשׁוּכָה. בְּאֶמְצַע הַלַּיְלָה זָרַק אֶבֶן קְטַנָּה עַל אֶחָד הָעֲנָקִים הַיְּשֵׁנִים.

הָעֲנָק הָרִאשׁוֹן הִתְעוֹרֵר בְּכַעַס וְצָעַק עַל חֲבֵרוֹ הַיָּשֵׁן לְיָדוֹ: "לָמָּה הִכִּיתָ אוֹתִי? אַתָּה חוֹשֵׁב שֶׁאַתָּה חָזָק מִמֶּנִּי?" הָעֲנָק הַשֵּׁנִי, מְבֻלְבָּל, עָנָה בְּכַעַס: "אֲנִי חָזָק מִמְּךָ בְּהַרְבֵּה, לֹא אַתָּה מִמֶּנִּי!" שְׁנֵיהֶם קָפְצוּ עַל רַגְלֵיהֶם וְהֵחֵלּוּ לְהִתְכַּתֵּשׁ בְּכָל כֹּחָם, עַד שֶׁנָּפְלוּ שְׁנֵיהֶם עֲיֵפִים וּפְצוּעִים.

בְּעוֹד הָעֲנָקִים מִתְכַּתְּשִׁים, הַחַיָּט חָמַק בְּשֶׁקֶט מִן הַמְּעָרָה וְהִמְשִׁיךְ בְּדַרְכּוֹ, צוֹחֵק לְעַצְמוֹ: "הֵם חֲזָקִים מִמֶּנִּי בְּכֹחַ הַגּוּף, אֲבָל אֲנִי חָכָם מֵהֶם."

שְׁמוּעוֹת עַל הַחַיָּט הָאַמִּיץ שֶׁהִכָּה "שִׁבְעָה בְּמַכָּה אַחַת" הִקְדִּימוּ אוֹתוֹ אֶל אַרְמוֹן הַמֶּלֶךְ. הַמֶּלֶךְ, שֶׁפָּחַד מִפְּנֵי גִּבּוֹר כָּזֶה, כִּנֵּס אֶת יוֹעֲצָיו וְאָמַר בִּדְאָגָה: "אִם הַחַיָּט הַזֶּה חָזָק מִשְּׁנֵי עֲנָקִים בְּיַחַד, הוּא וַדַּאי חָזָק מֵאִתָּנוּ כֻּלָּנוּ!" הַיּוֹעֲצִים הִצִּיעוּ תָּכְנִית: לִשְׁלֹחַ אֶת הַחַיָּט לִמְשִׂימוֹת מְסֻכָּנוֹת, וְאִם יַצְלִיחַ - יִנָּשֵׂא לַנְּסִיכָה.

הַמְּשִׂימָה הָרִאשׁוֹנָה הָיְתָה לְהַבְרִיחַ שְׁנֵי עֲנָקִים מְסֻכָּנִים שֶׁהֵצִיקוּ לְתוֹשָׁבֵי הַיַּעַר. הַחַיָּט מָצָא אוֹתָם יְשֵׁנִים תַּחַת עֵץ, וְשׁוּב זָרַק אֲבָנִים קְטַנּוֹת מֵעַל רֹאשָׁם, עַד שֶׁהֵחֵלּוּ לְהַאֲשִׁים זֶה אֶת זֶה וּלְהִלָּחֵם בֵּינֵיהֶם, בְּדִיּוּק כְּפִי שֶׁקָּרָה בַּמְּעָרָה. כְּשֶׁהִתְעַיְּפוּ, גֵּרֵשׁ אוֹתָם הַחַיָּט מִן הַיַּעַר לְעוֹלָמִים.

הַמְּשִׂימָה הַשְּׁנִיָּה הָיְתָה לִלְכֹּד חַד־קֶרֶן פָּרוּעַ שֶׁהִשְׁתּוֹלֵל בַּיַּעַר. הַחַיָּט נִכְנַס לַיַּעַר, וּכְשֶׁהַחַיָּה הִסְתָּעֲרָה לְעֶבְרוֹ בְּקֶרֶן חַדָּה, קָפַץ הַצִּדָּה בִּזְרִיזוּת - וְהַחַד־קֶרֶן נָעַץ אֶת קַרְנוֹ עָמֹק בְּגֶזַע עֵץ עָבֶה, וְנִתְקַע בּוֹ הֵיטֵב. כָּךְ תָּפַס אוֹתוֹ הַחַיָּט בְּקַלּוּת, בְּלִי לְהָרִים אֲפִלּוּ אֶצְבַּע בְּכֹחַ.

הַמְּשִׂימָה הַשְּׁלִישִׁית הָיְתָה לִלְכֹּד חֲזִיר בָּר עֲנָק וּמְסֻכָּן. הַחַיָּט רָץ אֶל תּוֹךְ קָפֶּלָה קְטַנָּה בַּיַּעַר, וְהַחֲזִיר הַזּוֹעֵם רָץ אַחֲרָיו פְּנִימָה. בְּרֶגַע הָאַחֲרוֹן קָפַץ הַחַיָּט הַצִּדָּה דֶּרֶךְ חַלּוֹן קָטָן, וְנָעַל בִּמְהִירוּת אֶת דֶּלֶת הַקָּפֶּלָה - וְהַחֲזִיר נִשְׁאַר לָכוּד בִּפְנִים.

לַמְרוֹת שְׁלוֹשֶׁת הַנִּצְחוֹנוֹת, הַמַּלְכָּה עֲדַיִן פִּקְפְּקָה. הִיא פָּנְתָה אֶל בִּתָּהּ הַנְּסִיכָה וְשָׁאֲלָה בְּסָפֵק: "אַתְּ בֶּאֱמֶת חוֹשֶׁבֶת שֶׁהוּא חָכָם מִמֵּךְ?" הַנְּסִיכָה חִיְּכָה וְעָנְתָה: "הוּא חָכָם מִמֶּנִּי, אִמָּא, וְזֶה בְּדִיּוּק מַה שֶּׁאֲנִי אוֹהֶבֶת בּוֹ - לֹא הַכֹּחַ שֶׁלּוֹ, אֶלָּא הַשֵּׂכֶל שֶׁלּוֹ."

הַחַיָּט וְהַנְּסִיכָה נִשְּׂאוּ בַּחֲתֻנָּה גְּדוֹלָה וּמְפֹאָרָה, וְהַמֶּלֶךְ הֵבִין לְבַסּוֹף אֶת הָאֱמֶת: מֵעוֹלָם לֹא הָיָה הַחַיָּט חָזָק בְּכֹחַ הַגּוּף מֵהָעֲנָקִים אוֹ מֵהַחַיּוֹת שֶׁלָּכַד - אֲבָל הַחָכְמָה שֶׁלּוֹ הָיְתָה שָׁוָה הַרְבֵּה יוֹתֵר מִכָּל כֹּחַ שֶׁבָּעוֹלָם.

מְסֻפָּר עַל־פִּי סִפּוּר הָעַם הַגֶּרְמָנִי "הַחַיָּט הַקָּטָן הָאַמִּיץ" (Das tapfere Schneiderlein), שֶׁפִּרְסְמוּ הָאַחִים גְּרִים בְּשָׁנַת 1812, מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ.

כִּנּוּיֵי גוּף חֲבוּרִים עִם מִלַּת הַיַּחַס "מִן"

בְּשִׁיעוּר 54 לָמַדְנוּ אֶת כִּנּוּיֵי הַגּוּף שֶׁל "לְ-" (לִי, לְךָ, לוֹ...), בְּשִׁיעוּר 57 אֶת כִּנּוּיֵי הַגּוּף שֶׁל "אֶת" (אוֹתִי, אוֹתְךָ, אוֹתוֹ...), וּבְשִׁיעוּר 74 אֶת כִּנּוּיֵי הַגּוּף שֶׁל "עִם" וְ"עַל". הַיּוֹם נִלְמַד אֶת כִּנּוּיֵי הַגּוּף שֶׁל מִלַּת הַיַּחַס "מִן" (בִּקְצָרָה: "מִ-"), שֶׁיֵּשׁ לָהּ שְׁתֵּי מַשְׁמָעֻיּוֹת חֲשׁוּבוֹת.

מַשְׁמָעוּת 1 - מָקוֹם אוֹ מָקוֹר: "מִן" מְצַיֶּנֶת מֵאֵיפֹה מַשֶּׁהוּ יוֹצֵא אוֹ בָּא. לְמָשָׁל: "הַחַיָּט יָצָא מִן הָעִיר", "הוּא חָמַק מִן הַמְּעָרָה".

מַשְׁמָעוּת 2 - הַשְׁוָאָה: "מִן" מְצָרֶפֶת לְמִלַּת "יוֹתֵר" (אוֹ בִּמְקוֹמָהּ) כְּדֵי לְהַשְׁווֹת בֵּין שְׁנֵי דְּבָרִים. לְמָשָׁל: "אֲנִי חָזָק מִמְּךָ" פֵּרוּשׁוֹ "אֲנִי חָזָק יוֹתֵר מִמְּךָ".

🔑 כְּלָל זָהָב - זְהִירוּת מֵ"אֲרוּרַת" הַהוֹמוֹגְרָף: הַצּוּרָה "מִמֶּנּוּ" מְשַׁמֶּשֶׁת בְּעִקָּרוֹן גַּם לְ"הוּא" (מִמֶּנּוּ = מִמֶּנּוּ, הַחַיָּט) וְגַם לְ"אֲנַחְנוּ" (מִמֶּנּוּ = מֵאִתָּנוּ, מִכֻּלָּנוּ) - אוֹתָהּ מִלָּה בְּדִיּוּק, בְּלִי שׁוּם הֶבְדֵּל בְּנִיקוּד אוֹ בְּהִגּוּי! כְּדֵי לְהִמָּנַע מִבִּלְבּוּל, בְּשִׁיעוּר זֶה וּבְעִבְרִית מְדֻבֶּרֶת נְקִיָּה נִשְׁתַּמֵּשׁ בַּצּוּרָה "מִמֶּנּוּ" רַק בְּמוֹבָן "מִמֶּנּוּ, הוּא", וּבַצּוּרָה הַבְּרוּרָה "מֵאִתָּנוּ" בְּמוֹבָן "מִמֶּנּוּ, אֲנַחְנוּ".
💡 טִיפ: אֶפְשָׁר לְהַבְחִין בֵּין שְׁתֵּי הַמַּשְׁמָעֻיּוֹת לְפִי הַהֶקְשֵׁר בַּמִּשְׁפָּט - "יָצָא מִן הַבַּיִת" (מָקוֹם), לְעֻמַּת "גָּדוֹל מִמֶּנִּי" (הַשְׁוָאָה). שִׂימוּ לֵב גַּם: לִפְנֵי מִלָּה שֶׁמַּתְחִילָה בְּ-ה' הַיְּדִיעָה, "מִן" הוֹפֶכֶת לְ"מֵ-" (מִן + הַבַּיִת = מֵהַבַּיִת).
