📖 קְרִיאָה

בְּיוֹם סְתָו אֶחָד הָלְכָה תַּרְנְגֹלֶת קְטַנָּה בַּיַּעַר וְחִפְּשָׂה גַּרְעִינִים. בַּדֶּרֶךְ עָמַד אַלּוֹן גָּבוֹהַּ, וְעַל עֲנָפָיו הִתְנַדְנְדוּ עָלִים צְהֻבִּים. פִּתְאוֹם — טַק! — בַּלּוּט קָשֶׁה נָפַל מִן הָעֵץ וְנָחַת עַל רֹאשָׁהּ. הַתַּרְנְגֹלֶת קָפְצָה וְצָעֲקָה: ״הַשָּׁמַיִם נוֹפְלִים! אֲנִי חַיֶּבֶת לְסַפֵּר לַמֶּלֶךְ!״ וְהִיא רָצָה בְּכָל כֹּחָהּ.

בַּדֶּרֶךְ פָּגְשָׁה אֶת הַבַּרְוָזָה. ״לְאָן אַתְּ רָצָה?״ שָׁאֲלָה הַבַּרְוָזָה. ״הַשָּׁמַיִם נוֹפְלִים! הֵם נָפְלוּ עַל הָרֹאשׁ שֶׁלִּי!״ נִבְהֲלָה הַבַּרְוָזָה וְאָמְרָה: ״אִם כָּךְ, אֲנִי בָּאָה אִתֵּךְ.״ אַחֲרֵיהֶן הִצְטָרְפוּ גַּם הָאַוָּזָה וְתַרְנְגֹלֶת־הַהֹדּוּ. כֻּלָּן הָיוּ מְבֹהָלוֹת, וְכָל אַחַת צָעֲקָה חָזָק יוֹתֵר מֵחֲבֵרָתָהּ.

עַל גֶּזַע עֵץ יָשַׁב שׁוּעָל עִם חִיּוּךְ חַלָּק. ״שָׁמַעְתִּי שֶׁהַשָּׁמַיִם נוֹפְלִים,״ אָמַר. ״הַמְּעָרָה שֶׁלִּי בְּטוּחָה וְחַמָּה. בּוֹאוּ לְהִתְחַבֵּא אֶצְלִי.״ הַתַּרְנְגֹלֶת וַחֲבֶרוֹתֶיהָ הָלְכוּ אַחֲרָיו, וְהוּא לִקֵּק אֶת שְׂפָתָיו.

מֵעַל הָעֲנָפִים הִסְתַּכֵּל יַנְשׁוּף זָקֵן, וְהוּא רָאָה אֶת הַלִּקּוּק. ״עִצְרוּ!״ קָרָא. ״הַשּׁוּעָל רוֹצֶה לֶאֱכֹל אֶתְכֶם!״ אַחַר כָּךְ יָרַד אֶל הַתַּרְנְגֹלֶת וְשָׁאַל בְּשֶׁקֶט: ״מָה בְּדִיּוּק נָפַל עָלַיִךְ?״ הִיא הֶרְאֲתָה לוֹ אֶת הַבַּלּוּט. ״זֶה בַּלּוּט מִן הָאַלּוֹן,״ חִיֵּךְ הַיַּנְשׁוּף. ״הַשָּׁמַיִם עֲדַיִן בִּמְקוֹמָם: עֲנָנִים, שֶׁמֶשׁ וְכוֹכָבִים.״

הַשּׁוּעָל בָּרַח אֶל הַיַּעַר, וְהַתַּרְנְגֹלֶת הֵבִינָה מָה קָרָה. מֵאָז, לִפְנֵי שֶׁהִיא נִבְהֶלֶת, הִיא מִתְבּוֹנֶנֶת וּבוֹדֶקֶת מָה נָפַל בֶּאֱמֶת.

בַּסִּפּוּר מֻדְגָּשׁוֹת מִלִּים מִן הַשָּׂדוֹת ״עֵץ״ וְ״פַּחַד״, וְגַם הַפֹּעַל ״לִקֵּק״. מְעֻבָּד לְפִי סִפּוּר עַם אַנְגְּלִי מְפֻרְסָם.

שָׂדֶה סֶמַנְטִי

שָׂדֶה סֶמַנְטִי הוּא קְבוּצָה שֶׁל מִלִּים הַקְּשׁוּרוֹת לְאוֹתוֹ נוֹשֵׂא. הַמִּלִּים בַּשָּׂדֶה יְכוֹלוֹת לִהְיוֹת שֵׁמוֹת עֶצֶם, פְּעָלִים אוֹ שְׁמוֹת תֹּאַר, וְאֵין צֹרֶךְ שֶׁיִּהְיֶה לָהֶן אוֹתוֹ שֹׁרֶשׁ.

הַבְדֵּל מִמִּשְׁפַּחַת מִלִּים: מִשְׁפַּחַת מִלִּים כּוֹלֶלֶת מִלִּים מֵאוֹתוֹ שֹׁרֶשׁ (שָׁמַר, שׁוֹמֵר, מִשְׁמֶרֶת). שָׂדֶה סֶמַנְטִי כּוֹלֵל מִלִּים מִשָּׁרָשִׁים שׁוֹנִים שֶׁקְּשׁוּרוֹת בַּנּוֹשֵׂא. הַבְדֵּל מִנִּרְדָּפוֹת: נִרְדָּפוֹת הֵן מִלִּים שֶׁמַּשְׁמָעוּתָן דּוֹמָה, וְאִלּוּ מִלִּים בַּשָּׂדֶה אֵינָן דּוֹמוֹת: עָנָף וְעָלֶה הֵם חֲלָקִים שׁוֹנִים שֶׁל הָעֵץ.

מוּשָׂג־עַל הוּא שֵׁם כּוֹלֵל לְכָל הַשָּׂדֶה. לְמָשָׁל, לַמִּלִּים תַּרְנְגֹלֶת, בַּרְוָזָה וְאַוָּזָה יֵשׁ מוּשָׂג־עַל: עוֹפוֹת. כְּדֵי לִמְצֹא מִלָּה שֶׁאֵינָהּ שַׁיֶּכֶת לַשָּׂדֶה, שׁוֹאֲלִים: ״אֵיזוֹ מִלָּה לֹא קְשׁוּרָה לַנּוֹשֵׂא?״
