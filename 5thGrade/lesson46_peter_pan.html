פִּיטֶר פֶּן

יֶלֶד שֶׁמֵּעוֹלָם לֹא גָּדַל טָס עִם וֶנְדִּי וְאַחֶיהָ אֶל אִי קָסוּם - וְזֶהוּ שִׁיעוּר חֲזָרָה עַל אוֹתִיּוֹת הַשִּׁמּוּשׁ מש"ה וכל"ב

📖 קְרִיאָה

בְּבַיִת קָטָן בְּלוֹנְדוֹן, בְּחֶדֶר הַיְלָדִים, גָּרוּ שְׁלֹשָׁה אַחִים: וֶנְדִּי, אָחִיהָ הַבֵּינוֹנִי גֹּ'ן וְאָחִיהָ הַקָּטָן מַייקְל. כָּל עֶרֶב, אַחֲרֵי נְשִׁיקַת לַיְלָה טוֹב מֵאִמָּם, שָׁמְרָה עֲלֵיהֶם נָנָה הַכֶּלְבָּה הַנֶּאֱמָנָה, שֶׁיָּשְׁנָה לְיַד הַמִּטּוֹת שֶׁלָּהֶם כְּמוֹ שׁוֹמֶרֶת אֲמִתִּית.

לַיְלָה אֶחָד, כְּשֶׁכָּל הַבַּיִת יָשַׁן, נִכְנַס דֶּרֶךְ הַחַלּוֹן הַפָּתוּחַ יֶלֶד מוּזָר בְּבִגְדֵי עָלִים יְרֻקִּים. זֶה הָיָה פִּיטֶר פֶּן - יֶלֶד שֶׁמֵּעוֹלָם לֹא גָּדַל, וְשֶׁטָּס בָּאֲוִיר בְּלִי שׁוּם עֵזֶר. הוּא חִפֵּשׂ בְּכָל הַחֶדֶר אֶת הַצֵּל שֶׁלּוֹ, שֶׁנָּפַל מִמֶּנּוּ בְּבִקּוּר קוֹדֵם וְנִשְׁאַר סָגוּר בְּתוֹךְ מְגֵרָה. וֶנְדִּי הִתְעוֹרְרָה מִשְּׁנָתָהּ, וּבְמַחַט וָחוּט תָּפְרָה לוֹ בְּעֶדְנָה אֶת הַצֵּל בַּחֲזָרָה אֶל כַּפּוֹת הָרַגְלַיִם.

פִּיטֶר שָׂמַח כָּל כָּךְ, עַד שֶׁהוֹצִיא מִכִּיסוֹ חָלִיל קָטָן וְנִגֵּן בּוֹ נִגּוּן קֶסֶם, שֶׁגָּרַם לְכָל מִי שֶׁשָּׁמַע אוֹתוֹ לְהַרְגִּישׁ קַל כְּמוֹ נוֹצָה. מִיָּד רִחְפָה בָּאֲוִיר פֵּיָה זְעִירָה בְּשֵׁם פַּעֲמוֹנִית, שֶׁנָּצְצָה כְּמוֹ כּוֹכָב קָטָן. פִּיטֶר הִזְמִין אֶת שְׁלֹשֶׁת הָאַחִים לָטוּס אִתּוֹ אֶל אֶרֶץ־לְעוֹלָם־לֹא, אִי קָסוּם שֶׁבּוֹ יְלָדִים אֵינָם צְרִיכִים לְעוֹלָם לִגְדֹּל. פַּעֲמוֹנִית פִּזְּרָה עֲלֵיהֶם קֶמַח נוֹצֵץ מְיֻחָד - אֲבַק קֶסֶם - וּבְבַת אַחַת הֵחֵלּוּ הָאַחִים לָצוּף בָּאֲוִיר, מֵעַל לַמִּטּוֹת, מֵעַל לַגַּג וּמֵעַל לְכָל גַּגּוֹת לוֹנְדוֹן הַנּוֹצְצִים בְּאוֹר הַיָּרֵחַ.

הֵם טָסוּ יָמִים וְלֵילוֹת, מֵעַל עֲנָנִים וְכוֹכָבִים, עַד שֶׁלְּבַסּוֹף הוֹפִיעַ מִתַּחְתֵּיהֶם אִי יָרֹק וּמָלֵא הַרְפַּתְקָאוֹת: יַעַר עָבֹת, מְעָרוֹת סוֹדִיּוֹת מִתַּחַת לָאֲדָמָה וּמִפְרָץ שָׁקֵט שֶׁבּוֹ עָגְנָה סְפִינָה שְׁחֹרָה עִם מִפְרָשִׂים קְרוּעִים. בָּאִי חָיָה פִּיטֶר עִם קְבוּצַת יְלָדִים בְּשֵׁם "הַנְּעָרִים הָאֲבוּדִים" - יְלָדִים שֶׁנָּפְלוּ פַּעַם מֵעֶגְלוֹת הַתִּינוֹקוֹת שֶׁלָּהֶם וְאִישׁ לֹא בָּא לְחַפֵּשׂ אוֹתָם.

בְּתוֹךְ הַמִּפְרָץ עָגְנָה סְפִינַת שׁוֹדְדֵי הַיָּם, בְּפִקּוּדוֹ שֶׁל קַבַּרְנִיט אַכְזָרִי שֶׁכֻּלָּם כִּנּוּ אוֹתוֹ "קַבַּרְנִיט קֶרֶס" - כִּי בִּמְקוֹם יָד יְמָנִית הָיָה לוֹ קֶרֶס חַד וּמַבְרִיק מִבַּרְזֶל. פַּעַם, בִּקְרָב עִם פִּיטֶר, נָשַׁךְ תַּנִּין רָעֵב אֶת יָדוֹ שֶׁל הַקַּבַּרְנִיט וּבָלַע אוֹתָהּ יַחַד עִם שָׁעוֹן מְעוֹרֵר שֶׁל אֶחָד הַמַּלָּחִים. מֵאָז, בְּכָל פַּעַם שֶׁהַתַּנִּין הִתְקָרֵב, נִשְׁמַע "טִיק־טָק, טִיק־טָק" מִתּוֹךְ בִּטְנוֹ - וְקוֹל זֶה הִזְהִיר אֶת הַקַּבַּרְנִיט לִבְרֹחַ, כִּי הַתַּנִּין רָצָה לִטְעֹם גַּם אֶת שְׁאָר גּוּפוֹ.

קַבַּרְנִיט קֶרֶס לֹא שָׁכַח וְלֹא סָלַח לְפִיטֶר עַל יָדוֹ הָאֲבוּדָה, וְתִכְנֵן נְקָמָה. הוּא וְאַנְשֵׁי הַשּׁוֹדְדִים תָּפְסוּ אֶת חֲבֵרֵי פִּיטֶר - אֶת הַנְּעָרִים הָאֲבוּדִים, וְגַם אֶת וֶנְדִּי וְאַחֶיהָ - וְכָלְאוּ אוֹתָם עָמֹק בְּתוֹךְ הַסְּפִינָה.

פִּיטֶר לֹא פָּחַד. הוּא טִפֵּס בְּשֶׁקֶט עַל צַד הַסְּפִינָה, וְנִלְחַם בַּקַּבַּרְנִיט בְּחֶרֶב מוּל חֶרֶב, מֵעַל לַסִּפּוּן הָרוֹעֵד. בְּדִיּוּק כְּשֶׁקַּבַּרְנִיט קֶרֶס הִתְקָרֵב לְנַצֵּחַ, נִשְׁמַע מֵרָחוֹק הַצְּלִיל הַמֻּכָּר: "טִיק־טָק, טִיק־טָק!" הַתַּנִּין הָרָעֵב הִתְקָרֵב אֶל הַסְּפִינָה. קַבַּרְנִיט קֶרֶס נִבְהַל כָּל כָּךְ, עַד שֶׁקָּפַץ בְּעַצְמוֹ מֵהַסִּפּוּן אֶל תּוֹךְ הַמַּיִם - וְנֶעֱלַם לְעוֹלָמִים בֵּין הַגַּלִּים.

בְּאוֹתוֹ עֶרֶב, כְּשֶׁכֻּלָּם חָגְגוּ אֶת הַנִּצָּחוֹן, גִּלָּה פִּיטֶר שֶׁאוֹרָהּ שֶׁל פַּעֲמוֹנִית הוֹלֵךְ וְדוֹעֵךְ - הִיא הָיְתָה חֲלוּשָׁה מְאֹד. "אוֹר שֶׁל פֵּיָה דּוֹעֵךְ," לָחַשׁ פִּיטֶר, "רַק כְּשֶׁיְּלָדִים מַפְסִיקִים לְהַאֲמִין בְּפֵיוֹת." הוּא קָרָא בְּקוֹל גָּדוֹל: "מִי שֶׁמַּאֲמִין בְּפֵיוֹת - שֶׁיִּמְחָא כַּפַּיִם!" וֶנְדִּי וְאַחֶיהָ וְהַנְּעָרִים הָאֲבוּדִים מָחֲאוּ כַּף בְּכָל כֹּחָם, וּלְשִׂמְחַת כֻּלָּם - אוֹרָהּ שֶׁל פַּעֲמוֹנִית חָזַר לְהַבְהֵב חָזָק וּבָהִיר כְּמוֹ קֹדֶם.

וֶנְדִּי, שֶׁכְּבָר הִתְגַּעְגְּעָה מְאֹד אֶל אִמָּהּ, אָמְרָה לְפִיטֶר שֶׁהִגִּיעַ הַזְּמַן לַחֲזֹר הַבַּיְתָה עִם אַחֶיהָ. פִּיטֶר הִתְעַצֵּב, אֲבָל הֵבִין. הוּא לִוָּה אֶת שְׁלֹשֶׁת הָאַחִים בְּטִיסָה אֲרֻכָּה בַּחֲזָרָה אֶל הַחַלּוֹן שֶׁל חֶדֶר הַיְלָדִים בְּלוֹנְדוֹן, בְּדִיּוּק כְּשֶׁהַשֶּׁמֶשׁ הֵחֵלָּה לַעֲלוֹת.

פִּיטֶר הִבְטִיחַ לָשׁוּב מִדֵּי אָבִיב, לָקַחַת אֶת וֶנְדִּי לְנִקּוּי אָבִיב בְּאֶרֶץ־לְעוֹלָם־לֹא. הַשָּׁנִים חָלְפוּ, וֶנְדִּי גָּדְלָה וְהָפְכָה לְאִשָּׁה, אֲבָל פִּיטֶר פֶּן נִשְׁאַר תָּמִיד אוֹתוֹ יֶלֶד קָטָן - כִּי הוּא הָיָה הַיֶּלֶד הַיָּחִיד בָּעוֹלָם שֶׁמֵּעוֹלָם לֹא רָצָה לִגְדֹּל.

תַּזְכֹּרֶת: שֶׁבַע אוֹתִיּוֹת הַשִּׁמּוּשׁ - מש"ה וכל"ב

בְּשִׁיעוּר 26 לָמַדְנוּ שֶׁיֵּשׁ בְּעִבְרִית שֶׁבַע אוֹתִיּוֹת שֶׁמִּתְחַבְּרוֹת יְשִׁירוֹת אֶל תְּחִלַּת הַמִּלָּה, בְּלִי רֶוַח: מ, ש, ה, ו, כ, ל, ב - וְקַל לִזְכֹּר אוֹתָן דֶּרֶךְ הַמִּלִּים "מֹשֶׁה" וְ"כָּלֵב". הַיּוֹם נַזְכִּיר אֶת מַשְׁמָעוּת כָּל אוֹת, וְנִלְמַד מָה קוֹרֶה כְּשֶׁמְּחַבְּרִים שְׁתַּיִם מֵהֶן יַחַד בְּתוֹךְ אוֹתָהּ מִלָּה.
