📖 קְרִיאָה

פַּעַם הָיָה טוֹחֵן עָנִי שֶׁגָּר בִּכְפָר קָטָן עִם בִּתּוֹ הַיָּפָה. יוֹם אֶחָד עָבַר בַּכְּפָר הַמֶּלֶךְ, וְהַטּוֹחֵן, שֶׁרָצָה לְהַרְשִׁים אוֹתוֹ, אָמַר בְּלִי לַחְשֹׁב פַּעֲמַיִם: "לְבִתִּי יֵשׁ כִּשְׁרוֹן נִפְלָא - הִיא יוֹדַעַת לִטְווֹת קַשׁ וְלַהֲפֹךְ אוֹתוֹ לְזָהָב טָהוֹר!"

הַמֶּלֶךְ, שֶׁאָהַב זָהָב יוֹתֵר מִכָּל דָּבָר אַחֵר בָּעוֹלָם, שָׂמַח מְאֹד לִשְׁמֹעַ זֹאת. הוּא צִוָּה לְהָבִיא אֶת בַּת הַטּוֹחֵן אֶל אַרְמוֹנוֹ, סָגַר אוֹתָהּ בְּחֶדֶר מָלֵא קַשׁ עַד הַתִּקְרָה, וְנָתַן לָהּ פֶּלֶך לִטְוִיָּה. "עַד הַבֹּקֶר," אָמַר בְּקוֹל קָשֶׁה, "כָּל הַקַּשׁ הַזֶּה חַיָּב לַהֲפֹךְ לְזָהָב - וְאִם לֹא, לֹא תֵּצְאִי מִן הַחֶדֶר הַזֶּה."

בַּת הַטּוֹחֵן נִשְׁאֲרָה לְבַדָּהּ בַּחֶדֶר, בּוֹכָה בְּיֵאוּשׁ, שֶׁכֵּן כַּמּוּבָן לֹא יָדְעָה כְּלָל אֵיךְ לִטְווֹת זָהָב. פִּתְאֹם נִפְתְּחָה הַדֶּלֶת, וְנִכְנַס אִישׁ זָעִיר וּמְשֻׁנֶּה, לֹא גָּבֹהַּ מֵאֶצְבַּע. "לָמָּה אַתְּ בּוֹכָה?" שָׁאַל בְּקוֹל דַּקִּיק. כְּשֶׁסִּפְּרָה לוֹ עַל צָרָתָהּ, חִיֵּךְ הָאִישׁ הַזָּעִיר וְאָמַר: "מָה תִּתְּנִי לִי, אִם אֶטְוֶה בִּשְׁבִילֵךְ אֶת כָּל הַקַּשׁ הַזֶּה?"

"אֶת טַבַּעְתִּי," עָנְתָה הַנַּעֲרָה, וְהוֹרִידָה מֵאֶצְבָּעָהּ טַבַּעַת קְטַנָּה וּפְשׁוּטָה. הָאִישׁ הַזָּעִיר לָקַח אוֹתָהּ בְּשִׂמְחָה, יָשַׁב לְיַד הַפֶּלֶך, וְטָוָה בִּמְהִירוּת מַפְתִּיעָה כָּל הַלַּיְלָה, עַד שֶׁכָּל עֲרֵמַת הַקַּשׁ הָפְכָה לְחוּטֵי זָהָב מַבְרִיקִים. בַּבֹּקֶר, כְּשֶׁרָאָה הַמֶּלֶךְ אֶת הַזָּהָב, עֵינָיו נִצְנְצוּ מֵאֹשֶׁר - וְהוּא רָצָה עוֹד.

הוּא הֶעֱבִיר אֶת בַּת הַטּוֹחֵן לְחֶדֶר גָּדוֹל עוֹד יוֹתֵר, מָלֵא קַשׁ עַד הָאֲרֻבָּה, וְחָזַר עַל אוֹתָהּ הַדְּרִישָׁה. שׁוּב הוֹפִיעַ הָאִישׁ הַזָּעִיר, וְשׁוּב שָׁאַל מָה תִּתֵּן לוֹ הַפַּעַם. הַנַּעֲרָה הוֹרִידָה אֶת הַשַּׁרְשֶׁרֶת שֶׁעַל צַוָּארָהּ. הָאִישׁ הַזָּעִיר לָקַח אוֹתָהּ בְּחִיּוּךְ, וְעַד הַבֹּקֶר הָיָה כָּל הַקַּשׁ בַּחֶדֶר לְזָהָב מַבְרִיק.

בַּלַּיְלָה הַשְּׁלִישִׁי הֵבִיא הַמֶּלֶךְ אֶת הַנַּעֲרָה לְחֶדֶר עוֹד יוֹתֵר גָּדוֹל, וְהִבְטִיחַ שֶׁאִם תַּצְלִיחַ גַּם הַפַּעַם, יִשָּׂא אוֹתָהּ לְאִשָּׁה וְתִהְיֶה לְמַלְכָּה. אַךְ לַנַּעֲרָה כְּבָר לֹא נוֹתַר דָּבָר לָתֵת לָאִישׁ הַזָּעִיר כְּשֶׁהוֹפִיעַ שׁוּב. "אִם תִּהְיִי מַלְכָּה," אָמַר, "תַּבְטִיחִי לִי אֶת בְּנֵךְ הַבְּכוֹר, כְּשֶׁיִּוָּלֵד." הַנַּעֲרָה הִתְלַבְּטָה, אֲבָל לֹא רָאֲתָה בְּרֵרָה אַחֶרֶת, וְהִסְכִּימָה.

הָאִישׁ הַזָּעִיר טָוָה שׁוּב אֶת כָּל הַקַּשׁ לְזָהָב, וְהַמֶּלֶךְ, מְאֻשָּׁר, נָשָׂא אֶת בַּת הַטּוֹחֵן לְאִשָּׁה. הִיא הָיְתָה לְמַלְכָּה, וְשָׁכְחָה כִּמְעַט לְגַמְרֵי אֶת הַהַבְטָחָה שֶׁנָּתְנָה בִּשְׁעַת צָרָה.

כַּעֲבֹר שָׁנָה נוֹלַד לַמַּלְכָּה בֵּן בָּרִיא וְיָפֶה, וְשִׂמְחָתָהּ הָיְתָה גְּדוֹלָה. אַךְ לַיְלָה אֶחָד, בְּלִי אַזְהָרָה, הוֹפִיעַ הָאִישׁ הַזָּעִיר בְּחַדְרָהּ. "הִגִּיעַ הַזְּמַן," אָמַר, "תְּנִי לִי אֶת בְּנֵךְ, כְּפִי שֶׁהִבְטַחְתְּ." הַמַּלְכָּה נִבְהֲלָה וּבָכְתָה, וְהִצִּיעָה לוֹ כָּל אוֹצְרוֹת הַמַּמְלָכָה תְּמוּרַת הַיֶּלֶד. אֲבָל הָאִישׁ הַזָּעִיר סֵרֵב בְּתֹקֶף.

לְבַסּוֹף רִחֵם עָלֶיהָ הָאִישׁ הַזָּעִיר, וְנָתַן לָהּ הִזְדַּמְּנוּת אַחֲרוֹנָה: "אֶתֵּן לָךְ שְׁלֹשָׁה יָמִים. אִם בְּתוֹךְ שְׁלֹשָׁה יָמִים תְּנַחֲשִׁי מַה שְּׁמִי, תּוּכְלִי לִשְׁמֹר אֶת בְּנֵךְ אֶצְלֵךְ."

בַּלַּיְלָה הָרִאשׁוֹן וּבַלַּיְלָה הַשֵּׁנִי שָׁלְחָה הַמַּלְכָּה שְׁלִיחִים לְכָל פִּנָּה בַּמַּמְלָכָה, כְּדֵי לֶאֱסֹף כָּל שֵׁם מְשֻׁנֶּה שֶׁאֶפְשָׁר לִמְצֹא. אַךְ בְּכָל פַּעַם שֶׁנִּחֲשָׁה, עָנָה הָאִישׁ הַזָּעִיר בְּחִיּוּךְ קָטָן: "לֹא, זֶה לֹא שְׁמִי."

בַּיּוֹם הַשְּׁלִישִׁי חָזַר אֶחָד הַשְּׁלִיחִים מִן הַיַּעַר וְסִפֵּר סִפּוּר מוּזָר: הוּא רָאָה בַּיִת קָטָן וּמוּזָר בְּלֵב הַיַּעַר, וּלְיָדוֹ מְדוּרָה קְטַנָּה. סָבִיב הַמְּדוּרָה רָקַד אִישׁ זָעִיר וְשָׁר לְעַצְמוֹ: "הַיּוֹם אֹפֶה, מָחָר אֲבַשֵּׁל, וּמָחֳרָתַיִם אֶקַּח אֶת בֶּן הַמַּלְכָּה. כַּמָּה טוֹב שֶׁאִישׁ לֹא יוֹדֵעַ שֶׁשְּׁמִי הוּא... חִזְקוּזִי!"

הַשָּׁלִיחַ מִהֵר בַּחֲזָרָה אֶל הָאַרְמוֹן וְסִפֵּר לַמַּלְכָּה מָה שָׁמַע. בָּעֶרֶב, כְּשֶׁהוֹפִיעַ הָאִישׁ הַזָּעִיר וְשָׁאַל בְּגַאֲוָה: "וּבְכֵן, מַה שְּׁמִי?", חִיְּכָה הַמַּלְכָּה וְאָמְרָה בְּשֶׁקֶט: "אוּלַי קוֹרְאִים לְךָ דָּן?" "לֹא!" צָחַק הָאִישׁ הַזָּעִיר. "אוּלַי מֹשֶׁה?" "לֹא וְלֹא!" צָחַק שׁוּב. אָז אָמְרָה הַמַּלְכָּה בְּחִיּוּךְ קָטָן: "אָז אוּלַי - שִׁמְךָ הוּא חִזְקוּזִי?"

הָאִישׁ הַזָּעִיר הִשְׁתַּתֵּק, פָּנָיו הֶאֱדִימוּ מִזַּעַם, וְהוּא רָקַע בְּרַגְלוֹ עַל הָרִצְפָּה בְּכָל כֹּחוֹ, עַד שֶׁרַגְלוֹ שָׁקְעָה עָמֹק בָּאֲדָמָה. הוּא מָשַׁךְ וּמָשַׁךְ, כָּעוּס וְנָבוֹךְ, וְלִבְסוֹף בָּרַח מִן הָאַרְמוֹן וְלֹא חָזַר עוֹד. הַמַּלְכָּה חִבְּקָה אֶת בְּנָהּ בְּאֹשֶׁר, וּמֵאָז חָיְתָה עִם בַּעְלָהּ וּבְנָהּ בְּשָׁלוֹם רַב.

מְסֻפָּר עַל־פִּי אַגָּדַת עַם גֶּרְמָנִית שֶׁאָסְפוּ הָאַחִים גְּרִים (הַמֵּאָה הַ-19) - מְעֻבֶּדֶת

מָה זֶה כִּנּוּי קִנְיָן?

כְּשֶׁרוֹצִים לוֹמַר שֶׁמַּשֶּׁהוּ שַׁיָּךְ לְמִישֶׁהוּ, יֵשׁ בְּעִבְרִית שְׁתֵּי דְּרָכִים. הַדֶּרֶךְ הַמֻּכֶּרֶת: מִלַּת "שֶׁל" + כִּנּוּי גוּף - "הַבַּיִת שֶׁלִּי". דֶּרֶךְ נוֹסֶפֶת, קְצָרָה וְעָשִׁירָה יוֹתֵר: מְצָרְפִים סִיּוֹמֶת קְטַנָּה יָשָׁר לְסוֹף שֵׁם הָעֶצֶם עַצְמוֹ, בְּלִי הַמִּלָּה "שֶׁל" בִּכְלָל - "בֵּיתִי". הַסִּיּוֹמֶת הַזֹּאת נִקְרֵאת כִּנּוּי קִנְיָן, וְהִיא מִשְׁתַּנָּה לְפִי מִי הַבְּעָלִים.
