📖 קְרִיאָה

בִּכְפָר קָטָן בְּרוּסְיָה גָּרוּ שְׁלוֹשָׁה אַחִים. שְׁנֵי הָאַחִים הַגְּדוֹלִים הָיוּ חֲרוּצִים, וְעָבְדוּ מֵהַבֹּקֶר עַד הָעֶרֶב. אֲבָל הָאָח הַצָּעִיר, יֶמֶלְיָה, הָיָה עַצְלָן גָּדוֹל. כָּל הַיּוֹם הוּא שָׁכַב עַל הַתַּנּוּר הֶחָם שֶׁבַּבַּיִת — תַּנּוּר רוּסִי גָּדוֹל מֵאֲבָנִים, שֶׁאֶפְשָׁר לִישֹׁן עָלָיו — וְלֹא רָצָה לָזוּז מִשָּׁם.

יוֹם אֶחָד יָצְאוּ הָאַחִים לָעִיר לִמְכֹּר אֶת הַסְּחוֹרָה שֶׁלָּהֶם, וְהַנָּשִׁים שֶׁלָּהֶם, הַגִּיסוֹת שֶׁל יֶמֶלְיָה, נִשְׁאֲרוּ אִתּוֹ בַּבַּיִת. הַגִּיסוֹת הָיוּ רַגְזָנִיּוֹת, וְכָל דָּבָר קָטָן הִרְגִּיז אוֹתָן. "יֶמֶלְיָה, קוּם וְלֵךְ לַנָּהָר לְהָבִיא מַיִם!" צָעֲקוּ. "לֹא רוֹצֶה," רָטַן יֶמֶלְיָה מֵעַל הַתַּנּוּר. "אִם תֵּלֵךְ, נֶאֱפֶה לְךָ עוּגָה," הִבְטִיחוּ הַגִּיסוֹת. רַק אָז יָרַד יֶמֶלְיָה מֵהַתַּנּוּר, לָקַח גַּרְזֶן וּשְׁנֵי דְּלָיִים, וְיָצָא.

הָיָה חֹרֶף, וְהַנָּהָר הָיָה קָפוּא. יֶמֶלְיָה חָצַב בַּגַּרְזֶן חֹר בַּקֶּרַח וּמִלֵּא אֶת הַדְּלָיִים. פִּתְאוֹם רָאָה בְּתוֹךְ אֶחָד הַדְּלָיִים דָּג גָּדוֹל — זְאֵב מַיִם. "אֵיזֶה מָרָק טָעִים יֵצֵא מִמֶּנּוּ!" שָׂמַח יֶמֶלְיָה. אֲבָל הַדָּג פָּתַח אֶת פִּיו וְדִבֵּר: "יֶמֶלְיָה, הַחְזֵר אוֹתִי לַמַּיִם, וַאֲנִי אֲמַלֵּא כָּל מִשְׁאָלָה שֶׁלְּךָ."

יֶמֶלְיָה אוּלַי הָיָה עַצְלָן, אֲבָל הוּא לֹא הָיָה פַּחְדָּן, וְגַם לֹא הָיָה תָּמִים. "וּמִי אוֹמֵר לִי שֶׁאַתָּה לֹא שַׁקְרָן?" שָׁאַל. "קֹדֶם תּוֹכִיחַ!" עָנָה הַדָּג: "אֱמֹר אֶת הַמִּלִּים הָאֵלֶּה: 'לְפִי פְּקֻדַּת הַדָּג וּלְפִי רְצוֹנִי' — וּבַקֵּשׁ מָה שֶׁתִּרְצֶה." אָמַר יֶמֶלְיָה: "לְפִי פְּקֻדַּת הַדָּג וּלְפִי רְצוֹנִי — דְּלָיִים, לְכוּ הַבַּיְתָה לְבַד!" וּמִיָּד הִתְחִילוּ הַדְּלָיִים לִצְעֹד בְּעַצְמָם בַּשְּׁבִיל הַמּוֹשְׁלָג, בְּלִי לִשְׁפֹּךְ אַף טִפָּה.

יֶמֶלְיָה הֶחְזִיר אֶת הַדָּג לַנָּהָר, וְהָלַךְ אַחֲרֵי הַדְּלָיִים. אַנְשֵׁי הַכְּפָר הַסַּקְרָנִים יָצְאוּ מֵהַבָּתִּים וְהִסְתַּכְּלוּ בְּפֶה פָּתוּחַ. וְיֶמֶלְיָה? הוּא נִכְנַס הַבַּיְתָה וְטִפֵּס בַּחֲזָרָה אֶל הַתַּנּוּר.

מֵאָז, כְּשֶׁהַגִּיסוֹת בִּקְּשׁוּ עֵצִים לַתַּנּוּר, הַגַּרְזֶן חָטַב אוֹתָם לְבַד, וְהָעֵצִים נִכְנְסוּ לַבַּיִת בְּעַצְמָם. וּכְשֶׁהָיָה צָרִיךְ לְהָבִיא עֵצִים מֵהַיַּעַר, יֶמֶלְיָה יָשַׁב עַל מִזְחֶלֶת — וְהַמִּזְחֶלֶת נָסְעָה בְּלִי סוּס, מַהֵר כָּל כָּךְ שֶׁכָּל הָאֲנָשִׁים בַּדֶּרֶךְ קָפְצוּ הַצִּדָּה.

הַסִּפּוּר עַל הַמִּזְחֶלֶת שֶׁנּוֹסַעַת לְבַד הִגִּיעַ עַד הַמֶּלֶךְ. הַמֶּלֶךְ שָׁלַח קָצִין לְהָבִיא אֶת יֶמֶלְיָה אֶל הָאַרְמוֹן. אֲבָל יֶמֶלְיָה הָיָה עַקְשָׁן: "טוֹב לִי עַל הַתַּנּוּר, וְלֹא אֵלֵךְ לְשׁוּם מָקוֹם!" רַק כְּשֶׁהַקָּצִין הִבְטִיחַ לוֹ מַגָּפַיִם חֲדָשִׁים וּמְעִיל אָדֹם, אָמַר יֶמֶלְיָה: "לְפִי פְּקֻדַּת הַדָּג וּלְפִי רְצוֹנִי — תַּנּוּר, סַע אֶל הַמֶּלֶךְ!" וְהַתַּנּוּר יָצָא מֵהַבַּיִת וְנָסַע בַּדֶּרֶךְ, וְיֶמֶלְיָה שׁוֹכֵב עָלָיו.

בָּאַרְמוֹן, הַנְּסִיכָה מַרְיָה הַבַּיְשָׁנִית הֵצִיצָה מִן הַחַלּוֹן. לָחַשׁ יֶמֶלְיָה אֶת מִלּוֹת הַקֶּסֶם, וּבִקֵּשׁ שֶׁהִיא תֹּאהַב אוֹתוֹ. וְכָךְ הָיָה. אֲבָל הַמֶּלֶךְ הָיָה כַּעֲסָן, וּכְשֶׁשָּׁמַע שֶׁבִּתּוֹ רוֹצָה לְהִתְחַתֵּן עִם הָעַצְלָן מֵהַכְּפָר, הִתְפָּרֵץ בְּזַעַם: הוּא צִוָּה לָשִׂים אֶת שְׁנֵיהֶם בְּתוֹךְ חָבִית גְּדוֹלָה וְלִזְרֹק אוֹתָהּ לַיָּם.

הֶחָבִית שָׁטָה עַל הַגַּלִּים, וְהַנְּסִיכָה בָּכְתָה. "לְפִי פְּקֻדַּת הַדָּג וּלְפִי רְצוֹנִי," אָמַר יֶמֶלְיָה, "שֶׁהֶחָבִית תַּגִּיעַ לְאִי, וְשֶׁיִּהְיֶה בּוֹ אַרְמוֹן יָפֶה." וּמִיָּד עָמְדוּ שְׁנֵיהֶם עַל אִי יָרֹק, מוּל אַרְמוֹן מִזָּהָב. בַּפַּעַם הָאַחֲרוֹנָה בִּקֵּשׁ יֶמֶלְיָה שֶׁיִּהְיֶה יָפֶה וְחָכָם — וְגַם זֶה קָרָה.

כְּשֶׁהַמֶּלֶךְ שָׁט לְיַד הָאִי וְרָאָה אֶת הָאַרְמוֹן הַנִּפְלָא, הוּא נִכְנַס לְבַקֵּר — וְלֹא הִכִּיר אֶת יֶמֶלְיָה. רַק בְּסוֹף הַסְּעוּדָה גִּלָּה לוֹ יֶמֶלְיָה מִי הוּא. הַמֶּלֶךְ הִתְבַּיֵּשׁ, בִּקֵּשׁ סְלִיחָה, וְעָרַךְ לָהֶם חֲתֻנָּה גְּדוֹלָה. וְאוֹמְרִים שֶׁגַּם בָּאַרְמוֹן, יֶמֶלְיָה לֹא נִפְרַד מֵהָעַצְלָנוּת שֶׁלּוֹ — רַק הַתַּנּוּר שֶׁלּוֹ נַעֲשָׂה קְצָת יוֹתֵר מְפֹאָר.

📜 עִבּוּד חָפְשִׁי לְסִפּוּר הָעַם הָרוּסִי "לְפִי פְּקֻדַּת הַדָּג" (По щучьему велению), שֶׁנֶּאֱסַף בַּמֵּאָה הַ־19 בְּאֹסֶף סִפּוּרֵי הָעַם שֶׁל אֲלֶכְּסַנְדֶּר אֲפָנַסְיֶב.

📚 דִּקְדּוּק — מִשְׁקַל קַטְלָן

קַטְלָן — מִלִּים שֶׁמְּתָאֲרוֹת אָדָם לְפִי הַתְּכוּנָה שֶׁלּוֹ

בַּשִּׁיעוּרִים הַקּוֹדְמִים פָּגַשְׁנוּ מִשְׁקָלִים שֶׁל שֵׁם עֶצֶם: קַטָּל לְבַעֲלֵי מִקְצוֹעַ (נַגָּר), מַקְטֵל לְכֵלִים (מַפְתֵּחַ), מִקְטָל לִמְקוֹמוֹת (מִגְדָּל) וְקַטֶּלֶת לְמַחֲלוֹת (שַׁפַּעַת). הַיּוֹם נַכִּיר מִשְׁקָל שֶׁמְּתָאֵר אָדָם לְפִי הַתְּכוּנָה שֶׁלּוֹ, אוֹ לְפִי מַשֶּׁהוּ שֶׁהוּא עוֹשֶׂה הַרְבֵּה: מִשְׁקַל קַטְלָן.

אֵיךְ מְזַהִים? פַּתָּח בָּאוֹת הָרִאשׁוֹנָה שֶׁל הַשֹּׁרֶשׁ, שְׁוָא בַּשְּׁנִיָּה, קָמָץ בַּשְּׁלִישִׁית — וּבַסּוֹף הַסִּיּוֹמֶת ־ָן. לְמָשָׁל: שֹׁרֶשׁ ע־צ־ל ← עַצְלָן, שֹׁרֶשׁ פ־ח־ד ← פַּחְדָּן.

שִׂימוּ לֵב לְאוֹתִיּוֹת גְּרוֹנִיּוֹת: כְּשֶׁהָאוֹת הַשְּׁנִיָּה בַּשֹּׁרֶשׁ הִיא ע׳, בָּא תַּחְתֶּיהָ חֲטַף פַּתָּח בִּמְקוֹם שְׁוָא: כ־ע־ס ← כַּעֲסָן (אֲבָל תַּחַת ח׳ יֵשׁ שְׁוָא רָגִיל, כְּמוֹ בְּפַחְדָּן). וְאַחֲרֵי הַשְּׁוָא, אוֹתִיּוֹת בֹּגֶ״ד כֶּפֶ״ת (ב׳ ג׳ ד׳ כ׳ פ׳ ת׳) מְקַבְּלוֹת דָּגֵשׁ: דַּבְּרָן וַתְּרָן פַּחְדָּן.

לֹא רַק תְּכוּנוֹת: לִפְעָמִים קַטְלָן מְתָאֵר מִי שֶׁעוֹשֶׂה פְּעֻלָּה הַרְבֵּה, אֲפִלּוּ כְּתַחְבִּיב אוֹ מִקְצוֹעַ: שַׂחְקָן רַקְדָּן שַׂחְיָן.

כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים עַל דְּמֻיּוֹת מֵהַסִּפּוּר אוֹ מֵהַחַיִּים שֶׁלָּכֶם. בְּכָל מִשְׁפָּט הִשְׁתַּמְּשׁוּ בְּמִלָּה אַחֶרֶת בְּמִשְׁקַל קַטְלָן — וְהַסְבִּירוּ בְּאוֹתוֹ מִשְׁפָּט לָמָּה הַדְּמוּת הִיא כָּזוֹ. דֻּגְמָה: ״יֶמֶלְיָה הָיָה עַצְלָן, כִּי הוּא שָׁכַב כָּל הַיּוֹם עַל הַתַּנּוּר.״ נַסּוּ לְשַׁלֵּב לְפָחוֹת פַּעַם אַחַת צוּרַת נְקֵבָה (־ָנִית) וּפַעַם אַחַת שֵׁם תְּכוּנָה (־וּת).
