רוּמְפֶּלְשְׁטִילְצְקִין

אִישׁ קָטָן וּמִסְתּוֹרִי, הַבְטָחָה שֶׁנִּשְׁכְּחָה, וְשֵׁם סוֹדִי שֶׁמַּצִּיל תִּינוֹק

📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת חַי בִּכְפָר קָטָן טַחָן עָנִי, שֶׁהָיָה גֵּאֶה מְאֹד בְּבִתּוֹ הַיָּפָה. יוֹם אֶחָד, כְּשֶׁפָּגַשׁ אֶת הַמֶּלֶךְ בַּדֶּרֶךְ, רָצָה הַטַּחָן לְהַרְשִׁים אוֹתוֹ, וְהִתְפָּאֵר: "הוֹד מַלְכוּתְךָ, יֵשׁ לִי בַּת חֲכָמָה כָּל כָּךְ, שֶׁהִיא יוֹדַעַת לִטְווֹת קַשׁ וְלַהֲפֹךְ אוֹתוֹ לְזָהָב טָהוֹר!"

הַמֶּלֶךְ שָׂמַח מְאֹד לִשְׁמֹעַ זֹאת, וּבוֹ בַּיּוֹם צִוָּה לְהָבִיא אֶת בַּת הַטַּחָן אֶל אַרְמוֹנוֹ. הוּא הוֹבִיל אוֹתָהּ אֶל חֶדֶר קָטָן וְחָשׁוּךְ, שֶׁהָיָה מָלֵא קַשׁ מֵהָרִצְפָּה וְעַד הַתִּקְרָה, וְעָמַד בּוֹ גַּלְגַּל טְוִיָּה יָשָׁן. "אִם עַד הַבֹּקֶר לֹא יַהֲפֹךְ הַקַּשׁ הַזֶּה לְזָהָב," אָמַר הַמֶּלֶךְ בְּקוֹל תַּקִּיף, "תִּשָּׁאֲרִי סְגוּרָה בָּאַרְמוֹן הַזֶּה לְעוֹלָם." וְאָז נָעַל אֶת הַדֶּלֶת מֵאֲחוֹרָיו.

בַּת הַטַּחָן נִשְׁאֲרָה לְבַדָּהּ בַּחֶדֶר, וְלֹא יָדְעָה כְּלָל אֵיךְ לִטְווֹת קַשׁ לְזָהָב. הִיא יָשְׁבָה עַל הָרִצְפָּה וּבָכְתָה בְּמָרָה שְׁחוֹרָה. פִּתְאֹם נִפְתְּחָה הַדֶּלֶת בְּלִי קוֹל, וְנִכְנַס אִישׁ קָטַנְטַן וּמוּזָר, בַּעַל זָקָן אָרֹךְ וְעֵינַיִם מַבְרִיקוֹת. "לָמָּה אַתְּ בּוֹכָה?" שָׁאַל הָאִישׁ הַקָּטָן.

"עָלַי לַהֲפֹךְ אֶת כָּל הַקַּשׁ הַזֶּה לְזָהָב עַד הַבֹּקֶר," עָנְתָה הַנַּעֲרָה בִּדְמָעוֹת, "וַאֲנִי בִּכְלָל לֹא יוֹדַעַת אֵיךְ." הָאִישׁ הַקָּטָן חִיֵּךְ בְּעָרְמָה. "אֲנִי יָכוֹל לַעֲשׂוֹת זֹאת בִּשְׁבִילֵךְ," אָמַר, "אִם תִּתְּנִי לִי אֶת שַׁרְשֶׁרֶת הַזָּהָב שֶׁעַל צַוָּארֵךְ." הַנַּעֲרָה, שֶׁלֹּא הָיְתָה לָהּ בְּרֵרָה אַחֶרֶת, הֵסִירָה אֶת הַשַּׁרְשֶׁרֶת וְנָתְנָה לוֹ אוֹתָהּ. הָאִישׁ הַקָּטָן יָשַׁב לְיַד גַּלְגַּל הַטְּוִיָּה, וְטָוָה כָּל הַלַּיְלָה - וִיר, וִיר, וִיר - עַד שֶׁכָּל הַקַּשׁ בַּחֶדֶר הָפַךְ לְגַלִּים גַּלִּים שֶׁל חוּט זָהָב נוֹצֵץ.

בַּבֹּקֶר מָצָא הַמֶּלֶךְ אֶת הַחֶדֶר מָלֵא זָהָב, וְשָׂמַח מְאֹד - אֲבָל גַּם נַעֲשָׂה חַמְדָּן יוֹתֵר. הוּא הוֹבִיל אֶת הַנַּעֲרָה לְחֶדֶר גָּדוֹל עוֹד יוֹתֵר, מָלֵא קַשׁ עַד הַתִּקְרָה, וְחָזַר עַל אוֹתָהּ הַדְּרִישָׁה. וְשׁוּב, בַּלַּיְלָה, הוֹפִיעַ הָאִישׁ הַקָּטָן. הַפַּעַם הוּא בִּקֵּשׁ אֶת הַטַּבַּעַת שֶׁעַל אֶצְבָּעָהּ, וְהַנַּעֲרָה נָתְנָה לוֹ אוֹתָהּ בְּרָצוֹן - וְשׁוּב טָוָה הָאִישׁ הַקָּטָן אֶת כָּל הַקַּשׁ לְזָהָב מַבְרִיק.

בַּבֹּקֶר הַשְּׁלִישִׁי הָיָה הַמֶּלֶךְ נִרְגָּשׁ מִכָּל פַּעַם קֹדֶם. הוּא הוֹבִיל אֶת הַנַּעֲרָה לְחֶדֶר עֲנָקִי, גָּדוֹל מִכָּל הַחֲדָרִים הַקּוֹדְמִים, וְאָמַר: "אִם תַּהַפְכִי גַּם אֶת הַקַּשׁ הַזֶּה לְזָהָב, אֶשָּׂא אוֹתָךְ לְאִשָּׁה וְתִהְיִי לְמַלְכָּה!" וְשׁוּב הוֹפִיעַ הָאִישׁ הַקָּטָן בַּלַּיְלָה. אַךְ הַפַּעַם לֹא נוֹתַר לַנַּעֲרָה שׁוּם תַּכְשִׁיט לָתֵת לוֹ. "אֵין לִי דָּבָר יוֹתֵר," אָמְרָה בְּיֵאוּשׁ. "אָז תַּבְטִיחִי לִי," אָמַר הָאִישׁ הַקָּטָן בְּקוֹל שָׁקֵט, "שֶׁכַּאֲשֶׁר תִּהְיִי מַלְכָּה, תִּתְּנִי לִי אֶת יַלְדֵּךְ הַבְּכוֹר." הַנַּעֲרָה חָשְׁבָה שֶׁזֶּה יִקְרֶה רַק בְּעָתִיד רָחוֹק, וּבְלִי בְּרֵרָה הִבְטִיחָה. הָאִישׁ הַקָּטָן טָוָה אֶת כָּל הַקַּשׁ לְזָהָב, וְנֶעֱלַם.

הַמֶּלֶךְ עָמַד בְּהַבְטָחָתוֹ, וְנָשָׂא אֶת בַּת הַטַּחָן לְאִשָּׁה. הִיא הָפְכָה לְמַלְכָּה, וְשָׁכְחָה בַּהֲדָרָגָה אֶת הַהַבְטָחָה שֶׁהִבְטִיחָה לָאִישׁ הַקָּטָן וְהַמּוּזָר. כַּעֲבֹר כִּמְעַט שָׁנָה נוֹלַד לַמַּלְכָּה תִּינוֹק, וְהִיא הָיְתָה הָאִשָּׁה הַמְאֻשֶּׁרֶת בְּיוֹתֵר בָּעוֹלָם. אַךְ לַיְלָה אֶחָד נִכְנַס לְחַדְרָהּ הָאִישׁ הַקָּטָן שֶׁכְּבָר שָׁכְחָה, וְאָמַר: "הִגִּיעַ הַזְּמַן שֶׁתְּקַיְּמִי אֶת הַבְטָחָתֵךְ."

הַמַּלְכָּה נִבְהֲלָה וְהִתְחַנְּנָה, וְהִצִּיעָה לוֹ כָּל אוֹצְרוֹת הַמַּמְלָכָה תְּמוּרַת הַתִּינוֹק. הָאִישׁ הַקָּטָן סֵרֵב, אַךְ בְּסוֹפוֹ שֶׁל דָּבָר רִחֵם עָלֶיהָ מְעַט וְאָמַר: "אֶתֵּן לָךְ הִזְדַּמְּנוּת אַחַת. אִם תְּנַחֲשִׁי אֶת שְׁמִי הָאֲמִתִּי בְּתוֹךְ שְׁלוֹשָׁה יָמִים, יוּכַל הַתִּינוֹק לְהִשָּׁאֵר אִתָּךְ."

הַמַּלְכָּה שָׁלְחָה מִיָּד שְׁלִיחִים לְכָל פִּנּוֹת הַמַּמְלָכָה, כְּדֵי שֶׁיֶּאֶסְפוּ כָּל שֵׁם מוּזָר וְנָדִיר שֶׁיּוּכְלוּ לִמְצֹא. בַּיּוֹם הָרִאשׁוֹן נִסְּתָה שֵׁמוֹת רְגִילִים כְּמוֹ "כַּסְפָּר" וְ"מֶלְכִּיאוֹר", וּבַיּוֹם הַשֵּׁנִי נִסְּתָה שֵׁמוֹת מוּזָרִים כְּמוֹ "צֶלַע-הַכֶּבֶשׂ" וְ"רַגְלֵי-הָעוֹרֵב" - אַךְ בְּכָל פַּעַם הֵנִיד הָאִישׁ הַקָּטָן בְּרֹאשׁוֹ וְאָמַר: "לֹא, זֶה לֹא שְׁמִי."

בַּיּוֹם הַשְּׁלִישִׁי חָזַר אֶחָד הַשְּׁלִיחִים מִיַּעַר רָחוֹק וְסִפֵּר סִפּוּר מוּזָר. הוּא רָאָה, לְיַד מְדוּרָה קְטַנָּה בֵּין הָעֵצִים, אֶת הָאִישׁ הַקָּטָן שֶׁחִפְּשׂוּ, מְרַקֵּד וְקוֹפֵץ בְּשִׂמְחָה סְבִיב הָאֵשׁ וְשָׁר לְעַצְמוֹ בְּקוֹל צָלוּל:
"הַיּוֹם אֲבַשֵּׁל, מָחָר אֶאֱפֶה,
מָחֳרָתַיִם אָבִיא אֶת יֶלֶד הַמַּלְכָּה הַבַּיְתָה!
כַּמָּה טוֹב שֶׁאַף אֶחָד לֹא יוֹדֵעַ
שֶׁרוּמְפֶּלְשְׁטִילְצְקִין שְׁמִי!"

הַשָּׁלִיחַ מִהֵר לָאַרְמוֹן וְסִפֵּר לַמַּלְכָּה אֶת מָה שֶׁשָּׁמַע. בְּאוֹתוֹ עֶרֶב, כְּשֶׁחָזַר הָאִישׁ הַקָּטָן לְקַבֵּל אֶת הַתְּשׁוּבָה הָאַחֲרוֹנָה, שָׁאַל אוֹתָהּ בְּחִיּוּךְ עַרְמוּמִי: "נוּ, מַלְכָּתִי, מָה שְׁמִי?" הַמַּלְכָּה הֶעֱמִידָה פָּנִים שֶׁהִיא מְהַסֶּסֶת, וְנִסְּתָה שְׁנֵי שֵׁמוֹת שְׁגוּיִים, וְרַק אָז אָמְרָה בְּשֶׁקֶט: "אוּלַי... שִׁמְךָ הוּא רוּמְפֶּלְשְׁטִילְצְקִין?"

הָאִישׁ הַקָּטָן הִזְדַּעֲזֵעַ. פָּנָיו הֶאֱדִימוּ מִכַּעַס, וְהוּא רָקַע בְּרַגְלוֹ הַיְמָנִית בְּכָל כֹּחוֹ - כָּל כָּךְ בְּחָזְקָה, שֶׁרַגְלוֹ שָׁקְעָה עָמֹק בָּרִצְפָּה וְנִתְקְעָה בָּהּ. הוּא צָעַק בְּזַעַם, מָשַׁךְ וּמָשַׁךְ, וְלַבְסוֹף הִצְלִיחַ לְהִשְׁתַּחְרֵר - וּבָרַח מֵהָאַרְמוֹן בְּרִיצָה מְהִירָה, וְאִישׁ לֹא רָאָה אוֹתוֹ שׁוּב אַף פַּעַם.

הַמַּלְכָּה חִבְּקָה אֶת תִּינוֹקָהּ בְּאֹשֶׁר, וְלָמְדָה לֶקַח חָשׁוּב: אָסוּר לְהַבְטִיחַ הַבְטָחוֹת בִּפְזִיזוּת, גַּם כְּשֶׁנִּדְמֶה שֶׁהֶעָתִיד רָחוֹק. וְעוֹד דָּבָר: לִפְעָמִים הַכֹּחַ הַגָּדוֹל בְּיוֹתֵר טָמוּן פָּשׁוּט בִּידִיעַת הָאֱמֶת - כְּמוֹ הַשֵּׁם הָאֲמִתִּי שֶׁל מִישֶׁהוּ.

מְסֻפָּר עַל־פִּי אַגָּדַת הָאַחִים גְּרִים "רוּמְפֶּלְשְׁטִילְצְחֶן" (Rumpelstilzchen), פֻּרְסְמָה לָרִאשׁוֹנָה בְּשָׁנַת 1812, מְעֻבֶּדֶת וּמְסֻפֶּרֶת מֵחָדָשׁ (עִם סוֹף מְרֻכָּךְ).

📚 דִּקְדּוּק — מִשְׁפַּט זִיקָה

מָה זוֹ פְּסוּקִית זִיקָה?

פְּסוּקִית זִיקָה הִיא חֵלֶק שֶׁל מִשְׁפָּט שֶׁמּוֹסִיף מֵידָע עַל שֵׁם עֶצֶם שֶׁכְּבָר הֻזְכַּר, וְתָמִיד מַתְחִילָה בְּמִלַּת הַזִּיקָה "שֶׁ-", הַמְחֻבֶּרֶת לַמִּלָּה הַבָּאָה אַחֲרֶיהָ בְּלִי רֶוַח. בַּסִּפּוּר עַל רוּמְפֶּלְשְׁטִילְצְקִין יֵשׁ הַרְבֵּה דֻּגְמָאוֹת: הַמַּלְכָּה כָּל הַזְּמַן מְחַפֶּשֶׂת אֶת "הָאִישׁ שֶׁ..." וְאֶת "הַשֵּׁם שֶׁ...".

"שֶׁ-" יָכוֹל לְהַחְלִיף נוֹשֵׂא אוֹ מוּשָׂא

בְּשִׁיעוּר 71 לָמַדְנוּ עַל מוּשָׂא יָשִׁיר וְעָקִיף. בִּפְסוּקִית זִיקָה, מִלַּת הַזִּיקָה "שֶׁ-" יְכוֹלָה לְהַחְלִיף אֶת הַנּוֹשֵׂא אוֹ אֶת הַמּוּשָׂא שֶׁל הַפֹּעַל בִּפְסוּקִית.

מִשְׁפָּטמָה "שֶׁ-" מַחֲלִיף
הָאִישׁ שֶׁטָּוָה זָהָב נֶעֱלַם.נוֹשֵׂא (הָאִישׁ הוּא זֶה שֶׁטָּוָה)
הַזָּהָב שֶׁטָּוְתָה הַנַּעֲרָה הִדְהִים אֶת הַמֶּלֶךְ.מוּשָׂא (הַנַּעֲרָה טָוְתָה אֶת הַזָּהָב)
הַשָּׁלִיחַ שֶׁמָּצָא אֶת הָאִישׁ בַּיַּעַר חָזַר.נוֹשֵׂא (הַשָּׁלִיחַ הוּא זֶה שֶׁמָּצָא)

מִיקוּם פְּסוּקִית הַזִּיקָה בַּמִּשְׁפָּט

פְּסוּקִית הַזִּיקָה חַיֶּבֶת לָבוֹא מִיָּד אַחֲרֵי שֵׁם הָעֶצֶם שֶׁהִיא מְתָאֶרֶת, אַחֶרֶת הַמִּשְׁפָּט נִשְׁמָע מְבֻלְבָּל אוֹ שָׁגוּי: "הַמַּלְכָּה פָּגְשָׁה אֶת הָאִישׁ שֶׁטָּוָה זָהָב בַּיַּעַר" - נָכוֹן. "הַמַּלְכָּה פָּגְשָׁה אֶת הָאִישׁ בַּיַּעַר שֶׁטָּוָה זָהָב" - מְבֻלְבָּל, כִּי נִדְמֶה שֶׁ"שֶׁטָּוָה זָהָב" מְתָאֵר אֶת הַיַּעַר.

🔑 כְּלָל זָהָב: פְּסוּקִית זִיקָה תָּמִיד פּוֹתַחַת בְּ"שֶׁ-" וּבָאָה מִיָּד אַחֲרֵי שֵׁם הָעֶצֶם שֶׁהִיא מְתָאֶרֶת. "שֶׁ-" יָכוֹל לַעֲמֹד בִּמְקוֹם הַנּוֹשֵׂא אוֹ בִּמְקוֹם הַמּוּשָׂא שֶׁל הַפֹּעַל בַּפְּסוּקִית.
💡 טִיפ: לֹא כָּל "שֶׁ-" הוּא מִלַּת זִיקָה! בְּמִשְׁפָּט כְּמוֹ "הַמַּלְכָּה יָדְעָה שֶׁהָאִישׁ כּוֹעֵס", "שֶׁ-" רַק מְחַבֵּר אֶת מָה שֶׁנֶּאֱמַר אוֹ נוֹדַע - הוּא לֹא מְתָאֵר שֵׁם עֶצֶם מְסֻיָּם. פְּסוּקִית זִיקָה תָּמִיד בָּאָה מִיָּד אַחֲרֵי שֵׁם עֶצֶם שֶׁהִיא מוֹסִיפָה עָלָיו מֵידָע.
