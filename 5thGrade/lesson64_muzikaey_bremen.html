מוּזִיקָאֵי בְּרֶמֶן

אַרְבַּע חַיּוֹת זְקֵנוֹת שֶׁכֻּלָּם רָצוּ לְהִפָּטֵר מֵהֶן יוֹצְאוֹת יַחַד לְדֶרֶךְ חֲדָשָׁה - וְשִׁיעוּר חֲזָרָה עַל שֵׁם הַפֹּעַל, הַפַּעַם בְּכָל הַבִּנְיָנִים

📖 קְרִיאָה

הָיָה הָיָה חֲמוֹר זָקֵן שֶׁעָבַד שָׁנִים רַבּוֹת בְּמֶשֶׁק חַקְלָאִי, וְנָשָׂא שַׂקֵּי קֶמַח כְּבֵדִים מִן הַבֹּקֶר וְעַד הָעֶרֶב. יוֹם אֶחָד שָׁמַע אֶת בַּעַל הַמֶּשֶׁק אוֹמֵר: "הֶחָמוֹר הַזֶּה כְּבָר זָקֵן מִדַּי לָעֲבוֹדָה - אֵין לִי יוֹתֵר שׁוּם צֹרֶךְ בּוֹ." הֶחָמוֹר נִבְהַל וְהֶחְלִיט לִבְרֹחַ מֵהַמֶּשֶׁק בְּלֵב שָׁקֵט אֲבָל נָחוּשׁ: "אֵלֵךְ לְעִיר בְּרֶמֶן וְאֶהְיֶה שָׁם מוּזִיקַאי - יֵשׁ לִי קוֹל חָזָק, וְזֶה כָּל מַה שֶּׁצָּרִיךְ!"

בַּדֶּרֶךְ אֶל בְּרֶמֶן פָּגַשׁ הֶחָמוֹר כֶּלֶב זָקֵן שֶׁשָּׁכַב עַל הַכְּבִישׁ וְנָשַׁם בִּכְבֵדוּת. "מָה קָרָה לְךָ?" שָׁאַל הֶחָמוֹר. "אֲנִי כְּבָר זָקֵן מִדַּי כְּדֵי לָצוּד," עָנָה הַכֶּלֶב בְּעֶצֶב, "וְהַבְּעָלִים שֶׁלִּי רָצוּ לְהִפָּטֵר מִמֶּנִּי. לָכֵן בָּרַחְתִּי, אֲבָל אֵין לִי לְאָן לָלֶכֶת." "בֹּא אִתִּי אֶל בְּרֶמֶן," הִצִּיעַ הֶחָמוֹר, "שָׁם נוּכַל לְנַגֵּן יַחַד בָּרְחוֹב - אַתָּה תְּנַבַּח בְּקֶצֶב, וַאֲנִי אֶזְעַק בְּקוֹל." הַכֶּלֶב שָׂמַח מְאֹד וְהִצְטָרֵף אֶל הֶחָמוֹר.

הֵם הִמְשִׁיכוּ יַחַד בַּדֶּרֶךְ, וְעַד מְהֵרָה פָּגְשׁוּ חָתוּל זָקֵן שֶׁיָּשַׁב עַל גָּדֵר וּפָנָיו עֲצוּבוֹת. "מָה קָרָה?" שָׁאֲלוּ הֶחָמוֹר וְהַכֶּלֶב. "שִׁנַּי כְּבָר לֹא חַדּוֹת כְּמוֹ פַּעַם," הֵשִׁיב הֶחָתוּל, "וְאֵינֶנִּי מַצְלִיחַ יוֹתֵר לָצוּד עַכְבָּרִים. בְּעָלַי הֶחְלִיטוּ לְהִפָּטֵר מִמֶּנִּי, אָז בָּרַחְתִּי בְּלֵב כָּבֵד." "בֹּא אִתָּנוּ אֶל בְּרֶמֶן," אָמְרוּ שְׁנֵי הַחֲבֵרִים, "גַּם קוֹלְךָ יָפֶה לְשִׁירָה בַּלַּיְלָה!" הֶחָתוּל שָׂמַח וְהִצְטָרֵף מִיָּד.

עוֹד מְעַט הִגִּיעוּ הַשְּׁלֹשָׁה אֶל חָצֵר שֶׁבָּהּ עָמַד תַּרְנְגוֹל וְקָרָא בְּקוֹל גָּדוֹל וְנוֹאָשׁ. "לָמָּה אַתָּה צוֹעֵק כָּל כָּךְ?" שָׁאַל הֶחָמוֹר. "מָחָר יָבוֹאוּ אוֹרְחִים לְבֵית הַחַוָּה," עָנָה הַתַּרְנְגוֹל, "וְשָׁמַעְתִּי שֶׁבַּעֲלַת הַבַּיִת מִתְכַּוֶּנֶת לְבַשֵּׁל אוֹתִי לַאֲרוּחַת הָעֶרֶב שֶׁלָּהֶם!" "אָז בֹּא אִתָּנוּ מַהֵר!" קָרְאוּ הַשְּׁלֹשָׁה בְּיַחַד. "שָׁם, בָּעִיר הַגְּדוֹלָה, תּוּכַל לְקַרְקֵר בַּבֹּקֶר, וְאִישׁ לֹא יְבַשֵּׁל אוֹתְךָ." הַתַּרְנְגוֹל לֹא הִסֵּס אֲפִלּוּ רֶגַע, פָּרַשׂ אֶת כְּנָפָיו וְעָף לְיַד חֲבֵרָיו הַחֲדָשִׁים.

אַרְבַּעְתָּם הִמְשִׁיכוּ בַּדֶּרֶךְ, וְכַאֲשֶׁר יָרַד הַלַּיְלָה מָצְאוּ עַצְמָם בְּתוֹךְ יַעַר חָשׁוּךְ. הֵם הֶחְלִיטוּ לָנוּחַ שָׁם עַד הַבֹּקֶר, אֲבָל אָז הִבְחִין הֶחָמוֹר בְּאוֹר קָטָן מִבַּעַד לָעֵצִים. "בֹּאוּ נִבְדֹּק מָה יֵשׁ שָׁם," הִצִּיעַ, "אוּלַי נִמְצָא מָקוֹם חַם יוֹתֵר לִישֹׁן בּוֹ." הֵם הָלְכוּ בְּשֶׁקֶט לְעֵבֶר הָאוֹר וְגִלּוּ בַּיִת קָטָן בְּתוֹךְ הַיַּעַר.

הַחֲבֵרִים הֵצִיצוּ פְּנִימָה דֶּרֶךְ הַחַלּוֹן וְרָאוּ שֻׁלְחָן עָמוּס בְּמַאֲכָלִים טְעִימִים, וְסָבִיב לוֹ יָשְׁבוּ כַּמָּה שׁוֹדְדִים מְסֻכָּנִים שֶׁאָכְלוּ וְשָׁתוּ בְּשִׂמְחָה. "אֲנִי כָּל כָּךְ רָעֵב," לָחַשׁ הֶחָמוֹר, "אֲבָל אֵיךְ נַצְלִיחַ לְהַפְחִיד שׁוֹדְדִים מְסֻכָּנִים כָּאֵלֶּה?" הֶחָתוּל חִיֵּךְ בְּעָרְמָה וְלָחַשׁ תָּכְנִית: "בֹּאוּ נַעֲמֹד אֶחָד עַל גַּבֵּי הַשֵּׁנִי, לְיַד הַחַלּוֹן, וְנַתְחִיל לְנַגֵּן בְּיַחַד בְּקוֹל רַעַשׁ נוֹרָא!"

הֶחָמוֹר עָמַד רִאשׁוֹן לְיַד הַחַלּוֹן, הַכֶּלֶב עָלָה עַל גַּבּוֹ, הֶחָתוּל טִפֵּס עַל הַכֶּלֶב, וְהַתַּרְנְגוֹל עָף וְנֶחַת עַל רֹאשׁ הֶחָתוּל. בְּאוֹת מוּסְכָּם הֵחֵלּוּ כֻּלָּם לְהַשְׁמִיעַ אֶת קוֹלָם בְּבַת אַחַת: הֶחָמוֹר נָעַר בְּקוֹל עָצוּם, הַכֶּלֶב נָבַח בְּזַעַם, הֶחָתוּל יִלֵּל בְּחֻצְפָּה, וְהַתַּרְנְגוֹל קִרְקֵר בְּכָל כֹּחוֹ. יַחַד נִשְׁמַע הַקּוֹל כְּמוֹ שֶׁל מִפְלֶצֶת אַחַת עֲנָקִית וּמְפַחִידָה.

הַשּׁוֹדְדִים נִבְהֲלוּ עַד מָוֶת. הֵם קָפְצוּ מִן הַשֻּׁלְחָן, שָׁבְרוּ אֶת הַחַלּוֹן בִּבְרִיחָתָם, וְרָצוּ אֶל תּוֹךְ הַיַּעַר הֶחָשׁוּךְ בְּלִי לְהַבִּיט לְאָחוֹר. אַרְבַּעַת הַחֲבֵרִים נִכְנְסוּ בְּשֶׁקֶט אֶל הַבַּיִת הָרֵיק, אָכְלוּ עַד שֶׂבַע, וְלָאַחַר מִכֵּן כִּבּוּ אֶת הָאוֹרוֹת וְנִרְדְּמוּ - כָּל אֶחָד בְּמָקוֹם שֶׁהִתְאִים לוֹ בְּיוֹתֵר.

בְּאֶמְצַע הַלַּיְלָה הֵעֵז אֶחָד הַשּׁוֹדְדִים לַחֲזֹר אֶל הַבַּיִת כְּדֵי לִבְדֹּק אִם הַסַּכָּנָה חָלְפָה. בַּחֲשֵׁכָה נִתְקַל בְּעֵינֵי הֶחָתוּל הַנּוֹצְצוֹת וְחָשַׁב שֶׁהֵן גֶּחָלִים בּוֹעֲרוֹת - הֶחָתוּל שָׂרַט אֶת פָּנָיו. הַכֶּלֶב נָשַׁךְ אוֹתוֹ בְּרַגְלוֹ, הֶחָמוֹר בָּעַט בּוֹ בְּעֹז, וְהַתַּרְנְגוֹל קָרָא מֵעַל הַגַּג: "קוּ-קוּ-רִי-קוּ!" הַשּׁוֹדֵד הַנִּבְהָל בָּרַח בְּצַעֲקוֹת אֶל חֲבֵרָיו וְסִפֵּר לָהֶם: "בַּבַּיִת יוֹשֶׁבֶת מְכַשֵּׁפָה נוֹרָאָה שֶׁשָּׂרְטָה אוֹתִי בְּצִפָּרְנֶיהָ, וְעוֹד יְצוּר עִם שִׁנַּיִם חַדּוֹת נָשַׁךְ אֶת רַגְלִי!"

מֵאוֹתוֹ לַיְלָה, הַשּׁוֹדְדִים לֹא הֵעֵזּוּ לַחֲזֹר לְעוֹלָם אֶל הַבַּיִת בַּיַּעַר. אַרְבַּעַת הַחֲבֵרִים - הֶחָמוֹר, הַכֶּלֶב, הֶחָתוּל וְהַתַּרְנְגוֹל - מָצְאוּ אֶת עַצְמָם כָּל כָּךְ מְרֻצִּים מֵהַבַּיִת הֶחָדָשׁ, שֶׁהֵם הֶחְלִיטוּ לְהִשָּׁאֵר בּוֹ. הֵם מֵעוֹלָם לֹא הִגִּיעוּ בְּפֹעַל אֶל בְּרֶמֶן, אֲבָל מָצְאוּ מַשֶּׁהוּ טוֹב בְּהַרְבֵּה - בַּיִת חַם וַחֲבֵרוּת אֲמִתִּית שֶׁתִּמְשֹׁךְ לְעוֹלָמֵי עַד.

מְסֻפָּר עַל־פִּי "מוּזִיקָאֵי בְּרֶמֶן" (Die Bremer Stadtmusikanten) מֵאֵת הָאַחִים גְרִים (גֶּרְמַנְיָה, פֻּרְסַם לָרִאשׁוֹנָה בִּשְׁנַת 1819) - מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ.

📚 דִּקְדּוּק — שֵׁם הַפֹּעַל בְּכָל הַבִּנְיָנִים

חֲזָרָה קְצָרָה: מָה זֶה שֵׁם הַפֹּעַל?

בְּשִׁיעוּר 50 לָמַדְנוּ שֶׁשֵּׁם הַפֹּעַל הוּא הַצּוּרָה הַבְּסִיסִית שֶׁל הַפֹּעַל - זוֹ שֶׁמּוֹפִיעָה בַּמִּלּוֹן - וְשֶׁהִיא כִּמְעַט תָּמִיד מַתְחִילָה בְּ-ל: לָלֶכֶת, לִכְתֹּב, לֶאֱכֹל. הַיּוֹם נַרְחִיב וְנִבְדֹּק אֵיךְ נִרְאֶה שֵׁם הַפֹּעַל בְּכָל אֶחָד מִשִּׁבְעַת הַבִּנְיָנִים: בִּנְיַן קַל, שֶׁמְּלַוֶּה אוֹתָנוּ כְּבָר מֵהַתְחָלָה, וְעוֹד שִׁשָּׁה בִּנְיָנִים שֶׁלָּמַדְנוּ בְּשִׁיעוּרִים 10-18.

שֵׁם הַפֹּעַל אַחֲרֵי פְּעָלִים מְסֻיָּמִים

שֵׁם הַפֹּעַל מוֹפִיעַ הַרְבֵּה אַחֲרֵי פְּעָלִים כְּמוֹ רוֹצֶה, יָכוֹל, צָרִיךְ, מַתְחִיל, מַפְסִיק וּמֵעֵז - כִּי הוּא מַשְׁלִים אֶת הַפְּעֻלָּה שֶׁהֵם מְתָאֲרִים.

🔑 כְּלָל זָהָב (מִשִּׁיעוּר 50): "הֶחָמוֹר הֵעֵז לִבְרֹחַ", "הַשּׁוֹדְדִים לֹא הֵעֵזּוּ לַחֲזֹר", "הֶחָתוּל הִתְחִיל לְנַגֵּן". שִׂימוּ לֵב: אַחֲרֵי הַפֹּעַל הָרִאשׁוֹן (רוֹצֶה/יָכוֹל/צָרִיךְ/מֵעֵז...) בָּא תָּמִיד שֵׁם הַפֹּעַל, וְלֹא צוּרַת פֹּעַל רְגִילָה - אָסוּר לוֹמַר "הֵעֵז בָּרַח" אוֹ "הֵעֵז בּוֹרֵחַ".
