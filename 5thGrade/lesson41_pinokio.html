📖 קְרִיאָה

בְּעִיר קְטַנָּה בְּאִיטַלְיָה, לִפְנֵי שָׁנִים רַבּוֹת, חַי נַגָּר זָקֵן וְטוֹב לֵב בְּשֵׁם גֶ'פֶּטוֹ. הוּא גָּר לְבַדּוֹ בְּסַדְנָה קְטַנָּה וּמְלֵאָה בִּנְסֹרֶת עֵץ, וְחָלַם שֶׁיִּהְיֶה לוֹ יוֹם אֶחָד בֵּן מִשֶּׁלּוֹ.

יוֹם אֶחָד מָצָא גֶ'פֶּטוֹ גֶּזַע עֵץ מְיֻחָד, שֶׁצָּחַק וּבָכָה כִּמְעַט כְּמוֹ יֶלֶד קָטָן. בְּשִׂמְחָה רַבָּה חָרַט מִמֶּנּוּ גֶ'פֶּטוֹ בֻּבַּת עֵץ, וְקָרָא לָהּ פִּינוֹקְיוֹ.

לְהַפְתָּעַת גֶ'פֶּטוֹ, בְּרֶגַע שֶׁסִּיֵּם לַחְרֹט אֶת הָרַגְלַיִם, פָּתְחָה הַבֻּבָּה אֶת עֵינֶיהָ וְהֵחֵלָּה לְדַבֵּר וְלָרוּץ בַּסַּדְנָה. גֶ'פֶּטוֹ שָׂמַח מְאֹד, אַךְ פִּינוֹקְיוֹ הַקָּטָן הָיָה שׁוֹבָב, וְלֹא תָּמִיד צִיֵּת.

כְּדֵי שֶׁפִּינוֹקְיוֹ יִלְמַד לִקְרֹא וְלִכְתֹּב וְיִגְדַּל לְיֶלֶד טוֹב, מָכַר גֶ'פֶּטוֹ אֶת מְעִילוֹ הַיָּחִיד וְקָנָה בַּכֶּסֶף סֵפֶר לִמּוּד. "לֵךְ לְבֵית הַסֵּפֶר," אָמַר לוֹ בְּחִבָּה, "וּלְמַד הֵיטֵב."

אֲבָל בַּדֶּרֶךְ לְבֵית הַסֵּפֶר פָּגַשׁ פִּינוֹקְיוֹ שׁוּעָל עַרְמוּמִי וְחָתוּל עִוֵּר, שֶׁשִּׁכְנְעוּ אוֹתוֹ לִמְכֹּר אֶת סֵפֶר הַלִּמּוּד שֶׁלּוֹ וְלָלֶכֶת אִתָּם לְתֵאַטְרוֹן הַבֻּבּוֹת, בִּמְקוֹם לְבֵית הַסֵּפֶר.

בְּדַרְכּוֹ חֲזָרָה הוֹפִיעָה לְפָנָיו פֵּיָה כְּחֻלָּה וּמְלֵאַת קֶסֶם. הִיא שָׁאֲלָה אֶת פִּינוֹקְיוֹ אֵיפֹה סֵפֶר הַלִּמּוּד וְהַכֶּסֶף, וְהוּא, מִפַּחַד, סִיפֵּר לָהּ שֶׁקֶר.

בְּאוֹתוֹ רֶגַע הֵחֵל אַפּוֹ שֶׁל פִּינוֹקְיוֹ לִגְדֹּל וְלִגְדֹּל, אָרֹךְ וְאָרֹךְ, עַד שֶׁלֹּא יָכוֹל הָיָה לָצֵאת דֶּרֶךְ דֶּלֶת הַחֶדֶר. "כָּל פַּעַם שֶׁתֹּאמַר שֶׁקֶר," אָמְרָה הַפֵּיָה בְּרֹךְ, "יִגְדַּל אַפְּךָ עוֹד קְצָת. רַק הָאֱמֶת תַּחְזִיר אוֹתוֹ לְקָדְמוּתוֹ."

פִּינוֹקְיוֹ נִבְהַל מְאֹד וְסִפֵּר לַפֵּיָה אֶת הָאֱמֶת כֻּלָּהּ, וְאַפּוֹ חָזַר לְגָדְלוֹ הָרָגִיל. מֵאוֹתוֹ יוֹם הֵחֵל פִּינוֹקְיוֹ לִלְמֹד וּלְהִשְׁתַּדֵּל לִהְיוֹת יֶלֶד יָשָׁר, שֶׁשּׁוֹמֵר עַל הַבְטָחוֹתָיו.

יוֹם אֶחָד שָׁמַע פִּינוֹקְיוֹ שֶׁגֶ'פֶּטוֹ יָצָא לְחַפֵּשׂ אוֹתוֹ בַּיָּם וְנִקְלַע לְסַכָּנָה גְּדוֹלָה. בְּלִי לְחַשֵּׁב פַּעֲמַיִם, שָׂחָה פִּינוֹקְיוֹ בְּאֹמֶץ בַּמַּיִם הַסּוֹעֲרִים וְהִצִּיל אֶת גֶ'פֶּטוֹ הַזָּקֵן מִן הַגַּלִּים.

הַפֵּיָה הַכְּחֻלָּה, שֶׁרָאֲתָה כַּמָּה אַמִּיץ וְיָשָׁר נַעֲשָׂה פִּינוֹקְיוֹ, הֶחְלִיטָה לְגַמֵּל אוֹתוֹ עַל כָּךְ. בְּנֶגִיעַת שַׁרְבִיטָהּ הַקְּסוּמָה הִיא הָפְכָה אוֹתוֹ לְיֶלֶד אֲמִתִּי, שֶׁל בָּשָׂר וָדָם.

גֶ'פֶּטוֹ וּפִינוֹקְיוֹ חִבְּקוּ זֶה אֶת זֶה בְּשִׂמְחָה גְּדוֹלָה, וְחָיוּ מֵאָז בְּאֹשֶׁר בַּסַּדְנָה הַקְּטַנָּה, כְּאָב וָבֵן אֲמִתִּיִּים.

מַהוּ שֹׁרֶשׁ?

כָּל מִלָּה בְּעִבְרִית בְּנוּיָה בְּדֶרֶךְ כְּלָל סְבִיב שֹׁרֶשׁ - בְּדֶרֶךְ כְּלָל שָׁלוֹשׁ אוֹתִיּוֹת שֶׁנּוֹשְׂאוֹת אֶת הַמַּשְׁמָעוּת הָעִקָּרִית שֶׁל הַמִּלָּה. מְעַנְיֵן לָדַעַת: הַמִּלָּה "שֹׁרֶשׁ" מְתָאֶרֶת גַּם אֶת הַשֹּׁרֶשׁ שֶׁל עֵץ! וּבְדִיּוּק כְּמוֹ שֶׁמִּשֹּׁרֶשׁ אֶחָד שֶׁל עֵץ צוֹמְחִים גֶּזַע וְעָלִים רַבִּים, כָּךְ מִשֹּׁרֶשׁ אֶחָד שֶׁל מִלָּה "צוֹמְחוֹת" מִלִּים רַבּוֹת - פְּעָלִים, שְׁמוֹת עֶצֶם וְתוֹאֲרִים - שֶׁכֻּלָּן קְשׁוּרוֹת בְּמַשְׁמָעוּתָן.

מִשְׁפַּחַת מִלִּים

קְבוּצַת מִלִּים שֶׁחוֹלְקוֹת אוֹתוֹ שֹׁרֶשׁ נִקְרֵאת מִשְׁפַּחַת מִלִּים. בַּסִּיפּוּר עַל פִּינוֹקְיוֹ אֶפְשָׁר לִמְצֹא כַּמָּה מִשְׁפָּחוֹת כָּאֵלֶּה. לְמָשָׁל, הַשֹּׁרֶשׁ ל-מ-ד: פִּינוֹקְיוֹ אָמוּר "לִלְמֹד" בְּבֵית הַסֵּפֶר, סֵפֶר "הַלִּמּוּד" שֶׁלּוֹ, וְהוּא עַצְמוֹ הָיָה אָמוּר לִהְיוֹת "תַּלְמִיד" טוֹב - שָׁלוֹשׁ מִלִּים שׁוֹנוֹת לְגַמְרֵי בְּתַפְקִידָן, אֲבָל כֻּלָּן מֵאוֹתוֹ שֹׁרֶשׁ.

בַּחֲרוּ אֶת אַרְבַּעַת הַשָּׁרָשִׁים שֶׁלָּמַדְנוּ הַיּוֹם (ל-מ-ד, ש-מ-ר, כ-ת-ב, ג-ד-ל), וְכִתְבוּ מִלָּה חֲדָשָׁה אַחַת מִכָּל מִשְׁפָּחָה (לֹא מִלָּה שֶׁכְּבָר מוֹפִיעָה בַּטַּבְלָה לְמַעְלָה) - וּבַסּוֹף מִשְׁפָּט שָׁלֵם עִם אַחַת מֵהַמִּלִּים שֶׁכְּתַבְתֶּם. לְדֻגְמָה: מִשֹּׁרֶשׁ ש-מ-ר אֶפְשָׁר לִכְתֹּב אֶת הַמִּלָּה "שְׁמִירָה".
