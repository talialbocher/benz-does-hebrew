📖 קְרִיאָה

הָיוּ פַּעַם שְׁנֵי אַחִים. הָאָח הַגָּדוֹל הָיָה עָשִׁיר מְאֹד, וְהָאָח הַקָּטָן הָיָה עָנִי כָּל כָּךְ, שֶׁלֹּא הָיָה לוֹ אֲפִלּוּ פֵּרוּר שֶׁל לֶחֶם.

בְּעֶרֶב חַג אֶחָד הָלַךְ הָאָח הֶעָנִי אֶל בֵּיתוֹ שֶׁל אָחִיו וּבִקֵּשׁ מִמֶּנּוּ מְעַט אֹכֶל. הֶעָשִׁיר הִתְרַגֵּז, זָרַק לוֹ חֲתִיכַת בָּשָׂר וְאָמַר: "קַח וְלֵךְ מִכָּאן, וְאַל תַּטְרִיד אוֹתִי יוֹתֵר!"

הָאָח הֶעָנִי לֹא נֶעֱלַב. הוּא הוֹדָה לְאָחִיו וְיָצָא לַדֶּרֶךְ עִם הַבָּשָׂר. בְּאֶמְצַע הַיַּעַר פָּגַשׁ זָקֵן עִם זָקָן לָבָן וְאָרֹךְ. אָמַר לוֹ הַזָּקֵן: "בְּתוֹךְ הָהָר גָּרִים גַּמָּדִים, וְהֵם אוֹהֲבִים בָּשָׂר יוֹתֵר מִכֹּל. אַל תְּבַקֵּשׁ מֵהֶם כֶּסֶף. בַּקֵּשׁ אֶת הַטַּחֲנָה הַקְּטַנָּה שֶׁעוֹמֶדֶת מֵאֲחוֹרֵי הַדֶּלֶת."

הָאָח הֶעָנִי נִכְנַס לַמְּעָרָה, וְהַגַּמָּדִים שָׂמְחוּ מְאֹד עַל הַבָּשָׂר. הֵם נָתְנוּ לוֹ אֶת הַטַּחֲנָה וְהוֹסִיפוּ: "זוֹ טַחֲנָה קְסוּמָה. הִיא טוֹחֶנֶת כֹּל מָה שֶׁתְּבַקֵּשׁ מִמֶּנָּה, וּכְדֵי לַעֲצֹר אוֹתָהּ צָרִיךְ לוֹמַר: 'דַּי, טַחֲנָה, דַּי!'"

הָאָח הֶעָנִי חָזַר הַבַּיְתָה וְאָמַר: "טַחֲנָה, אֲנִי רוֹצֶה לֶחֶם, גְּבִינָה וְדָגִים!" מִיָּד הִתְחִילָה הַטַּחֲנָה לִטְחֹן, וְעַל הַשֻּׁלְחָן הִצְטַבְּרוּ כָּל הַמַּאֲכָלִים הַטּוֹבִים. הוּא אָכַל כַּמָּה שֶׁרָצָה, וְאַחַר כָּךְ בִּקֵּשׁ מֵהַטַּחֲנָה בְּגָדִים חַמִּים, וַאֲפִלּוּ בַּיִת חָדָשׁ עִם גַּג זָהָב.

אָחִיו הֶעָשִׁיר בָּא לְבַקֵּר, רָאָה אֶת הַבַּיִת הַחָדָשׁ וְאֶת הַטַּחֲנָה וְאָמַר בְּבוּשָׁה: "סְלַח לִי, אָחִי." הָאָח הֶעָנִי סָלַח לוֹ בְּלֵב שָׁלֵם.

יוֹם אֶחָד הִגִּיעַ לַנָּמָל קַפִּיטָן שֶׁל סְפִינָה גְּדוֹלָה. כְּשֶׁשָּׁמַע עַל הַטַּחֲנָה, רָצָה אוֹתָהּ לְעַצְמוֹ. הוּא נָתַן לָאָח שַׂק מָלֵא זָהָב, וּמֵרֹב הִתְלַהֲבוּת לֹא הִקְשִׁיב כְּשֶׁהָאָח הֶעָנִי הִתְחִיל לְהַסְבִּיר לוֹ אֵיךְ עוֹצְרִים אוֹתָהּ.

בַּלַּיְלָה, בְּאֶמְצַע הַיָּם, חָשַׁב הַקַּפִּיטָן: "אֶמְכֹּר מֶלַח וְאֶהְיֶה עָשִׁיר מְאֹד!" הוּא עָמַד עַל הַסִּפּוּן לְיַד הַטַּחֲנָה וְצָעַק: "אֲנִי רוֹצֶה מֶלַח!" הַטַּחֲנָה טָחֲנָה וְטָחֲנָה, וְהַמֶּלַח נָפַל כְּמוֹ שֶׁלֶג לָבָן.

"עֲצֹר!" צָעַק הַקַּפִּיטָן. "דַּי! מַסְפִּיק!" אֲבָל הוּא לֹא יָדַע אֶת הַמִּלִּים הַנְּכוֹנוֹת, וְהַטַּחֲנָה לֹא הִפְסִיקָה. הַמֶּלַח הִצְטַבֵּר וְהִצְטַבֵּר, עַד שֶׁהַסְּפִינָה נִהְיְתָה כְּבֵדָה מִדַּי וְשָׁקְעָה בַּיָּם.

אוֹמְרִים שֶׁהַטַּחֲנָה נִמְצֵאת עַל קַרְקַע הַיָּם וְטוֹחֶנֶת שָׁם מֶלַח עַד הַיּוֹם. לָכֵן מֵי הַיָּם מְלוּחִים.

הַסִּפּוּר מְבֻסָּס עַל סִפּוּר עַם נוֹרְבֶגִי מְפֻרְסָם, "לָמָּה מֵי הַיָּם מְלוּחִים", מֵאוֹסַף אַסְבְּיוֹרְנְסֶן וּמוֹ (נוֹרְבֶגְיָה, הַמֵּאָה ה־19), מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ.

📚 דִּקְדּוּק — שִׁמּוּשׁ בַּמִּלּוֹן

אֵיךְ מְחַפְּשִׂים מִלָּה בַּמִּלּוֹן?

מִלּוֹן הוּא סֵפֶר שֶׁמְּסַדֵּר מִלִּים לְפִי הָאָלֶפְבֵּית וּמַסְבִּיר מָה פֵּרוּשָׁן. כְּדֵי לִמְצֹא מִלָּה מַהֵר, יֵשׁ לָלֶכֶת בְּכַמָּה צְעָדִים.

צַעַד 1 — הָאוֹת הָרִאשׁוֹנָה. מְחַפְּשִׂים אֶת הַחֵלֶק שֶׁל הָאוֹת הָרִאשׁוֹנָה בַּמִּלָּה. אִם שְׁתֵּי מִלִּים מַתְחִילוֹת בְּאוֹתָהּ אוֹת, מַשְׁוִים אֶת הָאוֹת הַשְּׁנִיָּה, אַחַר כָּךְ אֶת הַשְּׁלִישִׁית, וְכֵן הָלְאָה.

צַעַד 2 — מָה לֹא מַשְׁפִּיעַ עַל הַסֵּדֶר. הַנִּקּוּד לֹא קוֹבֵעַ אֶת הַסֵּדֶר. גַּם הָאוֹתִיּוֹת הַסּוֹפִיּוֹת (ךְ, ם, ן, ף, ץ) נִסְפָּרוֹת כְּמוֹ אוֹתִיּוֹתֵיהֶן הָרְגִילוֹת: הַמִּלָּה יָם נִמְצֵאת לִפְנֵי יַעַר, כִּי מ קוֹדֶמֶת לְע.

צַעַד 3 — הַצּוּרָה שֶׁל הַמִּלָּה. בַּמִּלּוֹן מְחַפְּשִׂים אֶת הַצּוּרָה הַבָּסִיסִית, לֹא אֶת הַצּוּרָה שֶׁבַּמִּשְׁפָּט.

פְּעָלִים. בְּמִלּוֹנִים רַבִּים הַפֹּעַל כָּתוּב בְּצוּרַת עָבָר, גּוּף שְׁלִישִׁי, זָכָר, יָחִיד (לְמָשָׁל: הוּא כָּתַב), וּבְמִלּוֹנִים אֲחֵרִים הוּא כָּתוּב בְּשֵׁם הַפֹּעַל (לִכְתֹּב). הַהַקְדָּמָה שֶׁל הַמִּלּוֹן מַסְבִּירָה אֵיזוֹ צוּרָה בָּחֲרוּ.

מִלּוֹת רֹאשׁ. בְּרֹאשׁ כָּל דַּף כְּתוּבוֹת שְׁתֵּי מִלִּים: הַמִּלָּה הָרִאשׁוֹנָה בַּדַּף וְהָאַחֲרוֹנָה בַּדַּף. אִם הַמִּלָּה שֶׁלָּנוּ בֵּין שְׁתֵּיהֶן — הִגַּעְנוּ לַדַּף הַנָּכוֹן!

הַרְבֵּה מַשְׁמָעֻיּוֹת. לְמִלָּה אַחַת יֵשׁ לִפְעָמִים כַּמָּה פֵּרוּשִׁים, וְהֵם מְמֻסְפָּרִים: עַיִן — 1. אֵיבָר הָרְאִיָּה 2. מַעְיָן. בּוֹחֲרִים אֶת הַפֵּרוּשׁ שֶׁמַּתְאִים לַמִּשְׁפָּט.
