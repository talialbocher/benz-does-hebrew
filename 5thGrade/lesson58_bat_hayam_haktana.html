בַּת הַיָּם הַקְּטַנָּה

אַהֲבָה אֲמִתִּית שָׁוָה יוֹתֵר מִכָּל אוֹצָר בְּמַעֲמַקֵּי הַיָּם

📖 קְרִיאָה

פַּעַם, עָמֹק בְּלֵב הַיָּם הַכָּחֹל, עָמַד אַרְמוֹן מֻפְלָא שֶׁל מֶלֶךְ הַיָּם. לַמֶּלֶךְ הָיוּ שֵׁשׁ בָּנוֹת יָפוֹת, וְהַצְּעִירָה שֶׁבֵּינֵיהֶן נִקְרְאָה "בַּת הַיָּם הַקְּטַנָּה". לְכָל אַחַת מֵהֶן הָיָה זָנָב כָּסוּף נוֹצֵץ בִּמְקוֹם רַגְלַיִם, אֲבָל קוֹלָהּ שֶׁל בַּת הַיָּם הַקְּטַנָּה הָיָה הַיָּפֶה מִכֻּלָּם - מִי שֶׁשָּׁמַע אוֹתָהּ שָׁרָה לֹא שָׁכַח אֶת הַקּוֹל לְעוֹלָם.

הַנְּסִיכָה הַקְּטַנָּה אָהֲבָה לָשֶׁבֶת בְּגִנַּת הָאַלְמֻגִּים שֶׁל אָבִיהָ וּלְהַקְשִׁיב לְסִפּוּרִים עַל הָעוֹלָם שֶׁמֵּעַל לַמַּיִם. מַלָּח זָקֵן שֶׁטָּבַע לִפְנֵי שָׁנִים רַבּוֹת סִפֵּר לָהּ עַל עָרִים גְּדוֹלוֹת, עַל אֳנִיּוֹת מִפְרָשׂ וְעַל שָׁמַיִם כְּחֻלִּים בְּלִי סוֹף. אָבִיהָ הִבְטִיחַ: "כְּשֶׁתִּמְלְאִי חֲמֵשׁ עֶשְׂרֵה, מֻתָּר יִהְיֶה לָךְ לַעֲלוֹת אֶל פְּנֵי הַיָּם וְלִרְאוֹת אֶת הָעוֹלָם בְּעֵינַיִךְ."

הַשָּׁנִים חָלְפוּ, וְיוֹם אֶחָד הִגִּיעַ יוֹם הֻלֶּדֶת הַחֲמִשָּׁה עָשָׂר שֶׁל בַּת הַיָּם הַקְּטַנָּה. הִיא עָלְתָה אֶל פְּנֵי הַמַּיִם בְּפַעַם הָרִאשׁוֹנָה, וְרָאֲתָה סְפִינָה גְּדוֹלָה מְקֻשֶּׁטֶת בְּפַנָּסִים - חֲגִיגַת יוֹם הֻלֶּדֶת שֶׁל נָסִיךְ צָעִיר וְנָאֶה. לְפֶתַע פָּרְצָה סְעָרָה עַזָּה, וְהַסְּפִינָה טָבְעָה בַּגַּלִּים הַגְּבוֹהִים. בַּת הַיָּם הַקְּטַנָּה, בְּלִי לְהַרְהֵר, שָׂחֲתָה בְּכָל כֹּחָהּ אֶל הַנָּסִיךְ הַטּוֹבֵעַ וְהֶחֱזִיקָה אֶת רֹאשׁוֹ מֵעַל הַמַּיִם עַד שֶׁהִגִּיעָה עִמּוֹ אֶל חוֹף שָׁקֵט.

הִיא הִנִּיחָה אוֹתוֹ עַל הַחוֹל הָרַךְ, שָׁרָה לוֹ שִׁיר עָדִין, וְנֶעֶלְמָה מַהֵר בֵּין הַגַּלִּים כְּשֶׁרָאֲתָה אֲנָשִׁים מִתְקָרְבִים. הַנָּסִיךְ הֵקִיץ וְרָאָה רַק צֵל נֶעֱלָם, אֲבָל קוֹלָהּ נִשְׁאַר חָרוּת בְּלִבּוֹ. בַּת הַיָּם הַקְּטַנָּה חָזְרָה לְאַרְמוֹן אָבִיהָ, אֲבָל מֵאוֹתוֹ יוֹם לֹא הִפְסִיקָה לַחְשֹׁב עַל הַנָּסִיךְ וְעַל הָעוֹלָם שֶׁמֵּעַל לַמַּיִם.

חָמֵשׁ אֲחוֹתֶיהָ שָׂמוּ לֵב שֶׁמַּשֶּׁהוּ הִשְׁתַּנָּה בָּהּ, וְנִסּוּ לְשַׂמֵּחַ אוֹתָהּ בְּכָל דֶּרֶךְ - שָׁרוֹת אִתָּהּ, מְשַׂחֲקוֹת בְּגִנַּת הָאַלְמֻגִּים, מְסַפְּרוֹת לָהּ בְּדִיחוֹת. אֲבָל דָּבָר לֹא עָזַר. לְבַסּוֹף פָּנְתָה בַּת הַיָּם הַקְּטַנָּה אֶל מְכַשֵּׁפַת הַיָּם, שֶׁגָּרָה בִּמְעָרָה אֲפֵלָה מֵעֵבֶר לָאַלְמֻגִּים, וּבִקְּשָׁה מִמֶּנָּה זוּג רַגְלַיִם אֱנוֹשִׁיּוֹת.

"אֶתֵּן לָךְ רַגְלַיִם," אָמְרָה הַמְּכַשֵּׁפָה, "אֲבָל הַמְּחִיר יָקָר: אֶקַּח מִמֵּךְ אֶת קוֹלֵךְ הַנִּפְלָא, וְכָל צַעַד עַל הַיַּבָּשָׁה יַרְגִּישׁ קָשֶׁה כִּמְעַט כְּמוֹ הֲלִיכָה עַל חוֹל לוֹהֵט. וְאִם לֹא תִּזְכִּי בְּאַהֲבַת הַנָּסִיךְ עַד שֶׁתַּעֲלֶה הַשֶּׁמֶשׁ בַּיּוֹם הַשְּׁלִישִׁי, תָּשׁוּבִי לִהְיוֹת בַּת יָם, וְתַחְזְרִי אֶל אָבִיךְ וְאֶל אֲחוֹתַיִךְ בַּיָּם." בַּת הַיָּם הַקְּטַנָּה הִסְכִּימָה בְּלִי הִסּוּס.

בַּת הַיָּם הַקְּטַנָּה שָׁתְתָה אֶת שִׁקּוּי הַקֶּסֶם, וּכְשֶׁהִתְעוֹרְרָה, בִּמְקוֹם זְנָבָהּ הַכָּסוּף הָיוּ לָהּ שְׁתֵּי רַגְלַיִם דַּקּוֹת. הִיא עָלְתָה לְגַן אַרְמוֹן הַנָּסִיךְ, וְכַאֲשֶׁר רָאָה אוֹתָהּ הַנָּסִיךְ, שָׂמַח מְאֹד וְלָקַח אוֹתָהּ אֶל תּוֹךְ הָאַרְמוֹן. הִיא לֹא יָכְלָה לְדַבֵּר, אֲבָל רָקְדָה בִּפְנֵי הֶחָצֵר בְּחֵן כָּזֶה עַד שֶׁכֻּלָּם הִתְפַּעֲלוּ - אַף עַל פִּי שֶׁכָּל צַעַד דָּקַר כְּמוֹ סַכִּינִים קְטַנִּים.

יוֹם אַחַר יוֹם בִּלּוּ הַנָּסִיךְ וּבַת הַיָּם הַקְּטַנָּה יַחַד, וְהוּא הֵחֵל לֶאֱהֹב אוֹתָהּ מְאֹד - אַךְ חָשַׁב שֶׁהִיא רַק יַלְדָּה יְתוֹמָה שֶׁמָּצָא בַּגַּן. הוּא סִפֵּר לָהּ עַל נַעֲרָה מִסְתּוֹרִית שֶׁהִצִּילָה אֶת חַיָּיו בַּיָּם, וְעַל כַּוָּנָתוֹ לִמְצֹא אוֹתָהּ יוֹם אֶחָד וּלְהוֹדוֹת לָהּ. בַּת הַיָּם הַקְּטַנָּה חִיְּכָה בְּעֶצֶב, בְּיוֹדְעָהּ שֶׁהִיא בְּעַצְמָהּ אוֹתָהּ נַעֲרָה - אֲבָל בְּלִי קוֹל, לֹא יָכְלָה לְסַפֵּר לוֹ.

הַיּוֹם הַשְּׁלִישִׁי הִתְקָרֵב, וְלֵב בַּת הַיָּם הַקְּטַנָּה הָלַךְ וְהִתְמַלֵּא דְּאָגָה. אַךְ אָז קָרְתָה תַּפְנִית: מְשָׁרֶתֶת בָּאַרְמוֹן שֶׁרָאֲתָה אוֹתָהּ שָׁרָה לְעַצְמָהּ בְּלִי קוֹל, רַק בִּתְנוּעוֹת שְׂפָתַיִם, הֵבִינָה שֶׁמַּשֶּׁהוּ מוּזָר, וְסִפְּרָה לַנָּסִיךְ עַל נַעֲרָה שֶׁרָאֲתָה יוֹצֵאת מִן הַיָּם בַּלַּיְלָה הַהוּא. הַנָּסִיךְ מִהֵר אֶל הַחוֹף, וְשָׁם, לְאוֹר הַיָּרֵחַ, רָאָה אֶת אֲחוֹתֶיהָ שֶׁל בַּת הַיָּם הַקְּטַנָּה מְחַכּוֹת בַּמַּיִם, קוֹרְאוֹת אֵלֶיהָ בִּדְאָגָה.

הַנָּסִיךְ הֵבִין הַכֹּל בְּבַת אַחַת: הַנַּעֲרָה שֶׁגָּרָה אִתּוֹ בָּאַרְמוֹן וְהַנַּעֲרָה שֶׁהִצִּילָה אֶת חַיָּיו הֵן אוֹתָהּ נַעֲרָה מַמָּשׁ. הוּא רָץ בַּחֲזָרָה, אָחַז בְּיָדָהּ שֶׁל בַּת הַיָּם הַקְּטַנָּה וְקָרָא: "אַתְּ הִיא זוֹ שֶׁהִצִּילָה אוֹתִי! וְאַתְּ זוֹ שֶׁאֲנִי אוֹהֵב!" בְּרֶגַע שֶׁל אַהֲבָה אֲמִתִּית, קָרַס כִּשְׁפָהּ שֶׁל הַמְּכַשֵּׁפָה מֵאֵלָיו: קוֹלָהּ שֶׁל בַּת הַיָּם הַקְּטַנָּה שָׁב אֵלֶיהָ בְּבַת אַחַת, צָלוּל וְיָפֶה כְּשֶׁהָיָה.

הִיא סִפְּרָה לַנָּסִיךְ אֶת כָּל הַסִּפּוּר, וְהוּא הִבְטִיחַ לָשֵׂאת אוֹתָהּ לְאִשָּׁה. הַחֲתֻנָּה נֶעֶרְכָה עַל שְׂפַת הַיָּם: מֶלֶךְ הַיָּם הוֹפִיעַ מִן הַמַּיִם בְּעִטּוּר כִּתְרוֹ הַמֻּזְהָב וּבֵרֵךְ אֶת בִּתּוֹ, וְכָל אֲחוֹתֶיהָ שָׁרוּ שִׁירֵי שִׂמְחָה. מֵאָז, בְּכָל עֶרֶב שֶׁל קַיִץ, יוֹשֶׁבֶת בַּת הַיָּם הַקְּטַנָּה עִם בַּעֲלָהּ עַל הַחוֹף, וְשָׁרָה עִם אֲחוֹתֶיהָ שֶׁמַּגִּיעוֹת מִן הַיָּם לְבַקֵּר - קוֹלָהּ שׁוֹמֵר עַל אוֹתוֹ יֹפִי מֻפְלָא כְּמוֹ בַּיּוֹם שֶׁבּוֹ נוֹלְדָה.

מְסֻפָּר עַל־פִּי "בַּת הַיָּם הַקְּטַנָּה" (The Little Mermaid) מֵאֵת הַנְס כְּרִיסְטְיָאן אַנְדֶּרְסֶן (דֶּנְמַרְק, 1837), מְעֻבָּד וּמְרֻכָּךְ לְסִיּוּם חַם.

תַּזְכֹּרֶת: כִּנּוּיֵי קִנְיָן עַל שֵׁם עֶצֶם יָחִיד (שִׁיעוּר 44)

בְּשִׁיעוּר 44 לָמַדְנוּ שֶׁכִּנּוּי קִנְיָן הוּא סִיּוֹמֶת שֶׁמִּצְטָרֶפֶת יָשָׁר לְסוֹף שֵׁם עֶצֶם, בְּלִי הַמִּלָּה "שֶׁל" - כְּמוֹ בְּ"בֵּיתִי" (=הַבַּיִת שֶׁלִּי). בַּסִּפּוּר שֶׁלָּנוּ מוֹפִיעוֹת דֻּגְמָאוֹת רַבּוֹת עַל שֵׁם עֶצֶם יָחִיד: קוֹלָהּ (=הַקּוֹל שֶׁלָּהּ), זְנָבָהּ (=הַזָּנָב שֶׁלָּהּ), אָבִיהָ (=הָאָב שֶׁלָּהּ).

קוֹלָהּ זְנָבָהּ אָבִיהָ כִּתְרוֹ

הַיּוֹם: כִּנּוּיֵי קִנְיָן עַל שֵׁם עֶצֶם בָּרַבִּים!

מָה קוֹרֶה כְּשֶׁרוֹצִים לוֹמַר "הָאֲחָיוֹת שֶׁלָּהּ" בְּלִי הַמִּלָּה "שֶׁל"? שֵׁם עֶצֶם בָּרַבִּים מְקַבֵּל סֵט אַחֵר שֶׁל סִיּוֹמוֹת - דּוֹמוֹת קְצָת לְאֵלֶּה שֶׁל הַיָּחִיד, אֲבָל לֹא זֵהוֹת לָהֶן, וְכֻלָּן מַכִילוֹת אוֹת יוֹ"ד. הַמִּלָּה "אֲחָיוֹת" (רַבִּים שֶׁל "אָחוֹת") הוֹפֶכֶת ל"אֲחוֹתֶיהָ" (=הָאֲחָיוֹת שֶׁלָּהּ), בְּדִיּוּק כְּפִי שֶׁמּוֹפִיעַ בַּסִּפּוּר.

💡 שִׂימוּ לֵב: הַצּוּרָה הָעַצְמָאִית שֶׁל "רַבּוֹת אֲחָיוֹת" (בְּלִי כִּנּוּי קִנְיָן) הִיא אֲחָיוֹת - אֲבָל בְּרֶגַע שֶׁמּוֹסִיפִים כִּנּוּי קִנְיָן, חוֹזְרִים לַצּוּרָה הַקְּצָרָה יוֹתֵר "אֲחוֹת" וּמוֹסִיפִים אֶת סִיּוֹמֶת הָרַבִּים: אֲחוֹתַי, אֲחוֹתֶיךָ, אֲחוֹתֶיהָ וְכוּ' (וְלֹא "אֲחָיוֹתַי"). זֶה קוֹרֶה גַּם בְּמִלִּים דּוֹמוֹת אַחֵרוֹת, כְּמוֹ סְפָרִים (=סְפָרַי, סְפָרֶיךָ, סְפָרַיִךְ, סְפָרָיו, סְפָרֶיהָ, סְפָרֵינוּ...).
