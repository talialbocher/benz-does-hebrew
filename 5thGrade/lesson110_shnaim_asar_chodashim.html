📖 קְרִיאָה

בִּכְפָר קָטָן לְרַגְלֵי הֶהָרִים גָּרָה יַלְדָּה יְתוֹמָה בְּשֵׁם מָרוּשְׁקָה. הִיא גָּרָה עִם אִמָּהּ הַחוֹרֶגֶת וְעִם אֲחוֹתָהּ הַחוֹרֶגֶת, הוֹלֶנָה. מָרוּשְׁקָה הָיְתָה טוֹבַת לֵב וְחָרוּצָה, וְהוֹלֶנָה הָיְתָה עַצְלָנִית וּמְפֻנֶּקֶת.

הָיָה יוֹם חָרְפִּי, וְהַכְּפָר כֻּלּוֹ הָיָה לָבָן וְשַׁלְגִּי. פִּתְאוֹם אָמְרָה הוֹלֶנָה: ״אֲנִי רוֹצָה לְהָרִיחַ סִגְלִיּוֹת!״ הָאֵם הִסְתַּכְּלָה עַל מָרוּשְׁקָה וְאָמְרָה: ״לְכִי אֶל הָהָר וְהָבִיאִי לָנוּ זֵר סִגְלִיּוֹת. אַל תַּחְזְרִי בְּלִי הַסִּגְלִיּוֹת!״

מָרוּשְׁקָה לָבְשָׁה מְעִיל דַּק וְיָצְאָה אֶל הַיַּעַר. הָרוּחַ נָשְׁבָה בְּחָזְקָה, וְהַשֶּׁלֶג הִגִּיעַ עַד בִּרְכֶּיהָ. אַחֲרֵי שָׁעוֹת רַבּוֹת שֶׁל הֲלִיכָה הִבְחִינָה בְּאוֹר קָטָן בְּרֹאשׁ הָהָר.

סְבִיב מְדוּרָה גְּדוֹלָה יָשְׁבוּ שְׁנֵים עָשָׂר אֲנָשִׁים: שְׁלוֹשָׁה זְקֵנִים עִם זָקָן לָבָן, שְׁלוֹשָׁה צְעִירִים, שְׁלוֹשָׁה גְּבָרִים חֲסוֹנִים וּשְׁלוֹשָׁה בַּחוּרִים שְׁזוּפֵי שֶׁמֶשׁ. הֵם הָיוּ שְׁנֵים עָשָׂר חֳדָשֵׁי הַשָּׁנָה! הַזָּקֵן שֶׁבָּהֶם, בַּעַל הַמַּטֶּה, הָיָה חֹדֶשׁ יָנוּאָר.

מָרוּשְׁקָה סִפְּרָה לָהֶם בְּנִימוּס אֶת סִפּוּרָהּ. יָנוּאָר הִקְשִׁיב לָהּ בְּסַבְלָנוּת וְאָמַר: ״חֲבֵרִים, הַיַּלְדָּה הַזֹּאת זְקוּקָה לְעֶזְרָה!״ הוּא הִגִּישׁ אֶת מַטֵּהוּ לְמַרְס, אָחִיו הַצָּעִיר, וְאָמַר: ״אָחִי, שֵׁב בִּמְקוֹמִי.״

מַרְס הִכָּה בָּאֵשׁ בַּמַּטֶּה, וּפִתְאוֹם הַשֶּׁלֶג נָמַס, הַשָּׁמַיִם הִתְבַּהֲרוּ, וְהָאֲוִיר נַעֲשָׂה חַם וַאֲבִיבִי. מִתַּחַת לָעֵצִים פָּרְחוּ סִגְלִיּוֹת. מָרוּשְׁקָה קָטְפָה סִגְלִיּוֹת רַבּוֹת, הוֹדְתָה לַחֳדָשִׁים וְרָצָה הַבַּיְתָה.

הָאֵם וְהוֹלֶנָה הִתְפַּלְּאוּ מְאוֹד, אֲבָל הוֹלֶנָה כְּבָר רָצְתָה דָּבָר חָדָשׁ: תּוּתֵי שָׂדֶה! מָרוּשְׁקָה חָזְרָה אֶל הָהָר, וְיוּנִי לָקַח אֶת הַמַּטֶּה. תּוֹךְ רֶגַע הָיָה הָאֲוִיר קֵיצִי וְחַם, וְעַל הָאֲדָמָה הִתְמַלְּאוּ תּוּתִים אֲדֻמִּים.

בַּפַּעַם הַשְּׁלִישִׁית רָצְתָה הוֹלֶנָה תַּפּוּחִים. סֶפְּטֶמְבֶּר הֶחֱזִיק בַּמַּטֶּה, וְהַגִּנָּה הָפְכָה סְתָוִית: הָעֵצִים הָיוּ עֲמוּסִים בְּתַפּוּחִים אֲדֻמִּים.

הָאֵם וְהוֹלֶנָה לֹא הִסְתַּפְּקוּ. הֵן הָלְכוּ בְּעַצְמָן אֶל הָהָר וְדָרְשׁוּ בְּחֻצְפָּה עוֹד וְעוֹד. יָנוּאָר כָּעַס וְהִכָּה בַּמַּטֶּה, וְסוּפַת שֶׁלֶג חֲזָקָה כִּסְּתָה אֶת הָהָר. הֵן חָזְרוּ הַבַּיְתָה קְפוּאוֹת וְרֵיקוֹת.

מֵאוֹתוֹ יוֹם לֹא הִצִּיקוּ עוֹד לְמָרוּשְׁקָה. וּבְכָל פַּעַם שֶׁשָּׁמְעָה אֶת הָרוּחַ בֶּהָרִים, הִיא זָכְרָה אֶת שְׁנֵים עָשָׂר חֲבֵרֶיהָ מִסְּבִיב לַמְּדוּרָה.

בַּסִּפּוּר הָיָה יוֹם חָרְפִּי וְהָיָה אֲוִיר אֲבִיבִי. הַתֹּאֲרִים הָאֵלֶּה נוֹצְרוּ מִשֵּׁם עֶצֶם: חֹרֶף ← חָרְפִּי, אָבִיב ← אֲבִיבִי. כְּדֵי לִיצֹר תֹּאַר כָּזֶה מוֹסִיפִים לְשֵׁם הָעֶצֶם אֶת הַסִּיּוֹמֶת ־ִי, וּבְדֶרֶךְ כְּלָל הַנִּקּוּד בְּתוֹךְ הַמִּלָּה מִשְׁתַּנֶּה (שֶׁלֶג ← שַׁלְגִּי, קַיִץ ← קֵיצִי). הַתֹּאַר עוֹנֶה עַל הַשְּׁאֵלָה ״אֵיזֶה?״ וְ״אֵיזוֹ?״.

כְּלָל 3: לֹא כָּל תֹּאַר נִגְזָר מִשֵּׁם עֶצֶם. תֹּאֲרִים כְּמוֹ קַר חַם גָּדוֹל הֵם תֹּאֲרִים בְּסִיסִיִּים, בְּלִי סִיּוֹמֶת ־ִי.
