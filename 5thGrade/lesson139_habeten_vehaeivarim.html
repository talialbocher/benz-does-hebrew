הַבֶּטֶן וְהָאֵיבָרִים

מָשָׁל שֶׁל אִיזוֹפּוֹס עַל אֵיבְרֵי הַגּוּף שֶׁהֶחְלִיטוּ לִשְׁבּוֹת

📖 קְרִיאָה

בְּעִיר קְטַנָּה לְיַד הַיָּם גָּר חַיָּט חָרוּץ בְּשֵׁם אַבְנֵר. מִן הַבֹּקֶר וְעַד הָעֶרֶב יָשַׁב לְיַד הַשֻּׁלְחָן הַגָּדוֹל שֶׁלּוֹ, גָּזַר בַּד בְּמִסְפָּרַיִם וְתָפַר מִכְנָסַיִם וּמְעִילִים לְכָל אַנְשֵׁי הָעִיר. עַל אַפּוֹ יָשְׁבוּ תָּמִיד מִשְׁקָפַיִם עֲגֻלִּים, כְּדֵי שֶׁיּוּכַל לִרְאוֹת כָּל תֶּפֶר קָטָן.

יוֹם אֶחָד נִכְנַס לַחֲנוּת שֶׁלּוֹ חָתָן צָעִיר. "בְּעוֹד שְׁבוּעַיִם תִּהְיֶה הַחֲתֻנָּה שֶׁלִּי," אָמַר, "וַאֲנִי צָרִיךְ מִכְנָסַיִם חֲגִיגִיִּים. תּוּכַל לְהַסְפִּיק?" אַבְנֵר חִיֵּךְ: "בְּוַדַּאי. אֶעֱבֹד שְׁעָתַיִם בַּבֹּקֶר וּשְׁעָתַיִם בָּעֶרֶב, וְהֵם יִהְיוּ מוּכָנִים בַּזְּמַן."

אֲבָל בְּאוֹתוֹ לַיְלָה, כְּשֶׁאַבְנֵר יָשַׁן, הִתְאַסְּפוּ אֵיבְרֵי הַגּוּף שֶׁלּוֹ לַאֲסֵפָה סוֹדִית. הָרִאשׁוֹנוֹת לְדַבֵּר הָיוּ הַיָּדַיִם: "נִמְאַס לָנוּ! כָּל הַיּוֹם אֲנַחְנוּ גּוֹזְרוֹת, תּוֹפְרוֹת וְסוֹחֲבוֹת. וְהַבֶּטֶן? הִיא רַק יוֹשֶׁבֶת בָּאֶמְצַע וְאוֹכֶלֶת!"

"צוֹדְקוֹת!" קָרְאוּ הָרַגְלַיִם. "פַּעֲמַיִם בְּכָל יוֹם אֲנַחְנוּ הוֹלְכוֹת לַשּׁוּק וּבַחֲזָרָה, וְהַבֶּטֶן רַק נֶהֱנֵית מִן הָאֹכֶל." הָעֵינַיִם הוֹסִיפוּ: "וַאֲנַחְנוּ מִסְתַּכְּלוֹת כָּל הַיּוֹם בַּתְּפָרִים הַקְּטַנִּים, עַד שֶׁאֲנַחְנוּ עֲיֵפוֹת." גַּם הָאָזְנַיִם וְהַשִּׁנַּיִם הִסְכִּימוּ: "גַּם אֲנַחְנוּ עוֹבְדוֹת קָשֶׁה, וְהַבֶּטֶן — שׁוּם דָּבָר!"

אָז הֶחְלִיטוּ כֻּלָּן יַחַד: "מֵעַכְשָׁו אֲנַחְנוּ שׁוֹבְתוֹת! הַיָּדַיִם לֹא יָבִיאוּ אֹכֶל אֶל הַפֶּה, וְהַשִּׁנַּיִם לֹא יִלְעֲסוּ אַף פֵּרוּר. נִרְאֶה אֵיךְ הַבֶּטֶן הָעַצְלָנִית תִּסְתַּדֵּר בְּלִי עֶזְרָה!"

בַּבֹּקֶר קָם אַבְנֵר, וְהַיָּדַיִם שֶׁלּוֹ לֹא הֵרִימוּ אֲפִלּוּ פְּרוּסַת לֶחֶם. כָּל הַיּוֹם לֹא אָכַל דָּבָר. הַבֶּטֶן קִרְקְרָה בְּשֶׁקֶט, אֲבָל הָאֵיבָרִים לֹא הִקְשִׁיבוּ לָהּ. וְכָךְ עָבְרוּ יוֹמַיִם בְּלִי אֹכֶל.

בַּיּוֹם הַשְּׁלִישִׁי הִרְגִּישׁוּ הָאֵיבָרִים שֶׁמַּשֶּׁהוּ לֹא בְּסֵדֶר. הָרַגְלַיִם רָעֲדוּ וְלֹא הִצְלִיחוּ לָלֶכֶת לַשּׁוּק. הַיָּדַיִם הָיוּ חַלָּשׁוֹת כָּל כָּךְ, עַד שֶׁלֹּא הִצְלִיחוּ לְהַחְזִיק אֶת הַמִּסְפָּרַיִם. וְהָעֵינַיִם רָאוּ הַכֹּל מְטֻשְׁטָשׁ — גַּם דֶּרֶךְ הַמִּשְׁקָפַיִם. הַמִּכְנָסַיִם שֶׁל הֶחָתָן נִשְׁאֲרוּ עַל הַשֻּׁלְחָן, חֲצִי תְּפוּרִים.

אָז דִּבְּרָה הַבֶּטֶן בְּקוֹל חַלָּשׁ: "חֲבֵרוֹת יְקָרוֹת, אַתֶּן חוֹשְׁבוֹת שֶׁאֲנִי רַק אוֹכֶלֶת. אֲבָל אֶת כָּל מָה שֶׁאֲנִי מְקַבֶּלֶת, אֲנִי מְעַכֶּלֶת וְשׁוֹלַחַת בַּחֲזָרָה אֲלֵיכֶן — כֹּחַ לַיָּדַיִם, כֹּחַ לָרַגְלַיִם וְאוֹר לָעֵינַיִם. כְּשֶׁאַתֶּן מַרְעִיבוֹת אוֹתִי, אַתֶּן מַרְעִיבוֹת אֶת עַצְמְכֶן."

הָאֵיבָרִים הִתְבַּיְּשׁוּ מְאוֹד. מִיָּד הֵבִיאוּ הַיָּדַיִם קְעָרָה שֶׁל מָרָק חַם, וְהַשִּׁנַּיִם לָעֲסוּ פְּרוּסַת לֶחֶם טְרִיָּה. כַּעֲבֹר שְׁעָתַיִם כְּבָר הָיוּ הָרַגְלַיִם חֲזָקוֹת, וְהַיָּדַיִם הֵרִימוּ שׁוּב אֶת הַמִּסְפָּרַיִם.

אַבְנֵר תָּפַר וְתָפַר, וְהַמִּכְנָסַיִם הָיוּ מוּכָנִים בְּדִיּוּק בַּזְּמַן לַחֲתֻנָּה. מֵאָז לֹא רָבוּ עוֹד אֵיבְרֵי הַגּוּף. הֵם הֵבִינוּ שֶׁבְּגוּף אֶחָד — כְּמוֹ בְּכִיתָּה אוֹ בְּמִשְׁפָּחָה — לְכָל חֵלֶק יֵשׁ תַּפְקִיד, וְאַף חֵלֶק אֵינוֹ יָכוֹל לְהִסְתַּדֵּר לְבַד.

מְבֻסָּס עַל הַמָּשָׁל "הַבֶּטֶן וְהָאֵיבָרִים" שֶׁל אִיזוֹפּוֹס (יָוָן הָעַתִּיקָה), בְּעִבּוּד חָפְשִׁי.

כְּלָל 1 — זְמַן וּמִסְפָּרִים: צוּרַת הַזּוּג פֵּרוּשָׁהּ בְּדִיּוּק שְׁנַיִם. מִשָּׁלוֹשׁ וָמַעְלָה חוֹזְרִים לָרַבִּים הָרְגִילִים:

יוֹמַיִם ← שְׁלוֹשָׁה יָמִים שְׁעָתַיִם ← שָׁלוֹשׁ שָׁעוֹת פַּעֲמַיִם ← שָׁלוֹשׁ פְּעָמִים

כְּלָל 2 — אֵיבְרֵי הַגּוּף: כָּאן צוּרַת הַזּוּג מְשַׁמֶּשֶׁת גַּם כְּרַבִּים, בְּכָל מִסְפָּר: לַכֶּלֶב אַרְבַּע רַגְלַיִם לָעַכָּבִישׁ שְׁמוֹנֶה רַגְלַיִם גַּם שִׁנַּיִם בְּצוּרַת הַזּוּג — אַף שֶׁיֵּשׁ לָנוּ הַרְבֵּה יוֹתֵר מִשְּׁתַּיִם! אֵיבְרֵי הַגּוּף הָאֵלֶּה הֵם נְקֵבָה, וְלָכֵן הַתֹּאַר בִּנְקֵבָה רַבּוֹת: עֵינַיִם כְּחֻלּוֹת יָדַיִם חֲזָקוֹת

כְּלָל 3 — סְמִיכוּת: בַּסְּמִיכוּת הַסִּיּוֹמֶת ־ַיִם הוֹפֶכֶת לְ־ֵי: יְדֵי הַחַיָּט עֵינֵי הֶחָתוּל רַגְלֵי הַשֻּׁלְחָן

⚠️ זְהִירוּת: לֹא כָּל מִלָּה שֶׁנִּגְמֶרֶת בְּ־ַיִם הִיא צוּרַת זוּג. בַּמִּלִּים מַיִם, שָׁמַיִם, צָהֳרַיִם וּבִשְׁמוֹת מְקוֹמוֹת כְּמוֹ יְרוּשָׁלַיִם וּמִצְרַיִם אֵין "שְׁנַיִם" בִּכְלָל.

כְּתוֹב חֲמִשָּׁה מִשְׁפָּטִים עַל הַיּוֹם שֶׁל אַבְנֵר הַחַיָּט (אוֹ עַל הַיּוֹם שֶׁלְּךָ), לְפִי הַהוֹרָאָה בְּכָל שׁוּרָה. דֻּגְמָה: "בְּמֶשֶׁךְ יוֹמַיִם הָיוּ הַיָּדַיִם שֶׁל אַבְנֵר חַלָּשׁוֹת."
