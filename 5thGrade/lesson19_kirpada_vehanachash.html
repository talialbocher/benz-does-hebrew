📖 קְרִיאָה

בִּבְרֵיכָה גְּדוֹלָה וִירוּקָּה, בִּקְצֵה יַעַר עָבוֹת בְּהוֹדּוּ, חַיָּה קְהִילַּת צְפַרְדְּעִים שְׁלֵמָה, בְּרֹאשׁוּת מֶלֶךְ זָקֵן וְטוֹב לֵב. יוֹם אֶחָד הִגִּיעַ לִשְׂפַת הַבְּרֵיכָה נָחָשׁ אָרוֹךְ וְכָחוּשׁ, שֶׁנִּרְאָה עָיֵיף מְאוֹד.

צִיפּוֹר עֲנָקִית שָׁבְרָה אֶת שִׁינָּיו שֶׁל הַנָּחָשׁ בִּקְרָב קָשֶׁה בַּגָּ'וּנְגֶל, וּמֵאָז שִׁינָּיו נִשְׁבְּרוּ הוּא לֹא הִצְלִיחַ לָצוּד לְבַד. "אֲנִי מְבַקֵּשׁ רַק פִּינָּה שְׁקֵטָה לָנוּחַ לְיַדְכֶם," אָמַר לַצְּפַרְדְּעִים בְּקוֹל חַלָּשׁ.

הַצְּפַרְדְּעִים נִבְהֲלוּ לְמַרְאֵה הַנָּחָשׁ, אֲבָל מֶלֶךְ הַבְּרֵיכָה רִיחֵם עָלָיו וְהִסְכִּים שֶׁיִּישָּׁאֵר. בִּתְמוּרָה הִצִּיעַ הַנָּחָשׁ טוֹבָה: הוּא יִישָּׂא עַל גַּבּוֹ צְפַרְדְּעִים עֲיֵיפוֹת אוֹ זְקֵנוֹת, מִבְּרֵיכָה לִבְרֵיכָה, בְּלִי לִפְגּוֹעַ בָּהֶן.

בַּהַתְחָלָה כּוּלָּם הָיוּ מְרוּצִּים. צְפַרְדְּעִים זְקֵנוֹת וְחַלָּשׁוֹת הוּבְאוּ עַל גַּב הַנָּחָשׁ לִמְקוֹמוֹת רְחוֹקִים, בְּלִי שֶׁיִּצְטָרְכוּ לִקְפּוֹץ כִּבְרַת דֶּרֶךְ אֲרוּכָּה בְּעַצְמָן. סִיפּוּרִים עַל "הַנָּחָשׁ הַטּוֹב" סוּפְּרוּ מִבְּרֵיכָה לִבְרֵיכָה בַּיַּעַר כּוּלּוֹ. מִפַּעַם לְפַעַם הוּא גַּם הֵבִיא לַצְּפַרְדְּעִים עֲלֵי אוֹכֶל טְרִיִּים מֵהַיַּעַר, כְּמֶחֱוָוה שֶׁל יְדִידוּת.

רַק צְפַרְדְּעָה קְטַנָּה אַחַת, וּשְׁמָהּ קַרְפָּדָה, לֹא הִצְטָרְפָה לַשִּׂמְחָה הַכְּלָלִית. הִיא שָׂמָה לֵב שֶׁמִּדֵּי כַּמָּה יָמִים נֶעֱלֶמֶת צְפַרְדְּעָה אַחַת - וְאַף אֶחָד לֹא זוֹכֵר בְּדִיּוּק לְאָן.

יוֹם אֶחָד, כְּשֶׁכּוּלָּם יָשְׁנוּ, קַרְפָּדָה הִתְחַבְּאָה מֵאֲחוֹרֵי אֶבֶן גְּדוֹלָה לְיַד קֵן הַנָּחָשׁ, וְחִיכְּתָה בְּסַבְלָנוּת. הִיא רָאֲתָה אֵיךְ הַנָּחָשׁ, כְּשֶׁחָשַׁב שֶׁאַף אֶחָד לֹא רוֹאֶה, בּוֹלֵעַ בְּשֶׁקֶט אֶת הַצְּפַרְדְּעָה הַחַלָּשָׁה שֶׁהוּא "עָזַר לָהּ לִנְסוֹעַ".

קַרְפָּדָה רָצָה בְּבֶהָלָה אֶל מֶלֶךְ הַבְּרֵיכָה וְסִיפְּרָה לוֹ אֶת מָה שֶׁרָאֲתָה. בַּהַתְחָלָה הַמֶּלֶךְ לֹא רָצָה לְהַאֲמִין: "הוּא כָּל כָּךְ אָדִיב אֵלֵינוּ!" אָמַר. אֲבָל קַרְפָּדָה עָמְדָה עַל שֶׁלָּהּ, וּבַסּוֹף שִׁכְנְעָה אוֹתוֹ.

יַחַד הֵם קָבְעוּ תּוֹכְנִית: מֶלֶךְ הַבְּרֵיכָה בִּיקֵּשׁ מֵהַנָּחָשׁ שֶׁיִּישָּׂא אוֹתוֹ הַפַּעַם בְּעַצְמוֹ, "כְּדֵי לִבְדּוֹק דֶּרֶךְ חֲדָשָׁה וּקְצָרָה לַבְּרֵיכָה הַגְּדוֹלָה". הַנָּחָשׁ, שֶׁלֹּא יָדַע שֶׁהִתְגַּלָּה, הִסְכִּים בְּשִׂמְחָה.

בְּאֶמְצַע הַדֶּרֶךְ, כְּשֶׁהַנָּחָשׁ פָּתַח אֶת פִּיו כְּדֵי לִתְפּוֹס אֶת הַמֶּלֶךְ, כָּל שְׁאָר הַצְּפַרְדְּעִים - שֶׁקַּרְפָּדָה כְּבָר הִזְהִירָה מֵרֹאשׁ - קָפְצוּ יַחַד מִכָּל עֵבֶר וְהִרְעִישׁוּ כָּל כָּךְ, שֶׁהַנָּחָשׁ נִבְהַל וּבָרַח מֵהַיַּעַר לְתָמִיד.

מֵאוֹתוֹ יוֹם נִקְרְאָה הַבְּרֵיכָה "בְּרֵיכַת קַרְפָּדָה הַחֲכָמָה", וְקַרְפָּדָה עַצְמָהּ זָכְתָה לְכָבוֹד רַב מִכָּל תּוֹשָׁבֵי הַבְּרֵיכָה. הַצְּפַרְדְּעִים לָמְדוּ שִׁיעוּר חָשׁוּב: מִי שֶׁמַּצִּיעַ עֶזְרָה גְּדוֹלָה מִדַּי, בְּלִי שֶׁמְּבַקְשִׁים מִמֶּנּוּ, כְּדַאי לִבְדּוֹק הֵיטֵב לִפְנֵי שֶׁסּוֹמְכִים עָלָיו - וְלִפְעָמִים תְּבוּנָה קְטַנָּה מְנַצַּחַת כּוֹחַ גָּדוֹל.

מָשָׁל בְּהַשְׁרָאַת אוֹסֶף הַפַּנְצָ'טַנְטְרָה הַהוֹדִּי (Panchatantra) - עִיבּוּד חוֹפְשִׁי
