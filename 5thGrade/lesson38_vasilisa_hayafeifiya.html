📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת, בִּכְפָר רָחוֹק בְּרוּסְיָה, גָּרוּ סוֹחֵר, אִשְׁתּוֹ וּבִתָּם הַקְּטַנָּה וָסִילִיסָה, בְּבֵּית הַסּוֹחֵר הַגָּדוֹל שֶׁבִּקְצֵה הַכְּפָר. וָסִילִיסָה הָיְתָה יַלְדָּה יָפָה, חֲכָמָה וְאֲדִיבָה, וְאִמָּהּ אָהֲבָה אוֹתָהּ מִכָּל לִבָּהּ.

יוֹם אֶחָד חָלְתָה הָאֵם קָשֶׁה, וְיָדְעָה שֶׁקִּצָּהּ מִתְקָרֵב. לִפְנֵי מוֹתָהּ קָרְאָה לְוָסִילִיסָה, נָתְנָה לָהּ בֻּבַּת עֵץ קְטַנָּה וְלָחֲשָׁה: "שִׁמְרִי עַל הַבֻּבָּה הַזֹּאת בְּסוֹד. אִם תִּסְתַּבְּכִי בְּצָרָה - תְּנִי לָהּ קְצָת אֹכֶל, וּבִקְשִׁי מִמֶּנָּה עֵצָה. הִיא תֵּדַע לַעֲזֹר לָךְ." זְמַן קָצָר אַחַר כָּךְ, הָאֵם מֵתָה.

וָסִילִיסָה שָׁמְרָה אֶת הַבֻּבָּה תָּמִיד בְּכִיסָהּ, הֶאֱכִילָה אוֹתָהּ בַּלַּיְלָה כְּשֶׁכֻּלָּם יָשְׁנוּ, וְסִיפְּרָה לָהּ כָּל מַה שֶּׁעָבַר עָלֶיהָ בְּאוֹתוֹ יוֹם. עִם הַזְּמַן הִתְחַתֵּן אָבִיהָ עִם אִשָּׁה שֶׁהָיְתָה לָהּ שְׁתֵּי בָּנוֹת מִשֶּׁלָּהּ, אֲבָל אֵשֶׁת הָאָב וּבְנוֹתֶיהָ שָׂנְאוּ אֶת וָסִילִיסָה וְקִנְּאוּ בְּיָפְיָהּ.

כְּשֶׁהָאָב יָצָא לִנְסֹעַ לַעֲסָקִים בְּעִיר רְחוֹקָה, הֵחֵלָּה אֵשֶׁת הָאָב לְהַעֲבִיד אֶת וָסִילִיסָה בְּכָל עֲבוֹדוֹת הַבַּיִת הַקָּשׁוֹת - לִשְׁטֹף, לְבַשֵּׁל, לְכַבֵּס וּלְנַקּוֹת - בְּתִקְוָה שֶׁהָעֲבוֹדָה הַקָּשָׁה תְּקַלְקֵל אֶת יָפְיָהּ. אֲבָל בְּכָל לַיְלָה עָזְרָה הַבֻּבָּה לְוָסִילִיסָה בְּסֵתֶר, וְוָסִילִיסָה רַק הָלְכָה וְהִתְיַפְּתָה.

יוֹם אֶחָד שָׁלְחָה אֵשֶׁת הָאָב אֶת שְׁלֹשׁ הַבָּנוֹת לִתְפֹּר לְאוֹר הַנֵּר - אֲבָל דָּאֲגָה מֵרֹאשׁ שֶׁלֹּא יִישָׁאֵר בַּבַּיִת אַף גַּפְרוּר נוֹסָף. "כְּשֶׁהַנֵּר יִכְבֶּה," אָמְרָה בְּלַחַשׁ לִבְנוֹתֶיהָ, "תִּשְׁלְחוּ אֶת וָסִילִיסָה לַיַּעַר, אֶל בֵּית הַמְּכַשֵּׁפָה אַבָּא יַאגָה, כְּדֵי לְהָבִיא אֵשׁ. אוּלַי הִיא לֹא תַּחֲזֹר."

וְכָךְ קָרָה. הַנֵּר כָּבָה, וְהַבַּיִת הֶחְשִׁיךְ. וָסִילִיסָה יָצְאָה לְבַדָּהּ אֶל תּוֹךְ הַיַּעַר הָאָפֵל, מְפֻחֶדֶת אַךְ עִם הַבֻּבָּה הַקְּטַנָּה בְּכִיסָהּ. "אַל תִּפְחֲדִי," לָחֲשָׁה הַבֻּבָּה, "אֲנִי אִתָּךְ." לְאֹרֶךְ הַדֶּרֶךְ חָלְפוּ עַל פָּנֶיהָ שְׁלֹשָׁה פָּרָשִׁים - אֶחָד לָבָן, אֶחָד אָדֹם וְאֶחָד שָׁחֹר - פָּרָשׁ הַשַּׁחַר, פָּרָשׁ הַצָּהֳרַיִם וּפָרָשׁ הַלַּיְלָה, כְּאִלּוּ הַזְּמַן עַצְמוֹ רוֹכֵב בַּיַּעַר.

לְבַסּוֹף הִגִּיעָה וָסִילִיסָה אֶל בִּקְתָּה מוּזָרָה, עוֹמֶדֶת עַל רַגְלֵי תַּרְנְגֹלֶת, מֻקֶּפֶת בְּגְּדַר עֲצָמוֹת. עַל כָּל עַמּוּד בַּגָּדֵר יָשְׁבָה גֻּלְגֹּלֶת, וּמֵעֵינֵי הַגֻּלְגֹּלֶת בָּקְעָה אֵשׁ קָרָה וּמַפְחִידָה. זוֹ הָיְתָה בִּקְתָּתָהּ שֶׁל אַבָּא יַאגָה, הַמְּכַשֵּׁפָה הַזְּקֵנָה.

אַבָּא יַאגָה יָצְאָה מֵהַבִּקְתָּה וְאָמְרָה: "אִם תִּרְצִי אֵשׁ - עָלַיִךְ לַעֲבֹד אֶצְלִי יוֹם שָׁלֵם. אִם תַּעֲשִׂי אֶת עֲבוֹדָתֵךְ הֵיטֵב - תְּקַבְּלִי אֵשׁ. אִם לֹא - אֹכַל אוֹתָךְ!" הִיא הֶרְאֲתָה לְוָסִילִיסָה עֲרֵמַת גַּרְגִּירֵי תִּירָס מְעֹרָבִים בְּעָפָר, וּבִיקְּשָׁה שֶׁתַּפְרִיד אֶת הַגַּרְגִּירִים הַטּוֹבִים מֵהֶעָפָר עַד הַבֹּקֶר, וְגַם שֶׁתְּנַקֶּה אֶת הַבַּיִת מֵהַמַּסָּד וְעַד הַטְּפָחוֹת.

בַּלַּיְלָה, כְּשֶׁהַמְּכַשֵּׁפָה נִרְדְּמָה, הוֹצִיאָה וָסִילִיסָה אֶת הַבֻּבָּה, נָתְנָה לָהּ קְצָת אֹכֶל וְסִיפְּרָה לָהּ עַל הַמְּשִׂימָה הַבִּלְתִּי אֶפְשָׁרִית. "לְכִי לִישֹׁן," אָמְרָה הַבֻּבָּה, "בַּבֹּקֶר הַכֹּל יִהְיֶה מוּכָן." וְכָךְ הָיָה - עַד הַזְּרִיחָה כְּבָר עָמְדָה עֲרֵמַת הַתִּירָס הַנָּקִי מְסֻדֶּרֶת, וְהַבַּיִת נָקִי וּמַבְרִיק מִקָּצֶה לְקָצֶה.

אַבָּא יַאגָה הוּפְתְּעָה מְאֹד, אֲבָל חָשְׁדָה. "אֵיךְ הִצְלַחַתְּ?" שָׁאֲלָה. וָסִילִיסָה עָנְתָה בִּזְהִירוּת: "בְּעֶזְרַת בִּרְכַּת אִמִּי." כְּשֶׁשָּׁמְעָה זֹאת הַמְּכַשֵּׁפָה נִבְהֲלָה - הִיא לֹא אָהֲבָה אֲנָשִׁים מְבֹרָכִים בְּבֵיתָהּ - וּמִיהֲרָה לִשְׁלֹחַ אֶת וָסִילִיסָה חֲזָרָה הַבַּיְתָה, וְנָתְנָה לָהּ גֻּלְגֹּלֶת עִם עֵינַיִם בּוֹעֲרוֹת כְּאֵשׁ, כְּדֵי שֶׁתָּאִיר לָהּ בַּדֶּרֶךְ וְגַם תִּהְיֶה הָאֵשׁ שֶׁבִּיקְּשָׁה אֵשֶׁת הָאָב.

וָסִילִיסָה חָזְרָה בַּיַּעַר הֶחָשׁוּךְ כְּשֶׁהַגֻּלְגֹּלֶת הַבּוֹעֶרֶת מְאִירָה אֶת דַּרְכָּהּ, עַד שֶׁהִגִּיעָה סוֹף סוֹף הַבַּיְתָה. אֲבָל בַּבַּיִת חִיכְּתָה לָהּ הַפְתָּעָה: מֵאָז שֶׁיָּצְאָה, לֹא הִצְלִיחוּ אֵשֶׁת הָאָב וּבְנוֹתֶיהָ לְהַדְלִיק שׁוּם אֵשׁ בַּבַּיִת - כָּל נֵר שֶׁהִדְלִיקוּ כָּבָה מִיָּד. כְּשֶׁשָּׁבָה וָסִילִיסָה עִם הַגֻּלְגֹּלֶת הַבּוֹעֶרֶת, הִבִּיטוּ בָּהּ עֵינֵי הַגֻּלְגֹּלֶת, וּלְפֶתַע הָאֵשׁ שֶׁבָּהֶן פָּרְצָה וְהִקִּיפָה אֶת אֵשֶׁת הָאָב וְאֶת בְּנוֹתֶיהָ, עַד שֶׁלֹּא נוֹתַר מֵהֶן דָּבָר. וָסִילִיסָה, לְעֻמָּתָן, לֹא נִיזּוֹקָה כְּלָל.

וָסִילִיסָה קָבְרָה אֶת הַגֻּלְגֹּלֶת עָמֹק בָּאֲדָמָה, כְּדֵי שֶׁלֹּא תַּזִּיק לְאִישׁ נוֹסָף, וְעָזְבָה אֶת הַבַּיִת לָעַד. הִיא הָלְכָה אֶל עִיר הַמֶּלֶךְ הַקְּרוֹבָה, מָצְאָה מַחְסֶה אֵצֶל אִשָּׁה זְקֵנָה וְטוֹבַת לֵב, וּבְעֶזְרַת הַבֻּבָּה לָמְדָה לֶאֱרֹג בַּד מֶשִׁי עָדִין וְיָפֶה כְּמוֹ שֶׁאִישׁ בָּעִיר לֹא רָאָה. שְׁמוּעוֹת עַל בִּגְדֵי הַמֶּשִׁי שֶׁאָרְגָה הִגִּיעוּ עַד לָאַרְמוֹן, וְהַמֶּלֶךְ עַצְמוֹ בִּיקֵּשׁ לִפְגֹּשׁ אֶת הָאוֹרֶגֶת הַמֻּפְלָאָה. כְּשֶׁרָאָה אֶת וָסִילִיסָה, הִתְאַהֵב בָּהּ מִיָּד, וּבְקָרוֹב נִישְׂאוּ הַשְּׁנַיִם וְשָׂמְחוּ יָמִים רַבִּים.

אַגֶּדֶת עַם מֻכֶּרֶת מֵרוּסְיָה - "וָסִילִיסָה הַיְּפֵהפִיָּה" - מְעֻבֶּדֶת

מה זו סמיכות?

סמיכות היא צירוף של שתי מילים (או יותר) שיוצר יחד משמעות אחת חדשה, כמו "בית ספר" או "עוגת שוקולד". במקום להשתמש במילת הקישור "של" (למשל "בית של ספר"), מחברים את שתי המילים זו לזו ישירות, בלי מילת קישור באמצע.

המילה הראשונה בצירוף נקראת נסמך, והמילה השנייה נקראת סומך. הסומך הוא זה שקובע את המשמעות המדויקת - "בית ספר" הוא בית מסוג מסוים (מקום שלומדים בו), ולא סתם בית.

איך בונים סמיכות

לפעמים המילה הנסמכת משנה קצת את צורתה כשהיא נכנסת לצירוף. הכלל הנפוץ ביותר: ברבים, הסיומת "ים" הופכת ל"י". למשל: "עצים" הופך ל"עצי" בצירוף "עצי היער", ו"עיניים" הופך ל"עיני" בצירוף "עיני הגולגולת".

איפה שמים את ה' הידיעה?

כלל חשוב: בצירוף סמיכות, ה' הידיעה לא מתווספת למילה הראשונה (הנסמך) - היא מתווספת רק למילה השנייה (הסומך), ומגדירה את כל הצירוף. לכן אומרים "עצי היער" ולא "העצי היער", ו"בית המכשפה" ולא "הבית המכשפה".

סמיכות לעומת "של"

אפשר להביע כמעט אותו רעיון בשתי דרכים: עם סמיכות ("עוגת דני") או עם המילה "של" ("העוגה של דני"). ההבדל: בסמיכות אין מילת קישור, שתי המילים דבוקות זו לזו; עם "של" יש שלוש מילים נפרדות, וזה נשמע יומיומי יותר. סמיכות נפוצה יותר בביטויים קבועים כמו "בית ספר" ו"גן ילדים", שאי אפשר בכלל להגיד אותם עם "של".
