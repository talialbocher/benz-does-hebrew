📖 קְרִיאָה

טוֹחֵן זָקֵן מֵת וְהִשְׁאִיר לִשְׁלֹשֶׁת בָּנָיו יְרֻשָּׁה: לַבֵּן הַבְּכוֹר נָתַן אֶת הַטַּחֲנָה, לַבֵּן הָאֶמְצָעִי נָתַן חֲמוֹר וְלַבֵּן הַצָּעִיר לֹא נִשְׁאַר דָּבָר - חוּץ מֵחָתוּל אֶחָד.

הַבֵּן הַצָּעִיר הִתְיָאֵשׁ: "מָה אֲנִי אֶעֱשֶׂה עִם חָתוּל? כְּשֶׁאֲסַיֵּם לֶאֱכֹל אוֹתוֹ, לֹא יִישָׁאֵר לִי כְּלוּם." אֲבָל הֶחָתוּל, שֶׁהָיָה חָכָם בִּמְיֻחָד, שָׁמַע אֶת הַדְּבָרִים וּפָנָה אֶל הָאָדוֹן הַצָּעִיר שֶׁלּוֹ.

"אַל תִּדְאַג," אָמַר הֶחָתוּל, "רַק תֵּן לִי זוּג מַגָּפַיִם וְשַׂק בַּד, וְתִרְאֶה שֶׁגַּם יְרֻשָּׁה קְטַנָּה יְכוֹלָה לְהָבִיא הַרְבֵּה מַזָּל."

הַבֵּן הַצָּעִיר, שֶׁלֹּא הָיָה לוֹ מָה לְהַפְסִיד, נָתַן לֶחָתוּל אֶת מַה שֶּׁבִּיקֵּשׁ. הֶחָתוּל נָעַל אֶת הַמַּגָּפַיִם, תָּלָה אֶת הַשַּׂק עַל כְּתֵפוֹ, וְיָצָא לַיַּעַר.

בְּתוֹךְ הַשַּׂק שָׂם הֶחָתוּל גֶּזֶר וְעָלֶה יָרוֹק, שָׁכַב בְּשֶׁקֶט עַל הַקַּרְקַע וְהֶעֱמִיד פְּנֵי מֵת - עַד שֶׁאַרְנָב סַקְרָן קָפַץ פְּנִימָה כְּדֵי לִטְעֹם מֵהָאֹכֶל. הֶחָתוּל סָגַר אֶת הַשַּׂק בִּמְהִירוּת וְתָפַס אוֹתוֹ.

מִשָּׁם הָלַךְ הֶחָתוּל הַיְשֵׁר אֶל אַרְמוֹן הַמֶּלֶךְ, קָד קִידָה עֲמֻקָּה, וְאָמַר: "הוֹד מַלְכוּתְךָ, זוֹהִי מַתָּנָה מֵהַמַּרְקִיז דֶּה קַרַבָּס." זֶה הָיָה תֹּאַר שֶׁהֶחָתוּל הִמְצִיא בִּשְׁבִיל אֲדוֹנוֹ, כְּדֵי שֶׁיִּשָּׁמַע חָשׁוּב וּמְכֻבָּד.

בְּמֶשֶׁךְ כַּמָּה שָׁבוּעוֹת הִמְשִׁיךְ הֶחָתוּל לָצוּד וּלְהָבִיא מַתָּנוֹת לָאַרְמוֹן בְּשֵׁם הַמַּרְקִיז דֶּה קַרַבָּס, עַד שֶׁהַמֶּלֶךְ הִכִּיר אֶת הַשֵּׁם הֵיטֵב וְהִתְרַשֵּׁם מְאֹד מֵהַנְּדִיבוּת שֶׁלּוֹ.

יוֹם אֶחָד שָׁמַע הֶחָתוּל שֶׁהַמֶּלֶךְ וּבִתּוֹ הַיָּפָה עוֹמְדִים לָצֵאת לְטִיּוּל בְּמִרְכָּבָה לְאֹרֶךְ הַנָּהָר. הוּא מִיהֵר אֶל אֲדוֹנוֹ הַצָּעִיר וְאָמַר לוֹ: "לֵךְ תִּרְחַץ בַּנָּהָר, בְּדִיּוּק בַּמָּקוֹם שֶׁאֲנִי אַגִּיד לְךָ - וְהַשְּׁאָר תַּשְׁאִיר לִי."

כְּשֶׁעָבְרָה מִרְכֶּבֶת הַמֶּלֶךְ, קָרָא הֶחָתוּל בְּקוֹל גָּדוֹל: "הַצִּילוּ! הַצִּילוּ! לַמַּרְקִיז דֶּה קַרַבָּס נִגְנְבוּ כָּל בְּגָדָיו בִּזְמַן שֶׁרָחַץ!" הַמֶּלֶךְ צִיוָּה מִיָּד לְהָבִיא בְּגָדִים מְפֹאָרִים מֵהָאַרְמוֹן, וְהַבֵּן הַצָּעִיר, לָבוּשׁ עַכְשָׁו כְּמוֹ אָצִיל אֲמִתִּי, הוּזְמַן לַעֲלוֹת לַמִּרְכָּבָה.

בְּעוֹדָם נוֹסְעִים, רָץ הֶחָתוּל הַנּוֹעָז קָדִימָה וּפָגַשׁ אִכָּרִים שֶׁעָבְדוּ בְּשָׂדֶה רָחָב. "כְּשֶׁהַמֶּלֶךְ יִשְׁאַל לְמִי שַׁיָּךְ הַשָּׂדֶה," אָמַר לָהֶם, "תַּעֲנוּ שֶׁהוּא שַׁיָּךְ לַמַּרְקִיז דֶּה קַרַבָּס - אַחֶרֶת יִקְרֶה לָכֶם רַע מְאֹד." הָאִכָּרִים הַנִּבְהָלִים הִסְכִּימוּ, וְהַמֶּלֶךְ הִתְרַשֵּׁם עוֹד יוֹתֵר מִגֹּדֶל הָאֲחֻזָּה שֶׁל הַמַּרְקִיז.

לְבַסּוֹף הִגִּיעַ הֶחָתוּל לְטִירָה עֲנָקִית, שֶׁשַּׁיֶּכֶת לְעֲנָק מַפְחִיד. הוּא נִכְנַס פְּנִימָה בְּבִטָּחוֹן וּבִיקֵּשׁ מֵהָעֲנָק לְהוֹכִיחַ אֶת כֹּחוֹתָיו: "שָׁמַעְתִּי שֶׁאַתָּה יָכוֹל לַהֲפֹךְ אֶת עַצְמְךָ לְכָל חַיָּה שֶׁתִּרְצֶה - אֲפִלּוּ לְאַרְיֵה!" הָעֲנָק, גֵּאֶה בְּכוֹחוֹ, הָפַךְ מִיָּד לְאַרְיֵה שׁוֹאֵג.

"מַרְשִׁים מְאֹד," אָמַר הֶחָתוּל, "אֲבָל בֶּטַח שֶׁאַתָּה לֹא מְסֻגָּל לַהֲפֹךְ לְחַיָּה קְטַנָּה, כְּמוֹ עַכְבָּר." הָעֲנָק, שֶׁרָצָה לְהוֹכִיחַ שֶׁהוּא בֶּאֱמֶת מְסֻגָּל לְכָל דָּבָר, הָפַךְ מִיָּד לְעַכְבָּר קָטָן - וּבְרֶגַע הַזֶּה קָפַץ עָלָיו הֶחָתוּל וְתָפַס אוֹתוֹ.

כְּשֶׁהִגִּיעָה מִרְכֶּבֶת הַמֶּלֶךְ אֶל שַׁעֲרֵי הַטִּירָה, כְּבָר לֹא הָיָה בָּהּ עֲנָק, וּכְבָר חִיכָּה שָׁם הֶחָתוּל, שֶׁהִכְרִיז בְּגַאֲוָה: "בְּרוּכִים הַבָּאִים לְטִירַת הַמַּרְקִיז דֶּה קַרַבָּס!" הַמֶּלֶךְ הִתְפַּעֵל כָּל כָּךְ מֵהַטִּירָה, מֵהָאוֹצָרוֹת וּמֵהַנִּמּוּסִים הַטּוֹבִים שֶׁל הַבֵּן הַצָּעִיר, עַד שֶׁהִצִּיעַ לוֹ אֶת יַד בִּתּוֹ לְנִישּׂוּאִים.

כָּךְ הָפַךְ הַבֵּן הַצָּעִיר, שֶׁחָשַׁב שֶׁלֹּא נִשְׁאַר לוֹ כְּלוּם חוּץ מֵחָתוּל אֶחָד, לְחָתָן שֶׁל הַנְּסִיכָה - וְכָל זֶה בִּזְכוּת חָתוּל חָכָם אֶחָד שֶׁיָּדַע לְהִשְׁתַּמֵּשׁ בָּרֹאשׁ, לֹא רַק בַּצִּפֳּרְנַיִם.

עִבּוּד שֶׁל סִפּוּר הָעָם "הֶחָתוּל בַּמַּגָּפַיִם" מֵאֵת שַׁארְל פֶּרוֹ (צָרְפַת, הַמֵּאָה הַ-17)

חמישה חלקים קבועים במכתב אישי

מכתב אישי הוא מכתב שכותבים לאדם קרוב - בן משפחה, חבר או מכר - כדי לספר, לשאול או לשתף. כל מכתב אישי בנוי מחמישה חלקים קבועים, תמיד באותו סדר:

איך בונים את גוף המכתב?

גוף המכתב בנוי בדרך כלל משלושה חלקים קטנים: פתיחה קצרה (למשל, מה שלומכם, או למה אני כותב), התוכן העיקרי (הסיפור, החדשות או הבקשה) וסיום קטן (שאלה לקורא, או איחול). כך גוף המכתב לא קופץ ממחשבה למחשבה, אלא זורם בסדר הגיוני.

איך מתאימים את הפנייה והסיום למקבל המכתב?

הפנייה והסיום צריכים להתאים לקשר עם מי שמקבל את המכתב. לחבר קרוב או לבן משפחה אפשר לכתוב "יקירי" או "אחים יקרים"; למישהו שמכירים פחות אפשר לכתוב "שלום רב". גם הסיום משתנה: "באהבה רבה" לבן משפחה קרוב, "בברכה" למישהו רשמי יותר.

כתבו מכתב אישי קצר, כאילו אתם הבן הצעיר מהסיפור, ששולחים מכתב לאחיכם ומספרים להם על ההרפתקה המדהימה שעברתם - איך החתול עזר לכם להכיר את המלך, ואיך הגעתם בסוף לטירה ולנישואים עם הנסיכה. כתבו כל חלק במכתב בשורה נפרדת, בדיוק לפי חמשת החלקים שלמדנו.
