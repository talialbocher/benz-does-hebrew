הַקּוֹף וְהַתַּנִּין

📖 קְרִיאָה

עַל גְּדַת נָהָר רָחָב, בְּאֶרֶץ הֹדּוּ הָרְחוֹקָה, צָמַח פַּעַם עֵץ תְּאֵנָה עָנָק וְעָבֹת עָלִים. עַל עַנְפֵי הָעֵץ הַגְּבוֹהִים גָּר קוֹף פִּקֵּחַ וְזָרִיז, שֶׁאָהַב לְטַפֵּס מֵעָנָף לְעָנָף וּלְקַטֵּף אֶת הַתְּאֵנִים הַמְּתוּקוֹת שֶׁגָּדְלוּ שָׁם. בַּמַּיִם הַכֵּהִים שֶׁל הַנָּהָר, מַמָּשׁ מִתַּחַת לָעֵץ, גָּר תַּנִּין גָּדוֹל וְשָׁקֵט.

יוֹם אֶחָד יָרַד הַקּוֹף עַד לְעָנָף הַנָּמוּךְ בְּיוֹתֵר, וְזָרַק לַתַּנִּין כַּמָּה תְּאֵנִים בְּשֵׁלוֹת וּמְתוּקוֹת. הַתַּנִּין שֶׁטָּעַם אוֹתָן הִתְלַהֵב מְאֹד, וּמֵאוֹתוֹ הַיּוֹם וָהָלְאָה הָיוּ הַשְּׁנַיִם לְלֹא נִפְרָדִים: בְּכָל בֹּקֶר יָרַד הַקּוֹף אֶל הָעָנָף הַנָּמוּךְ, וְזָרַק לְחֶבְרוֹ שְׁמוֹנָה פֵּרוֹת בְּשֵׁלִים בְּדִיּוּק. הֵם שׂוֹחֲחוּ שָׁלֹשׁ שָׁעוֹת אֲרֻכּוֹת עַל גְּדוֹת הַנָּהָר - עַל הַיַּעַר, עַל הַשָּׁמַיִם, וְעַל כָּל מַה שֶּׁקּוֹרֶה בָּעוֹלָם.

יוֹם אֶחָד לָקַח הַתַּנִּין כַּמָּה תְּאֵנִים בְּפִיו, וְשָׂחָה אִתָּן עַד לְבֵיתוֹ שֶׁמֵּעֵבֶר לַנָּהָר - מְעָרַת אֶבֶן שְׁקוּפָה מִתַּחַת לַמַּיִם. שָׁם חִכְּתָה לוֹ אִשְׁתּוֹ, שֶׁיָּשְׁבָה עַל אֶחָד מִשְּׁלֹשָׁה כִּסְאוֹת אֶבֶן לְיַד שֻׁלְחָן קָטָן, מוּל שְׁנֵי חַלּוֹנוֹת שֶׁדֶּרֶךְ בָּהֶם נִכְנַס אוֹר יָרֹק וּמְרַצֵּד. "טַעֲמִי אֶת זֶה," אָמַר הַתַּנִּין בְּגַאֲוָה, "חָבֵר שֶׁלִּי, קוֹף פִּקֵּחַ שֶׁגָּר עַל הָעֵץ, נוֹתֵן לִי אוֹתָן כָּל בֹּקֶר."

אִשְׁתּוֹ שֶׁל הַתַּנִּין טָעֲמָה תְּאֵנָה אַחַת, וְחִיּוּךְ עָרוּם עָלָה עַל פָּנֶיהָ. "אִם הַפְּרִי הַזֶּה כָּל כָּךְ מָתוֹק," חָשְׁבָה לְעַצְמָהּ, "בְּטַח לִבּוֹ שֶׁל הַקּוֹף שֶׁאוֹכֵל אוֹתוֹ כָּל יוֹם מָתוֹק שִׁבְעָתַיִם." בְּלִבָּהּ טָוְתָה מְזִמָּה. לְמָחֳרָת הֶעֱמִידָה פָּנִים שֶׁהִיא חוֹלָה מְאֹד, שָׁכְבָה בַּמִּטָּה וְלֹא קָמָה. "רַק דָּבָר אֶחָד יְרַפֵּא אוֹתִי," לָחֲשָׁה לְבַעְלָהּ בְּקוֹל חַלָּשׁ, "הָבֵא לִי אֶת לִבּוֹ שֶׁל הַקּוֹף חֲבֵרְךָ - כָּךְ אָמַר לִי דָּג זָקֵן וְחָכָם שֶׁפָּגַשְׁתִּי בַּנָּהָר."

הַתַּנִּין הִתְלַבֵּט יוֹמַיִם שְׁלֵמִים: הוּא אָהַב אֶת אִשְׁתּוֹ, אֲבָל גַּם אֶת חֲבֵרוֹ הַקּוֹף. לְבַסּוֹף הֶחְלִיט לְנַסּוֹת, וּבַבֹּקֶר הַשְּׁלִישִׁי שָׂחָה אֶל עֵץ הַתְּאֵנָה וְקָרָא: "חָבֵר, בּוֹא נַעֲשֶׂה מַשֶּׁהוּ מְיֻחָד הַיּוֹם - עֲלֵה עַל גַּבִּי, וְאֶשָּׂא אוֹתְךָ אֶל בֵּיתִי, שֶׁתַּכִּיר סוֹף סוֹף אֶת אִשְׁתִּי!" הוּא לֹא הָיָה צָרִיךְ אֲפִלּוּ לְפַתּוֹת אוֹתוֹ - הַקּוֹף שָׂמַח מְאֹד, כִּי מֵעוֹלָם לֹא שָׂחָה בַּנָּהָר, וְקָפַץ בְּלִי הִסּוּס עַל גַּב הַתַּנִּין.

בְּאֶמְצַע הַנָּהָר, כְּשֶׁהַמַּיִם כְּבָר הָיוּ עֲמֻקִּים, הֵחֵל הַתַּנִּין לְהַרְגִּישׁ רַע עִם עַצְמוֹ. הוּא עָצַר בַּמָּקוֹם וְאָמַר בְּקוֹל נָמוּךְ: "חָבֵר... אֲנִי חַיָּב לוֹמַר לְךָ אֶת הָאֱמֶת. אִשְׁתִּי חוֹלָה, וְהִיא בִּקְּשָׁה מִמֶּנִּי לְהָבִיא לָהּ אֶת לִבְּךָ כִּרְפוּאָה. סְלַח לִי." הַקּוֹף הֵבִין בְּבַת אַחַת שֶׁחֲבֵרוֹ הַטּוֹב נִסָּה לְרַמּוֹת אוֹתוֹ - אֲבָל הוּא לֹא צָעַק וְלֹא נִבְהַל.

בְּקֹר רוּחַ מַפְתִּיעַ עָנָה הַקּוֹף: "אוֹי, אֵיזֶה חֲבָל שֶׁלֹּא אָמַרְתָּ לִי לִפְנֵי שֶׁיָּרַדְתִּי מֵהָעֵץ! כָּל קוֹף חָכָם מַשְׁאִיר אֶת לִבּוֹ תָּלוּי בְּבִטְחָה עַל עָנָף גָּבוֹהַּ, וְלוֹקֵחַ אוֹתוֹ אִתּוֹ רַק כְּשֶׁהוּא חוֹזֵר הַבַּיְתָה. הַלֵּב שֶׁלִּי נִשְׁאָר שָׁם עַל הָעֵץ בְּדִיּוּק עַכְשָׁו! בּוֹא נָשׁוּב מַהֵר, וְאֶתֵּן לְךָ אוֹתוֹ בְּעַצְמִי." הַתַּנִּין, שֶׁלֹּא חָשַׁד בְּדָבָר, שָׂמַח מְאֹד וְשָׂחָה בַּחֲזָרָה בִּמְהִירוּת אֶל הָעֵץ.

בְּרֶגַע שֶׁהִגִּיעוּ אֶל הָעֵץ, קָפַץ הַקּוֹף בִּזְרִיזוּת מֵעַל גַּב הַתַּנִּין וְטִפֵּס בִּמְהִירוּת עַד לְעָנָף הַגָּבוֹהַּ בְּיוֹתֵר. מִשָּׁם הִבִּיט לְמַטָּה וְאָמַר: "יְקִירִי, קוֹפִים לְעוֹלָם לֹא מַשְׁאִירִים אֶת לִבָּם עַל עֵץ - זֶה סִפּוּר שֶׁהִמְצֵאתִי בְּרֶגַע. עַכְשָׁו אֲנִי מֵבִין שֶׁלֹּא אוּכַל לָתֵת בְּךָ אֱמוּן שׁוּב."

הַתַּנִּין הוֹרִיד אֶת רֹאשׁוֹ בְּבֹשֶׁה. "צָדַקְתָּ לְהִתְרַחֵק מִמֶּנִּי," אָמַר בְּעֶצֶב, "אַהֲבַת בַּיִת אֶחָד לֹא צְרִיכָה לָבוֹא עַל חֶשְׁבּוֹן חָבֵר טוֹב. אֲנִי מִצְטַעֵר, וַאֲקַוֶּה שֶׁיּוֹם אֶחָד תִּסְלַח לִי." הַקּוֹף לֹא עָנָה מִיָּד, אַךְ גַּם לֹא הִתְרַחֵק לְגַמְרֵי - הוּא הוֹסִיף לָשֶׁבֶת עַל הָעֵץ, קָרוֹב אֲבָל זָהִיר, וְלָמַד לִבְחֹר בִּזְהִירוּת בְּמִי רָאוּי לֶאֱמוּנוֹ.

מֵאוֹתוֹ הַיּוֹם וָהָלְאָה סִפְּרוּ בְּכָל הַיַּעַר אֶת סִפּוּר הַקּוֹף הַפִּקֵּחַ, שֶׁלִּמֵּד אֶת כֻּלָּם: חֲבֵרוּת אֲמִתִּית בְּנוּיָה עַל אֱמוּן הֲדָדִי, וְלֹא כָּל מִי שֶׁמְּחַיֵּךְ אֵלֶיךָ בִּידִידוּת רוֹצֶה בֶּאֱמֶת בְּטוֹבָתְךָ - אֲבָל גַּם קֹר רוּחַ וּמַחְשָׁבָה מְהִירָה יְכוֹלִים לְהַצִּיל אֶת מִי שֶׁיּוֹדֵעַ לְהִשְׁתַּמֵּשׁ בָּהֶם.

מְסֻפָּר עַל־פִּי מָשָׁל הֹדִּי עַתִּיק מֵאוֹסֶף הַפַּנְצֶ'טַנְטְרָה (Panchatantra), הֹדּוּ הָעַתִּיקָה, מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ.

הַיּוֹצֵא מִן הַכְּלָל: הַמִּסְפָּר "שְׁמוֹנֶה"

לְרֹב הַמִּסְפָּרִים יֵשׁ שְׁתֵּי צוּרוֹת שׁוֹנוֹת מְאֹד - זָכָר וּנְקֵבָה, עִם אוֹ בְּלִי ה' בַּסּוֹף. אֲבָל הַמִּסְפָּר "שְׁמוֹנֶה" מְיֻחָד: הוּא הַיָּחִיד (מִבֵּין 2 עַד 10) שֶׁנִּכְתָּב בְּאוֹתָן הָאוֹתִיּוֹת בְּדִיּוּק בִּשְׁנֵי הַמִּינִים, בְּלִי שֶׁה' מִתְוַסֶּפֶת אוֹ נֶעֱלֶמֶת. בַּסִּפּוּר זָרַק הַקּוֹף לַתַּנִּין "שְׁמוֹנָה פֵּרוֹת" (זָכָר) - וְאוֹתָן הָאוֹתִיּוֹת מְשַׁמְּשׁוֹת גַּם עִם שֵׁם עֶצֶם נְקֵבָה, אַף שֶׁהַנִּקּוּד מִשְׁתַּנֶּה קְצָת: קָמָץ בַּזָּכָר (שְׁמוֹנָה), סֶגּוֹל בַּנְּקֵבָה (שְׁמוֹנֶה).

שֵׁמוֹת עֶצֶם כְּמוֹ שֻׁלְחָן, כִּסֵּא וְחַלּוֹן הֵם זָכָר בְּיָחִיד - אֲבָל הָרַבִּים שֶׁלָּהֶם מִסְתַּיֵּם בְּ־וֹת (שֻׁלְחָנוֹת, כִּסְאוֹת, חַלּוֹנוֹת), בְּדִיּוּק כְּמוֹ רַבִּים שֶׁל שֵׁם עֶצֶם נְקֵבָה! בַּסִּפּוּר יָשְׁבָה אֵשֶׁת הַתַּנִּין עַל אֶחָד "מִשְּׁלֹשָׁה כִּסְאוֹת" וּמוּל "שְׁנֵי חַלּוֹנוֹת" - בְּצוּרַת זָכָר, לַמְרוֹת הַסּוֹף בְּ־וֹת, כִּי שֵׁם הַמִּסְפָּר תָּמִיד מַתְאִים לְמִין שֵׁם הָעֶצֶם בַּיָּחִיד, וְלֹא לְצוּרַת הָרַבִּים.
