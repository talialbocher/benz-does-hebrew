📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת מְאֹד, לְמַרְגְּלוֹת הַר גָּבֹהַ וְקָרוֹב לְחוֹף הַיָּם, שָׁכַן כְּפָר קָטָן וְשָׁלֵו בְּסִין הָעַתִּיקָה. תּוֹשָׁבֵי הַכְּפָר עָבְדוּ בַּשָּׂדוֹת מִן הַבֹּקֶר וְעַד הָעֶרֶב, גִּדְּלוּ אֹרֶז וִירָקוֹת וְחָיוּ בְּשָׁלוֹם - עַד לַלַּיְלָה הָאַחֲרוֹן שֶׁל כָּל שָׁנָה.

כִּי בְּדִיּוּק בַּלַּיְלָה הַזֶּה, פַּעַם בְּשָׁנָה, יָצְאָה מִן הָהָר מִפְלֶצֶת עֲנָקִית וּמְפַחִידָה בְּשֵׁם "נְיָאן". לַנְּיָאן הָיָה רֹאשׁ אָרֹךְ וְקַרְנַיִם חַדּוֹת, שִׁנַּיִם גְּדוֹלוֹת וּשְׁאָגָה שֶׁהִרְעִידָה אֶת הָאֲדָמָה. כָּל שָׁנָה, בְּאוֹתוֹ הַלַּיְלָה בְּדִיּוּק, הָיְתָה הַמִּפְלֶצֶת יוֹרֶדֶת אֶל הַכְּפָר, טוֹרֶפֶת בְּהֵמוֹת, הוֹרֶסֶת שָׂדוֹת וּמַפְחִידָה אֶת הַתּוֹשָׁבִים.

מֵאָז שֶׁזָּכְרוּ תּוֹשָׁבֵי הַכְּפָר, נָהֲגוּ לִבְרֹחַ מִדֵּי שָׁנָה, יוֹמַיִם לִפְנֵי הַלַּיְלָה הַמְּסֻכָּן, אֶל הֶהָרִים הָרְחוֹקִים, וְשָׁם הִסְתַּתְּרוּ בִּמְעָרוֹת עַד שֶׁחָלְפָה הַסַּכָּנָה. הֵם נָעֲלוּ אֶת בָּתֵּיהֶם, לָקְחוּ אִתָּם רַק אֶת הַחָשׁוּב בְּיוֹתֵר וְהִשְׁאִירוּ אֶת הַכְּפָר רֵיק וְשָׁקֵט - מְחַכֶּה שֶׁהַנְּיָאן תָּבוֹא וְתֵלֵךְ.

אֲבָל בְּשָׁנָה מְסֻיֶּמֶת, בְּדִיּוּק כְּשֶׁכָּל הַתּוֹשָׁבִים מִהֲרוּ הַחוּצָה מִן הַכְּפָר אֶל הֶהָרִים, הוֹפִיעַ פִּתְאֹם זָקֵן זָר, לָבוּשׁ בְּבִגְדֵי קַבְּצָן וּמַקֵּל אָרֹךְ בְּיָדוֹ. הוּא בִּקֵּשׁ לָלוּן בַּכְּפָר לַיְלָה אֶחָד בִּלְבַד. אִשָּׁה זְקֵנָה שֶׁגָּרָה קָרוֹב לְשַׁעַר הַכְּפָר נֶעֶצְרָה לְרֶגַע וְהִזְהִירָה אוֹתוֹ: "לֵךְ מִכָּאן מַהֵר! הַלַּיְלָה תָּבוֹא מִפְלֶצֶת נוֹרָאָה!" אַךְ הַזָּקֵן חִיֵּךְ בְּשֶׁקֶט וְהֵשִׁיב: "אַל תִּדְאֲגִי לִי. אֲנִי אֶשָּׁאֵר כָּאן הַלַּיְלָה."

הָאִשָּׁה, שֶׁלֹּא הָיָה לָהּ זְמַן לְהִתְוַכֵּחַ, בָּרְחָה יַחַד עִם כָּל הַשְּׁאָר אֶל הֶהָרִים. הַזָּקֵן נִשְׁאַר לְבַדּוֹ בַּכְּפָר הָרֵיק. הוּא הִסְתּוֹבֵב בֵּין הַבָּתִּים, אָסַף נְיָר אָדֹם שֶׁמָּצָא בְּבַיִת אֶחָד וְתָלָה אוֹתוֹ עַל דְּלָתוֹת רַבּוֹת. אַחַר כָּךְ הִדְלִיק נֵרוֹת בְּכָל חַלּוֹן וְעָרַם לְיַד שַׁעַר הַכְּפָר עֲרֵמָה גְּדוֹלָה שֶׁל עַנְפֵי בַּמְבּוּק יְבֵשִׁים.

בְּדִיּוּק בַּחֲצוֹת הַלַּיְלָה, כְּמוֹ בְּכָל שָׁנָה, יָצְאָה הַנְּיָאן מִן הָהָר וְיָרְדָה אֶל הַכְּפָר. הִיא צָעֲדָה לְאַט לְעֵבֶר הַשַּׁעַר, מְצַפָּה לִמְצֹא כְּפָר רֵיק וְחָשׁוּךְ כְּמוֹ בְּכָל שָׁנָה. אַךְ הַפַּעַם, מַשֶּׁהוּ הָיָה שׁוֹנֶה לְגַמְרֵי.

כְּשֶׁהִגִּיעָה הַנְּיָאן עַד הַשַּׁעַר, הִצִּית הַזָּקֵן בִּמְהִירוּת אֶת עֲרֵמַת הַבַּמְבּוּק. הָאֵשׁ הִתְלַקְּחָה בְּקוֹל נְפִיצוֹת חֲזָקוֹת, וְהָאוֹר הַבּוֹעֵר הֵאִיר אֶת הַנְּיָר הָאָדֹם עַל הַדְּלָתוֹת. הַמִּפְלֶצֶת נֶעֶצְרָה בְּפֶתַע, נִבְהֶלֶת מִן הָאוֹר הַבּוֹהֵק, מִן הַצֶּבַע הָאָדֹם וּמִן הָרַעַשׁ הַפִּתְאוֹמִי. הִיא נָסוֹגָה אֲחוֹרַנִּית, הִסְתּוֹבְבָה וּבָרְחָה בַּחֲזָרָה אֶל תּוֹךְ הָהָר בִּמְהִירוּת רַבָּה - וּמֵאָז לֹא חָזְרָה אַף פַּעַם.

לְמָחֳרָת בַּבֹּקֶר, כְּשֶׁחָזְרוּ תּוֹשָׁבֵי הַכְּפָר מִן הֶהָרִים בִּזְהִירוּת, הֵם נִדְהֲמוּ לִמְצֹא אֶת הַכְּפָר שָׁלֵם וְשָׁקֵט, בְּלִי שׁוּם נֶזֶק. הָאִשָּׁה הַזְּקֵנָה סִפְּרָה לְכֻלָּם עַל הַזָּקֵן הַמּוּזָר, עַל הַנְּיָר הָאָדֹם, עַל הַנֵּרוֹת וְעַל הַבַּמְבּוּק הַבּוֹעֵר. אֲבָל הַזָּקֵן עַצְמוֹ כְּבָר לֹא הָיָה שָׁם - הוּא נֶעֱלַם, כְּפִי שֶׁהוֹפִיעַ, בְּלִי לְהִשָּׁאֵר אֲפִלּוּ לְתוֹדָה.

מֵאָז אוֹתוֹ לַיְלָה, נוֹהֲגִים תּוֹשָׁבֵי הַכְּפָר, וְאַחֲרֵיהֶם כָּל סִין, לַעֲשׂוֹת בְּכָל שָׁנָה, בְּלֵיל הַשָּׁנָה הַחֲדָשָׁה, בְּדִיּוּק אֶת מַה שֶּׁעָשָׂה הַזָּקֵן: תּוֹלִים נְיָר אָדֹם עַל הַדְּלָתוֹת, מַדְלִיקִים אוֹרוֹת בְּכָל בַּיִת וּמַדְלִיקִים זִיקוּקִים רוֹעֲשִׁים בְּכָל רְחוֹב. מֵאָז וְעַד הַיּוֹם, הַנְּיָאן לֹא חָזְרָה אַף פַּעַם - וּבְסִינִית קוֹרְאִים לְחַג הַשָּׁנָה הַחֲדָשָׁה "גוֹ נְיָאן" (過年), שֶׁפֵּרוּשׁוֹ "לַעֲבֹר אֶת הַנְּיָאן" אוֹ "לְנַצֵּחַ אֶת הַנְּיָאן".

מְסֻפָּר עַל־פִּי אֲגָדַת "הַנְּיָאן" (年獸), אַחַת הָאֲגָדוֹת הָעַתִּיקוֹת בְּיוֹתֵר בְּסִין עַל מְקוֹר חַג הַשָּׁנָה הַחֲדָשָׁה הַסִּינִית, מְעֻבֶּדֶת וּמְסֻפֶּרֶת מֵחָדָשׁ.

מָה זֶה תֵּאוּרֵי מָקוֹם וּזְמַן?

בְּשִׁיעוּר 77 לָמַדְנוּ עַל תֹּאַר הַפֹּעַל - מִלִּים שֶׁמְּתָאֲרוֹת אֵיךְ נַעֲשֵׂית הַפְּעֻלָּה (לְמָשָׁל: "רָץ מַהֵר", "תָּלָה בִּזְהִירוּת"). הַיּוֹם נַכִּיר שְׁנֵי סוּגִים נוֹסָפִים שֶׁל תֵּאוּרֵי פֹּעַל, בְּמוּבָן רָחָב: מִלִּים וְצֵרוּפִים שֶׁלֹּא עוֹנִים עַל הַשְּׁאֵלָה "אֵיךְ?", אֶלָּא עַל הַשְּׁאֵלוֹת "אֵיפֹה?" (תֵּאוּר מָקוֹם) וְ"מָתַי?" (תֵּאוּר זְמַן). שְׁלָשְׁתָּם יַחַד - אֵיךְ, אֵיפֹה וּמָתַי - מוֹסִיפִים פְּרָטִים חֲשׁוּבִים עַל הַפְּעֻלָּה בַּמִּשְׁפָּט, בְּלִי לְתָאֵר שֵׁם עֶצֶם.
