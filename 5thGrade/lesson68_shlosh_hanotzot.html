שְׁלֹשׁ הַנּוֹצוֹת

מֶלֶךְ זָקֵן נוֹפֵחַ שָׁלֹשׁ נוֹצוֹת בָּרוּחַ כְּדֵי לִבְחֹר יוֹרֵשׁ - מִי מִשְּׁלֹשֶׁת הַבָּנִים יָבִיא אֶת הַשָּׁטִיחַ, הַטַּבַּעַת וְהַכַּלָּה הֲכִי יָפִים?

📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת מָלַךְ בְּמַמְלָכָה רְחוֹקָה מֶלֶךְ זָקֵן וְטוֹב לֵב. לַמֶּלֶךְ הָיוּ שְׁלוֹשָׁה בָּנִים: הַבֵּן הָרִאשׁוֹן וְהַבֵּן הַשֵּׁנִי נֶחְשְׁבוּ לַחֲכָמִים וְלִנְבוֹנִים, אֲבָל עַל הַבֵּן הַשְּׁלִישִׁי, הַצָּעִיר מִכֻּלָּם, אֲחָיו נָהֲגוּ לְהִסְתַּכֵּל בְּזִלְזוּל וְלַחְשֹׁב שֶׁהוּא פָּחוֹת פִּקֵּחַ מֵהֶם.

כְּשֶׁהִזְקִין הַמֶּלֶךְ מְאֹד, הֶחְלִיט שֶׁהִגִּיעַ הַזְּמַן לִבְחֹר מִי מִבָּנָיו יִירַשׁ אֶת הַכֶּתֶר. הוּא קָרָא לִשְׁלוֹשְׁתָּם וְאָמַר: "מִי שֶׁיָּבִיא לִי אֶת הַשָּׁטִיחַ הֲכִי יָפֶה בָּעוֹלָם - יִהְיֶה לְמֶלֶךְ אַחֲרָי."

כְּדֵי לִבְחֹר לְכָל בֵּן לְאָן לָלֶכֶת, עָמַד הַמֶּלֶךְ בְּשַׁעַר הַטִּירָה וְנָפַח שָׁלֹשׁ נוֹצוֹת אֶל הָרוּחַ. הַנּוֹצָה הָרִאשׁוֹנָה עָפָה רָחוֹק מִזְרָחָה, וְהַבֵּן הָרִאשׁוֹן יָצָא בְּעִקְבוֹתֶיהָ. הַנּוֹצָה הַשְּׁנִיָּה עָפָה רָחוֹק מַעֲרָבָה, וְהַבֵּן הַשֵּׁנִי יָצָא בְּעִקְבוֹתֶיהָ. אַךְ הַנּוֹצָה הַשְּׁלִישִׁית כִּמְעַט לֹא זָזָה מִמְּקוֹמָהּ - הִיא צָנְחָה יָשָׁר לְמַטָּה, מַמָּשׁ לְיַד הַשַּׁעַר.

אֶחָיו שֶׁל הַבֵּן הַשְּׁלִישִׁי צָחֲקוּ: "אֵיךְ תָּבִיא שָׁטִיחַ מְיֻחָד אִם אַתָּה אֲפִלּוּ לֹא יוֹצֵא מֵהֶחָצֵר?" הַבֵּן הַשְּׁלִישִׁי הִצְטַעֵר מְעַט, יָשַׁב לְיַד הַמָּקוֹם שֶׁבּוֹ נָפְלָה הַנּוֹצָה - וְגִלָּה שָׁם, בֵּין הָאֲבָנִים, דֶּלֶת עֵץ קְטַנָּה עִם מַדְרֵגוֹת יוֹרְדוֹת אֶל הָאֲדָמָה.

הוּא יָרַד בַּמַּדְרֵגוֹת וּפָגַשׁ שָׁם קִרְפָּדָה גְּדוֹלָה וּטוֹבַת לֵב, מֻקֶּפֶת בְּקִרְפָּדוֹת קְטַנּוֹת. כְּשֶׁסִּפֵּר לָהּ עַל הַמְּשִׂימָה, קָרְאָה הַקִּרְפָּדָה לְאַחַת הַקְּטַנּוֹת, וְזוֹ הֵבִיאָה תֵּבַת עֵץ. בְּתוֹךְ הַתֵּבָה הָיָה מֻנָּח שָׁטִיחַ אָרוּג וּמַרְהִיב, יָפֶה מִכָּל שָׁטִיחַ שֶׁרָאָה בְּחַיָּיו.

בֵּינְתַיִם, הָאַחִים הַגְּדוֹלִים חָשְׁבוּ לְעַצְמָם שֶׁאֲחִיהֶם הַצָּעִיר, שֶׁכִּמְעַט לֹא יָצָא מֵהֶחָצֵר, בְּטַח לֹא יָבִיא כְּלוּם. לָכֵן הֵם אֲפִלּוּ לֹא טָרְחוּ לְחַפֵּשׂ הֵיטֵב - כָּל אֶחָד מֵהֶם לָקַח בְּחִפָּזוֹן שָׁטִיחַ פָּשׁוּט מֵאֵשֶׁת רוֹעֶה כִּמְעַט רִאשׁוֹנָה שֶׁפָּגַשׁ בַּדֶּרֶךְ.

כְּשֶׁחָזְרוּ שְׁלוֹשְׁתָּם אֶל הַמֶּלֶךְ, הִצִּיגוּ הַבֵּן הָרִאשׁוֹן וְהַבֵּן הַשֵּׁנִי אֶת הַשְּׁטִיחִים הַפְּשׁוּטִים שֶׁלָּהֶם. אָז הוֹצִיא הַבֵּן הַשְּׁלִישִׁי אֶת הַשָּׁטִיחַ הַמַּרְהִיב מֵהַקִּרְפָּדָה - וְכֻלָּם נֶאֶלְמוּ דּוֹם מֵרֹב יֹפִי. אֶחָיו כָּעֲסוּ וְטָעֲנוּ שֶׁלֹּא יִתָּכֵן שֶׁדַּוְקָא הַצָּעִיר יִירַשׁ אֶת הַכֶּתֶר, וּבִקְּשׁוּ מְשִׂימָה נוֹסֶפֶת. הַמֶּלֶךְ הִסְכִּים וְקָבַע: הַפַּעַם - מִי שֶׁיָּבִיא אֶת הַטַּבַּעַת הֲכִי יָפָה יְנַצֵּחַ.

שׁוּב נָפַח הַמֶּלֶךְ אֶת שָׁלֹשׁ הַנּוֹצוֹת, וְשׁוּב קָרָה בְּדִיּוּק אוֹתוֹ דָּבָר: נוֹצַת הַבֵּן הָרִאשׁוֹן עָפָה מִזְרָחָה, נוֹצַת הַבֵּן הַשֵּׁנִי עָפָה מַעֲרָבָה, וְנוֹצַת הַבֵּן הַשְּׁלִישִׁי צָנְחָה שׁוּב לְיַד אוֹתָהּ הַדֶּלֶת הַקְּטַנָּה. הוּא יָרַד שׁוּב אֶל הַקִּרְפָּדָה, וְזוֹ הֶעֱנִיקָה לוֹ הַפַּעַם טַבַּעַת נוֹצֶצֶת, שֶׁנִּצְנְצָה בְּאוֹר הַשֶּׁמֶשׁ יוֹתֵר מִכָּל תַּכְשִׁיט שֶׁצּוֹרֵף מְיֻמָּן יָכוֹל הָיָה לְעַצֵּב.

הָאַחִים הַגְּדוֹלִים, כְּמוֹ בַּפַּעַם הַקּוֹדֶמֶת, לֹא טָרְחוּ לְחַפֵּשׂ רָחוֹק - הֵם חִלְּצוּ טַבָּעוֹת יְשָׁנוֹת מִגַּלְגַּל עֲגָלָה שֶׁמָּצְאוּ בְּצַד הַדֶּרֶךְ. כְּשֶׁהִשְׁווּ אֶת שְׁלוֹשׁ הַטַּבָּעוֹת, נִצְּחָה שׁוּב טַבַּעְתּוֹ שֶׁל הַבֵּן הַשְּׁלִישִׁי בְּקַלּוּת. הָאַחִים, שֶׁעֲדַיִן לֹא הֶאֱמִינוּ שֶׁאֲחִיהֶם הַצָּעִיר רָאוּי לַכֶּתֶר, בִּקְּשׁוּ מְשִׂימָה שְׁלִישִׁית וְאַחֲרוֹנָה: מִי שֶׁיָּבִיא הַבַּיְתָה אֶת הַכַּלָּה הֲכִי יָפָה יִהְיֶה לְמֶלֶךְ.

בַּפַּעַם הַשְּׁלִישִׁית קָרָה שׁוּב בְּדִיּוּק אוֹתוֹ דָּבָר עִם הַנּוֹצוֹת, וְהַבֵּן הַשְּׁלִישִׁי יָרַד שׁוּב אֶל הַקִּרְפָּדָה. הַפַּעַם הִיא נָתְנָה לוֹ לֶפֶת חֲלוּלָה, רְתוּמָה לְשִׁשָּׁה עַכְבָּרִים קְטַנִּים. הוּא הִתְאַכְזֵב מְעַט - אֵיךְ תַּעֲזֹר לוֹ לֶפֶת לְהָבִיא כַּלָּה? - אַךְ הַקִּרְפָּדָה חִיְּכָה וְשָׂמָה אַחַת מִקִּרְפְּדוֹתֶיהָ הַקְּטַנּוֹת בְּתוֹךְ הַלֶּפֶת.

בְּרֶגַע שֶׁהֵחֵל הַבֵּן הַשְּׁלִישִׁי לִמְשֹׁךְ אֶת הַלֶּפֶת בַּמַּדְרֵגוֹת - קָרָה קֶסֶם: הַלֶּפֶת הָפְכָה לְעֶגְלַת זָהָב מַבְרִיקָה, שֵׁשֶׁת הָעַכְבָּרִים הָפְכוּ לְשִׁשָּׁה סוּסִים לְבָנִים, וְהַקִּרְפָּדָה הַקְּטַנָּה הָפְכָה לְנַעֲרָה יָפָה וְטוֹבַת לֵב, שֶׁחִיְּכָה אֵלָיו בְּחֹם.

כְּשֶׁהִגִּיעוּ שְׁלוֹשְׁתָּם אֶל אַרְמוֹן הַמֶּלֶךְ, הָאַחִים הַגְּדוֹלִים הֵבִיאוּ עִמָּם שְׁתֵּי נָעֲרוֹת פְּשׁוּטוֹת מִן הַכְּפָר הַסָּמוּךְ. אֲבָל כְּשֶׁיָּרְדָה הַנַּעֲרָה הַזּוֹהֶרֶת מֵהָעֶגְלָה הַמֻּזְהֶבֶת, הֵבִין הַמֶּלֶךְ מִיָּד מִי נִצַּח. הוּא חִבֵּק בְּשִׂמְחָה אֶת בְּנוֹ הַשְּׁלִישִׁי וְהִכְרִיז: "אַתָּה הֵבֵאתָ אֶת הַשָּׁטִיחַ הֲכִי יָפֶה, אֶת הַטַּבַּעַת הֲכִי יָפָה, וְעַכְשָׁו גַּם אֶת הַכַּלָּה הֲכִי יָפָה - אַתָּה תִּהְיֶה הַמֶּלֶךְ הַבָּא."

הָאַחִים הַגְּדוֹלִים הֵבִינוּ סוֹף סוֹף אֶת טָעוּתָם - הֵם מִהֲרוּ בְּלִי לְחַפֵּשׂ הֵיטֵב, בְּעוֹד אֲחִיהֶם הַצָּעִיר עָשָׂה כָּל מְשִׂימָה בְּסַבְלָנוּת וּבְלֵב פָּתוּחַ. הֵם חִבְּקוּ אוֹתוֹ וְאִחְלוּ לוֹ בְּלֵב שָׁלֵם. הַבֵּן הַשְּׁלִישִׁי מָלַךְ אַחֲרֵי אָבִיו שָׁנִים רַבּוֹת, בְּחָכְמָה וּבְטוּב לֵב, לְצַד הַכַּלָּה הַיָּפָה שֶׁמָּצָא בְּמַסָּעוֹ.

מְסֻפָּר עַל־פִּי "שְׁלֹשׁ הַנּוֹצוֹת" (Die drei Federn) מֵאֵת הָאַחִים גְרִים (גֶּרְמַנְיָה), מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ, עִם סוֹף מְרֻכָּךְ.

מָה זֶה מִסְפָּר סִידוּרִי?

מִסְפָּר מוֹנֶה (כְּמוֹ אֶחָד, שְׁתַּיִם, שָׁלוֹשׁ) עוֹנֶה עַל הַשְּׁאֵלָה "כַּמָּה?". לְעֻמָּתוֹ, מִסְפָּר סִידוּרִי (כְּמוֹ רִאשׁוֹן, שֵׁנִי, שְׁלִישִׁי) עוֹנֶה עַל הַשְּׁאֵלָה "בְּאֵיזֶה מָקוֹם בַּסֵּדֶר?" - הוּא מְצַיֵּן אֶת הַמָּקוֹם שֶׁל מַשֶּׁהוּ בְּרֶצֶף, לֹא אֶת הַכַּמּוּת שֶׁלּוֹ. בַּסִּפּוּר שֶׁלָּמַדְנוּ לֹא שָׁאַלְנוּ "כַּמָּה בָּנִים הָיוּ?" (שְׁלוֹשָׁה), אֶלָּא "מִי הֵבִיא אֶת הַשָּׁטִיחַ?" - וְהַתְּשׁוּבָה הָיְתָה "הַבֵּן הַשְּׁלִישִׁי".

🔑 כְּלָל זָהָב: מִסְפָּר סִידוּרִי מִתְנַהֵג כְּמוֹ שֵׁם תֹּאַר - הוּא בָּא אַחֲרֵי שֵׁם הָעֶצֶם וּמַתְאִים לוֹ בְּמִין (זָכָר/נְקֵבָה): "הַבֵּן הַשְּׁלִישִׁי" אֲבָל "הַמְּשִׂימָה הַשְּׁלִישִׁית". אִם שֵׁם הָעֶצֶם מְיֻדָּע (עִם ה' הַיְדִיעָה), גַּם הַמִּסְפָּר הַסִּידוּרִי מְקַבֵּל ה' הַיְדִיעָה.

עֻבְדָּה מַפְתִּיעָה: "רִאשׁוֹן" לֹא בָּא מִ-"אֶחָד"!

שִׂימוּ לֵב: רֹב הַמִּסְפָּרִים הַסִּידוּרִיִּים נִבְנִים מִתּוֹךְ הַמִּסְפָּר הַמּוֹנֶה הַמַּתְאִים - שָׁלוֹשׁ→שְׁלִישִׁי, אַרְבַּע→רְבִיעִי, עֶשֶׂר→עֲשִׂירִי. אֲבָל "רִאשׁוֹן" הוּא יוֹצֵא דֹּפֶן: הוּא לֹא קָשׁוּר לַמִּלָּה "אֶחָד", אֶלָּא בָּא מֵהַשֹּׁרֶשׁ רא"ש (כְּמוֹ בַּמִּלָּה "רֹאשׁ") - כִּי מִי שֶׁרִאשׁוֹן, נִמְצָא "בָּרֹאשׁ" הַשּׁוּרָה.

💡 טִיפ: לִפְנֵי שֶׁמַּחְלִיטִים אֵיזוֹ צוּרָה לִבְחֹר - זָכָר אוֹ נְקֵבָה - שַׁאֲלוּ קֹדֶם מַה מִּינוֹ שֶׁל שֵׁם הָעֶצֶם: "בֵּן" הוּא זָכָר (הַשֵּׁנִי), אֲבָל "נוֹצָה" הִיא נְקֵבָה (הַשְּׁנִיָּה).
הַבֵּן הָרִאשׁוֹן הַנּוֹצָה הַשְּׁנִיָּה הַמְּשִׂימָה הַשְּׁלִישִׁית הַפַּעַם הָרְבִיעִית
