הַדַּיָּג וְאִשְׁתּוֹ

📖 קְרִיאָה

פַּעַם הָיוּ דַּיָּג זָקֵן וְאִשְׁתּוֹ שֶׁגָּרוּ בְּבִקְתָּה יְשָׁנָה וַעֲלוּבָה, מַמָּשׁ עַל שְׂפַת הַיָּם. כָּל בֹּקֶר יָצָא הַדַּיָּג עִם חַכָּה וּדְלִי אֶל הַמַּיִם, וְדָג דָּגִים קְטַנִּים כְּדֵי לְפַרְנֵס אֶת שְׁנֵיהֶם.

יוֹם אֶחָד יָשַׁב הַדַּיָּג עַל הַסֶּלַע וְהִבְחִין שֶׁהַחַכָּה שֶׁלּוֹ מִתְכּוֹפֶפֶת בְּחָזְקָה. הוּא מָשַׁךְ בְּכָל כֹּחוֹ, וְעַל הַחוֹף נִשְׁלַף דָּג גָּדוֹל וּמוּזָר, שֶׁפָּתַח פֶּה וְדִבֵּר אֵלָיו בְּקוֹל אָדָם: "הַנַּח לִי לָשׁוּב אֶל הַיָּם - אֲנִי לֹא דָּג רָגִיל, אֶלָּא נָסִיךְ שֶׁקָּסְמוּ עָלָיו." הַדַּיָּג נִבְהַל וְהֵשִׁיב אֶת הַדָּג הַמְדַבֵּר לַמַּיִם, בְּלִי לְבַקֵּשׁ שׁוּם דָּבָר בַּתְּמוּרָה.

כְּשֶׁחָזַר הַדַּיָּג הַבַּיְתָה וְסִפֵּר לְאִשְׁתּוֹ מָה קָרָה, הִיא כָּעֲסָה מְאֹד. "דָּג קֶסֶם עָמַד מוּלְךָ וְלֹא בִּקַּשְׁתָּ מִמֶּנּוּ כְּלוּם?! לֵךְ מִיָּד וּתְבַקֵּשׁ מִמֶּנּוּ שֶׁיִּתֵּן לָנוּ בַּיִת נָאֶה בִּמְקוֹם הַבִּקְתָּה הַזֹּאת." הַדַּיָּג לֹא רָצָה לְהַטְרִיד אֶת הַדָּג שׁוּב, אֲבָל בְּסוֹפוֹ שֶׁל דָּבָר נִכְנַע וְיָצָא אֶל הַיָּם.

הוּא קָרָא לַדָּג, וְזֶה מִיָּד הוֹפִיעַ. "אִשְׁתִּי מְבַקֶּשֶׁת שֶׁתִּתֵּן לָהּ בַּיִת יָפֶה," אָמַר הַדַּיָּג בְּבֹשֶׁת פָּנִים. "לֵךְ הַבַּיְתָה - כְּבָר נָתַתִּי לָהּ אוֹתוֹ," עָנָה הַדָּג. וְאָכֵן, בִּמְקוֹם הַבִּקְתָּה הָעֲלוּבָה עָמַד עַכְשָׁו בַּיִת קָטָן וְנָקִי, עִם גִּנָּה קְטַנָּה מִסָּבִיב.

בַּהַתְחָלָה שָׂמְחָה הָאִשָּׁה מְאֹד, אֲבָל כַּעֲבֹר שָׁבוּעַ הִיא כְּבָר לֹא הִסְתַּפְּקָה בַּבַּיִת הַקָּטָן. "לֵךְ אֶל הַדָּג," אָמְרָה לְבַעֲלָהּ, "וּתְבַקֵּשׁ שֶׁיִּבְנֶה לָנוּ אַרְמוֹן גָּדוֹל." הַדַּיָּג נֶאֱנַח, אַךְ יָצָא שׁוּב אֶל הַיָּם - הַפַּעַם הַשָּׁמַיִם הָיוּ אֲפֹרִים, וְהַגַּלִּים כְּבָר לֹא הָיוּ שְׁקֵטִים כְּמוֹ קֹדֶם.

הַדָּג מִלֵּא גַּם אֶת הַבַּקָּשָׁה הַזֹּאת, וְהָאִשָּׁה זָכְתָה בְּאַרְמוֹן עִם מְשָׁרְתִים רַבִּים. אַךְ עוֹד לִפְנֵי סוֹף הַחֹדֶשׁ הִיא כְּבָר רָצְתָה יוֹתֵר: "אֲנִי רוֹצָה לִהְיוֹת מַלְכָּה!" קָרְאָה. הַדַּיָּג הִתְנַגֵּד בְּכָל כֹּחוֹ, אֲבָל בַּסּוֹף שׁוּב נִכְנַע וְהָלַךְ אֶל הַיָּם הַסּוֹעֵר לְבַקֵּשׁ עֲבוּרָהּ.

גַּם הַבַּקָּשָׁה הַזֹּאת הִתְמַלְּאָה, אֲבָל הָאִשָּׁה לֹא נָחָה. אַחֲרֵי כַּמָּה יָמִים שָׁלְחָה אוֹתוֹ שׁוּב, הַפַּעַם כְּדֵי לְבַקֵּשׁ שֶׁתִּהְיֶה קֵיסָרִית. הַיָּם הָיָה עַכְשָׁו שָׁחֹר וְסוֹעֵר, וְהַגַּלִּים הִכּוּ בַּסְּלָעִים בְּזַעַם. הַדָּג נֵעָנָה גַּם הַפַּעַם, אַךְ בְּקוֹל עָצוּב יוֹתֵר.

לְבַסּוֹף, כְּשֶׁגַּם זֶה לֹא הִסְפִּיק לָהּ, בִּקְּשָׁה הָאִשָּׁה שֶׁהַדָּג יִתֵּן לָהּ שְׁלִיטָה עַל הַשֶּׁמֶשׁ וְעַל הַיָּרֵחַ עַצְמָם. הַדַּיָּג יָדַע שֶׁזֹּאת בַּקָּשָׁה שֶׁאִי אֶפְשָׁר לְבַקֵּשׁ, אֲבָל אִשְׁתּוֹ לֹא הִסְכִּימָה לְוַתֵּר, וְהוּא יָצָא בְּפַעַם הָאַחֲרוֹנָה אֶל יָם סוֹעֵר וְשָׁחֹר.

הַפַּעַם לֹא עָנָה הַדָּג בְּמִלִּים. סְעָרָה עֲנָקִית פָּרְצָה, וּכְשֶׁחָזַר הַדַּיָּג הַבַּיְתָה - הָאַרְמוֹן נֶעֱלַם, וְהַכֹּל שָׁב לִהְיוֹת כְּמוֹ שֶׁהָיָה בַּהַתְחָלָה: הַבִּקְתָּה הַיְּשָׁנָה וְהָעֲלוּבָה, וְאִשְׁתּוֹ יָשְׁבָה עַל הַסַּף, בְּלִי כְּלוּם מִכָּל מָה שֶׁקִּבְּלָה.

הַסִּפּוּר הַזֶּה מְלַמֵּד אוֹתָנוּ שֶׁמִּי שֶׁתָּמִיד רוֹצֶה עוֹד וָעוֹד, בְּלִי לְהַכִּיר תּוֹדָה עַל מָה שֶׁיֵּשׁ לוֹ - עָלוּל לְאַבֵּד גַּם אֶת מָה שֶׁכְּבָר קִבֵּל.

מְסֻפָּר עַל־פִּי מַעֲשִׂיַּת "הַדַּיָּג וְאִשְׁתּוֹ" (Der Fischer und seine Frau) מֵאֵת הָאַחִים גְרִים (גֶּרְמַנְיָה, 1812), מְעֻבָּד.

📚 דִּקְדּוּק — כִּנּוּיֵי גוּף חֲבוּרִים עִם "לְ"

מָה זֶה כִּנּוּי חָבוּר?

מִלַּת הַיַּחַס "לְ" (כְּמוֹ בְּ"נָתַן לְדָנִי") יְכוֹלָה לְהִתְחַבֵּר יָשָׁר לְכִנּוּיֵי גּוּף, בְּלִי מִלָּה נִפְרֶדֶת - וְאָז נוֹצֶרֶת מִלָּה אַחַת חֲדָשָׁה: "לִי", "לְךָ", "לוֹ" וְכֵן הָלְאָה. בַּסִּפּוּר, כְּשֶׁהַדָּג אוֹמֵר "הַנַּח לִי לָשׁוּב", הַכִּנּוּי "לִי" פֵּרוּשׁוֹ "לְ" + "אֲנִי".

לְמָה זֶה חָשׁוּב?

כִּנּוּיִים חֲבוּרִים חוֹסְכִים מִלִּים וְעוֹזְרִים לַמִּשְׁפָּט לִזְרֹם בְּטִבְעִיּוּת. שִׂימוּ לֵב שֶׁלִּפְעָמִים אוֹתוֹ כִּנּוּי חָבוּר נִשְׁמָע דּוֹמֶה לְאַחֵר, אֲבָל הַכְּתִיב שׁוֹנֶה - "לוֹ" (הוּא) לְעֻמַּת "לָהּ" (הִיא), אוֹ "לָךְ" (אַתְּ) לְעֻמַּת "לְךָ" (אַתָּה).

עָנָה לוֹ (= לְ + הוּא) נָתַן לָהּ (= לְ + הִיא) תֵּן לָנוּ (= לְ + אֲנַחְנוּ)
🔑 כְּלָל זָהָב: כְּדֵי לְזַהוֹת כִּנּוּי חָבוּר עִם "לְ" - שׁוֹאֲלִים "לְמִי?" וּמְחַפְּשִׂים לְאֵיזֶה גּוּף (אֲנִי, אַתָּה, הוּא...) הַסִּיּוֹמֶת מִתְאִימָה בַּטַּבְלָה.
