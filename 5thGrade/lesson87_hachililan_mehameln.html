📖 קְרִיאָה

בְּעִיר קְטַנָּה בְּשֵׁם הָמֶלִין הִתְרַבּוּ פִּתְאֹם עַכְבָּרִים. הֵם אָכְלוּ אֶת הַלֶּחֶם בַּמַּאֲפִיּוֹת, אֶת הַגְּבִינָה בַּחֲנוּיוֹת וְאֶת הָאֹכֶל מִצַּלְחוֹת הַיְלָדִים. "מָה נַעֲשֶׂה?" שָׁאֲלוּ הַתּוֹשָׁבִים אֶת רֹאשׁ הָעִיר. "אֲנִי לֹא יוֹדֵעַ!" צָעַק רֹאשׁ הָעִיר. "הִפָּטְרוּ מֵהָעַכְבָּרִים!"

בְּיוֹם אֶחָד הִגִּיעַ לָעִיר אִישׁ זָר, לָבוּשׁ בְּבִגְדֵי צְבָעִים, וּבְיָדוֹ חָלִיל. "אֲנִי חַלִּילָן," אָמַר לְרֹאשׁ הָעִיר, "וַאֲנִי יָכוֹל לְסַלֵּק אֶת כָּל הָעַכְבָּרִים. מָה תְּשַׁלֵּם לִי?" "אֶלֶף מַטְבְּעוֹת זָהָב!" הִבְטִיחַ רֹאשׁ הָעִיר. "רַק סַלֵּק אוֹתָם!"

הַחַלִּילָן יָצָא לָרְחוֹב וְהֵחֵל לְנַגֵּן מַנְגִּינָה מְתוּקָה. מִכָּל פִּנָּה יָצְאוּ עַכְבָּרִים וְהָלְכוּ אַחֲרָיו, אֶחָד אַחַר הַשֵּׁנִי. הוּא הוֹבִיל אוֹתָם הָרְחֵק מֵהָעִיר, מֵעֵבֶר לַנָּהָר, וְהֵם לֹא חָזְרוּ יוֹתֵר. "אֵין עוֹד עַכְבָּרִים בָּעִיר!" קָרְאוּ הַתּוֹשָׁבִים בְּשִׂמְחָה.

אַחַר כָּךְ בָּא הַחַלִּילָן אֶל רֹאשׁ הָעִיר וְאָמַר: "עַכְשָׁו תְּשַׁלֵּם לִי!" אֲבָל רֹאשׁ הָעִיר חִיֵּךְ בְּעָרְמָה. "אֶלֶף מַטְבְּעוֹת בִּשְׁבִיל כַּמָּה שִׁירִים עַל חָלִיל? אַתָּה צוֹחֵק!" "אַתָּה הִבְטַחְתָּ," אָמַר הַחַלִּילָן בְּשֶׁקֶט. "הַבְטָחָה צָרִיךְ לְקַיֵּם!" רֹאשׁ הָעִיר לֹא הִקְשִׁיב, וְהַחַלִּילָן יָצָא מִן הָעִיר בְּלִי מַטְבְּעוֹת.

בַּבֹּקֶר שֶׁאַחֲרֵי כֵן, כְּשֶׁהַמְּבֻגָּרִים הָיוּ בָּעֲבוֹדָה, יָצָא הַחַלִּילָן שׁוּב לָרְחוֹב וְנִגֵּן מַנְגִּינָה אַחֶרֶת, עַלִּיזָה וּמְרַקֶּדֶת. הַיְלָדִים שָׁמְעוּ וְרָצוּ אַחֲרָיו, צוֹחֲקִים וְרוֹקְדִים. הוּא הוֹבִיל אוֹתָם אֶל עֵמֶק יָרֹק מֵעֵבֶר לַגִּבְעָה, וְשָׁם שִׂחֲקוּ עַד הָעֶרֶב.

הַהוֹרִים חָזְרוּ הַבַּיְתָה וְלֹא מָצְאוּ אֶת הַיְלָדִים. "הֵיכָן הַיְלָדִים?" קָרְאוּ בְּפַחַד. "מִי רָאָה אוֹתָם?" כֻּלָּם חִפְּשׂוּ בְּכָל מָקוֹם, וּפִתְאֹם הֵבִינוּ מִי לָקַח אוֹתָם. רֹאשׁ הָעִיר רָץ אֶל הַגִּבְעָה וְצָעַק: "חַלִּילָן, חֲזֹר! אֲנִי אֲשַׁלֵּם לְךָ אֶת כָּל מַה שֶׁהִבְטַחְתִּי!"

הַחַלִּילָן חָזַר עִם הַיְלָדִים, וְרֹאשׁ הָעִיר שִׁלֵּם לוֹ אֶת כָּל הַמַּטְבְּעוֹת. מִן הַיּוֹם הַהוּא אָמְרוּ בְּהָמֶלִין: "אָסוּר לְהַבְטִיחַ וְלֹא לְקַיֵּם!" וְהָעִיר לָמְדָה שֶׁמִּי שֶׁנּוֹתֵן אֶת מִלָּתוֹ, חַיָּב לִשְׁמֹר עָלֶיהָ.

מְסֻפָּר עַל־פִּי אַגָּדַת "הֶחָלִילָן מֵהָמֶלִין" (Der Rattenfänger von Hameln), אַגָּדָה גֶּרְמָנִית עַתִּיקָה שֶׁנֶּאֶסְפָה בֵּין הַשְּׁאָר בְּאֹסֶף הָאַגָּדוֹת שֶׁל הָאַחִים גְּרִים (1816), מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ עִם סוֹף מְרֻכָּךְ.

📚 דִּקְדּוּק — סוּגֵי מִשְׁפָּטִים (חֲזָרָה)

אַרְבָּעָה סוּגֵי מִשְׁפָּטִים — מָה הַמִּשְׁפָּט עוֹשֶׂה?

בְּשִׁיעוּר 36 לָמַדְנוּ שֶׁלְּכָל מִשְׁפָּט יֵשׁ תַּפְקִיד. הַיּוֹם נַחֲזֹר עַל אַרְבַּעַת הַסּוּגִים, וְנִרְאֶה כֵּיצַד הֵם מוֹפִיעִים בְּסִפּוּר עַל הַחַלִּילָן. שִׂימוּ לֵב: סִימַן הַפִּסּוּק עוֹזֵר, אֲבָל מָה שֶׁקּוֹבֵעַ הוּא מָה הַמִּשְׁפָּט עוֹשֶׂה.

הַסּוּגמָה הוּא עוֹשֶׂה?סִימָן בַּסּוֹףדֻּגְמָה מִן הַסִּפּוּר
מִשְׁפָּט חִוּוּימְסַפֵּר אוֹ מוֹסֵר עֻבְדָּהנְקֻדָּה .הָעַכְבָּרִים אָכְלוּ אֶת הַלֶּחֶם.
מִשְׁפָּט שְׁאֵלָהשׁוֹאֵל וּמְחַכֶּה לִתְשׁוּבָהסִימַן שְׁאֵלָה ?הֵיכָן הַיְלָדִים?
מִשְׁפָּט קְרִיאָהמַבִּיעַ רֶגֶשׁ: שִׂמְחָה, כַּעַס אוֹ פַּחַדסִימַן קְרִיאָה !אֵין עוֹד עַכְבָּרִים בָּעִיר!
מִשְׁפָּט צִוּוּיאוֹמֵר לְמִישֶׁהוּ מָה לַעֲשׂוֹתסִימַן קְרִיאָה ! אוֹ נְקֻדָּה .חַלִּילָן, חֲזֹר!
🔑 אֵיךְ מַבְחִינִים? (1) מִשְׁפָּט שֶׁבּוֹ שׁוֹאֲלִים וְיֵשׁ לוֹ תְּשׁוּבָה הוּא שְׁאֵלָה. (2) מִשְׁפָּט שֶׁאוֹמֵר לְמִישֶׁהוּ לַעֲשׂוֹת מַשֶּׁהוּ הוּא צִוּוּי, וְהַפֹּעַל בּוֹ בִּלְשׁוֹן צִוּוּי: סַלֵּק, חֲזֹר, הִפָּטְרוּ. (3) מִשְׁפָּט שֶׁמְּבַטֵּא רֶגֶשׁ חָזָק, בְּלִי לְבַקֵּשׁ דָּבָר, הוּא קְרִיאָה. (4) כֹּל שְׁאָר הַמִּשְׁפָּטִים, שֶׁמְּסַפְּרִים בָּהֶם עֻבְדָּה, הֵם חִוּוּי.
💡 שִׂימוּ לֵב: סִימַן קְרִיאָה מוֹפִיעַ גַּם בִּקְרִיאָה וְגַם בְּצִוּוּי. לָכֵן "חֲזֹר!" הוּא צִוּוּי, כִּי הוּא אוֹמֵר לַחַלִּילָן מָה לַעֲשׂוֹת, וְ"אֵין עוֹד עַכְבָּרִים!" הוּא קְרִיאָה, כִּי הוּא מַבִּיעַ שִׂמְחָה. וְיֵשׁ גַּם שְׁאֵלוֹת שֶׁאֵין מְחַכִּים לָהֶן לִתְשׁוּבָה, כְּמוֹ "מִי לֹא רוֹצֶה גְּלִידָה?" - זוֹ שְׁאֵלָה, אֲבָל הַכַּוָּנָה הִיא לוֹמַר שֶׁכֻּלָּם רוֹצִים.
מָה נַעֲשֶׂה? אֲנִי לֹא יוֹדֵעַ! סַלֵּק אוֹתָם! הַחַלִּילָן חָזַר.
