📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת חַי בְּיַעַר צָפוּף בְּצָרְפַת חוֹטֵב עֵצִים עָנִי, שֶׁעָבַד קָשֶׁה מִבֹּקֶר עַד עֶרֶב וְאַף פַּעַם לֹא הִתְלוֹנֵן עַל גּוֹרָלוֹ. יוֹם אֶחָד, כְּשֶׁחָתַךְ עֵצִים בְּתוֹךְ הַיַּעַר, נֶאֱנַח בְּקוֹל: "מִי יִתֵּן וְהָיְתָה לִי, לְפָחוֹת פַּעַם אַחַת, קְצָת יוֹתֵר מִזֶּה."

לְפֶתַע הוֹפִיעָה לְפָנָיו רוּחַ הַיַּעַר - יְצוּר קָסוּם שֶׁל אוֹר וְעָלִים יְרֻקִּים. "שָׁמַעְתִּי אֶת אַנְחָתְךָ," אָמְרָה רוּחַ הַיַּעַר, "וּבִזְכוּת הַסַּבְלָנוּת שֶׁלְּךָ כָּל הַשָּׁנִים הָאֵלֶּה, אֲנִי מַעֲנִיקָה לְךָ וּלְאִשְׁתְּךָ שָׁלוֹשׁ מִשְׁאָלוֹת. בַּקְּשׁוּ כָּל מַה שֶּׁתִּרְצוּ - אֲבָל בִּזְהִירוּת, כִּי אַחֲרֵי שָׁלוֹשׁ מִשְׁאָלוֹת לֹא יִהְיֶה עוֹד קֶסֶם."

חוֹטֵב הָעֵצִים רָץ הַבַּיְתָה בְּשִׂמְחָה עֲצוּמָה וְסִפֵּר לְאִשְׁתּוֹ עַל הַמַּתָּנָה הַמֻּפְלָאָה. "אִם נְבַקֵּשׁ בִּזְהִירוּת," אָמַר לָהּ בְּהִתְרַגְּשׁוּת, "נִהְיֶה עֲשִׁירִים מְאֹד עַד סוֹף הַשָּׁבוּעַ!" שְׁנֵיהֶם הֶחְלִיטוּ לִישֹׁן עַל זֶה, וּלְהַחְלִיט בְּיַחַד בַּבֹּקֶר מָה בֶּאֱמֶת הֲכִי חָשׁוּב לָהֶם.

בָּעֶרֶב יָשְׁבוּ שְׁנֵיהֶם לְיַד הָאֵשׁ הַקְּטַנָּה בַּבַּיִת, מְחַכִּים לַאֲרוּחַת עֶרֶב דַּלָּה. רֵיחַ נִפְלָא שֶׁל נַקְנִיקִיָּה צְלוּיָה עָלָה מִבֵּית הַשְּׁכֵנִים וְנִכְנַס דֶּרֶךְ הַחַלּוֹן הַפָּתוּחַ.

חוֹטֵב הָעֵצִים הָיָה כׇּל כָּךְ רָעֵב, שֶׁבְּלִי לַחְשֹׁב אֲפִלּוּ רֶגַע קָרָא בְּקוֹל: "אֶה, הַלְוַאי שֶׁהָיְתָה לָנוּ נַקְנִיקִיָּה אֲרֻכָּה כָּזֹאת עַל הָאֵשׁ שֶׁלָּנוּ!" בְּאוֹתוֹ רֶגַע מַמָּשׁ, נַקְנִיקִיָּה עֲנָקִית וּמְעַשֶּׁנֶת צָנְחָה מִן הָאֲרֻבָּה יָשָׁר לְתוֹךְ הַסִּיר.

אִשְׁתּוֹ נִבְהֲלָה וְכָעֲסָה נוֹרָא. "אֵיךְ יָכֹלְתָּ לְבַזְבֵּז מִשְׁאָלָה שְׁלֵמָה עַל נַקְנִיקִיָּה?! הָיִינוּ יְכוֹלִים לְבַקֵּשׁ אַרְמוֹן, אוֹ שַׂק זָהָב!" הִיא הִמְשִׁיכָה לִצְעֹק עָלָיו בִּפְּזִיזוּת, עַד שֶׁקָּרְאָה בְּלִי לְהִתְאַפֵּק: "הַלְוַאי שֶׁהַנַּקְנִיקִיָּה הַזֹּאת תִּדָּבֵק לְךָ יָשָׁר עַל הָאַף!" וּמִיָּד - כֵּן, בְּדִיּוּק כָּכָה קָרָה.

עַכְשָׁו נוֹתְרָה לָהֶם רַק מִשְׁאָלָה אַחַת, וְנַקְנִיקִיָּה תְּלוּיָה עַל אַף הַבַּעַל. הֵם יָשְׁבוּ וְחָשְׁבוּ יַחַד: אִם יְבַקְּשׁוּ עֹשֶׁר רַב בַּמִּשְׁאָלָה הָאַחֲרוֹנָה, יִהְיוּ עֲשִׁירִים - אֲבָל עִם נַקְנִיקִיָּה עַל הָאַף לְעוֹלָם. "אִלּוּ חָשַׁבְנוּ רֶגַע לִפְנֵי שֶׁדִּבַּרְנוּ," אָמְרָה הָאִשָּׁה בְּצַעַר, "הָיִינוּ עֲשִׁירִים עַכְשָׁו בְּלִי שׁוּם צָרָה עַל הַפָּנִים."

"לוּלֵא הָיִינוּ כׇּל כָּךְ רְעֵבִים וְכוֹעֲסִים," הוֹסִיף הַבַּעַל בְּחָכְמָה חֲדָשָׁה, "לֹא הָיִינוּ מְבַזְבְּזִים שְׁתֵּי מִשְׁאָלוֹת שְׁלֵמוֹת. בֹּא נִשְׁתַּמֵּשׁ בַּמִּשְׁאָלָה הָאַחֲרוֹנָה בְּחָכְמָה אֲמִתִּית."

בְּסוֹפוֹ שֶׁל דָּבָר בִּקְּשׁוּ בַּמִּשְׁאָלָה הַשְּׁלִישִׁית וְהָאַחֲרוֹנָה רַק דָּבָר אֶחָד: שֶׁהַנַּקְנִיקִיָּה תֵּעָלֵם מִן הָאַף. הֵם נִשְׁאֲרוּ עֲנִיִּים כְּשֶׁהָיוּ - אֲבָל מְאֻשָּׁרִים, וְעִם לֶקַח חָשׁוּב: לְבַקֵּשׁ בִּזְהִירוּת, וְלַחְשֹׁב לִפְנֵי שֶׁמְּדַבְּרִים.

מְסֻפָּר עַל־פִּי "הַמִּשְׁאָלוֹת הַמְּגֻחָכוֹת" (Les Souhaits Ridicules) מֵאֵת שַׁארְל פֶּרוֹ (צָרְפַת, 1697), מֻכָּר גַּם בְּשֵׁם "שָׁלוֹשׁ הַמִּשְׁאָלוֹת", מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ.

📚 דִּקְדּוּק — מִשְׁפַּט תְּנַאי (חֲזָרָה)

מִשְׁפַּט תְּנַאי — אֲמִתִּי לְעֻמַּת לֹא-אֲמִתִּי, וְהַמִּלָּה "לוּלֵא"

בְּשִׁיעוּר 39 לָמַדְנוּ שֶׁיֵּשׁ שְׁנֵי סוּגֵי מִשְׁפַּט תְּנַאי: תְּנַאי אֲמִתִּי עִם הַמִּלָּה "אִם" וְעָתִיד (מַשֶּׁהוּ שֶׁעֲדַיִן אֶפְשָׁר לְהַגְשִׁים), וּתְנַאי לֹא-אֲמִתִּי עִם "אִלּוּ" אוֹ "לוּ" וְעָבָר (מַשֶּׁהוּ שֶׁכְּבָר לֹא אֶפְשָׁרִי, וּמְבִיעִים בּוֹ חֲרָטָה). הַיּוֹם נַכִּיר גַּם מִלָּה שְׁלִישִׁית: "לוּלֵא" - מִלָּה אַחַת שֶׁפֵּרוּשָׁהּ "אִלּוּ לֹא", וּמְשַׁמֶּשֶׁת לִתְנַאי לֹא-אֲמִתִּי בְּשָׁלִילָה.

🔑 כְּלָל: "אִם" בָּא עִם עָתִיד, כִּי הַתְּנַאי עֲדַיִן פָּתוּחַ: "אִם נְבַקֵּשׁ בִּזְהִירוּת, נִהְיֶה עֲשִׁירִים." לְעֻמַּת זֹאת, "אִלּוּ", "לוּ" וְ"לוּלֵא" בָּאִים עִם עָבָר, כִּי הָרֶגַע לְהַחְלִיט אַחֶרֶת כְּבָר חָלַף: "אִלּוּ חָשַׁבְנוּ קֹדֶם, הָיִינוּ עֲשִׁירִים עַכְשָׁו." "לוּלֵא" נוֹתֵן אֶת אוֹתָהּ הַצּוּרָה בְּדִיּוּק, רַק בְּשָׁלִילָה: "לוּלֵא הָיִינוּ רְעֵבִים, לֹא הָיִינוּ מְבַזְבְּזִים אֶת הַמִּשְׁאָלָה."
💡 שִׂימוּ לֵב: מִשְׁפַּט תְּנַאי שׁוֹנֶה מִמִּשְׁפַּט תּוֹצָאָה (שִׁיעוּר 83, "כׇּל כָּךְ... עַד שֶׁ..."). מִשְׁפַּט תְּנַאי מְתָאֵר אֶפְשָׁרוּת - אֲמִתִּית אוֹ לֹא-אֲמִתִּית - וּמִשְׁפַּט תּוֹצָאָה מְתָאֵר מָה שֶׁבֶּאֱמֶת קָרָה בִּגְלַל עֹצֶם גָּבוֹהַּ.

כִּתְבוּ שְׁלוֹשָׁה מִשְׁפְּטֵי תְּנַאי: אֶחָד עִם "אִם" (עַל מַשֶּׁהוּ שֶׁעֲדַיִן אֶפְשָׁר לְהַגְשִׁים), אֶחָד עִם "אִלּוּ" אוֹ "לוּ" (עַל מַשֶּׁהוּ שֶׁכְּבָר לֹא אֶפְשָׁרִי), וְאֶחָד עִם "לוּלֵא". לְדֻגְמָה: "אִם" - "אִם אֶתְאַמֵּן הַיּוֹם, אֶהְיֶה מוּכָן לַמִּשְׂחָק מָחָר"; "אִלּוּ" - "אִלּוּ יָדַעְתִּי עַל הַמִּבְחָן, הָיִיתִי מִתְכּוֹנֵן יוֹתֵר טוֹב"; "לוּלֵא" - "לוּלֵא יָרַד גֶּשֶׁם אֶתְמוֹל, הָיִינוּ הוֹלְכִים לְטַיֵּל בַּפָּרְק."
