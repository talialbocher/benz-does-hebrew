פֶּרְסֵאוּס וּמֶדוּסָה

אֵיךְ מָגֵן מַבְרִיק, סַנְדָּלִים מְכֻנָּפִים וְשַׂק קֶסֶם עָזְרוּ לְנַעַר עָנִי לְנַצֵּחַ אֶת הַגּוֹרְגּוֹנָה הַמְּפַחִידָה בְּיוֹתֵר בְּיָוָן הָעַתִּיקָה

📖 קְרִיאָה

בְּמַמְלֶכֶת אַרְגּוֹס שֶׁבְּיָוָן הָעַתִּיקָה מָלַךְ פַּעַם מֶלֶךְ בְּשֵׁם אַקְרִיסְיוֹס, שֶׁלֹּא הָיָה לוֹ בֵּן - רַק בַּת יְחִידָה בְּשֵׁם דָּנָאָה. יוֹם אֶחָד רָצָה הַמֶּלֶךְ לָדַעַת מָה צָפוּי לוֹ בֶּעָתִיד, וְקִבֵּל נְבוּאָה מַפְחִידָה: "בְּנָהּ שֶׁל בִּתְּךָ דָּנָאָה יַהֲרֹג אוֹתְךָ יוֹם אֶחָד!" הַמֶּלֶךְ נִבְהָל, וּמֵאָז לֹא הִרְפְּתָה מִמֶּנּוּ הַנְּבוּאָה הַמַּפְחִידָה הַזֹּאת. הוּא כָּלָא אֶת דָּנָאָה בְּמִגְדָּל נְחֹשֶׁת גָּבֹהַּ, בְּלִי דֶּלֶת וּבְלִי חַלּוֹן, כְּדֵי שֶׁלְּעוֹלָם לֹא יִוָּלֵד לָהּ בֵּן שֶׁיּוּכַל לְאַיֵּם עָלָיו.

אֲבָל זֵאוּס, אֲבִי הָאֵלִים, רָאָה אֶת דָּנָאָה מִן הַשָּׁמַיִם וְהִתְאַהֵב בָּהּ, וְהִצְלִיחַ לְהִכָּנֵס אֵלֶיהָ בְּצוּרַת גֶּשֶׁם שֶׁל זָהָב - וְכָךְ נוֹלַד לָהּ בֵּן קָטָן, פֶּרְסֵאוּס. כְּשֶׁגִּלָּה אַקְרִיסְיוֹס אֶת הַתִּינוֹק, נִבְהַל מְאֹד וְחָשַׁשׁ שֶׁקִּלְלַת הָאֵלִים תָּחוּל עָלָיו אִם יִפְגַּע בְּבִתּוֹ וּבְנֶכְדוֹ בְּיָדָיו מַמָּשׁ - וּבִמְקוֹם זֹאת נָעַל אוֹתָם בְּתוֹךְ תֵּבַת עֵץ וְהִשְׁלִיךְ אוֹתָהּ אֶל הַיָּם. הַגַּלִּים סָחֲפוּ אֶת הַתֵּבָה יָמִים רַבִּים, עַד שֶׁהִגִּיעָה אֶל חוֹפוֹ שֶׁל אִי קָטָן בְּשֵׁם סֶרִיפוֹס.

דַּיָּג טוֹב לֵב בְּשֵׁם דִּיקְטִיס מָצָא אֶת הַתֵּבָה עַל הַחוֹף, וְגִדֵּל אֶת פֶּרְסֵאוּס יַחַד עִם דָּנָאָה, כְּאִלּוּ הָיוּ מִשְׁפַּחְתּוֹ שֶׁלּוֹ מַמָּשׁ; דָּנָאָה גָּרָה עִמּוֹ בַּבַּיִת הַקָּטָן שֶׁלּוֹ שָׁנִים רַבּוֹת, וְהַנַּעַר גָּדַל וְהָיָה לְבָחוּר חָזָק וְאַמִּיץ. אַךְ אָחִיו שֶׁל דִּיקְטִיס, הַמֶּלֶךְ פּוֹלִידֶקְטֶס, רָצָה לָשֵׂאת אֶת דָּנָאָה לְאִשָּׁה בְּנִגּוּד לִרְצוֹנָהּ, וּפֶרְסֵאוּס הַבּוֹגֵר תָּמִיד עָמַד בֵּינֵיהֶם וְהֵגֵן עָלֶיהָ.

פּוֹלִידֶקְטֶס תִּכְנֵן תַּחְבּוּלָה כְּדֵי לְהִפָּטֵר מִפֶּרְסֵאוּס. הוּא הִכְרִיז שֶׁהוּא עוֹמֵד לָשֵׂאת אִשָּׁה אַחֶרֶת, וּבִקֵּשׁ מִכָּל אֲנָשָׁיו מַתָּנוֹת יְקָרוֹת לַחֲתֻנָּה. פֶּרְסֵאוּס הַגֵּאֶה הִבְטִיחַ לוֹ בְּקוֹל רָם: "אָבִיא לְךָ כָּל מַתָּנָה שֶׁתִּרְצֶה - אֲפִלּוּ אֶת רֹאשָׁהּ שֶׁל הַגּוֹרְגּוֹנָה מֶדוּסָה!" זֶה בְּדִיּוּק מַה שֶּׁפּוֹלִידֶקְטֶס חִכָּה לִשְׁמֹעַ, וּמִיָּד תָּפַס אוֹתוֹ בְּמִלָּתוֹ.

מֶדוּסָה הָיְתָה מִפְלֶצֶת מַפְחִידָה בְּיוֹתֵר - אִשָּׁה עִם נְחָשִׁים חַיִּים בִּמְקוֹם שְׂעָרוֹת, שֶׁכָּל מִי שֶׁמַּבִּיט בְּעֵינֶיהָ הוֹפֵךְ מִיָּד לְאֶבֶן. פֶּרְסֵאוּס הֵבִין שֶׁהַמְּשִׂימָה בִּלְתִּי אֶפְשָׁרִית לְבַדּוֹ - וְאָז הוֹפִיעוּ לְפָנָיו שְׁנֵי אֵלִים כְּדֵי לַעֲזֹר לוֹ: אֲתֶנָה, אֵלַת הַחָכְמָה, נָתְנָה לוֹ מָגֵן מַבְרִיק כְּמַרְאָה, וְהֶרְמֵס, שְׁלִיחָם הַמָּהִיר שֶׁל הָאֵלִים, נָתַן לוֹ חֶרֶב חַדָּה בְּיוֹתֵר. "לְעוֹלָם אַל תַּבִּיט יָשָׁר בְּעֵינֵי מֶדוּסָה," הִזְהִירָה אוֹתוֹ אֲתֶנָה, "אֶלָּא רַק בַּבָּבוּאָה שֶׁלָּהּ בְּתוֹךְ הַמָּגֵן."

כְּדֵי לִמְצֹא אֶת מְעוֹן הַנִּימְפוֹת, שֶׁאֶצְלָן שָׁמוּר כּוֹבַע קֶסֶם הַמַּעֲלִים אֶת לוֹבְשׁוֹ, סַנְדָּלִים מְכֻנָּפִים וְשַׂק קֶסֶם חָזָק, פָּגַשׁ פֶּרְסֵאוּס שָׁלֹשׁ אֲחָיוֹת זְקֵנוֹת שֶׁחָלְקוּ בֵּינֵיהֶן עַיִן אַחַת וְשֵׁן אַחַת בִּלְבַד. הוּא חָטַף בְּעָרְמָה אֶת הָעַיִן בָּרֶגַע שֶׁעָבְרָה מִיָּד לְיָד, וְהֵשִׁיב אוֹתָהּ לָהֶן רַק אַחֲרֵי שֶׁגִּלּוּ לוֹ אֶת הַדֶּרֶךְ אֶל הַנִּימְפוֹת - וְאֵלֶּה שְׂמֵחוֹת נָתְנוּ לוֹ אֶת שְׁלוֹשֶׁת חַפְצֵי הַקֶּסֶם.

בְּעֶזְרַת הַסַּנְדָּלִים הַמְּכֻנָּפִים טָס פֶּרְסֵאוּס עַד מְעָרַת הַגּוֹרְגּוֹנוֹת, שָׁם יָשְׁנוּ מֶדוּסָה וּשְׁתֵּי אֲחוֹתֶיהָ. הוּא הִבִּיט רַק בַּמָּגֵן הַמַּבְרִיק, זִהָּה בּוֹ אֶת בָּבוּאַת מֶדוּסָה הַיְּשֵׁנָה, וּבִמְכַת חֶרֶב אַחַת קָטַע אֶת רֹאשָׁהּ - בְּלִי לְהַבִּיט בָּהּ וְלוּ לְרֶגַע. הוּא תָּחַב אֶת הָרֹאשׁ אֶל תּוֹךְ שַׂק הַקֶּסֶם, חָבַשׁ אֶת כּוֹבַע הַהֵעָלְמוּת, וְנִמְלַט בַּמְּעוֹפוֹ מֵאֲחוֹתֶיהָ שֶׁהִתְעוֹרְרוּ בְּזַעַם - הֵן חִפְּשׂוּ אוֹתוֹ בְּכָל מָקוֹם, אֲבָל לֹא רָאוּ אוֹתוֹ כְּלָל.

בְּדַרְכּוֹ חֲזָרָה הַבַּיְתָה, טָס פֶּרְסֵאוּס מֵעַל חוֹף רָחוֹק וְרָאָה נַעֲרָה יָפָה בְּשֵׁם אַנְדְּרוֹמֶדָה כְּבוּלָה בְּשַׁרְשְׁרוֹת אֶל סֶלַע, מֻקְרֶבֶת לְמִפְלֶצֶת יָם עֲנָקִית. פֶּרְסֵאוּס הוֹצִיא מֵהַשַּׂק אֶת רֹאשׁ מֶדוּסָה וְהֶרְאָה אוֹתוֹ לַמִּפְלֶצֶת, שֶׁהָפְכָה מִיָּד לְאֶבֶן. הוּא שִׁחְרֵר אֶת אַנְדְּרוֹמֶדָה מֵהַשַּׁרְשְׁרוֹת, וּבְקָרוֹב נָשְׂאָה לוֹ לְאִשָּׁה.

כְּשֶׁחָזַר פֶּרְסֵאוּס אֶל הָאִי סֶרִיפוֹס, גִּלָּה שֶׁאִמּוֹ עֲדַיִן מִסְתַּתֶּרֶת מִפְּנֵי פּוֹלִידֶקְטֶס הָרָע, שֶׁלֹּא חָדַל לְהָצִיק לָהּ. פֶּרְסֵאוּס נִכְנַס לְאַרְמוֹנוֹ שֶׁל הַמֶּלֶךְ, הֵרִים אֶת רֹאשׁ מֶדוּסָה מוּל עֵינָיו הַמֻּפְתָּעוֹת שֶׁל פּוֹלִידֶקְטֶס - וְהַמֶּלֶךְ הָרָע וְכָל אֲנָשָׁיו הָפְכוּ בְּבַת אַחַת לִפְסָלִים שֶׁל אֶבֶן. אֶת כֶּתֶר הָאִי נָתַן פֶּרְסֵאוּס לְדִיקְטִיס הַדַּיָּג טוֹב הַלֵּב, שֶׁגִּדֵּל אוֹתוֹ בְּאַהֲבָה.

פֶּרְסֵאוּס הֵשִׁיב אֶת הַסַּנְדָּלִים הַמְּכֻנָּפִים, אֶת הַכּוֹבַע וְאֶת שַׂק הַקֶּסֶם לְהֶרְמֵס וְלַנִּימְפוֹת, וְאֶת רֹאשׁ מֶדוּסָה נָתַן בְּמַתָּנָה לַאֲתֶנָה, שֶׁקָּבְעָה אוֹתוֹ עַל מְגִנָּהּ שֶׁלָּהּ לְעוֹלְמֵי עַד. הוּא חָזַר עִם אַנְדְּרוֹמֶדָה וְעִם דָּנָאָה אִמּוֹ אֶל מַמְלֶכֶת אַרְגּוֹס, וְחַי שָׁם בְּאֹשֶׁר וּבְשָׁלוֹם רַבּוֹת בַּשָּׁנִים.

מְסֻפָּר עַל־פִּי מִיתוֹס פֶּרְסֵאוּס וּמֶדוּסָה מִן הַמִּיתוֹלוֹגְיָה הַיְּוָנִית הָעַתִּיקָה, מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ (בְּעִבּוּד זֶה, בְּלִי הַנְּבוּאָה הַסּוֹפִית עַל אַקְרִיסְיוֹס, כְּדֵי לְסַיֵּם בְּסוֹף שָׂמֵחַ).

בְּאַרְבָּעָה שִׁיעוּרִים קוֹדְמִים לָמַדְנוּ אֵיךְ כִּנּוּיֵי הַגּוּף (אֲנִי, אַתָּה, הוּא...) מִתְחַבְּרִים לְאַרְבַּע מִלּוֹת יַחַס שׁוֹנוֹת: "לְ" (שִׁיעוּר 54), "אֶת" (שִׁיעוּר 57), "עִם" וְ"עַל" (שִׁיעוּר 74) וּ"מִן" (שִׁיעוּר 79). הַיּוֹם נַרְכִּיב אֶת כֻּלָּם יַחַד בִּשְׁתֵּי טַבְלָאוֹת, כְּדֵי לְתַרְגֵּל אֵיךְ לְהַבְחִין בֵּינֵיהֶן.

🔑 אֵיךְ מַחְלִיטִים אֵיזוֹ מִלַּת יַחַס מַתְאִימָה? "לְ" - לְמִי מְיֹעֶדֶת הַפְּעֻלָּה אוֹ הַמַּתָּנָה ("נָתְנָה לוֹ מָגֵן"). "אֶת" - עַל מִי אוֹ עַל מָה נַעֲשֵׂית הַפְּעֻלָּה בְּאֹפֶן יָשִׁיר, בְּלִי מִלַּת יַחַס אַחֶרֶת ("רָאָה אוֹתָהּ"). "עִם" - בְּיַחַד עִם מִישֶׁהוּ ("טָס עִמָּם"). "עַל" - נוֹשֵׂא לְדִבּוּר, אוֹ מַשֶּׁהוּ שֶׁמַּשְׁפִּיעַ (לְטוֹב אוֹ לְרַע) ("הַתָּכְנִית הִתְהַפְּכָה עָלָיו"). "מִן" - מָקוֹר שֶׁמִּמֶּנּוּ בָּאִים, אוֹ הַשְׁוָאָה בֵּין שְׁנֵי דְּבָרִים ("בָּרַח מֵהַמְּעָרָה", "חָזָק מִמֶּנִּי").
💡 תַּזְכֹּרֶת מִשִּׁיעוּר 79: "מִמֶּנּוּ" (הוּא) וּ"מֵאִתָּנוּ" (אֲנַחְנוּ) הֵן שְׁתֵּי מִלִּים שׁוֹנוֹת לְגַמְרֵי - אָסוּר לְבַלְבֵּל בֵּינֵיהֶן!

כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים מְקוֹרִיִּים מִשֶּׁלָּכֶם - כָּל מִשְׁפָּט עִם כִּנּוּי גּוּף חָבוּר אַחֵר, אֶחָד מִכָּל אַחַת מֵחֲמֵשׁ מִלּוֹת הַיַּחַס שֶׁלָּמַדְנוּ: "לְ" (כְּמוֹ "נָתַתִּי לוֹ"), "אֶת" (כְּמוֹ "רָאִיתִי אוֹתָהּ"), "עִם" (כְּמוֹ "שִׂחַקְתִּי עִמָּם"), "עַל" (כְּמוֹ "סָמַכְתִּי עָלֶיךָ") וּ"מִן" (כְּמוֹ "אֲנִי גָּבוֹהַּ מִמֶּנּוּ"). לְדֻגְמָה: "אֲנִי גָּבוֹהַּ מֵאָחִי הַקָּטָן."
