📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת, בְּבַיִת קָטָן עַל שְׂפַת הַכְּפָר, גָּרוּ זָקֵן וְזִקְנָה שֶׁלֹּא הָיוּ לָהֶם יְלָדִים. בְּבֹקֶר חֹרֶף קַר אָפְתָה הַזְּקֵנָה עוּגַת זַנְגְּבִיל בְּצוּרַת אִישׁ קָטָן. הִיא הִדְבִּיקָה לוֹ שְׁתֵּי עֵינַיִם מִצִּמּוּקִים, פֶּה מִסֻּכָּר וּשְׁלוֹשָׁה כַּפְתּוֹרִים, הִנִּיחָה אוֹתוֹ עַל מַגָּשׁ וְהִכְנִיסָה אֶת הַמַּגָּשׁ לַתַּנּוּר.

אַחֲרֵי זְמַן קָצָר נִשְׁמַע קוֹל דַּקִּיק: "אֲנִי חַם! אֲנִי רוֹצֶה לָצֵאת!" הַזְּקֵנָה פָּתְחָה אֶת דֶּלֶת הַתַּנּוּר, וְאִישׁ הַזַּנְגְּבִיל קָפַץ מִן הַמַּגָּשׁ אֶל הָרִצְפָּה וְהִתְחִיל לָרוּץ. "עֲצֹר!" קָרְאָה הַזְּקֵנָה, וְהַזָּקֵן רָץ אַחֲרָיו. אֲבָל הָאִישׁ הַקָּטָן צָחַק וְשָׁר: "רוּצוּ, רוּצוּ, כַּמָּה שֶׁתּוּכְלוּ — לֹא תַּשִּׂיגוּ אוֹתִי, אֲנִי אִישׁ הַזַּנְגְּבִיל!"

אִישׁ הַזַּנְגְּבִיל רָץ מִן הַבַּיִת אֶל הַשָּׂדֶה, וְשָׁם פָּגַשׁ פָּרָה גְּדוֹלָה. "עֲצֹר, אִישׁ זַנְגְּבִיל!" אָמְרָה הַפָּרָה. "הָרֵיחַ שֶׁלְּךָ טָעִים כָּל כָּךְ! אֲנִי רוֹצָה לִטְעֹם אוֹתְךָ!" הָאִישׁ הַקָּטָן הִתְפָּאֵר: "בָּרַחְתִּי מִזָּקֵן וּמִזְּקֵנָה, וְאֶבְרַח גַּם מִמֵּךְ!" הוּא שָׁר שׁוּב אֶת שִׁירוֹ וְהִמְשִׁיךְ לָרוּץ.

בַּדֶּרֶךְ פָּגַשׁ אִישׁ הַזַּנְגְּבִיל סוּס חָזָק. "עֲצֹר!" צָהַל הַסּוּס. אֲבָל אִישׁ הַזַּנְגְּבִיל לֹא עָצַר, כִּי הָיָה בָּטוּחַ בְּעַצְמוֹ. הוּא צָעַק: "בָּרַחְתִּי מִזָּקֵן, מִזְּקֵנָה וּמִפָּרָה, וְאֶבְרַח גַּם מִמְּךָ!" עַכְשָׁיו רָצוּ אַחֲרָיו הַזָּקֵן, הַזְּקֵנָה, הַפָּרָה וְהַסּוּס, וְכֻלָּם נָשְׁמוּ בִּכְבֵדוּת.

אַחַר כָּךְ עָבַר הָאִישׁ הַקָּטָן לְיַד חַקְלָאִים שֶׁעָבְדוּ בַּשָּׂדֶה. הַחַקְלָאִים הִצְטָרְפוּ לַמִּרְדָּף, וְאִישׁ הַזַּנְגְּבִיל שָׁר לָהֶם אֶת שִׁירוֹ בְּקוֹל רָם יוֹתֵר. עַד מְהֵרָה הִגִּיעַ לְנָהָר רָחָב. הוּא עָמַד עַל הַגָּדָה וְלֹא יָדַע אֵיךְ לַחֲצוֹת אוֹתוֹ, כִּי עוּגָה אֵינָהּ יוֹדַעַת לִשְׂחוֹת. מֵאָחוֹר כְּבָר נִשְׁמְעוּ צְעָדִים רַבִּים.

עַל הַגָּדָה עָמַד שׁוּעָל עָרוּם, וְהוּא רָאָה אֶת הָאִישׁ הַקָּטָן מְפֻחָד. "אַל תִּפְחַד", אָמַר הַשּׁוּעָל בְּחִיּוּךְ. "אֲנִי שׂוֹחֶה הֵיטֵב. עֲלֵה עַל גַּבִּי, וַאֲנִי אַעֲבִיר אוֹתְךָ אֶת הַנָּהָר." אִישׁ הַזַּנְגְּבִיל הִסְתַּכֵּל אָחוֹרָה, רָאָה אֶת הַמִּרְדָּף מִתְקָרֵב, וְקָפַץ עַל גַּבּוֹ שֶׁל הַשּׁוּעָל.

הַשּׁוּעָל הִתְחִיל לִשְׂחוֹת, וְהַמַּיִם עָלוּ וְעָלוּ. "הַמַּיִם מַגִּיעִים לְגַבִּי", אָמַר הַשּׁוּעָל. "עֲלֵה עַל כְּתֵפִי, כְּדֵי שֶׁלֹּא תִּרְטַב." אִישׁ הַזַּנְגְּבִיל עָלָה עַל הַכָּתֵף. כַּעֲבֹר רֶגַע אָמַר הַשּׁוּעָל: "עֲלֵה עַל חָטְמִי, כְּדֵי שֶׁלֹּא תִּרְטַב." אִישׁ הַזַּנְגְּבִיל הִבְחִין שֶׁשִּׁנֵּי הַשּׁוּעָל מַבְרִיקוֹת, וְלִבּוֹ דָּפַק בְּחוּזְקָה.

בָּרֶגַע הָאַחֲרוֹן, כְּשֶׁהַשּׁוּעָל פָּתַח אֶת פִּיו, הֵבִין אִישׁ הַזַּנְגְּבִיל אֶת הַסַּכָּנָה. הוּא דָּחַף בְּכָל כֹּחוֹ וְקָפַץ מֵחָטְמוֹ שֶׁל הַשּׁוּעָל אֶל הַגָּדָה הַקְּרוֹבָה! הַשּׁוּעָל נִשְׁאַר עִם פֶּה רֵיק, וְכָל הַמִּרְדָּף עָצַר בְּתִמָּהוֹן.

אִישׁ הַזַּנְגְּבִיל עָמַד נִרְעָד עַל הַגָּדָה, וְהַזְּקֵנָה וְהַזָּקֵן הִגִּיעוּ אֵלָיו מִתְנַשְּׁמִים. הוּא הִבְטִיחַ שֶׁלֹּא יִתְפָּאֵר יוֹתֵר. הַזְּקֵנָה חִבְּקָה אוֹתוֹ בְּעֶדְנָה וְאָמְרָה: "לֹא אָפִיתִי אוֹתְךָ כְּדֵי לֶאֱכֹל אוֹתְךָ, אֶלָּא כְּדֵי שֶׁיִּהְיֶה לָנוּ יֶלֶד." מֵאָז גָּר אִישׁ הַזַּנְגְּבִיל בַּבַּיִת הַקָּטָן, וְהוּא יָדַע: מִי שֶׁמְּחַנֵּף לְךָ בְּמִלִּים מְתוּקוֹת אֵינוֹ תָּמִיד חָבֵר.

מְסֻפָּר מֵחָדָשׁ בִּלְשׁוֹנֵנוּ עַל־פִּי "אִישׁ הַזַּנְגְּבִיל" (The Gingerbread Man), סִפּוּר עַם אַנְגְּלִי־אֲמֶרִיקָנִי מֻכָּר, שֶׁפֻּרְסַם בְּשָׁנָה 1875 בַּמַּגָּזִין לִילָדִים "סֵנְט נִיקוֹלָס", נַחֲלַת הַכְּלָל, עִם סוֹף מְרֻכָּךְ.

📚 דִּקְדּוּק — נְקֻדַּת מַבָּט: מִי מְסַפֵּר אֶת הַסִּפּוּר?

גּוּף שְׁלִישִׁי וְגוּף רִאשׁוֹן

בְּכָל סִפּוּר יֵשׁ מְסַפֵּר — הַקּוֹל שֶׁמְּסַפֵּר לָנוּ מָה קָרָה. הַמְּסַפֵּר יָכוֹל לְסַפֵּר בִּשְׁתֵּי דְּרָכִים עִקָּרִיּוֹת. סִפּוּר אִישׁ הַזַּנְגְּבִיל שֶׁקָּרָאנוּ מְסֻפָּר בְּגוּף שְׁלִישִׁי: הַמְּסַפֵּר עוֹמֵד מִחוּץ לַסִּפּוּר וְאוֹמֵר "הוּא קָפַץ", "הִיא אָפְתָה", "הֵם רָצוּ".

מָה הַמְּסַפֵּר יוֹדֵעַ?
