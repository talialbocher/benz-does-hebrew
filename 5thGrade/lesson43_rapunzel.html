📖 קְרִיאָה

פַּעַם אַחַת חָיוּ אִישׁ וְאִשָּׁה שֶׁרָצוּ מְאֹד שֶׁיִּוָּלֵד לָהֶם יֶלֶד, אֲבָל שָׁנִים רַבּוֹת עָבְרוּ בְּלִי שֶׁזֶּה קָרָה. לְבַסּוֹף, לְשִׂמְחָתָם הַגְּדוֹלָה, הָאִשָּׁה נִכְנְסָה לְהֵרָיוֹן.

מֵאֲחוֹרֵי הַבַּיִת שֶׁלָּהֶם הָיְתָה גִּנָּה גְּדוֹלָה וְיָפָה, אֲבָל הִיא הָיְתָה שַׁיֶּכֶת לְמְכַשֵּׁפָה מַפְחִידָה, שֶׁאִישׁ לֹא הֵעֵז לְהִתְקָרֵב אֵלֶיהָּ. בַּגִּנָּה גָּדַל צֶמַח יָרֹק וְטָרִי בְּשֵׁם "רַפּוּנְזֶל", וְהָאִשָּׁה הַהָרָה הִבִּיטָה בּוֹ מִדֵּי יוֹם מֵהַחַלּוֹן, וְנִתְקְפָה תְּשׁוּקָה עַזָּה לֶאֱכֹל מִמֶּנּוּ.

בְּלַיְלָה אֶחָד הֵעֵז בַּעְלָהּ לְטַפֵּס עַל הַגָּדֵר וְלִקְטֹף מְעַט מִן הַצֶּמַח בִּשְׁבִיל אִשְׁתּוֹ. הוּא עָשָׂה זֹאת פַּעַם נוֹסֶפֶת, אֲבָל הַפַּעַם הַמְּכַשֵּׁפָה תָּפְסָה אוֹתוֹ בְּעַצְמָהּ. "אֶסְלַח לְךָ עַל הַגְּנֵבָה," אָמְרָה בְּקוֹל קָשֶׁה, "רַק בִּתְנַאי אֶחָד - כְּשֶׁתִּוָּלֵד הַבַּת שֶׁלָּכֶם, תִּתְּנוּ אוֹתָהּ לִי, וַאֲנִי אֲגַדֵּל אוֹתָהּ כְּאִלּוּ הָיְתָה בִּתִּי." הָאִישׁ, מְפֻחָד מְאֹד, הִסְכִּים.

כַּאֲשֶׁר נוֹלְדָה הַתִּינֹקֶת, בָּאָה הַמְּכַשֵּׁפָה וְלָקְחָה אוֹתָהּ מִיָּד, וְקָרְאָה לָהּ "רַפּוּנְזֶל", עַל שֵׁם הַצֶּמַח שֶׁגָּרַם לְכָל הַסִּפּוּר הַזֶּה. רַפּוּנְזֶל גָּדְלָה וְהָיְתָה לְיַלְדָּה הֲכִי יָפָה מִתַּחַת לַשֶּׁמֶשׁ, וּשְׂעָרָהּ הַזָּהֹב הָיָה אָרֹךְ וּמַבְרִיק כְּמוֹ מֶשִׁי.

כְּשֶׁמָּלְאוּ לְרַפּוּנְזֶל שְׁתֵּים־עֶשְׂרֵה שָׁנִים, סָגְרָה אוֹתָהּ הַמְּכַשֵּׁפָה בְּמִגְדָּל גָּבֹהַּ בְּתוֹךְ הַיַּעַר, מִגְדָּל בְּלִי דֶּלֶת וּבְלִי מַדְרֵגוֹת, וְרַק חַלּוֹן אֶחָד קָטָן בְּרֹאשׁוֹ.

בְּכָל פַּעַם שֶׁהַמְּכַשֵּׁפָה רָצְתָה לְבַקֵּר, הִיא עָמְדָה מִתַּחַת לַחַלּוֹן וְקָרְאָה: "רַפּוּנְזֶל, רַפּוּנְזֶל, הוֹרִידִי אֶת הַצַּמָּה שֶׁלָּךְ!" וְרַפּוּנְזֶל הָיְתָה קוֹשֶׁרֶת אֶת שְׂעָרָהּ הָאָרֹךְ סְבִיב וָו בַּחַלּוֹן, וּמְשַׁלְשֶׁלֶת אוֹתוֹ לְמַטָּה, וְהַמְּכַשֵּׁפָה הָיְתָה מְטַפֶּסֶת עָלָיו כְּמוֹ עַל סֻלָּם.

יוֹם אֶחָד עָבַר בֶּן־מֶלֶךְ בַּיַּעַר וְשָׁמַע שִׁיר יָפֶה בּוֹקֵעַ מֵרֹאשׁ הַמִּגְדָּל. הוּא חִפֵּשׂ דֶּלֶת וְלֹא מָצָא, אֲבָל כַּעֲבֹר זְמַן־מָה רָאָה אֶת הַמְּכַשֵּׁפָה קוֹרֵאת אֶת הַמִּלִּים הַסּוֹדִיּוֹת וְעוֹלָה עַל הַצַּמָּה הַזְּהֻבָּה. בְּעֶרֶב הַלַּיְלָה הַבָּא, כְּשֶׁהַמְּכַשֵּׁפָה כְּבָר הָלְכָה, קָרָא גַּם הוּא: "רַפּוּנְזֶל, רַפּוּנְזֶל, הוֹרִידִי אֶת הַצַּמָּה שֶׁלָּךְ!" וְטִפֵּס אֶל הַחַלּוֹן.

בַּתְּחִלָּה רַפּוּנְזֶל נִבְהֲלָה מְאֹד, שֶׁכֵּן מֵעוֹלָם לֹא רָאֲתָה אִישׁ זָר, אֲבָל בֶּן־הַמֶּלֶךְ דִּבֵּר אֵלֶיהָּ בְּרֹךְ, וְעַד מְהֵרָה נוֹצְרָה בֵּינֵיהֶם יְדִידוּת חַמָּה. הוּא הִגִּיעַ לְבַקֵּר שׁוּב וָשׁוּב, וְסִכְּמוּ יַחַד שֶׁבְּכָל פַּעַם הוּא יָבִיא אִתּוֹ פְּתִיל מֶשִׁי, כְּדֵי שֶׁרַפּוּנְזֶל תּוּכַל לִקְלֹעַ מִמֶּנּוּ סֻלָּם וְלָרֶדֶת סוֹף־סוֹף אֶל הָאֲדָמָה.

אֲבָל יוֹם אֶחָד, בְּלִי מַשִׂים, שָׁאֲלָה רַפּוּנְזֶל אֶת הַמְּכַשֵּׁפָה מַדּוּעַ הִיא הַרְבֵּה יוֹתֵר כְּבֵדָה מִבֶּן־הַמֶּלֶךְ הַצָּעִיר. הַמְּכַשֵּׁפָה הֵבִינָה מִיָּד אֶת הָאֱמֶת, זָעֲמָה מְאֹד, גָּזְרָה אֶת צַמָּתָהּ הָאֲרֻכָּה בְּמִסְפָּרַיִם, וְשָׁלְחָה אֶת רַפּוּנְזֶל לְמָקוֹם שָׁמֵם וְרָחוֹק, שֶׁתְּחַיֶּה שָׁם לְבַדָּהּ.

בָּעֶרֶב, כְּשֶׁבֶּן־הַמֶּלֶךְ קָרָא כְּרָגִיל אֶת הַמִּלִּים הַסּוֹדִיּוֹת, הוֹרִידָה הַמְּכַשֵּׁפָה בְּעַצְמָהּ אֶת הַצַּמָּה הַגְּזוּזָה, וְטִפֵּס בֶּן־הַמֶּלֶךְ בְּלִי לָדַעַת מַה מְּחַכֶּה לוֹ. כְּשֶׁהִגִּיעַ לְמַעְלָה וְרָאָה אֶת הַמְּכַשֵּׁפָה בִּמְקוֹם רַפּוּנְזֶל, נִבְהַל מְאֹד. "לְעוֹלָם לֹא תִּרְאֶה אוֹתָהּ שׁוּב!" צָעֲקָה הַמְּכַשֵּׁפָה בְּזַעַם. הֲמוּם וּמְיֹאָשׁ, בֶּן־הַמֶּלֶךְ קָפַץ מִן הַחַלּוֹן, וְנָפַל אֶל שִׂיחֵי קוֹצִים שֶׁצָּמְחוּ מִתַּחַת לַמִּגְדָּל. הַקּוֹצִים פָּצְעוּ אֶת עֵינָיו, וְהוּא לֹא הֵיטִיב לִרְאוֹת יוֹתֵר.

בֶּן־הַמֶּלֶךְ נָדַד בְּיֵאוּשׁ בַּיְעָרוֹת בְּמֶשֶׁךְ שָׁנִים רַבּוֹת, בְּלִי לִרְאוֹת דָּבָר. יוֹם אֶחָד שָׁמַע קוֹל שִׁיר מֻכָּר מִתּוֹךְ הַמִּדְבָּר - זֶה הָיָה קוֹלָהּ שֶׁל רַפּוּנְזֶל, שֶׁחָיְתָה שָׁם עִם שְׁנֵי יְלָדֶיהָּ הַתְּאוֹמִים. כְּשֶׁרַפּוּנְזֶל רָאֲתָה אוֹתוֹ, רָצָה אֵלָיו בִּבְכִי שֶׁל שִׂמְחָה, וְחִבְּקָה אוֹתוֹ בְּחֹזֶק. שְׁתֵּי דְּמָעוֹת חַמּוֹת נָפְלוּ מֵעֵינֶיהָּ יָשָׁר לְתוֹךְ עֵינָיו - וּפִתְאֹם, בְּנֵס גָּדוֹל, שָׁב אֵלָיו מְאוֹר עֵינָיו.

בֶּן־הַמֶּלֶךְ לָקַח אֶת רַפּוּנְזֶל וְאֶת יְלָדֵיהֶם אֶל מַמְלַכְתּוֹ, וְשָׁם חָיוּ כֻּלָּם בְּאֹשֶׁר וּבְשָׁלוֹם עַד סוֹף יְמֵיהֶם.

מְסֻפָּר עַל־פִּי אַגָּדַת עַם גֶּרְמָנִית שֶׁאָסְפוּ הָאַחִים גְּרִים (הַמֵּאָה הַ-19), "רַפּוּנְזֶל" - מְעֻבֶּדֶת

📚 דִּקְדּוּק — חֲזָרָה עַל ה' הַיְדִיעָה

תַּזְכֹּרֶת: מַהִי ה' הַיְדִיעָה?

ה' הַיְדִיעָה הִיא הָאוֹת ה' שֶׁמִּתְוַסֶּפֶת בִּתְחִלַּת שֵׁם עֶצֶם, כְּדֵי לְצַיֵּן דָּבָר מְסֻיָּם וְיָדוּעַ (בְּאַנְגְּלִית: "the"). לְדֻגְמָה: מִגְדָּל (a tower) לְעֻמַּת הַמִּגְדָּל (the tower) - הַמִּגְדָּל הַמְּסֻיָּם שֶׁבּוֹ יָשְׁבָה רַפּוּנְזֶל.

תַּזְכֹּרֶת: ה' הַיְדִיעָה "נִבְלַעַת" אַחֲרֵי ב, כ, ל

כְּשֶׁמְּצָרְפִים לְמִלָּה עִם ה' הַיְדִיעָה אֶת אַחַת מִמִּלּוֹת הַיַּחַס ב (בְּתוֹךְ), כ (כְּמוֹ), אוֹ ל (אֶל) - ה' הַיְדִיעָה "נִבְלַעַת": הִיא לֹא נִכְתֶּבֶת יוֹתֵר, אֲבָל מַשְׁאִירָה סִימָן - הָאוֹת שֶׁאַחֲרֶיהָ מְקַבֶּלֶת דָּגֵשׁ. לְדֻגְמָה: הַמִּגְדָּל + ב = בַּמִּגְדָּל (לֹא בְּהַמִּגְדָּל), הַיַּעַר + ל = לַיַּעַר (לֹא לְהַיַּעַר).

תַּזְכֹּרֶת: מָה קוֹרֶה עִם מ (מִן)?

לְעֻמַּת זֹאת, אַחֲרֵי מִלַּת הַיַּחַס מ (מִן) - ה' הַיְדִיעָה נִשְׁאֶרֶת וְנִרְאֵית בַּמִּלָּה, וְלֹא נִבְלַעַת. לְדֻגְמָה: הַמִּגְדָּל + מ = מֵהַמִּגְדָּל (לֹא מִמִּגְדָּל) - ה' נִשְׁאֶרֶת בִּכְתִיב.

תַּזְכֹּרֶת: אֵין ה' הַיְדִיעָה לִפְנֵי שֵׁם פְּרָטִי

ה' הַיְדִיעָה לֹא מִתְוַסֶּפֶת לִפְנֵי שֵׁם פְּרָטִי (שֵׁם שֶׁל אָדָם, עִיר אוֹ מְדִינָה) - כִּי הוּא כְּבָר מְסֻיָּם וְיָדוּעַ מֵעַצְמוֹ. לְדֻגְמָה: רַפּוּנְזֶל, גֶּרְמַנְיָה - בְּלִי ה' בַּהַתְחָלָה.

כְּשֶׁשֵּׁם עֶצֶם מְיֻדָּע (עִם ה' הַיְדִיעָה) מְלֻוֶּה בְּשֵׁם תֹּאַר, וְרוֹצִים לוֹמַר "צֵרוּף" אֶחָד (לֹא מִשְׁפָּט שָׁלֵם) - חַיָּב לְהִתְוַסֵּף ה' הַיְדִיעָה גַּם לְשֵׁם הַתֹּאַר. לְדֻגְמָה: "הַמִּגְדָּל הַגָּבֹהַּ" - שְׁתֵּי הַמִּלִּים מְקַבְּלוֹת ה'.

שִׂימוּ לֵב לַהֶבְדֵּל הַחָשׁוּב: "הַמִּגְדָּל הַגָּבֹהַּ" (צֵרוּף שֵׁם - מְתָאֵר מִגְדָּל מְסֻיָּם) לְעֻמַּת "הַמִּגְדָּל גָּבֹהַּ" (מִשְׁפָּט שָׁלֵם - אוֹמֵר עַל הַמִּגְדָּל שֶׁהוּא גָּבֹהַּ). כְּשֶׁרַק אֶחָד מֵהֶם מְקַבֵּל ה', הַמַּשְׁמָעוּת מִשְׁתַּנָּה!
