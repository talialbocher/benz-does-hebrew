גּוּלִיבֶר בְּאֶרֶץ לִילִיפּוּט

עַל פִּי "מַסְעוֹת גּוּלִיבֶר" מֵאֵת גּ׳וֹנָתָן סְוִיפְט, בְּעִבּוּד חׇפְשִׁי

📖 קְרִיאָה

לִפְנֵי כִּשְׁלֹשׁ מֵאוֹת שָׁנָה יָצָא רוֹפֵא אַנְגְּלִי צָעִיר בְּשֵׁם לֶמוּאֵל גּוּלִיבֶר לְמַסָּע אָרֹךְ בַּיָּם. בְּאֶמְצַע הַדֶּרֶךְ פָּרְצָה סְעָרָה נוֹרָאָה. הַגַּלִּים הָיוּ גְּבוֹהִים כׇּל כָּךְ, עַד שֶׁהַסְּפִינָה נִשְׁבְּרָה לַחֲתִיכוֹת. גּוּלִיבֶר שָׂחָה שָׁעוֹת אֲרֻכּוֹת בַּמַּיִם הַקָּרִים, עַד שֶׁרַגְלָיו נָגְעוּ סוֹף סוֹף בַּחוֹל.

הוּא הָיָה עָיֵף כׇּל כָּךְ, עַד שֶׁנִּרְדַּם עַל הַחוֹף מִיָּד וְיָשַׁן כׇּל הַלַּיְלָה. כְּשֶׁהִתְעוֹרֵר בַּבֹּקֶר, נִסָּה לָקוּם — וְלֹא הִצְלִיחַ. מֵאוֹת חוּטִים דַּקִּים קָשְׁרוּ אֶת יָדָיו, אֶת רַגְלָיו וַאֲפִלּוּ אֶת שְׂעָרוֹ אֶל הָאֲדָמָה.

פִּתְאוֹם הִרְגִּישׁ שֶׁמַּשֶּׁהוּ זָז עַל רַגְלוֹ. הוּא הֵרִים מְעַט אֶת רֹאשׁוֹ וְרָאָה אִישׁ זָעִיר, גָּבוֹהַּ כְּמוֹ כַּף יָד, וּבְיָדָיו קֶשֶׁת וְחֵץ. אַחֲרָיו טִפְּסוּ עָלָיו עוֹד עֲשָׂרוֹת אֲנָשִׁים קְטַנְטַנִּים. גּוּלִיבֶר צָעַק בְּקוֹל כׇּל כָּךְ חָזָק, עַד שֶׁכֻּלָּם נָפְלוּ מִפַּחַד אֶל הַדֶּשֶׁא.

כָּךְ הִגִּיעַ גּוּלִיבֶר לְלִילִיפּוּט — אִי שֶׁבּוֹ הַכֹּל קָטָן פִּי שְׁנֵים עָשָׂר: הָאֲנָשִׁים, הַבָּתִּים, הָעֵצִים וַאֲפִלּוּ הַסּוּסִים. הַלִּילִיפּוּטִים הֵבִינוּ שֶׁהָעֲנָק רָעֵב, וְהֵבִיאוּ לוֹ כׇּל כָּךְ הַרְבֵּה סַלִּים שֶׁל אֹכֶל, עַד שֶׁהַסַּלִּים יָצְרוּ הַר קָטָן. אֲבָל גּוּלִיבֶר הָיָה רָעֵב כׇּל כָּךְ, עַד שֶׁבָּלַע שָׁלֹשׁ כִּכָּרוֹת לֶחֶם בְּבַת אַחַת וְשָׁתָה שְׁתֵּי חָבִיּוֹת מַיִם.

הַקֵּיסָר שֶׁל לִילִיפּוּט צִוָּה לְהָבִיא אֶת הָעֲנָק אֶל הָעִיר הַבִּירָה. אֶלֶף וַחֲמֵשׁ מֵאוֹת סוּסִים מָשְׁכוּ עֲגָלָה עֲנָקִית, וְגוּלִיבֶר שָׁכַב עָלֶיהָ קָשׁוּר. הָרְחוֹבוֹת הָיוּ מְלֵאִים כׇּל כָּךְ, עַד שֶׁיְּלָדִים טִפְּסוּ עַל הַגַּגּוֹת כְּדֵי לְהָצִיץ בּוֹ. בָּעִיר כָּבְלוּ אוֹתוֹ בְּשַׁלְשְׁלָאוֹת לְיַד מִקְדָּשׁ יָשָׁן, כְּדֵי שֶׁלֹּא יִבְרַח.

עִם הַזְּמַן לָמַד גּוּלִיבֶר אֶת שְׂפַת לִילִיפּוּט, וְהַקֵּיסָר הִסְכִּים לְהָסִיר אֶת הַשַּׁלְשְׁלָאוֹת. גּוּלִיבֶר הִבְטִיחַ לִשְׁמֹר עַל הָאִי וּלְהִזָּהֵר שֶׁלֹּא לִדְרֹךְ עַל אַף אֶחָד. מֵאָז הָלַךְ בִּזְהִירוּת כָּזֹאת, עַד שֶׁאֲפִלּוּ הַתַּרְנְגֹלוֹת בָּרְחוֹב לֹא נִבְהֲלוּ מִמֶּנּוּ.

יוֹם אֶחָד בָּא הַקֵּיסָר מֻדְאָג. "אַנְשֵׁי בְּלֶפוּסְקוֹ, הָאִי שֶׁמֵּעֵבֶר לַמֵּצַר, הֵכִינוּ צִי שֶׁל חֲמִשִּׁים סְפִינוֹת מִלְחָמָה," אָמַר, "וְהֵם עוֹמְדִים לִתְקֹף אוֹתָנוּ!" גּוּלִיבֶר הֵכִין חֲמִשִּׁים חֲבָלִים עִם וָוִים, נִכְנַס לַיָּם וְצָעַד בַּמַּיִם אֶל הַצַּד הַשֵּׁנִי שֶׁל הַמֵּצַר. הַחַיָּלִים בַּסְּפִינוֹת יָרוּ בּוֹ אַלְפֵי חִצִּים, אֲבָל הַחִצִּים הָיוּ קְטַנִּים כׇּל כָּךְ, עַד שֶׁהִרְגִּישׁ רַק עֲקִיצוֹת קַלּוֹת, כְּמוֹ שֶׁל יַתּוּשִׁים.

גּוּלִיבֶר קָשַׁר חֶבֶל לְכׇל סְפִינָה, אָסַף אֶת כׇּל הַחֲבָלִים בְּיָדוֹ וּמָשַׁךְ אֶת הַצִּי כֻּלּוֹ אַחֲרָיו אֶל נְמַל לִילִיפּוּט. הָאֲנָשִׁים עַל הַחוֹף שָׂמְחוּ כׇּל כָּךְ, עַד שֶׁרָקְדוּ בָּרְחוֹבוֹת עַד הַלַּיְלָה.

אֲבָל הַקֵּיסָר רָצָה עוֹד. "חֲזֹר לִבְלֶפוּסְקוֹ," צִוָּה, "וְהָבֵא לִי גַּם אֶת שְׁאָר הַסְּפִינוֹת שֶׁלָּהֶם. אֲנִי רוֹצֶה שֶׁכׇּל אַנְשֵׁי בְּלֶפוּסְקוֹ יִהְיוּ עֲבָדִים שֶׁלִּי!" גּוּלִיבֶר סֵרֵב. "לֹא אֶעֱזֹר לְךָ לַהֲפֹךְ עַם חׇפְשִׁי לַעֲבָדִים," אָמַר בְּשֶׁקֶט. הַקֵּיסָר נֶעֱלַב כׇּל כָּךְ, עַד שֶׁהִפְסִיק לְדַבֵּר אִתּוֹ, וְהַשָּׂרִים שֶׁלּוֹ הִתְחִילוּ לְתַכְנֵן מְזִמָּה נֶגֶד גּוּלִיבֶר בַּסֵּתֶר.

גּוּלִיבֶר הֵבִין שֶׁהִגִּיעַ הַזְּמַן לְהִפָּרֵד. יוֹם אֶחָד רָאָה עַל הַחוֹף סִירָה אֲמִתִּית, בְּגֹדֶל רָגִיל, שֶׁהַגַּלִּים הֵבִיאוּ מֵרָחוֹק. הוּא תִּקֵּן אוֹתָהּ, הֶעֱמִיס עָלֶיהָ אֹכֶל וְגַם כַּמָּה פָּרוֹת וּכְבָשִׂים זְעִירִים לְמַזְכֶּרֶת, וְהִפְלִיג הַבַּיְתָה. בְּאַנְגְּלִיָּה הֶרְאָה אֶת הַכְּבָשִׂים לַחֲבֵרָיו. הַכְּבָשִׂים הָיוּ קְטַנִּים כׇּל כָּךְ, עַד שֶׁכֻּלָּם חָשְׁבוּ שֶׁהֵם צַעֲצוּעִים — עַד שֶׁאֶחָד מֵהֶם פָּעָה.

מָקוֹר: גּ׳וֹנָתָן סְוִיפְט (אִירְלַנְד, 1726), "מַסְעוֹת גּוּלִיבֶר", חֵלֶק א׳ — הַמַּסָּע לְלִילִיפּוּט, בְּעִבּוּד חׇפְשִׁי וּמְקֻצָּר.

"כׇּל כָּךְ... עַד שֶׁ..." — וְעוֹד דְּרָכִים לִבְנוֹת מִשְׁפַּט תּוֹצָאָה

בְּשִׁיעוּר 83 (פְּרוֹמֶתֵאוּס) לָמַדְנוּ: בְּמִשְׁפַּט תּוֹצָאָה יֵשׁ דָּבָר שֶׁקָּרָה בְּעֹצֶם גָּדוֹל, וְאַחֲרָיו — הַתּוֹצָאָה שֶׁל אוֹתוֹ עֹצֶם. מִלַּת הָעֹצֶם בָּאָה לִפְנֵי "עַד שֶׁ-", וְהַתּוֹצָאָה בָּאָה אַחֲרָיו. הַיּוֹם נַכִּיר כַּמָּה צוּרוֹת שֶׁל מִלַּת הָעֹצֶם, וּשְׁתֵּי מַלְכֹּדוֹת.

⚠️ מַלְכֹּדֶת 2 — "עַד שֶׁ" שֶׁל זְמַן (שִׁיעוּר 125): "גּוּלִיבֶר שָׂחָה שָׁעוֹת, עַד שֶׁרַגְלָיו נָגְעוּ בַּחוֹל." כָּאן אֵין מִלַּת עֹצֶם, וְהַפְּסוּקִית עוֹנָה עַל הַשְּׁאֵלָה "עַד מָתַי?" — זֶה מִשְׁפַּט זְמַן.

🔄 מִ"וְלָכֵן" לְמִשְׁפַּט תּוֹצָאָה: מַחְלִיפִים אֶת "מְאֹד" בְּ"כׇּל כָּךְ", וְאֶת "וְלָכֵן" בְּ"עַד שֶׁ-":

הַחִצִּים הָיוּ קְטַנִּים מְאֹד, וְלָכֵן הוּא כִּמְעַט לֹא הִרְגִּישׁ אוֹתָם.

      דַּמְיְנוּ שֶׁהִגַּעְתֶּם, כְּמוֹ גּוּלִיבֶר, לְאִי מוּזָר — אִי שֶׁל עֲנָקִים אוֹ אִי שֶׁל אֲנָשִׁים זְעִירִים. כִּתְבוּ חֲמִשָּׁה מִשְׁפְּטֵי תּוֹצָאָה עַל מָה שֶׁקָּרָה לָכֶם שָׁם. הִשְׁתַּמְּשׁוּ בְּמִלּוֹת עֹצֶם שׁוֹנוֹת (כׇּל כָּךְ, כׇּל כָּךְ הַרְבֵּה, כָּזֶה, כָּזֹאת, כָּאֵלֶּה), וְאַל תִּשְׁכְּחוּ פְּסִיק לִפְנֵי "עַד שֶׁ". לְמָשָׁל: "הַתַּפּוּחִים בָּאִי הָיוּ גְּדוֹלִים כׇּל כָּךְ, עַד שֶׁהָיִיתִי צָרִיךְ סֻלָּם כְּדֵי לִקְטֹף אֶחָד."
