📖 קְרִיאָה

הָיָה הָיָה מֶלֶךְ שֶׁהָיוּ לוֹ שְׁתֵּים עֶשְׂרֵה בָּנוֹת, וְכֻלָּן נְסִיכוֹת יָפוֹת וַחֲבִיבוֹת. שְׁתֵּים עֶשְׂרֵה הַנְּסִיכוֹת יָשְׁנוּ יַחַד בְּחֶדֶר אֶחָד גָּדוֹל, וְכָל עֶרֶב נָעַל הַמֶּלֶךְ בְּעַצְמוֹ אֶת דֶּלֶת הַחֶדֶר בְּמַפְתֵּחַ כָּבֵד. אַךְ בְּכָל בֹּקֶר, כְּשֶׁפָּתַח אֶת הַדֶּלֶת מֵחָדָשׁ, מָצָא שֶׁנַעֲלֵי הָרִקּוּד שֶׁל בְּנוֹתָיו קְרוּעוֹת לְגַמְרֵי, כְּאִלּוּ רָקְדוּ בָּהֶן כָּל הַלַּיְלָה. אִישׁ לֹא הֵבִין אֵיךְ זֶה קוֹרֶה - הַדֶּלֶת הָיְתָה נְעוּלָה, וְלֹא הָיְתָה שׁוּם דֶּרֶךְ אַחֶרֶת לָצֵאת מֵהַחֶדֶר.

הַמֶּלֶךְ הִכְרִיז הַכְרָזָה בְּכָל הַמַּמְלָכָה: כָּל נָסִיךְ שֶׁיַּצְלִיחַ לְגַלּוֹת אֶת הַסּוֹד תּוֹךְ שָׁלוֹשׁ לֵילוֹת - יוּכַל לָשֵׂאת לְאִשָּׁה אֶת הַנְּסִיכָה שֶׁיִּבְחַר, וְלָרֶשֶׁת אֶת הַכֶּתֶר אַחֲרָיו. אַךְ מִי שֶׁיְּנַסֶּה וְלֹא יַצְלִיחַ תּוֹךְ שָׁלוֹשׁ לֵילוֹת - יִצְטָרֵךְ לַעֲזֹב אֶת הַמַּמְלָכָה בְּלִי כְּלוּם. נְסִיכִים רַבִּים בָּאוּ מֵאֲרָצוֹת רְחוֹקוֹת כְּדֵי לְנַסּוֹת אֶת מַזָּלָם, אַךְ כֻּלָּם נִרְדְּמוּ בְּאֶמְצַע הַלַּיְלָה בְּלִי לָדַעַת לָמָּה, וְעָזְבוּ בְּבֹשֶׁת פָּנִים בַּבֹּקֶר.

בְּאוֹתָהּ מַמְלָכָה חַי גַּם חַיָּל זָקֵן וְעָנִי, שֶׁשָּׁב מֵהַמִּלְחָמָה בְּלִי רְכוּשׁ וּבְלִי מַזָּל. בְּדַרְכּוֹ אֶל אַרְמוֹן הַמֶּלֶךְ פָּגַשׁ בַּיַּעַר אִשָּׁה זְקֵנָה, שֶׁשָּׁאֲלָה אוֹתוֹ לְאָן הוּא הוֹלֵךְ. כְּשֶׁסִּפֵּר לָהּ עַל חִידַת הַנְּסִיכוֹת, יָעֲצָה לוֹ הַזְּקֵנָה: "אַל תִּשְׁתֶּה מֵהַיַּיִן שֶׁיַּגִּישׁוּ לְךָ בַּלַּיְלָה - יֵשׁ בּוֹ סַם שֵׁנָה. וְהִנֵּה לְךָ גְּלִימָה קְסוּמָה שֶׁתַּהֲפֹךְ אוֹתְךָ לְבִלְתִּי־נִרְאֶה." הַחַיָּל הוֹדָה לָהּ בְּכָל לִבּוֹ, וְהִמְשִׁיךְ בְּדַרְכּוֹ אֶל הָאַרְמוֹן.

בָּאַרְמוֹן קִבֵּל הַמֶּלֶךְ אֶת הַחַיָּל בְּשִׂמְחָה, וְנָתַן לוֹ חֶדֶר קָטָן סָמוּךְ לְחֶדֶר הַנְּסִיכוֹת. בַּלַּיְלָה הֵבִיאָה לוֹ הַבְּכִירָה שֶׁבַּנְּסִיכוֹת כּוֹס יַיִן, בְּדִיּוּק כְּפִי שֶׁהַזְּקֵנָה הִזְהִירָה. הַחַיָּל שָׁפַךְ אֶת הַיַּיִן בְּחֶשְׁאִי מִבְּלִי שֶׁתִּרְאֶה, וְהֶעֱמִיד פָּנִים שֶׁהוּא יָשֵׁן שֵׁנָה עֲמֻקָּה, עַד שֶׁהִשְׁמִיעַ נְחִירוֹת חֲזָקוֹת.

בַּחֲצוֹת הַלַּיְלָה שָׁמַע הַחַיָּל אֶת הַנְּסִיכוֹת קָמוֹת בְּחֶשְׁאִי. הַבְּכִירָה נָקְשָׁה עַל מִטָּתָהּ, וְזוֹ נִפְתְּחָה לְפֶתַע וְגִלְּתָה מַדְרֵגוֹת נִסְתָּרוֹת שֶׁיָּרְדוּ עָמֹק בָּאֲדָמָה. שְׁתֵּים עֶשְׂרֵה הַנְּסִיכוֹת לָבְשׁוּ שִׂמְלוֹת נוֹצְצוֹת וְיָרְדוּ בַּמַּדְרֵגוֹת אַחַת אַחֲרֵי הַשְּׁנִיָּה. הַחַיָּל עָטָה מַהֵר אֶת הַגְּלִימָה הַקְּסוּמָה, וְהָלַךְ אַחֲרֵיהֶן בְּלִי שֶׁתַּרְגִּישׁ אַף אַחַת.

הַדֶּרֶךְ עָבְרָה בֵּין שָׁלוֹשׁ חֻרְשׁוֹת קְסוּמוֹת: חֻרְשַׁת עֵצִים עִם עֲלֵי כֶּסֶף, חֻרְשַׁת עֵצִים עִם עֲלֵי זָהָב, וְחֻרְשַׁת עֵצִים עִם עֲלֵי יַהֲלוֹם. הַחַיָּל שָׁבַר בְּכָל חֻרְשָׁה עָנָף אֶחָד כְּהוֹכָחָה, וְכָל שְׁבִירָה הִשְׁמִיעָה קוֹל חַד. הַנְּסִיכָה הַצְּעִירָה בְּיוֹתֵר נִבְהֲלָה בְּכָל פַּעַם וְשָׁאֲלָה "מָה זֶה?", אַךְ הַבְּכִירָה הִרְגִּיעָה אוֹתָהּ: "זֶה רַק הֵד בֵּין הָעֵצִים, אַל תִּדְאֲגִי."

בְּסוֹף הַדֶּרֶךְ הִמְתִּין אֲגַם גָּדוֹל, וְעָלָיו שְׁתֵּים עֶשְׂרֵה סִירוֹת קְטַנּוֹת. בְּכָל סִירָה חִכָּה נָסִיךְ קָסוּם, וְכָל נְסִיכָה עָלְתָה עִם הַנָּסִיךְ שֶׁלָּהּ. הַחַיָּל טִפֵּס בְּחֶשְׁאִי אֶל סִירַת הַנְּסִיכָה הַצְּעִירָה, וְזוֹ תָּמְהָה מַדּוּעַ הַסִּירָה כְּבֵדָה כָּל כָּךְ הַלַּיְלָה, וּמַדּוּעַ הִיא צְרִיכָה לַחְתֹּר בְּכֹחַ רַב יוֹתֵר מִתָּמִיד.

מֵעֵבֶר לָאֲגַם עָמַד אַרְמוֹן מֻקְסָם, מוּאָר בְּאוֹרוֹת רַבִּים וּמָלֵא בְּמוּסִיקָה עַלִּיזָה. שָׁם רָקְדוּ הַנְּסִיכוֹת עִם נְסִיכֵיהֶן הַקְּסוּמִים עַד אוֹר הַבֹּקֶר, בְּלִי לְהַרְגִּישׁ עֲיֵפוּת, עַד שֶׁנַּעֲלֵיהֶן נִקְרְעוּ לְגַמְרֵי מֵרֹב רִקּוּד. הַחַיָּל צָפָה בַּכֹּל מֵהַצַּד, נִסְתָּר בְּתוֹךְ הַגְּלִימָה הַקְּסוּמָה, וְשָׁמַר הֵיטֵב כָּל פְּרָט בְּזִכְרוֹנוֹ.

לִפְנוֹת בֹּקֶר חָזְרוּ הַנְּסִיכוֹת בַּסִּירוֹת, עָלוּ בַּמַּדְרֵגוֹת הַנִּסְתָּרוֹת, וְשָׁכְבוּ לִישֹׁן כְּאִלּוּ שׁוּם דָּבָר לֹא קָרָה. הַחַיָּל חָזַר בְּשֶׁקֶט לְחַדְרוֹ, וּבַבֹּקֶר סִפֵּר לַמֶּלֶךְ עַל כָּל מַה שֶּׁרָאָה - וְהֵנִיחַ עַל הַשֻּׁלְחָן שְׁלוֹשָׁה עֲנָפִים: עֲנַף כֶּסֶף, עֲנַף זָהָב וַעֲנַף יַהֲלוֹם, כְּהוֹכָחָה לְדִבְרֵי אֱמֶת.

הַמֶּלֶךְ קָרָא לִשְׁתֵּים עֶשְׂרֵה בְּנוֹתָיו, וְאֵלֶּה, כְּשֶׁרָאוּ אֶת הָעֲנָפִים עַל הַשֻּׁלְחָן, הֵבִינוּ שֶׁהַסּוֹד נִתְגַּלָּה וְהוֹדוּ בַּכֹּל. הַמֶּלֶךְ שָׂמַח מְאֹד עַל פִּתְרוֹן הַחִידָה, וְאָמַר לַחַיָּל שֶׁיִּבְחַר לוֹ אִשָּׁה מִבֵּין הַנְּסִיכוֹת. הַחַיָּל, שֶׁכְּבָר לֹא הָיָה צָעִיר, בָּחַר בַּבְּכִירָה שֶׁבָּהֶן, וְהִיא חִיְּכָה אֵלָיו בְּחִבָּה. הַמֶּלֶךְ נָתַן לָהֶם אֶת בִּרְכָּתוֹ, וְהַחַיָּל הֶעָנִי הָפַךְ לְיוֹרֵשׁ הַכֶּתֶר.

מֵאָז וָהָלְאָה יָשְׁנוּ שְׁתֵּים עֶשְׂרֵה הַנְּסִיכוֹת שֵׁנָה שְׁקֵטָה כָּל לַיְלָה, וְנַעֲלֵיהֶן נִשְׁאֲרוּ שְׁלֵמוֹת בַּבֹּקֶר. כָּךְ נִפְתַּר סוֹד נַעֲלֵי הָרִקּוּד, בִּזְכוּת אֹמֶץ, סַבְלָנוּת וְעֵינַיִם פְּקוּחוֹת שֶׁל חַיָּל אֶחָד עָנִי וְחָכָם.

מְסֻפָּר עַל־פִּי "שְׁתֵּים עֶשְׂרֵה הַנְּסִיכוֹת הָרוֹקְדוֹת" (Die zertanzten Schuhe), אַגָּדַת עַם מוּכֶּרֶת שֶׁל הָאַחִים גְּרִים (גֶּרְמַנְיָה, פֻּרְסְמָה לָרִאשׁוֹנָה בְּ־1815), מְעֻבֶּדֶת וּמְסֻפֶּרֶת מֵחָדָשׁ עִם סוֹף מְרֻכָּךְ (הַחַיָּל בּוֹחֵר בַּבְּכִירָה בְּהֶסְכֵּמָה הֲדָדִית, בִּמְקוֹם עֹנֶשׁ קָשֶׁה לַנְּסִיכִים שֶׁנִּכְשְׁלוּ כְּבַגִּרְסָה הַמְּקוֹרִית).

📚 דִּקְדּוּק — סְמִיכוּת (חֲזָרָה)

תַּזְכֹּרֶת: מָה זוֹ סְמִיכוּת?

בְּשִׁיעוּר 38 לָמַדְנוּ שֶׁסְּמִיכוּת הִיא צֵרוּף שֶׁל שְׁתֵּי מִלִּים שֶׁיּוֹצֵר יַחַד מַשְׁמָעוּת חֲדָשָׁה, בְּלִי מִלַּת הַקִּשּׁוּר "שֶׁל" - כְּמוֹ "בֵּית סֵפֶר" אוֹ "נַעֲלֵי רִקּוּד". הַמִּלָּה הָרִאשׁוֹנָה נִקְרֵאת נִסְמָךְ, וְהַשְּׁנִיָּה נִקְרֵאת סוֹמֵךְ - וְה' הַיְּדִיעָה מִתְוַסֶּפֶת רַק לַסּוֹמֵךְ (לָכֵן "נַעֲלֵי הָרִקּוּד" וְלֹא "הַנַּעֲלֵי רִקּוּד").
