📖 קְרִיאָה

בְּאָחוּ יָרֹק לְיַד אֲגַם שָׁקֵט יָשְׁבָה בַּרְוָזָה אִמָּא עַל קֵן מָלֵא בֵּיצִים, וְחִכְּתָה בְּסַבְלָנוּת שֶׁהָאֶפְרוֹחִים שֶׁלָּה יִבְקְעוּ.

יוֹם אֶחָד בָּקְעוּ הַבֵּיצִים אַחַת אַחֲרֵי הַשְּׁנִיָּה, וּמִתּוֹכָן יָצְאוּ אֶפְרוֹחִים צְהֻבִּים וַחֲמוּדִים - חוּץ מִבֵּיצָה אַחַת גְּדוֹלָה וּמוּזָרָה, שֶׁנִּשְׁאֲרָה סְגוּרָה עוֹד זְמַן מָה.

לְבַסּוֹף בָּקְעָה גַּם הַבֵּיצָה הַגְּדוֹלָה, וּמִתּוֹכָהּ יָצָא אֶפְרוֹחַ אָפֹר, גָּדוֹל וּמְכֹעָר, שֶׁלֹּא דָּמָה בִּכְלָל לְאֶחָיו הַצְּהֻבִּים.

שְׁאָר הָאֶפְרוֹחִים וְהָאַוָּזִים בֶּחָצֵר צָחֲקוּ עַל הַבַּרְוָזוֹן הַמְּכֹעָר וְלָעֲגוּ לוֹ, וַאֲפִלּוּ אֶחָיו דָּחֲפוּ אוֹתוֹ הַצִּדָּה כָּל הַזְּמַן.

הַבַּרְוָזוֹן הָיָה עָצוּב וּבוֹדֵד מְאֹד, אֲבָל לֹא הֵבִין לָמָּה כֻּלָּם כָּל כָּךְ לֹא אוֹהֲבִים אוֹתוֹ.

יוֹם אֶחָד, כְּשֶׁכְּבָר לֹא יָכוֹל הָיָה לִסְבֹּל אֶת הַלַּעַג, בָּרַח הַבַּרְוָזוֹן מֵהֶחָצֵר וְנָדַד לְבַדּוֹ אֶל הַיַּעַר.

עָבַר עָלָיו חֹרֶף קַר וְקָשֶׁה בַּיַּעַר, וְהַבַּרְוָזוֹן הָיָה רָעֵב, קָפוּא וּבוֹדֵד מִתָּמִיד, אֲבָל הוּא לֹא וִיתֵּר וְהִמְשִׁיךְ לְחַפֵּשׂ מָקוֹם חַם.

כְּשֶׁהִגִּיעַ הָאָבִיב, שָׂחָה הַבַּרְוָזוֹן אֶל אֲגַם גָּדוֹל וּמָצָא בּוֹ בַּרְבּוּרִים לְבָנִים וִיפֵהפִיִּים, וְהִתְבַּיֵּשׁ לְהִתְקָרֵב אֲלֵיהֶם כִּי חָשַׁב שֶׁגַּם הֵם יִלְעֲגוּ לוֹ.

אֲבָל כְּשֶׁהִתְקָרֵב בְּכָל זֹאת, הוּא הִבִּיט בַּמַּיִם וְלֹא הֶאֱמִין לְמָה שֶׁרָאָה: הִשְׁתַּקְּפוּת שֶׁל בַּרְבּוּר צָעִיר וּמַרְהִיב - הוּא בְּעַצְמוֹ!

הִתְבָּרֵר שֶׁהַבַּרְוָזוֹן הַמְּכֹעָר כְּלָל לֹא הָיָה בַּרְוָז, אֶלָּא בֵּיצַת בַּרְבּוּר שֶׁהִתְגַּלְגְּלָה בְּטָעוּת אֶל קֵן הַבַּרְוָזִים.

מֵאָז חַי הַבַּרְבּוּר הַיָּפֶה בְּשִׂמְחָה בֵּין הַבַּרְבּוּרִים הָאֲחֵרִים, וְלָמַד שֶׁלִּפְעָמִים מִי שֶׁנִּרְאֶה שׁוֹנֶה - רַק עֲדַיִן לֹא גָּדַל לִהְיוֹת מִי שֶׁהוּא בֶּאֱמֶת.

מַעֲשִׂיָּה מֻכֶּרֶת מֵאֵת הַנְס כְּרִיסְטְיָאן אַנְדֶּרְסֶן (דֶּנְמַרְק) - מְעֻבֶּדֶת

📚 דִּקְדּוּק — חזרה: משפט פשוט ומשפט מורכב

תזכורת: מהו משפט פשוט?

משפט פשוט מכיל רעיון אחד בלבד: נושא אחד ונשוא אחד. לדוגמה: "הברווזון ברח." או: "האפרוחים צחקו."

תזכורת: מהו משפט מורכב?

משפט מורכב (או: משפט מחובר) נוצר כאשר מחברים שני משפטים פשוטים לתוך משפט אחד, בעזרת מילת חיבור. לדוגמה: "הברווזון היה עצוב, והאווזים לעגו לו." המשפט הזה מורכב משני משפטים פשוטים ("הברווזון היה עצוב" + "האווזים לעגו לו"), שחוברו בעזרת המילה "ו".
