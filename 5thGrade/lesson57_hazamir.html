הַזָּמִיר

שִׁיר אֶחָד אֲמִתִּי שָׁוֶה יוֹתֵר מֵאֶלֶף תַּכְשִׁיטִים נוֹצְצִים

📖 קְרִיאָה

פַּעַם, בְּאַרְמוֹן מְפֹאָר בְּסִין הָרְחוֹקָה, חַי קֵיסָר עָשִׁיר. סְבִיב הָאַרְמוֹן פָּרַשׂ גַּן קָסוּם, וּמֵעֵבֶר לַגַּן הִשְׁתָּרֵעַ יַעַר עָמֹק וְעַד הַיָּם עַצְמוֹ. בְּתוֹךְ הַיַּעַר, בֵּין עַנְפֵי הָעֵצִים, שָׁרָה בְּכָל לַיְלָה צִפּוֹר קְטַנָּה וְלֹא מְיֻחֶדֶת לְמַרְאֶה - הַזָּמִיר. קוֹלוֹ הָיָה כֹּה יָפֶה, שֶׁמְּטַיְּלִים זָרִים שֶׁבִּקְּרוּ בְּסִין כָּתְבוּ בְּסִפְרֵיהֶם: "הַדָּבָר הַיָּפֶה בְּיוֹתֵר בְּכָל הַמַּמְלָכָה הוּא שִׁירַת הַזָּמִיר בַּיַּעַר."

יוֹם אֶחָד קָרָא הַקֵּיסָר אֶחָד מִן הַסְּפָרִים הָאֵלֶּה, וְהִשְׁתּוֹמֵם מְאֹד: "צִפּוֹר כֹּה מְפֻרְסֶמֶת חַיָּה בְּתוֹךְ מַמְלַכְתִּי שֶׁלִּי - וַאֲנִי אֲפִלּוּ לֹא שָׁמַעְתִּי עָלֶיהָ!" הוּא צִוָּה עַל שָׂרָיו לְהָבִיא אֶת הַזָּמִיר לָאַרְמוֹן מִיָּד, אַךְ אַף אֶחָד בֶּחָצֵר לֹא יָדַע הֵיכָן לְחַפֵּשׂ אוֹתוֹ.

לְבַסּוֹף נִמְצְאָה נַעֲרַת מִטְבָּח קְטַנָּה שֶׁאָמְרָה: "אֲנִי מַכִּירָה אֶת הַזָּמִיר הֵיטֵב - הוּא שָׁר לִי בְּכָל עֶרֶב כְּשֶׁאֲנִי חוֹזֶרֶת הַבַּיְתָה. בּוֹאוּ אַחֲרַי, וַאֲנִי אֶקַּח אֶתְכֶם אֵלָיו!" הַשָּׂרִים הָלְכוּ אַחֲרֶיהָ אֶל תּוֹךְ הַיַּעַר הָאָפֵל, וּבַדֶּרֶךְ טָעוּ פַּעֲמַיִם - פַּעַם חָשְׁבוּ שֶׁגְּעִיַּת פָּרָה הִיא שִׁירַת הַזָּמִיר, וּפַעַם חָשְׁבוּ כָּךְ עַל קִרְקוּר צְפַרְדֵּעִים בַּבִּצָּה. לְבַסּוֹף הִצְבִּיעָה הַנַּעֲרָה עַל עָנָף נָמוּךְ: "הִנֵּה הוּא!"

הַזָּמִיר הִסְכִּים לָבוֹא. בָּאַרְמוֹן הֻכַן עֲבוּרוֹ עָנָף זָהָב מְיֻחָד, וְכָל הֶחָצֵר הִתְאַסְּפָה לִשְׁמֹעַ אוֹתוֹ. כְּשֶׁפָּתַח הַזָּמִיר בְּשִׁירָתוֹ, עֵינֵי הַקֵּיסָר עַצְמוֹ הִתְמַלְּאוּ דְּמָעוֹת שֶׁל הִתְרַגְּשׁוּת. מֵאוֹתוֹ יוֹם נִשְׁאַר הַזָּמִיר בָּאַרְמוֹן, גָּר בִּכְלוּב זָהָב מְקֻשָּׁט, וְהֻרְשָׁה לָצֵאת לְטַיֵּל רַק פַּעֲמַיִם בְּיוֹם וּפַעַם אַחַת בַּלַּיְלָה, כְּשֶׁשְּׁנֵים עָשָׂר מְשָׁרְתִים אוֹחֲזִים בְּסֶרֶט מֶשִׁי הַקָּשׁוּר לְרַגְלוֹ.

יוֹם אֶחָד שָׁלַח קֵיסַר יָפָּן לַקֵּיסָר מַתָּנָה מְיֻחֶדֶת: צִפּוֹר מְכָנִית, מְכֻסָּה כֻּלָּהּ זָהָב וָכֶסֶף וּמְקֻשֶּׁטֶת בְּתַכְשִׁיטִים נוֹצְצִים. כְּשֶׁמְּמַלְּאִים אוֹתָהּ, הִיא שָׁרָה שִׁיר אֶחָד יָפֶה, פַּעַם אַחַר פַּעַם, בְּדִיּוּק אוֹתוֹ הַשִּׁיר. כָּל הֶחָצֵר הֶעֱדִיפָה אֶת הַצִּפּוֹר הַחֲדָשָׁה - הִיא הָיְתָה נוֹצֶצֶת וּקְבוּעָה, וְאֶפְשָׁר הָיָה לָדַעַת בְּדִיּוּק אֵיזֶה שִׁיר תָּשִׁיר. "הַזָּמִיר הָאֲמִתִּי אֲפִלּוּ לֹא יוֹדֵעַ אֵיזֶה שִׁיר יָבוֹא לוֹ הַבָּא!" אָמְרוּ הַשָּׂרִים בְּזִלְזוּל.

הַזָּמִיר הָאֲמִתִּי נִפְגַּע עָמֹק. הוּא פָּרַשׂ כְּנָפָיו וְעָף בַּחֲזָרָה אֶל הַיַּעַר הַחָפְשִׁי, וְהֶחָצֵר גֵּרְשָׁה אוֹתוֹ רִשְׁמִית מִן הַמַּמְלָכָה. מֵאוֹתוֹ יוֹם נִגְּנָה הַצִּפּוֹר הַמְּכָנִית בְּכָל עֵת, וְכֻלָּם שִׁבְּחוּ אוֹתָהּ בְּלִי הֶפְסֵק.

לַיְלָה אֶחָד, בְּעֵת נְגִינָה, נִשְׁמַע פִּתְאוֹם קוֹל "קְרָץ!" מִתּוֹךְ הַצִּפּוֹר הַמְּכָנִית - אַחַד הַגַּלְגַּלִּים הַקְּטַנִּים בִּפְנִים נִשְׁבַּר. שָׁעָן מֻמְחֶה תִּקֵּן אוֹתָהּ, אַךְ הִזְהִיר: "מֵעַתָּה מֻתָּר לְנַגֵּן אוֹתָהּ רַק פַּעַם בַּשָּׁנָה, וּבִזְהִירוּת רַבָּה." חָמֵשׁ שָׁנִים חָלְפוּ, וְהַקֵּיסָר חָלָה מְאֹד. הָרוֹפְאִים חָשְׁבוּ שֶׁלֹּא יַחְלִים, וּבֶחָצֵר כְּבָר בָּחֲרוּ קֵיסָר חָדָשׁ בִּמְקוֹמוֹ.

הַקֵּיסָר שָׁכַב חִוֵּר וְקַר בְּמִטָּתוֹ, וְעַל חָזֵהוּ יָשַׁב הַמָּוֶת בְּעַצְמוֹ, חוֹבֵשׁ אֶת כִּתְרוֹ שֶׁל הַקֵּיסָר. "מוּסִיקָה!" לָחַשׁ הַקֵּיסָר בְּקוֹל חַלָּשׁ. "הָבִיאוּ לִי אֶת הַצִּפּוֹר הַמְּכָנִית, שֶׁתָּשִׁיר וְתַרְחִיק מִמֶּנִּי אֶת הַמָּוֶת!" אֲבָל אַף אֶחָד לֹא הָיָה שָׁם לְמַלֵּא אוֹתָהּ - כָּל הַמְּשָׁרְתִים כְּבָר עָבְרוּ לְבָרֵךְ אֶת הַקֵּיסָר הֶחָדָשׁ.

פִּתְאוֹם, מִבַּעַד לַחַלּוֹן, נִשְׁמַע קוֹל שִׁירָה מֻכָּר וּמַקְסִים. זֶה הָיָה הַזָּמִיר הָאֲמִתִּי! הוּא שָׁמַע שֶׁהַקֵּיסָר חוֹלֶה, וְעָף כָּל הַדֶּרֶךְ מִן הַיַּעַר כְּדֵי לָשִׁיר לוֹ. הַזָּמִיר שָׁר עַל הַגַּן הַשָּׁקֵט, עַל הַלַּיְלָה הַכּוֹכָבִי וְעַל יֹפִי הַחַיִּים - וְהַמָּוֶת, שֶׁשָּׁמַע אֶת הַשִּׁיר, הִתְרַגֵּשׁ כָּל כָּךְ עַד שֶׁקָּם וְעָזַב דֶּרֶךְ הַחַלּוֹן. הַקֵּיסָר נִרְדַּם שֵׁנָה עֲמֻקָּה וּמַרְפֵּאת, וּכְשֶׁהִתְעוֹרֵר בַּבֹּקֶר הוּא הָיָה בָּרִיא לְגַמְרֵי - הוּא הֶחֱלִים!

"הִצַּלְתָּ אֶת חַיַּי," אָמַר הַקֵּיסָר לַזָּמִיר בְּהַכָּרַת תּוֹדָה. "בַּקֵּשׁ מִמֶּנִּי כָּל מַה שֶּׁתִּרְצֶה!" "כְּבָר קִבַּלְתִּי אֶת הַשָּׂכָר הַכִּי יָקָר," עָנָה הַזָּמִיר, "רָאִיתִי דְּמָעוֹת בְּעֵינֶיךָ כְּשֶׁשַּׁרְתִּי לְךָ בַּפַּעַם הָרִאשׁוֹנָה. אֲבָל אַל תַּחְזִיר אוֹתִי לִכְלוּב זָהָב. תֵּן לִי לָעוּף חָפְשִׁי בַּיַּעַר, וּבְכָל עֶרֶב אָבוֹא לְבַד לָשִׁיר לְךָ עַל הַטּוֹב וְעַל הָרַע בְּמַמְלַכְתְּךָ - כִּי צִפּוֹר שֶׁטָּסָה בָּעוֹלָם רוֹאָה יוֹתֵר מִקֵּיסָר בְּתוֹךְ אַרְמוֹנוֹ." הַקֵּיסָר הִסְכִּים בְּשִׂמְחָה, וּכְשֶׁהַמְּשָׁרְתִים נִכְנְסוּ בַּבֹּקֶר, בְּטוּחִים שֶׁיִּמְצְאוּ אֶת הַקֵּיסָר מֵת, הֵם מָצְאוּ אוֹתוֹ עוֹמֵד וְאוֹמֵר בְּחִיּוּךְ: "בֹּקֶר טוֹב!"

מְסֻפָּר עַל־פִּי "הַזָּמִיר" (The Nightingale) מֵאֵת הַנְס כְּרִיסְטְיָאן אַנְדֶּרְסֶן (דֶּנְמַרְק, 1843), מְעֻבָּד.

מָה זֶה כִּנּוּי חָבוּר עִם "אֶת"?

כְּשֶׁפֹּעַל פּוֹעֵל יְשִׁירוֹת עַל מִישֶׁהוּ אוֹ מַשֶּׁהוּ, קוֹרְאִים לוֹ "מוֹשָׂא יָשִׁיר", וּמְצָרְפִים לוֹ אֶת הַמִּלָּה "אֶת": "רָאִיתִי אֶת הַזָּמִיר." כְּשֶׁרוֹצִים לְהַחְלִיף אֶת הַשֵּׁם בְּכִנּוּי גוּף, הַמִּלָּה "אֶת" מְקַבֶּלֶת סִיּוֹמֶת שֶׁל כִּנּוּי הַגּוּף: "רָאִיתִי אוֹתוֹ" (לֹא "רָאִיתִי אֶת הוּא").
