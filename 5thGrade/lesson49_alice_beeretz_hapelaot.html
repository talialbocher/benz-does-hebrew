אֵלִיס בְּאֶרֶץ הַפְּלָאוֹת

אֵלִיס עוֹקֶבֶת אַחֲרֵי אַרְנָב לָבָן לְתוֹךְ חוֹר עָמֹק, נוֹפֶלֶת, מִצְטַמֶּקֶת וּגְדֵלָה - וְחוֹזְרִים לְתַרְגֵּל אוֹתִיּוֹת סוֹפִיּוֹת, מנצפ"ך

📖 קְרִיאָה

אֵלִיס יָשְׁבָה עַל שְׂפַת הַנָּהָר עִם אֲחוֹתָהּ הַגְּדוֹלָה, בְּיוֹם קַיִץ חַם וְעַצְלָנִי, וְהִתְחִילָה לְהִשְׁתַּעֲמֵם. פִּתְאוֹם רָץ לְיָדָהּ אַרְנָב לָבָן עִם עֵינַיִם וְרֻדּוֹת, מוֹצִיא שָׁעוֹן מִכִּיס אֲפֻדָּתוֹ וְקוֹרֵא בְּקוֹל: "אוֹי, אוֹי, אֵאַחֵר!" אֵלִיס, סַקְרָנִית מְאֹד, קָמָה וְרָצָה אַחֲרָיו.

הָאַרְנָב נֶעֱלַם לְתוֹךְ חוֹר עָמֹק מִתַּחַת לְגָדֵר שִׂיחִים, וּבְלִי לַחְשֹׁב פַּעֲמַיִם קָפְצָה אֵלִיס אַחֲרָיו. הַחוֹר יָרַד וְיָרַד, כְּמוֹ בְּאֵר עֲמֻקָּה מְאֹד - וְאֵלִיס נָפְלָה לְאִטָּהּ בְּתוֹךְ מִסְדְּרוֹן אָרֹךְ, שֶׁעַל קִירוֹתָיו הִצְטַיְּרוּ אֲרוֹנוֹת סְפָרִים, מַדָּפִים עִם צִנְצָנוֹת רִיבָּה, וּמַפּוֹת עַל הַקִּיר. הָיָה לָהּ אֲפִלּוּ זְמַן לְהַבִּיט סָבִיב בְּעֵת הַנְּפִילָה הָאִטִּית.

לְבַסּוֹף נָחֲתָה אֵלִיס בְּרַכּוּת עַל עֲרֵמַת עָלִים וְזַרְדִּים, בְּלִי שֶׁנֶּחְבְּלָה כְּלָל. הָאַרְנָב הַלָּבָן כְּבָר נֶעֱלַם בְּמַעֲבָר צַר וְרָחוֹק. אֵלִיס מָצְאָה אֶת עַצְמָהּ בְּאוּלָם אָרֹךְ, שֶׁלְּאֹרֶךְ קִירוֹתָיו הִשְׂתָּרְכוּ דְּלָתוֹת רַבּוֹת - וְכֻלָּן נְעוּלוֹת. בְּאֶמְצַע הָאוּלָם עָמַד שֻׁלְחָן קָטָן עָשׂוּי זְכוּכִית, וְעָלָיו מַפְתֵּחַ זָהָב זָעִיר.

מֵאֲחוֹרֵי וִילוֹן נָמוּךְ מָצְאָה אֵלִיס דֶּלֶת קְטַנְטַנָּה, לֹא גְּבוֹהָה מֵאַרְבָּעִים סַנְטִימֶטֶר. הַמַּפְתֵּחַ הִתְאִים לַמַּנְעוּל בְּדִיּוּק, אַךְ אֵלִיס הָיְתָה גְּדוֹלָה מִדַּי לְהִכָּנֵס. דֶּרֶךְ הַדֶּלֶת הַקְּטַנָּה רָאֲתָה גַּן יָפֶהפֶה - עֲרוּגוֹת פְּרָחִים בְּצִבְעֵי הַקֶּשֶׁת וּמִזְרָקוֹת מַיִם קְרִירוֹת - וְנִכְסְפָה בְּכָל לִבָּהּ לְהַגִּיעַ לְשָׁם.

עַל הַשֻּׁלְחָן הַקָּטָן הוֹפִיעַ פִּתְאוֹם בַּקְבּוּק זָעִיר, שֶׁלֹּא הָיָה שָׁם קֹדֶם, וְעָלָיו כָּתוּב בְּאוֹתִיּוֹת יָפוֹת: "שְׁתִי אוֹתִי". אֵלִיס בָּדְקָה קֹדֶם שֶׁלֹּא כָּתוּב עָלָיו "רַעַל", וְאָז טָעֲמָה קְצָת. הַטַּעַם הָיָה נִפְלָא - כְּמוֹ עוּגַת דֻּבְדְּבָנִים, קֶצֶפֶת, אֲנָנָס וְטוֹסְט חַם בְּיַחַד - וּפִתְאוֹם הִתְחִילָה אֵלִיס לְהִתְכַּוֵּץ, עַד שֶׁנִּשְׁאֲרָה גְּבוֹהָה כְּעֶשְׂרִים וַחֲמִשָּׁה סַנְטִימֶטֶר בִּלְבַד!

עַכְשָׁו הָיְתָה אֵלִיס בְּדִיּוּק בַּגֹּדֶל הַנָּכוֹן לַדֶּלֶת הַקְּטַנְטַנָּה - אֲבָל הַמַּפְתֵּחַ נִשְׁאַר לְמַעְלָה עַל הַשֻּׁלְחָן, וְעַכְשָׁו הַשֻּׁלְחָן הָיָה גָּבֹהַּ מִדַּי בִּשְׁבִילָהּ. אֵלִיס יָשְׁבָה עַל הָרִצְפָּה וּבָכְתָה מֵרֹב תִּסְכּוּל.

לְפֶתַע שָׂמָה לֵב לְעוּגִיָּה קְטַנָּה מִתַּחַת לַשֻּׁלְחָן, וְעָלֶיהָ כָּתוּב: "אִכְלִי אוֹתִי". אֵלִיס אָכְלָה חֲתִיכָה, בְּתִקְוָה שֶׁתִּגְדַּל וְתוּכַל לְהַגִּיעַ אֶל הַמַּפְתֵּחַ. וְאָכֵן - הִיא גָּדְלָה, אֲבָל הַרְבֵּה יוֹתֵר מִדַּי! רֹאשָׁהּ נָגַע בַּתִּקְרָה, וְצַוָּארָהּ הִתְאָרֵךְ כְּמוֹ שֶׁל גִּ'ירָפָה.

עַכְשָׁו הָיְתָה אֵלִיס עֲנָקִית וּתְקוּעָה בָּאוּלָם, וְהִיא פָּרְצָה בִּבְכִי כָּבֵד. דְּמָעוֹת עֲנָקִיּוֹת נָפְלוּ כְּמוֹ גֶּשֶׁם וְיָצְרוּ בְּרֵכָה גְּדוֹלָה עַל הָרִצְפָּה. לְמַזָּלָהּ, מָצְאָה שָׁם מְנִיפָה שֶׁהָאַרְנָב הִשְׁאִיר מֵאָחוֹר. כְּשֶׁהֵנִיפָה אוֹתָהּ, הִתְחִילָה שׁוּב לְהִתְכַּוֵּץ - וְהִתְכַּוְּצָה כָּל כָּךְ, שֶׁלְּבַסּוֹף הֶחֱלִיקָה וְנָפְלָה יָשָׁר לְתוֹךְ יָם הַדְּמָעוֹת שֶׁל עַצְמָהּ!

בְּתוֹךְ הַיָּם הַקָּטָן שָׂחֲתָה לְיָדָהּ עַכְבָּרָה מְנֻמֶּסֶת, וּשְׁתֵּיהֶן שָׂחוּ יַחַד לְעֵבֶר הַחוֹף - שָׁם, מֵעֵבֶר לַדֶּלֶת הַקְּטַנְטַנָּה, חִכּוּ לָהּ הַגַּן הַיָּפֶה וְהָעֵצִים הַיְרֻקִּים. וְזוֹ הָיְתָה רַק הַתְחָלַת הַהַרְפַּתְקָה שֶׁל אֵלִיס בְּאֶרֶץ הַפְּלָאוֹת.

מְסֻפָּר עַל־פִּי הַסֵּפֶר "הַרְפַּתְקְאוֹת אֵלִיס בְּאֶרֶץ הַפְּלָאוֹת" מֵאֵת לוּאִיס קֶרוֹל (אַנְגְּלִיָּה, 1865) - מְעֻבָּד
