📖 קְרִיאָה

לִפְנֵי מֵאוֹת רַבּוֹת שֶׁל שָׁנִים, בְּעִיר קְטַנָּה בְּגֶרְמַנְיָה בְּשֵׁם הָאַמֶּלִּין, שֶׁשָּׁכְנָה עַל גְּדוֹת נָהָר רָחָב, הִתְרַבּוּ בְּבַת אַחַת אֲלָפֵי עַכְבְּרוֹשִׁים. הֵם רָצוּ בְּכָל רְחוֹב וְסִמְטָה, נִכְנְסוּ לְכָל בַּיִת וּמַחְסָן, וְאָכְלוּ אֶת כָּל הַתְּבוּאָה שֶׁהָיְתָה בָּעִיר.

תּוֹשָׁבֵי הָעִיר נִסּוּ הַכֹּל - חֲתוּלִים, מַלְכֹּדוֹת וּמַקְלוֹת - אֲבָל הָעַכְבְּרוֹשִׁים הָיוּ רַבִּים וְעַרְמוּמִיִּים מִדַּי. לְבַסּוֹף פָּנוּ אֶל רֹאשׁ הָעִירִיָּה בְּבַקָּשָׁה נוֹאָשֶׁת: "עֲשֵׂה מַשֶּׁהוּ, אַחֶרֶת נֵאָלֵץ לַעֲזֹב אֶת הָעִיר!" רֹאשׁ הָעִירִיָּה הִבְטִיחַ שָׂכָר גָּדוֹל בְּזָהָב לְכָל מִי שֶׁיַּצְלִיחַ לְגָרֵשׁ אֶת הָעַכְבְּרוֹשִׁים.

בְּאוֹתוֹ יוֹם עַצְמוֹ הוֹפִיעַ בָּעִיר אִישׁ זָר וְגָבוֹהַּ, לָבוּשׁ בִּגְדֵי טְלָאִים צִבְעוֹנִיִּים, וְנוֹשֵׂא בְּיָדוֹ חָלִיל עֵץ מְלֻטָּשׁ. "אֲנִי הַחַלִּילָן," אָמַר בְּחִיּוּךְ, "וַאֲנִי יוֹדֵעַ לְגָרֵשׁ עַכְבְּרוֹשִׁים בְּעֶזְרַת נִגּוּן בִּלְבַד. תְּשַׁלְּמוּ לִי אֶת הַשָּׂכָר שֶׁהֻבְטַח, וְהָעִיר תְּהֵא נְקִיָּה עוֹד הַלַּיְלָה." רֹאשׁ הָעִירִיָּה הֵשִׁיב בְּשִׂמְחָה: "כֵּן! כֵּן! רַק תַּעֲשֶׂה זֹאת!"

הֶחַלִּילָן צָעַד אֶל כִּכַּר הָעִיר וְהֵרִים אֶת הֶחָלִיל אֶל שְׂפָתָיו. הוּא נִגֵּן נִגּוּן פְּלָאִים - מָתוֹק וּמוּזָר בְּעֵת וּבְעוֹנָה אַחַת - וְהָעַכְבְּרוֹשִׁים, כְּאִלּוּ הוּקְסְמוּ, פָּרְצוּ מִכָּל חֹר וּמַחְסָן וְנֶאֶסְפוּ מֵאֲחוֹרָיו. הֶחַלִּילָן הוֹלִיךְ אֶת הָעַכְבְּרוֹשִׁים בִּרְחוֹבוֹת הָעִיר, צָעַד וְנִגֵּן, עַד אֶל הַנָּהָר הָרָחָב.

כְּשֶׁהִגִּיעוּ אֶל שְׂפַת הַנָּהָר, קָפְצוּ הָעַכְבְּרוֹשִׁים אֶחָד אַחֲרֵי הַשֵּׁנִי לְתוֹךְ הַמַּיִם הַזּוֹרְמִים וְטָבְעוּ. הֶחַלִּילָן הִפְסִיק לְנַגֵּן, הִסְתּוֹבֵב, וְחָזַר אֶל כִּכַּר הָעִיר לְקַבֵּל אֶת הַשָּׂכָר שֶׁהֻבְטַח לוֹ.

אֲבָל רֹאשׁ הָעִירִיָּה וְחַבְרֵי הַמּוֹעָצָה, שֶׁרָאוּ שֶׁהַסַּכָּנָה חָלְפָה, הִצְטַעֲרוּ פִּתְאוֹם עַל הַסְּכוּם הַגָּדוֹל שֶׁהִבְטִיחוּ. "עַכְבְּרוֹשִׁים בּוֹרְחִים מֵעִיר מְתֻקָּנָה בְּלָאו הָכֵי," אָמַר אֶחָד הַחֲבֵרִים, "לָמָּה שֶׁנְּשַׁלֵּם עַל כֵּן שָׂכָר כָּזֶה?" הֵם הֵפֵרוּ אֶת הַבְטָחָתָם וְשִׁלְּמוּ לַחַלִּילָן רַק חֶלְקִיק קָטָן מֵהַסְּכוּם.

הֶחַלִּילָן הִבִּיט בָּהֶם בְּעֵינַיִם קָרוֹת. "הֶסְכֵּם הוּא הֶסְכֵּם," אָמַר בְּשֶׁקֶט, "וּמִי שֶׁמֵּפֵר הַבְטָחָה - סוֹפוֹ לְשַׁלֵּם מְחִיר אַחֵר." בְּלִי מִלָּה נוֹסֶפֶת, הוּא עָזַב אֶת הָעִיר.

בְּיוֹם חַג, כְּשֶׁכָּל הַמְּבוּגָּרִים הָיוּ בְּבֵית הַתְּפִלָּה, שָׁב הֶחַלִּילָן אֶל הָעִיר - הַפַּעַם לָבוּשׁ בְּגָדִים אֲחֵרִים, וּמְנַגֵּן נִגּוּן שׁוֹנֶה, עַלִּיז וְקַל רֶגֶל. הַיְּלָדִים שֶׁשִּׁחֲקוּ בָּרְחוֹבוֹת שָׁמְעוּ אֶת הַנִּגּוּן, יָצְאוּ מִבֵּיתָם וְרָקְדוּ אַחֲרָיו בְּעֵינַיִם קְסוּמוֹת, כְּמוֹ הָעַכְבְּרוֹשִׁים לִפְנֵי כֵן. הֶחַלִּילָן הוֹבִיל אֶת הַיְּלָדִים אֶל שַׁעַר הָעִיר וּמִשָּׁם אֶל עֵבֶר הָהָר הַגָּבוֹהַ שֶׁמִּחוּץ לָעִיר. רַק יֶלֶד אֶחָד צוֹלֵעַ, שֶׁלֹּא הִצְלִיחַ לְהַדְבִּיק אֶת קֶצֶב הַנִּגּוּן, נִשְׁאַר מֵאָחוֹר וְצָפָה בַּמַּתְרַחֵשׁ בְּלֵב הוֹלֵם.

הַיֶּלֶד הַצּוֹלֵעַ רָץ בְּכָל כֹּחוֹ אֶל בֵּית הַתְּפִלָּה וְצָעַק בְּקוֹל: "הֶחַלִּילָן חוֹזֵר! הוּא לוֹקֵחַ אֶת כָּל הַיְּלָדִים!" הַתּוֹשָׁבִים פָּרְצוּ מִבֵּית הַתְּפִלָּה בְּבֶהָלָה. רֹאשׁ הָעִירִיָּה, אָחוּז חֲרָטָה, תָּפַס בִּמְהִירוּת אֶת שַׂק הַזָּהָב וְרָץ אַחֲרֵי הַשַּׁיָּרָה, צוֹעֵק: "עֲמֹד! עֲמֹד! נְשַׁלֵּם לְךָ אֶת מְלוֹא הַסְּכוּם שֶׁהֻבְטַח - סְלַח לָנוּ!"

הֶחַלִּילָן עָמַד בְּדִיּוּק בְּפֶתַח הָהָר, בְּטֶרֶם נִכְנְסוּ הַיְּלָדִים לְתוֹכוֹ. הוּא הִפְסִיק לְנַגֵּן, וְהַיְּלָדִים נֶעֶצְרוּ בְּבַת אַחַת, כְּאִלּוּ הִתְעוֹרְרוּ מֵחֲלוֹם. הוּא סָפַר אֶת מַטְבְּעוֹת הַזָּהָב בְּעֵינֵי בָּדִיקָה, הִנְהֵן, וְאָמַר: "עַכְשָׁו שֻׁלַּם לִי מַה שֶּׁהֻבְטַח - וַאֲנִי מַחְזִיר לָכֶם אֶת מַה שֶּׁשֶּׁלָּכֶם." בְּחִיּוּךְ קַל, נָפְנֵף בְּיָדוֹ, וְהַיְּלָדִים רָצוּ בְּשִׂמְחָה אֶל הוֹרֵיהֶם הַמְּחַבְּקִים.

מֵאוֹתוֹ יוֹם וָאֵילָךְ קִבְּלוּ עַל עַצְמָם תּוֹשָׁבֵי הָאַמֶּלִּין עֵצָה חֲדָשָׁה: הֶסְכֵּם הוּא הֶסְכֵּם, וְחוֹב שֶׁמַּבְטִיחִים לְשַׁלֵּם - יֵשׁ לְשַׁלֵּם עַד הַסּוֹף. בְּבֵית הָעִירִיָּה תָּלוּ תְּמוּנָה שֶׁל הֶחַלִּילָן בְּבִגְדָיו הַצִּבְעוֹנִיִּים, כְּתַזְכִּיר לַדּוֹרוֹת הַבָּאִים.

מְסֻפָּר עַל־פִּי אַגָּדָה גֶּרְמָנִית עַתִּיקָה עַל חַלִּילָן הָאַמֶּלִּין (Der Rattenfänger von Hameln), הַמֻּכֶּרֶת גַּם מֵאוֹסֶף "אַגָּדוֹת גֶּרְמָנִיָּה" שֶׁל הָאַחִים גְּרִים (גֶּרְמַנְיָה, 1816), מְעֻבֶּדֶת וּמְסֻפֶּרֶת מֵחָדָשׁ עִם סוֹף מְרֻכָּךְ.

בְּשִׁיעוּר 71 לָמַדְנוּ שֶׁמּוּשָׂא יָשִׁיר הוּא הַמִּלָּה שֶׁ"מְקַבֶּלֶת" אֶת הַפְּעֻלָּה שֶׁל הַפֹּעַל - עִם "אֶת" אוֹ בְּלִי. הַיּוֹם נִלְמַד שֶׁלֹּא לְכָל פֹּעַל יֵשׁ מוּשָׂא יָשִׁיר: יֵשׁ פְּעָלִים שֶׁ"יוֹצְאִים" אֶל מִישֶׁהוּ אוֹ מַשֶּׁהוּ, וְיֵשׁ פְּעָלִים שֶׁהַפְּעֻלָּה נִשְׁאֶרֶת אֵצֶל מִי שֶׁעוֹשֶׂה אוֹתָהּ.

טָבַעטָבְעוּ מָה? - אֵין תְּשׁוּבָה"הָעַכְבְּרוֹשִׁים טָבְעוּ"
רָקַדרָקְדוּ מָה? - אֵין תְּשׁוּבָה"הַיְּלָדִים רָקְדוּ אַחֲרָיו"
🔑 כְּלָל זָהָב: שׁוֹאֲלִים "מָה?" אוֹ "אֶת מִי?" אַחֲרֵי הַפֹּעַל. אִם יֵשׁ תְּשׁוּבָה טִבְעִית וּמְלֵאָה - זֶה פֹּעַל יוֹצֵא, וְהַתְּשׁוּבָה הִיא הַמּוּשָׂא הַיָּשִׁיר. אִם אֵין תְּשׁוּבָה - זֶה פֹּעַל עוֹמֵד.
💡 יֵשׁ פְּעָלִים שֶׁיּוֹדְעִים "לְהִתְחַפֵּשׂ" לִשְׁנֵי הַסּוּגִים, לְפִי הַמִּשְׁפָּט: "הֶחַלִּילָן נִגֵּן" (בְּלִי מוּשָׂא - פֹּעַל עוֹמֵד בַּמִּשְׁפָּט הַזֶּה) לְעֻמַּת "הֶחַלִּילָן נִגֵּן נִגּוּן פְּלָאִים" (עִם מוּשָׂא "נִגּוּן פְּלָאִים" - כָּאן הוּא פֹּעַל יוֹצֵא). הַבְּדִיקָה תָּמִיד עַל הַמִּשְׁפָּט הַשָּׁלֵם, לֹא רַק עַל הַפֹּעַל בְּעַצְמוֹ.
יוֹצֵא: הוֹלִיךְ אֶת... יוֹצֵא: הִבְטִיחַ... עוֹמֵד: צָעַד עוֹמֵד: טָבַע עוֹמֵד: רָקַד

      כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים לְפִי הַהַנְחָיוֹת בְּכָל שׁוּרָה. בְּכָל מִשְׁפָּט הַשְׁתַּמְּשׁוּ בְּפֹעַל אֶחָד, וְנַסּוּ לְהַחְלִיט אִם הוּא פֹּעַל עוֹמֵד (בְּלִי מוּשָׂא יָשִׁיר) אוֹ פֹּעַל יוֹצֵא (עִם מוּשָׂא יָשִׁיר) - בְּדִיּוּק כְּמוֹ שֶׁלָּמַדְנוּ עַל הֶחַלִּילָן וְהָעַכְבְּרוֹשִׁים.
