💡 לְחַץ עַל הַמִּלִּים הַמֻּדְגָּשׁוֹת כְּדֵי לִרְאוֹת מָה הֵן אוֹמְרוֹת
פֶּרֶק א׳ · הַגֶּשֶׁר וְהַטְּרוֹל

לְמַרְגְּלוֹת הַר גָּבֹהַ וְיָרֹק גָּרוּ שְׁלֹשָׁה תְּיָשִׁים מִמִּשְׁפַּחַת גְּרוּף - הַקָּטָן, הַבֵּינוֹנִי וְהַגָּדוֹל. בְּכָל בֹּקֶר הֵם הִבִּיטוּ בְּעֵינַיִם רְעֵבוֹת אֶל הַצַּד הַשֵּׁנִי שֶׁל הַנַּחַל, שָׁם הִשְׁתָּרֵעַ מִדְרוֹן מָלֵא בְּדֶּשֶׁא יָרֹק וְעָסִיסִי.

כְּדֵי לְהַגִּיעַ אֶל הַמִּדְרוֹן הַיָּרֹק, הָיָה צָרִיךְ לַחֲצוֹת גֶּשֶׁר עֵץ יָשָׁן מֵעַל הַנַּחַל. וּמִתַּחַת לַגֶּשֶׁר, בֵּין הָאֲבָנִים הָרְטֻבּוֹת, גָּר טְרוֹל עֲנָק, בַּעַל עֵינַיִם צְהֻבּוֹת וְאַף אָרֹךְ, שֶׁטָּרַף כָּל מִי שֶׁהֵעֵז לַעֲבֹר עַל הַגֶּשֶׁר שֶׁלּוֹ.

כָּל בַּעֲלֵי הַחַיִּים בַּסְּבִיבָה יָדְעוּ עַל הַטְּרוֹל הַמְּסֻכָּן, וּפָחֲדוּ לְהִתְקָרֵב אֶל הַגֶּשֶׁר. אֲבָל שְׁלֹשֶׁת הַתְּיָשִׁים הָיוּ רְעֵבִים מִדַּי כְּדֵי לְהִשָּׁאֵר בַּבַּיִת.

פֶּרֶק ב׳ · הַקָּטָן וְהַבֵּינוֹנִי

בְּבֹקֶר אֶחָד הֶחְלִיט הַתַּיִשׁ הַקָּטָן שֶׁהוּא כְּבָר לֹא יָכוֹל לִסְבֹּל אֶת הָרָעָב, וְיָצָא לְבַדּוֹ אֶל הַגֶּשֶׁר. טְרַפּ-טְרַפּ, טְרַפּ-טְרַפּ, נִשְׁמְעוּ פַּרְסוֹתָיו הַקְּטַנּוֹת עַל קוֹרוֹת הָעֵץ.

"מִי זֶה שֶׁהוֹלֵךְ עַל הַגֶּשֶׁר שֶׁלִּי?" שָׁאַג הַטְּרוֹל וְקָפַץ מִתַּחַת לַגֶּשֶׁר. "עַכְשָׁו אֲנִי הוֹלֵךְ לִבְלֹעַ אוֹתְךָ!" "אַל תֹּאכַל אוֹתִי," עָנָה הַתַּיִשׁ הַקָּטָן בְּקוֹל רוֹעֵד, "חַכֵּה לְאָחִי הַבֵּינוֹנִי - הוּא גָּדוֹל מִמֶּנִּי, וְיִהְיֶה לְךָ אֹכֶל טוֹב יוֹתֵר." הַטְּרוֹל הָרָעֵב חָשַׁב רֶגַע, וְהִסְכִּים לְחַכּוֹת.

כַּעֲבֹר שָׁעָה יָצָא גַּם הַתַּיִשׁ הַבֵּינוֹנִי אֶל הַגֶּשֶׁר. טְרַפּ-טְרַפּ, טְרַפּ-טְרַפּ. "מִי זֶה שֶׁהוֹלֵךְ עַל הַגֶּשֶׁר שֶׁלִּי?" שָׁאַג הַטְּרוֹל שׁוּב, רָעֵב עוֹד יוֹתֵר. גַּם הַתַּיִשׁ הַבֵּינוֹנִי הִצְלִיחַ לְשַׁכְנֵעַ אוֹתוֹ לְחַכּוֹת לְאָחִיו הַגָּדוֹל, הַשָּׁמֵן וְהֶחָזָק בַּמִּשְׁפָּחָה.

פֶּרֶק ג׳ · הַתַּיִשׁ הַגָּדוֹל

לְבַסּוֹף עָלָה עַל הַגֶּשֶׁר גַּם הַתַּיִשׁ הַגָּדוֹל, וְצַעֲדָיו הִרְעִידוּ אֶת קוֹרוֹת הָעֵץ: בּוּם-בּוּם, בּוּם-בּוּם. הַטְּרוֹל קָפַץ מִתַּחַת לַגֶּשֶׁר בְּזַעַם. "עַכְשָׁו אֲנִי הוֹלֵךְ לִבְלֹעַ אוֹתְךָ!" צָעַק.

אֲבָל הַתַּיִשׁ הַגָּדוֹל לֹא פָּחַד כְּלָל. הוּא הִרְכִּין אֶת רֹאשׁוֹ, הֵנִיף אֶת קַרְנָיו הַחַדּוֹת, וּבְכֹחַ רַב הֵחֵל לִנְגֹּחַ בַּטְּרוֹל שׁוּב וְשׁוּב, עַד שֶׁהֵטִיל אוֹתוֹ אֶל תּוֹךְ מֵי הַנַּחַל הַקָּרִים. הַטְּרוֹל נִסְחַף בַּזֶּרֶם וְנֶעֱלַם, וְאִישׁ לֹא רָאָה אוֹתוֹ שׁוּב.

פֶּרֶק ד׳ · הַמִּדְרוֹן הַיָּרֹק

שְׁלֹשֶׁת הַתְּיָשִׁים הִגִּיעוּ סוֹף סוֹף אֶל הַמִּדְרוֹן הַיָּרֹק, וְאָכְלוּ שָׁם דֶּשֶׁא טָרִי כָּל הַיּוֹם, עַד שֶׁהִשְׁמִינוּ וְנַעֲשׂוּ שְׁמֵנִים וּמְאֻשָּׁרִים.

מֵאָז לִמֵּד הַתַּיִשׁ הַקָּטָן אֶת כֻּלָּם שֶׁגַּם מִי שֶׁחַלָּשׁ יָכוֹל לְהִנָּצֵל בְּעֶזְרַת מִלִּים חֲכָמוֹת, וְהַתַּיִשׁ הַגָּדוֹל הוֹכִיחַ שֶׁלִּפְעָמִים צָרִיךְ פָּשׁוּט אֹמֶץ וְכֹחַ. כָּל אֶחָד מֵהֶם תָּרַם מַשֶּׁהוּ אַחֵר לְהַצְלָחַת הַמִּשְׁפָּחָה.

מְסֻפָּר עַל־פִּי אַגָּדַת עַם נוֹרְבֶגִית שֶׁאָסְפוּ פֶּטֶר כְּרִיסְטֶן אַסְבְּיֶרְנְסֶן וְיוֹרְגֶן מוֹ (הַמֵּאָה הַ-19) - "שְׁלֹשֶׁת הַתְּיָשִׁים גְּרוּף" - מְעֻבֶּדֶת
