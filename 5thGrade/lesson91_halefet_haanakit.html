📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת, בָּאָבִיב, בִּכְפָר קָטָן שֶׁעַל שְׂפַת הַיַּעַר, גָּרוּ סָבָא וְסָבְתָּא. הָיְתָה לָהֶם גִּנָּה גְּדוֹלָה, וְאִתָּם גָּרוּ נֶכְדָּתָם הַקְּטַנָּה, כֶּלֶב נֶאֱמָן וְחָתוּל אָפֹר. בָּאֶסֶם הַקָּרוֹב גָּר עַכְבָּר קְטַנְטָן, שֶׁכִּמְעַט אַף אֶחָד לֹא הִבְחִין בּוֹ.

בְּאוֹתוֹ אָבִיב הִטְמִין הַסָּבָא בָּאֲדָמָה זֶרַע קָטָן שֶׁל לֶפֶת וְלָחַשׁ: "גִּדְלִי, גִּדְלִי, לֶפֶת מְתוּקָה! גִּדְלִי חֲזָקָה וּגְדוֹלָה!" הַקַּיִץ עָבַר, הַסָּבָא הִשְׁקָה אֶת הַגִּנָּה בְּכָל בֹּקֶר, וְהַלֶּפֶת גָּדְלָה וְגָדְלָה.

בַּסְּתָו יָצָא הַסָּבָא לַגִּנָּה כְּדֵי לַעֲקֹר אֶת הַלֶּפֶת, וְנִדְהַם. הַלֶּפֶת הָיְתָה עֲנָקִית, גְּדוֹלָה כְּמוֹ חָבִית! הוּא אָחַז בֶּעָלִים הַיְרֻקִּים וּמָשַׁךְ בְּכָל כֹּחוֹ. הוּא מָשַׁךְ וּמָשַׁךְ, אֲבָל הַלֶּפֶת לֹא זָזָה מִמְּקוֹמָהּ.

"סָבְתָּא! בּוֹאִי לַעֲזֹר לִי!" קָרָא הַסָּבָא. סָבְתָּא יָצְאָה מֵהַבַּיִת וְתָפְסָה בַּסָּבָא, וְהַסָּבָא תָּפַס בֶּעָלִים. הֵם מָשְׁכוּ וּמָשְׁכוּ, וְהַלֶּפֶת עֲדַיִן לֹא זָזָה.

סָבְתָּא קָרְאָה לַנֶּכְדָּה. הַנֶּכְדָּה רָצָה אֶל הַגִּנָּה וְתָפְסָה בְּסָבְתָּא, וְסָבְתָּא תָּפְסָה בַּסָּבָא, וְהַסָּבָא תָּפַס בֶּעָלִים. כֻּלָּם מָשְׁכוּ וּמָשְׁכוּ וּמָשְׁכוּ — אֲבָל הַלֶּפֶת לֹא זָזָה.

הַנֶּכְדָּה קָרְאָה לַכֶּלֶב. הַכֶּלֶב בָּא בְּרִיצָה וְנָבַח: "הַב־הַב!" הוּא תָּפַס בְּשִׂמְלַת הַנֶּכְדָּה, הַנֶּכְדָּה תָּפְסָה בְּסָבְתָּא, סָבְתָּא בַּסָּבָא, וְהַסָּבָא בֶּעָלִים. כֻּלָּם מָשְׁכוּ בְּבַת אַחַת. הַלֶּפֶת רָעֲדָה קְצָת, אַךְ נִשְׁאֲרָה בָּאֲדָמָה.

הַכֶּלֶב קָרָא לֶחָתוּל, וְהֶחָתוּל הִגִּיעַ לְאַט, אָחַז בְּזָנָב הַכֶּלֶב וְהִצְטָרֵף אֶל הַשַּׁרְשֶׁרֶת. עַכְשָׁיו הָיוּ חֲמִשָּׁה: סָבָא, סָבְתָּא, הַנֶּכְדָּה, הַכֶּלֶב וְהֶחָתוּל. הֵם מָשְׁכוּ וְנָשְׁמוּ בִּכְבֵדוּת, וְהַלֶּפֶת עֲדַיִן לֹא יָצְאָה מִן הָאֲדָמָה.

הַסָּבָא נֶאֱנַח וְאָמַר: "אוּלַי זוֹ לֶפֶת שֶׁאִי אֶפְשָׁר לַעֲקֹר." בְּאוֹתוֹ רֶגַע הֵצִיץ הָעַכְבָּר הַקָּטָן מִן הָאֶסֶם וְהִסְתַּכֵּל בָּהֶם. הֶחָתוּל, שֶׁבְּדֶרֶךְ כְּלָל רָדַף אַחֲרָיו, הִרְגִּישׁ שֶׁהַיּוֹם אֵין זְמַן לְרִיבִים, וְקָרָא: "בּוֹא, עַכְבָּר! אֲנַחְנוּ צְרִיכִים אוֹתְךָ!"

הָעַכְבָּר רָץ, אָחַז בְּזָנָב הֶחָתוּל וּמָשַׁךְ בְּכָל כֹּחוֹ הַקָּטָן. הַסָּבָא סָפַר בְּקוֹל: "אַחַת, שְׁתַּיִם, שָׁלוֹשׁ — מוֹשְׁכִים!" וְכֻלָּם מָשְׁכוּ בְּבַת אַחַת. פִּתְאוֹם נִשְׁמַע קוֹל: "פּוֹפּ!" הָאֲדָמָה רָעֲדָה, וְהַלֶּפֶת הָעֲנָקִית קָפְצָה מִן הָאֲדָמָה! כֻּלָּם נָפְלוּ זֶה עַל זֶה.

הֵם שָׁכְבוּ עַל הַדֶּשֶׁא וְצָחֲקוּ עַד שֶׁדָּמְעוּ עֵינֵיהֶם. בָּעֶרֶב בִּשְּׁלָה סָבְתָּא מָרָק לֶפֶת חַם וּמָתוֹק, וְכֻלָּם — גַּם הָעַכְבָּר הַקָּטָן — יָשְׁבוּ יַחַד סְבִיב הַשֻּׁלְחָן. הֶחָתוּל הִבְטִיחַ שֶׁלֹּא יִרְדֹּף אַחֲרָיו יוֹתֵר. מֵאָז יָדְעוּ כֻּלָּם: כְּשֶׁעוֹבְדִים יַחַד, גַּם הָעֶזְרָה שֶׁל הַקָּטָן בְּיוֹתֵר חֲשׁוּבָה.

מְסֻפָּר מֵחָדָשׁ בִּלְשׁוֹנֵנוּ עַל־פִּי "הַלֶּפֶת" (Репка), סִפּוּר עַם רוּסִי מֵאוֹסֶף סִפּוּרֵי הָעָם שֶׁל אָפַנַסְיֶיב (רוּסְיָה, 1860 בְּעֵרֶךְ), נַחֲלַת הַכְּלָל.

כִּתְבוּ סִפּוּר קָצָר מֵאַרְבָּעָה חֲלָקִים, מִשְׁפָּט אוֹ שְׁנַיִם לְכָל חֵלֶק: פְּתִיחָה, עֲלִילָה, שִׂיא וְסוֹף. לְדֻגְמָה: "פְּתִיחָה: בְּיוֹם קַיִץ חַם, בְּחוֹף הַיָּם, גָּר סַרְטָן קָטָן בְּשֵׁם סַרְטָנִי. עֲלִילָה: הוּא אִבֵּד אֶת הַצֶּדֶף הַיָּפֶה שֶׁלּוֹ וְחִפֵּשׂ אוֹתוֹ בְּכָל מָקוֹם. שִׂיא: לְפֶתַע הִגִּיעַ גַּל גָּדוֹל וְהֵבִיא אֶת הַצֶּדֶף. סוֹף: סַרְטָנִי חִיֵּךְ וְחָזַר הַבַּיְתָה בְּשִׂמְחָה."
