הָאַלּוֹן וְהַקָּנֶה

מָשָׁל מֻכָּר מֵאֵת אֵיזוֹפּוֹס, בְּעִבּוּד

📖 קְרִיאָה

עַל גְּדַת נַחַל עָמַד אַלּוֹן עֲנָק, וּלְיָדוֹ צָמַח קָנֶה דַּק. הָאַלּוֹן הָיָה גָּבוֹהַּ וְחָזָק, וַעֲנָפָיו הָרַבִּים כִּמְעַט נָגְעוּ בֶּעָנָנִים. הַקָּנֶה, לְעֻמָּתוֹ, הָיָה דַּק וְגָּמִישׁ, וְגָבַהּ רַק כְּמֵטֶר אֶחָד מֵעַל הַמַּיִם.

בְּכָל בֹּקֶר, כְּשֶׁנָּשְׁבָה רוּחַ קַלָּה, הַקָּנֶה הִתְכּוֹפֵף וְאַחַר כָּךְ חָזַר וְהִזְדַּקֵּף. הָאַלּוֹן הִתְבּוֹנֵן בּוֹ וְצָחַק: ״אֵיזֶה חַלָּשׁ אַתָּה! כָּל רוּחַ קְטַנָּה מַטָּה אוֹתְךָ אֶל הַמַּיִם. אֲנִי עוֹמֵד זָקוּף וְגֵּאֶה, וְשׁוּם דָּבָר לֹא מַזִּיז אוֹתִי.״

הַקָּנֶה לֹא הִתְוַכֵּחַ. הוּא רַק אָמַר בְּשֶׁקֶט: ״אֲנִי לֹא פּוֹחֵד מִן הָרוּחַ.״

יוֹם אֶחָד הִשְׁחִירוּ הָעֲנָנִים בַּשָּׁמַיִם. הַנְּמָלִים מִהֲרוּ אֶל הַקֵּן, וְהַצִּפֳּרִים הִפְסִיקוּ לְצַפְצֵף וְעָפוּ הַרְחֵק. הַקָּנֶה הִרְגִּישׁ אֶת הַשִּׁנּוּי וְנִשְׁאַר שָׁקֵט, אֲבָל הָאַלּוֹן הִמְשִׁיךְ לְהִתְגָּאוֹת בְּכֹחוֹ.

פִּתְאוֹם פָּרְצָה סוּפָה אֲדִירָה. הָרוּחַ יִלְלָה בֵּין הָעֲנָפִים, וְהַמַּיִם סָעֲרוּ בַּנַּחַל. הַקָּנֶה הִתְכּוֹפֵף עַד הַמַּיִם, וְהָרוּחַ עָבְרָה מֵעָלָיו. הָאַלּוֹן, לְעֻמָּתוֹ, עָמַד זָקוּף וְהִתְנַגֵּד בְּכָל כֹּחוֹ. הָרוּחַ הִתְחַזְּקָה עוֹד וָעוֹד, וְאָז נִשְׁמַע קוֹל נֶפֶץ אַדִּיר.

אַחֲרֵי הַסּוּפָה יָצְאָה הַשֶּׁמֶשׁ. הַקָּנֶה הִזְדַּקֵּף לְאַט, דַּק וְיָרֹק כְּמוֹ קֹדֶם. עַל גְּדַת הַנַּחַל שָׁכַב הָאַלּוֹן הָעֲנָק, וְשָׁרָשָׁיו נִרְאוּ בַּחוּץ, מֵעַל הָאֲדָמָה. הַקָּנֶה הִסְתַּכֵּל עָלָיו וְלֹא אָמַר דָּבָר. הוּא לֹא צָחַק וְלֹא הִתְפָּאֵר.

אֵיזוֹפּוֹס (יָוָן הָעַתִּיקָה), ״הָאַלּוֹן וְהַקָּנֶה״, בְּעִבּוּד חָפְשִׁי.

📚 דִּקְדּוּק — הַסָּקָה: לִקְרֹא בֵּין הַשּׁוּרוֹת (חֲזָרָה)

מָה כָּתוּב, מָה אֶפְשָׁר לְהַסִּיק וּמָה רַק נִחוּשׁ?

בְּשִׁיעוּר 85 הִכַּרְנוּ אֶת הַהַסָּקָה: מַסְקָנָה שֶׁאֵינָהּ כְּתוּבָה בַּטֶּקְסְט בְּמֻפְרָשׁ, אֲבָל אֶפְשָׁר לְהָבִין אוֹתָהּ מִתּוֹךְ רְמָזִים. בַּסִּפּוּר הַזֶּה הַסּוֹפֵר לֹא כָּתַב שֶׁהַסּוּפָה מִתְקָרֶבֶת, וְגַם לֹא כָּתַב שֶׁהַקָּנֶה צָנוּעַ. אֲנַחְנוּ מַסִּיקִים אֶת זֶה בְּעַצְמֵנוּ.

מָה כָּתוּב בַּטֶּקְסְטמָה אֶפְשָׁר לְהַסִּיקלָמָּה?
הַנְּמָלִים מִהֲרוּ אֶל הַקֵּן וְהַצִּפֳּרִים הִפְסִיקוּ לְצַפְצֵףסוּפָה מִתְקָרֶבֶתשְׁנֵי רְמָזִים: בַּעֲלֵי חַיִּים מְחַפְּשִׂים מָקוֹם מָגֵן לִפְנֵי סוּפָה
הָאַלּוֹן הִתְנַגֵּד בְּכָל כֹּחוֹ וְנִשְׁמַע קוֹל נֶפֶץהָאַלּוֹן נָפַל בִּגְלַל שֶׁלֹּא הִתְגַּמֵּשׁעֵץ שֶׁלֹּא זָז מוּל רוּחַ חֲזָקָה עָלוּל לִפֹּל
הַקָּנֶה לֹא הִתְוַכֵּחַ וְלֹא הִתְפָּאֵרהַקָּנֶה צָנוּעַהוּא לֹא הִתְנַהֵג בְּגַאֲוָה, לֹא לִפְנֵי הַסּוּפָה וְלֹא אַחֲרֶיהָ
🔑 שְׁלוֹשָׁה סוּגִים שֶׁל מִשְׁפָּטִים עַל הַטֶּקְסְט: עֻבְדָּה — כָּתוּב בְּמֻפְרָשׁ (״הָאַלּוֹן שָׁכַב עַל הָאֲדָמָה״). הַסָּקָה — לֹא כָּתוּב, אֲבָל יֵשׁ לָהּ רֶמֶז אֶחָד לְפָחוֹת (״כַּנִּרְאֶה הָאַלּוֹן נָפַל בִּגְלַל הָרוּחַ״). נִחוּשׁ — אֵין לוֹ שׁוּם רֶמֶז בַּטֶּקְסְט (״הָאַלּוֹן בֶּטַח הָיָה בֶּן מֵאָה שָׁנָה״).

כְּלָלִים לִזְכֹּר

• הַסָּקָה טוֹבָה בְּנוּיָה עַל שְׁלוֹשָׁה צְעָדִים: (1) מָה כָּתוּב? (2) מָה אֲנִי כְּבָר יוֹדֵעַ עַל הָעוֹלָם? (3) מָה מִסְתַּבֵּר מִשְּׁנֵיהֶם יַחַד? שְׁנֵי רְמָזִים חֲזָקִים מֵרֶמֶז אֶחָד.

• מִלּוֹת שֶׁמְּסַמְּנוֹת הַסָּקָה: כַּנִּרְאֶה מִסְתַּבֵּר שֶׁ- אֶפְשָׁר לְהָבִין שֶׁ-. כְּשֶׁאֵין רֶמֶז, אוֹמְרִים אוּלַי אוֹ אֲנִי מְנַחֵשׁ.

• זָוִית חֲדָשָׁה: אֶת הָאֹפִי שֶׁל דְּמוּת מַסִּיקִים מִמַּעֲשִׂים וּמִדְּבָרִים שֶׁהִיא אוֹמֶרֶת, וְלֹא תָּמִיד הַסּוֹפֵר כּוֹתֵב עָלָיו. מוּסַר הַהַשְׂכֵּל בְּמָשָׁל הוּא כִּמְעַט תָּמִיד הַסָּקָה.
