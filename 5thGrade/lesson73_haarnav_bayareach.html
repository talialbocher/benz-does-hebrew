📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת מְאֹד, בְּיַעַר שָׁקֵט, חָיוּ בִּידִידוּת גְּדוֹלָה אַרְבַּע חַיּוֹת: אַרְנָב קָטָן, קוֹף זָרִיז, לוּטְרָה שֶׁאָהֲבָה לָדוּג דָּגִים בַּנָּהָר, וְתַן פִּקֵּחַ. הָאַרְבַּע הֶחְלִיטוּ שֶׁבְּכָל לֵיל יָרֵחַ מָלֵא יַעֲשׂוּ מַעֲשֶׂה טוֹב: יִתְּנוּ אֹכֶל לְכָל מִי שֶׁיִּפְגְּשׁוּ בַּדֶּרֶךְ, כִּי כָּךְ לִמְּדוּ שֶׁנְדִיבוּת הִיא הַמִּדָּה הֲכִי חֲשׁוּבָה.

בְּלֵיל יָרֵחַ מָלֵא אֶחָד, יָצְאוּ הָאַרְבַּע לְחַפֵּשׂ מִישֶׁהוּ לַעֲזֹר לוֹ. פִּתְאֹם פָּגְשׁוּ קַבְּצָן זָקֵן וְרָעֵב, לָבוּשׁ בְּגָדִים בְּלוּיִים, שֶׁיָּשַׁב לְיַד הַדֶּרֶךְ וּבִקֵּשׁ אֹכֶל. הַקּוֹף מִהֵר וְטִפֵּס עַל עֵץ גָּבֹהַּ, וְהֵבִיא לַזָּקֵן פֵּרוֹת מְתוּקִים. הַלּוּטְרָה קָפְצָה לַנָּהָר וְהֵבִיאָה לוֹ דָּגִים טְרִיִּים. הַתַּן הַפִּקֵּחַ, שֶׁלֹּא הָיָה יָשָׁר לְגַמְרֵי, גָּנַב מִמַּחְנֶה סָמוּךְ לְטָאָה וּקְצָת גְּבִינַת חָלָב, וְהֵבִיא גַּם הוּא.

הַזָּקֵן קִבֵּל בְּשִׂמְחָה אֶת כָּל הַמַּתָּנוֹת, וְאָז פָּנָה אֶל הָאַרְנָב הַקָּטָן: "וְאַתָּה, אַרְנָב? מָה תִּתֵּן לִי?" הָאַרְנָב הִצְטַעֵר מְאֹד. הוּא לֹא יָדַע לְטַפֵּס כְּמוֹ הַקּוֹף, וְלֹא לָדוּג כְּמוֹ הַלּוּטְרָה, וְגַם לֹא הָיָה זָרִיז וְעַרְמוּמִי כְּמוֹ הַתַּן. כָּל מָה שֶׁיָּדַע לַעֲשׂוֹת הוּא לֶאֱכֹל עֵשֶׂב - וְעֵשֶׂב, הוּא יָדַע, בְּנֵי אָדָם לֹא אוֹכְלִים.

הָאַרְנָב חָשַׁב רֶגַע, וְאָז אָמַר בְּקוֹל שָׁקֵט: "אֲנִי אֶתֵּן לְךָ אֶת גּוּפִי." הוּא בִּקֵּשׁ מִן הַזָּקֵן לְהַדְלִיק מְדוּרָה, וְהִסְבִּיר שֶׁהוּא מִתְכַּוֵּן לְהַקְרִיב אֶת עַצְמוֹ כְּמָזוֹן, כְּדֵי שֶׁלֹּא יִשָּׁאֵר רָעֵב. לִפְנֵי שֶׁקָּפַץ אֶל הָאֵשׁ, נִעֵר הָאַרְנָב בִּזְהִירוּת אֶת פַּרְוָתוֹ, כְּדֵי שֶׁאִם יֵשׁ בָּהּ חֶרֶק קָטָן, הוּא יִבְרַח וְלֹא יִפָּגַע יַחַד אִתּוֹ.

וְאָז קָפַץ הָאַרְנָב אֶל תּוֹךְ הַלֶּהָבוֹת. אַךְ קָרָה נֵס: הָאֵשׁ לֹא שָׂרְפָה אוֹתוֹ כְּלָל! הַזָּקֵן הַקַּבְּצָן הִתְגַּלָּה כְּשַׂקְרָה, מֶלֶךְ הָאֵלִים בַּמָּסֹרֶת הַהֹדִּית-בּוּדְהִיסְטִית הָעַתִּיקָה, שֶׁיָּרַד לָאָרֶץ בְּתַחְפֹּשֶׂת כְּדֵי לִבְחֹן אֶת נְדִיבוּת הַחַיּוֹת. הוּא הָיָה נָע וְנִרְגָּשׁ מְאֹד מִנְּכוֹנוּתוֹ שֶׁל הָאַרְנָב הַקָּטָן לְהַקְרִיב אֶת הַכֹּל, לְלֹא שׁוּם דָּבָר בַּתְּמוּרָה.

"אַרְנָב יָקָר," אָמַר שַׂקְרָה בְּקוֹל חַם, "הוֹכַחְתָּ שֶׁאַתָּה הַנָּדִיב מִכָּל הַחַיּוֹת, כִּי נָתַתָּ אֶת הַדָּבָר הַיָּחִיד שֶׁהָיָה בְּיָדְךָ לָתֵת. כְּדֵי שֶׁכֻּלָּם יִזְכְּרוּ אֶת מַעֲשֵׂךָ לָנֶצַח, אָשִׂים אוֹתְךָ בַּיָּרֵחַ עַצְמוֹ." וּמֵאָז, אוֹמְרִים בְּאוֹתָהּ מָסֹרֶת, כְּשֶׁמִּסְתַּכְּלִים בַּיָּרֵחַ הַמָּלֵא בְּעֵינַיִם טוֹבוֹת, אֶפְשָׁר לִרְאוֹת בְּתוֹכוֹ אֶת דְּמוּתוֹ שֶׁל הָאַרְנָב הַקָּטָן וְהַנָּדִיב.

הַקּוֹף, הַלּוּטְרָה וְהַתַּן שָׁמְעוּ אֶת הַסִּפּוּר וְהֵבִינוּ מַשֶּׁהוּ חָשׁוּב: לֹא הַגֹּדֶל שֶׁל הַמַּתָּנָה קוֹבֵעַ אֶת עֶרְכָּהּ, אֶלָּא כַּמָּה מִלֵּב נוֹתְנִים אוֹתָהּ. מֵאָז אוֹתוֹ לַיְלָה, בְּכָל פַּעַם שֶׁמַּבִּיטִים בַּיָּרֵחַ הַמָּלֵא, מִתְעוֹרֵר בְּלֵב הָאַרְבַּע זִכָּרוֹן חַם שֶׁל הָאַרְנָב הַקָּטָן שֶׁנָּתַן אֶת הַכֹּל.

מְסֻפָּר עַל־פִּי "שַׂשַׂה גַ'אטָקָה" (Sasa Jataka), אַחַת מֵאַגָּדוֹת חַיֵּי-קֶדֶם שֶׁל הַבּוּדְהָא בַּמָּסֹרֶת הַהֹדִּית-בּוּדְהִיסְטִית הָעַתִּיקָה, מְעֻבֶּדֶת וּמְסֻפֶּרֶת מֵחָדָשׁ.

בְּשִׁיעוּר 7 לָמַדְנוּ אֵיךְ חֲכָמִים בֶּעָבָר קָבְעוּ אֶת רָאשֵׁי הַחֳדָשִׁים. עַכְשָׁו נוֹסִיף זָוִית חֲדָשָׁה: כָּל חֹדֶשׁ עִבְרִי מַתְחִיל בְּיָרֵחַ חָדָשׁ (רֹאשׁ חֹדֶשׁ), וּמִסְתַּיֵּם כִּמְעַט בְּדִיּוּק כְּשֶׁהַיָּרֵחַ חוֹזֵר לִהְיוֹת "חָדָשׁ" שׁוּב - מַחְזוֹר שֶׁל בְּעֵרֶךְ 29 וָחֵצִי יָמִים. לָכֵן חֹדֶשׁ עִבְרִי הוּא תָּמִיד אוֹ 29 יוֹם (חֹדֶשׁ "חָסֵר") אוֹ 30 יוֹם (חֹדֶשׁ "מָלֵא") - וְלֹא 31 יוֹם כְּמוֹ בַּלּוּחַ הַלּוֹעֲזִי.

לָמָּה יֵשׁ שָׁנָה מְעֻבֶּרֶת? (זָוִית חֲדָשָׁה)

שְׁנֵים עָשָׂר חֳדָשִׁים שֶׁל יָרֵחַ אֹרֶךְ כָּל אֶחָד כְּ-29 וָחֵצִי יָמִים מִסְתַּכְּמִים לְכ-354 יָמִים בִּלְבַד - כְּ-11 יוֹם פָּחוֹת מִשָּׁנָה שֶׁל הַשֶּׁמֶשׁ (כ-365 יָמִים). אִלּוּ הָיִינוּ מִתְעַלְּמִים מֵהַפַּעַר הַזֶּה, פֶּסַח הָיָה "נוֹדֵד" בְּכָל שָׁנָה כַּמָּה יָמִים אֲחוֹרָה, עַד שֶׁבְּעוֹד כַּמָּה שָׁנִים הוּא הָיָה חָל בַּחֹרֶף בִּמְקוֹם בָּאָבִיב! כְּדֵי שֶׁהַחַגִּים תָּמִיד יַחוּלוּ בָּעוֹנָה הַנְּכוֹנָה, מוֹסִיפִים בְּעֵרֶךְ שֶׁבַע פְּעָמִים בְּכָל 19 שָׁנִים חֹדֶשׁ נוֹסָף - "אֲדָר ב׳" - בְּמָה שֶׁנִּקְרָא "שָׁנָה מְעֻבֶּרֶת".

🔑 כְּלָל זָהָב: הַלּוּחַ הָעִבְרִי הוּא לוּחַ יָרֵחַ-שֶׁמֶשׁ - הַחֳדָשִׁים הוֹלְכִים לְפִי הַיָּרֵחַ, אֲבָל הַשָּׁנָה כֻּלָּהּ מְתֻאֶמֶת לְפִי הַשֶּׁמֶשׁ (בְּעֶזְרַת עִבּוּר הַשָּׁנָה), כְּדֵי שֶׁהֶעוֹנוֹת וְהַחַגִּים תָּמִיד יִתְאִימוּ.
💡 טִיפ: הַלּוּחַ הַלּוֹעֲזִי (הַלּוּחַ הָאֶזְרָחִי, שֶׁמְּשַׁמֵּשׁ בָּעוֹלָם כֻּלּוֹ) הוּא לוּחַ שֶׁמֶשׁ בִּלְבַד - הַחֳדָשִׁים בּוֹ בְּאֹרֶךְ קָבוּעַ (30 אוֹ 31 יוֹם, וּפֶבְּרוּאָר 28 אוֹ 29) וְאֵין לָהֶם שׁוּם קֶשֶׁר לְמַחְזוֹר הַיָּרֵחַ.
