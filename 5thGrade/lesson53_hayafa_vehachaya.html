📖 קְרִיאָה

פַּעַם הָיָה סוֹחֵר עָשִׁיר שֶׁגָּר עִם שָׁלוֹשׁ בָּנוֹת בְּבַיִת גָּדוֹל לְיַד הַנָּמָל. הַבַּת הַצְּעִירָה, יָפָה, הָיְתָה שׁוֹנָה מֵאֲחוֹתֶיהָ: הִיא אָהֲבָה לִקְרֹא סְפָרִים וּלְטַיֵּל בַּגִּנָּה, וְלֹא הִתְעַנְיְנָה בְּתַכְשִׁיטִים וּבִשְׂמָלוֹת יְקָרוֹת כְּמוֹתָן.

יוֹם אֶחָד הִגִּיעָה אֶל הַסּוֹחֵר יְדִיעָה קָשָׁה: כָּל אֳנִיּוֹת הַמִּסְחָר שֶׁלּוֹ טָבְעוּ בְּסַעַר בְּלֵב יָם, וְהוּא אִבֵּד אֶת כָּל רְכוּשׁוֹ בְּבַת אַחַת. הַמִּשְׁפָּחָה נֶאֱלְצָה לַעֲזֹב אֶת הַבַּיִת הַגָּדוֹל וְלַעֲבֹר לְבַיִת קָטָן וּפָשׁוּט בַּכְּפָר. הָאֲחָיוֹת הַגְּדוֹלוֹת הִתְלוֹנְנוּ יוֹם וָלַיְלָה, אֲבָל יָפָה עָבְדָה בְּשֶׁקֶט וּבְשִׂמְחָה, וְלֹא הִתְאוֹנְנָה אַף פַּעַם.

כַּעֲבֹר זְמַן הִגִּיעָה שְׁמוּעָה שֶׁאַחַת מֵאֳנִיּוֹת הַסּוֹחֵר בְּכָל זֹאת נִצְּלָה, וְהוּא יָצָא לְדֶרֶךְ אֲרֻכָּה כְּדֵי לִבְדֹּק זֹאת בְּעַצְמוֹ. לִפְנֵי שֶׁיָּצָא, שָׁאַל אֶת בְּנוֹתָיו אֵיזוֹ מַתָּנָה לְהָבִיא לָהֶן. הָאֲחָיוֹת בִּקְּשׁוּ שִׂמְלוֹת וְתַכְשִׁיטִים, וְיָפָה בִּקְּשָׁה רַק וֶרֶד אֶחָד יָפֶה.

בְּדַרְכּוֹ חֲזָרָה תָּעָה הַסּוֹחֵר בְּיַעַר חָשׁוּךְ בְּתוֹךְ סוּפַת שֶׁלֶג, עַד שֶׁמָּצָא טִירָה מוּזָרָה וּנְטוּשָׁה. הוּא נִכְנַס בְּתִקְוָה לִמְצֹא מַחְסֶה, וּמָצָא אֵשׁ בּוֹעֶרֶת וְשֻׁלְחָן עָמוּס בְּאֹכֶל, אַךְ אַף אָדָם לֹא נִרְאָה בַּסְּבִיבָה. בַּבֹּקֶר, לִפְנֵי שֶׁיָּצָא, קָטַף מִן הַגִּנָּה וֶרֶד אָדֹם אֶחָד עֲבוּר יָפָה.

בְּאוֹתוֹ רֶגַע הוֹפִיעָה לְפָנָיו חַיָּה עֲנָקִית וּמְפַחִידָה, בְּפַרְוָה סְמִיכָה וּבְעֵינַיִם בּוֹעֲרוֹת. "גָּנַבְתָּ אֶת הַוֶּרֶד הֲכִי יָקָר לִי בָּעוֹלָם!" שָׁאֲגָה הַחַיָּה. "עַל כָּךְ תְּשַׁלֵּם בְּחַיֶּיךָ - אֶלָּא אִם כֵּן אַחַת מִבְּנוֹתֶיךָ תָּבוֹא לָגוּר כָּאן בִּמְקוֹמְךָ, מֵרְצוֹנָהּ הַחָפְשִׁי."

הַסּוֹחֵר חָזַר הַבַּיְתָה נִסְעָר וְסִפֵּר לְבָנוֹתָיו מָה קָרָה. יָפָה, שֶׁלֹּא רָצְתָה שֶׁאָבִיהָ יִסְבֹּל בִּגְלָלָהּ, הֶחְלִיטָה מִיָּד: "אֲנִי אֵלֵךְ אֶל הַטִּירָה בִּמְקוֹמְךָ, אַבָּא." לַמְרוֹת הַתְּחִנּוֹת שֶׁלּוֹ, הִיא עָמְדָה בְּהַחְלָטָתָהּ וְיָצְאָה לְבַדָּהּ אֶל הַטִּירָה הַמְּכֻשֶּׁפֶת.

בַּטִּירָה גִּלְּתָה יָפָה שֶׁהַחַיָּה, לַמְרוֹת מַרְאֶהָ הַמַּפְחִיד, אֵינָהּ אַכְזָרִית כְּלָל. בְּכָל עֶרֶב הָיוּ נִפְגָּשִׁים לְשׂוֹחֵחַ עַל סְפָרִים, עַל הָעוֹלָם וְעַל עַצְמָם, וְיָפָה גִּלְּתָה לְאַט לְאַט שֶׁמֵּאֲחוֹרֵי הַפַּרְוָה וְהַצִּפֳּרְנַיִם מִסְתַּתֵּר לֵב טוֹב וְעָדִין.

אַךְ יָפָה הִתְגַּעְגְּעָה מְאֹד לְמִשְׁפַּחְתָּהּ. הַחַיָּה, שֶׁרָאֲתָה אֶת עֶצְבּוֹנָהּ, נָתְנָה לָהּ מַרְאָה מְכֻשֶּׁפֶת וְאָמְרָה: "לְכִי לְבַקֵּר אֶת מִשְׁפַּחְתֵּךְ שָׁבוּעַ אֶחָד, אֲבָל שׁוּבִי בַּזְּמַן - כִּי אִם לֹא תַּחְזְרִי, אָמוּת מִצַּעַר." יָפָה הִבְטִיחָה לַחְזֹר בַּזְּמַן.

בְּבֵית מִשְׁפַּחְתָּהּ שָׂמְחוּ מְאֹד לִרְאוֹתָהּ, וְהָאֲחָיוֹת, מִקִּנְאָה, שִׁכְנְעוּ אוֹתָהּ לְהִשָּׁאֵר עוֹד וָעוֹד יָמִים. בְּלֵיל הַיּוֹם הַשְּׁמִינִי הִבִּיטָה יָפָה בַּמַּרְאָה הַמְּכֻשֶּׁפֶת - וְרָאֲתָה אֶת הַחַיָּה שׁוֹכֶבֶת בַּגִּנָּה, חַלְשָׁה וּקְרוֹבָה לְמָוֶת מֵרֹב גַּעְגּוּעִים.

יָפָה נִבְהֲלָה וְרָצָה בְּחָזְרָה אֶל הַטִּירָה מַהֵר כְּכָל שֶׁיָּכְלָה. הִיא מָצְאָה אֶת הַחַיָּה שׁוֹכֶבֶת בֵּין פִּרְחֵי הַגִּנָּה, וְכִמְעַט לֹא נוֹשֶׁמֶת. "אַל תָּמוּת!" בָּכְתָה יָפָה וְחִבְּקָה אוֹתָהּ. "גִּלִּיתִי שֶׁאֲנִי אוֹהֶבֶת אוֹתְךָ, בְּדִיּוּק כְּפִי שֶׁאַתָּה!"

בְּאוֹתוֹ רֶגַע נִשְׁבַּר הַקֶּסֶם. עֵינֵי הַחַיָּה הֵחֵלּוּ לְהַבְרִיק בְּאוֹר זָהָב, וּגְוִיָּתָהּ הִתְמַלְּאָה בְּאוֹר, עַד שֶׁלְּפָנֶיהָ עָמַד נָסִיךְ צָעִיר וְנָאֶה. הוּא סִפֵּר לְיָפָה שֶׁמְּכַשֵּׁפָה קָלְלָה אוֹתוֹ לִהְיוֹת חַיָּה מְכֹעֶרֶת עַד שֶׁמִּישֶׁהִי תֹּאהַב אוֹתוֹ עַל לִבּוֹ וְלֹא עַל מַרְאֵהוּ הַחִיצוֹנִי.

יָפָה וְהַנָּסִיךְ הִתְחַתְּנוּ וְחָיוּ בְּאֹשֶׁר בַּטִּירָה, וְכָל הַמִּשְׁפָּחָה הִצְטָרְפָה אֲלֵיהֶם. הַסִּפּוּר הַזֶּה מְלַמֵּד אוֹתָנוּ שֶׁהַיֹּפִי הָאֲמִתִּי נִמְצָא בַּלֵּב, וְלֹא בְּמַרְאֶה חִיצוֹנִי.

מְסֻפָּר עַל־פִּי הַמַּעֲשִׂיָּה "הַיָּפָה וְהַחַיָּה" (La Belle et la Bête), בְּעִקָּר לְפִי הַגִּרְסָה הַמֻּכֶּרֶת שֶׁל הַסּוֹפֶרֶת הַצָּרְפָתִיָּה זַ'אן-מָארִי לֶפְּרִנְס דֶה בּוֹמוֹן (1756), הַמְּבֻסֶּסֶת עַל סִפּוּר מֻקְדָּם וְאָרֹךְ יוֹתֵר שֶׁל גַּבְּרִיאֶל-סוּזָאן דֶּה וִילְנֶב (1740) - מְעֻבָּד.

📚 דִּקְדּוּק — נוֹשֵׂא וְנָשׂוּא

מָה זֶה נוֹשֵׂא וּמָה זֶה נָשׂוּא?

כָּל מִשְׁפָּט בָּעִבְרִית בָּנוּי מִשְּׁנֵי חֲלָקִים עִקָּרִיִּים: הַנּוֹשֵׂא - מִי אוֹ מָה שֶׁהַמִּשְׁפָּט מְדַבֵּר עָלָיו; וְהַנָּשׂוּא - מָה נֶאֱמָר עַל הַנּוֹשֵׂא (בְּדֶרֶךְ כְּלָל זֶה הַפֹּעַל, יַחַד עִם הַמִּלִּים שֶׁבָּאוֹת אַחֲרָיו).

נוֹשֵׂא מֻבְלָע בַּפֹּעַל

לִפְעָמִים הַנּוֹשֵׂא לֹא כָּתוּב בְּמִלָּה נִפְרֶדֶת, אֶלָּא מֻבְלָע בְּתוֹךְ הַפֹּעַל עַצְמוֹ - אֲנַחְנוּ מְזַהִּים אוֹתוֹ לְפִי הַסִּיּוֹמֶת אוֹ הַתְּחִלִּית שֶׁל הַפֹּעַל. לְמָשָׁל, בַּמִּשְׁפָּט "הִבְטַחְתִּי לַחְזֹר בַּזְּמַן" - אֵין מִלָּה נִפְרֶדֶת לְנוֹשֵׂא, אֲבָל הַסִּיּוֹמֶת "־תִּי" מְלַמֶּדֶת שֶׁהַנּוֹשֵׂא הַמֻּבְלָע הוּא "אֲנִי".

הִבְטַחְתִּי (= אֲנִי הִבְטַחְתִּי) גָּנַבְתָּ (= אַתָּה גָּנַבְתָּ) נֵלֵךְ (= אֲנַחְנוּ נֵלֵךְ)

נוֹשֵׂא מוּרְכָּב מִכַּמָּה מִלִּים

הַנּוֹשֵׂא יָכוֹל לִהְיוֹת גַּם צֵרוּף שֶׁל כַּמָּה מִלִּים - לֹא רַק שֵׁם עֶצֶם בּוֹדֵד, אֶלָּא שֵׁם עֶצֶם יַחַד עִם תֹּאַר אוֹ עִם שֵׁם עֶצֶם נוֹסָף שֶׁצָּמוּד אֵלָיו. לְמָשָׁל, בַּמִּשְׁפָּט "הָאֲחָיוֹת הַגְּדוֹלוֹת הִתְלוֹנְנוּ" - הַנּוֹשֵׂא הַמָּלֵא הוּא "הָאֲחָיוֹת הַגְּדוֹלוֹת", וּבַמִּשְׁפָּט "עֵינֵי הַחַיָּה הֵחֵלּוּ לְהַבְרִיק" - הַנּוֹשֵׂא הַמָּלֵא הוּא "עֵינֵי הַחַיָּה".

🔑 כְּלָל זָהָב: לְזִהּוּי הַנּוֹשֵׂא - שׁוֹאֲלִים "מִי?" אוֹ "מָה?" לִפְנֵי הַפֹּעַל. לְזִהּוּי הַנָּשׂוּא - שׁוֹאֲלִים "מָה נֶאֱמַר עַל הַנּוֹשֵׂא?" (בְּדֶרֶךְ כְּלָל זֶה הַפֹּעַל וְהַמִּלִּים שֶׁאַחֲרָיו).
