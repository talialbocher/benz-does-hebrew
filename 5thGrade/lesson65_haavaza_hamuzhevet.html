הָאַוָּזָה הַמֻּזְהֶבֶת

נַעַר טוֹב לֵב חוֹלֵק אֶת מְעַט הָאֹכֶל שֶׁלּוֹ עִם זָקֵן זָר בַּיַּעַר, וּמְגַלֶּה שֶׁטוּב לֵב מוֹבִיל רָחוֹק יוֹתֵר מִכֹּחַ - וְשִׁיעוּר חָדָשׁ עַל כְּתִיב מָלֵא וְכָתִיב חָסֵר

📖 קְרִיאָה

הָיָה הָיָה חוֹטֵב עֵצִים זָקֵן שֶׁהָיוּ לוֹ שְׁלֹשָׁה בָּנִים. שְׁנֵי הַבָּנִים הַגְּדוֹלִים נֶחְשְׁבוּ פִּקְחִים וְנוֹעֲזִים, וְהֵם תָּמִיד לָעֲגוּ לָאָח הַצָּעִיר בְּיוֹתֵר, שֶׁקָּרְאוּ לוֹ "הַתָּם", כִּי הָיָה תָּמִים וְטוֹב לֵב מִדַּי, לְדַעְתָּם. יוֹם אֶחָד יָצָא הַבֵּן הַגָּדוֹל לַיַּעַר לַחְטֹב עֵצִים, וְאִמּוֹ הֵכִינָה לוֹ עוּגָה טְעִימָה וּבַקְבּוּק יַיִן לַדֶּרֶךְ.

בַּיַּעַר פָּגַשׁ הַבֵּן הַגָּדוֹל זָקֵן קָטָן וְאָפֹר שֶׁבִּקֵּשׁ מִמֶּנּוּ פְּרוּסַת עוּגָה וְלֹגֶם יַיִן, כִּי הָיָה רָעֵב וְצָמֵא מְאֹד. "לָמָּה שֶׁאֶתֵּן לְךָ מִן הָאֹכֶל שֶׁלִּי?" עָנָה הַבֵּן בְּגַסּוּת, וְהִמְשִׁיךְ בְּדַרְכּוֹ בְּלִי לְהַבִּיט אֲחוֹרָה. כַּעֲבֹר רֶגַע הֵחֵל לַחְטֹב עֵץ עָבֶה, אַךְ הַגַּרְזֶן הֶחֱלִיק וּפָצַע אֶת יָדוֹ קָשׁוֹת, וְהוּא נֶאֱלַץ לַחֲזֹר הַבַּיְתָה בְּלִי אֲפִלּוּ עֵץ אֶחָד.

לְמָחֳרָת יָצָא הַבֵּן הַשֵּׁנִי לַיַּעַר, וְגַם הוּא פָּגַשׁ אֶת אוֹתוֹ הַזָּקֵן הָאָפֹר וּבִקֵּשׁ מִמֶּנּוּ קְצָת אֹכֶל. גַּם הוּא סֵרֵב בְּגַסּוּת - "מַסְפִּיק שֶׁיְּהֵא לִי לְעַצְמִי!" - וְגַם לוֹ קָרְתָה אוֹתָהּ תְּאוּנָה בְּדִיּוּק: הַגַּרְזֶן פָּגַע בְּרַגְלוֹ, וְהוּא חָזַר הַבַּיְתָה כּוֹאֵב וּמְאֻכְזָב.

גַּם הַתָּם רָצָה לְנַסּוֹת אֶת מַזָּלוֹ בַּיַּעַר, אַךְ אָבִיו לֹא הֶאֱמִין בּוֹ: "אַתָּה? הֲרֵי אֲפִלּוּ אֵינְךָ יוֹדֵעַ לְהַחְזִיק גַּרְזֶן כָּמוֹ שֶׁצָּרִיךְ!" בַּסּוֹף הִסְכִּים הָאָב, אַךְ נָתַן לוֹ רַק פְּרוּסַת עוּגָה יְבֵשָׁה וּבַקְבּוּק שֶׁל שֵׁכָר חָמוּץ - כָּל מַה שֶּׁנִּשְׁאַר בַּבַּיִת אַחֲרֵי שֶׁאֶחָיו לָקְחוּ אֶת הַמָּנוֹת הַטּוֹבוֹת. הַתָּם לֹא הִתְלוֹנֵן וְיָצָא לַדֶּרֶךְ בְּשִׂמְחָה.

כְּשֶׁפָּגַשׁ הַתָּם אֶת הַזָּקֵן הָאָפֹר בַּיַּעַר, וְהַזָּקֵן בִּקֵּשׁ מִמֶּנּוּ אֹכֶל וּמַשְׁקֶה, לֹא הִסֵּס הַתָּם אֲפִלּוּ רֶגַע: "בְּשִׂמְחָה! אֵין לִי הַרְבֵּה, אֲבָל מַה שֶּׁיֵּשׁ - נַחֲלֹק בֵּינֵינוּ." לְהַפְתָּעָתוֹ, הָעוּגָה הַיְּבֵשָׁה הָיְתָה טְעִימָה כְּמוֹ עוּגַת דְּבַשׁ, וְהַשֵּׁכָר הֶחָמוּץ הָיָה מָתוֹק כְּמוֹ יַיִן מְשֻׁבָּח. הַזָּקֵן חִיֵּךְ וְאָמַר: "מִפְּנֵי שֶׁלִּבְּךָ טוֹב, אַרְאֶה לְךָ סוֹד. רְאֵה אֶת הָעֵץ הַזָּקֵן שָׁם - חֲטֹב אוֹתוֹ, וְתִמְצָא בֵּין שָׁרָשָׁיו מַתָּנָה מְיֻחֶדֶת."

הַתָּם מִהֵר אֶל הָעֵץ, חָטַב אוֹתוֹ בְּזָרוֹעוֹתָיו הַצְּעִירוֹת, וְשָׁם, בֵּין הַשָּׁרָשִׁים, יָשְׁבָה אַוָּזָה מֻזְהֶבֶת שֶׁנּוֹצוֹתֶיהָ נוֹצְצוֹת כְּמוֹ זָהָב אֲמִתִּי. הוּא הֵרִים אוֹתָהּ בִּזְהִירוּת, שָׂם אוֹתָהּ תַּחַת זְרוֹעוֹ, וּמִיָּד יָצָא לְדַרְכּוֹ אֶל פֻּנְדָּק קָרוֹב, כְּדֵי לָלוּן שָׁם לַלַּיְלָה.

בַּפֻּנְדָּק גָּרוּ שָׁלֹשׁ בָּנוֹת שֶׁל בַּעַל הַבַּיִת, וְכֻלָּן הִתְאַהֲבוּ מִיָּד בְּנוֹצוֹת הַזָּהָב שֶׁל הָאַוָּזָה. בַּלַּיְלָה, כְּשֶׁהַתָּם נִרְדַּם, הִתְגַּנְּבָה הַבַּת הַגְּדוֹלָה כְּדֵי לִגְנֹב נוֹצָה קְטַנָּה אַחַת - אֲבָל בְּרֶגַע שֶׁנָּגְעָה בָּאַוָּזָה, יָדָהּ נִדְבְּקָה אֵלֶיהָ כְּאִלּוּ הָיְתָה דֶּבֶק! הִיא צָעֲקָה לְעֶזְרַת אֲחוֹתָהּ, אֲבָל בְּרֶגַע שֶׁזּוֹ נָגְעָה בָּהּ כְּדֵי לְמָשְׁכָהּ אֲחוֹרָה - גַּם הִיא נִדְבְּקָה! וְכָךְ קָרָה גַּם לַבַּת הַשְּׁלִישִׁית, עַד שֶׁשָּׁלֹשׁ הַבָּנוֹת עָמְדוּ דְּבוּקוֹת זוֹ לְזוֹ בְּשׁוּרָה אֲרֻכָּה, מֵאֲחוֹרֵי הָאַוָּזָה.

בַּבֹּקֶר קָם הַתָּם, לֹא שָׂם לֵב לְכָל מַה שֶּׁקָּרָה בַּלַּיְלָה, וְיָצָא לְדַרְכּוֹ עִם הָאַוָּזָה תַּחַת זְרוֹעוֹ - וְאַחֲרָיו נִגְרְרוּ שָׁלֹשׁ הַבָּנוֹת, דְּבוּקוֹת וּמְבֻיָּשׁוֹת. בַּדֶּרֶךְ פָּגְשׁוּ כֹּמֶר שֶׁכָּעַס עַל הַמַּחֲזֶה וְתָפַס בְּיַד הַבַּת הָאַחֲרוֹנָה כְּדֵי לְמָשְׁכָהּ הַצִּדָּה - וְנִדְבַּק גַּם הוּא. אַחֲרָיו נִדְבַּק שַׁמָּשׁ הַכְּנֵסִיָּה שֶׁנִּסָּה לְהַצִּיל אֶת הַכֹּמֶר, וְאַחֲרָיו עוֹד שְׁנֵי אִכָּרִים שֶׁעָבְרוּ בַּשָּׂדֶה וְנִסּוּ לְהַפְרִיד בֵּין כֻּלָּם. עַד מְהֵרָה נִגְרְרָה אַחֲרֵי הַתָּם שׁוּרָה שְׁלֵמָה שֶׁל שִׁבְעָה אֲנָשִׁים דְּבוּקִים זֶה לְזֶה, וְאִישׁ מֵהֶם לֹא הִצְלִיחַ לְהִשְׁתַּחְרֵר.

בְּדִיּוּק בְּאוֹתָהּ הָעִיר גָּרָה נְסִיכָה שֶׁמֵּעוֹלָם לֹא צָחֲקָה, וְהַמֶּלֶךְ אָבִיהָ הִבְטִיחַ שֶׁמִּי שֶׁיַּצְלִיחַ לְהַצְחִיק אוֹתָהּ יִשָּׂא אוֹתָהּ לְאִשָּׁה. כְּשֶׁהִבִּיטָה הַנְּסִיכָה מֵחַלּוֹן אַרְמוֹנָהּ וְרָאֲתָה אֶת הַשּׁוּרָה הַמְּשֻׁנָּה מִדַּי הַנִּגְרֶרֶת אַחֲרֵי הַתָּם וְהָאַוָּזָה, לֹא הִצְלִיחָה לְהִתְאַפֵּק וּפָרְצָה בִּצְחוֹק גָּדוֹל שֶׁנִּשְׁמַע בְּכָל הָאַרְמוֹן. הַמֶּלֶךְ, שֶׁלֹּא הָיָה מְרֻצֶּה כָּל כָּךְ מֵחָתָן פָּשׁוּט כָּזֶה, הִטִּיל עַל הַתָּם עוֹד שְׁתֵּי מְשִׂימוֹת קָשׁוֹת: לִשְׁתּוֹת אֶת כָּל הַיַּיִן שֶׁבְּמַרְתֵּף הָאַרְמוֹן, וְלֶאֱכֹל הַר שָׁלֵם שֶׁל לֶחֶם. הַתָּם דָּאַג, אֲבָל אוֹתוֹ הַזָּקֵן הָאָפֹר הוֹפִיעַ שׁוּב בַּסֵּתֶר וְעָזַר לוֹ לְהַשְׁלִים אֶת שְׁתֵּי הַמְּשִׂימוֹת בְּקַלּוּת.

כְּשֶׁרָאָה הַמֶּלֶךְ שֶׁהַתָּם עָמַד בְּכָל הַמִּבְחָנִים, לֹא הָיְתָה לוֹ בְּרֵרָה אֶלָּא לְקַיֵּם אֶת הַבְטָחָתוֹ. הַתָּם נָשָׂא אֶת הַנְּסִיכָה לְאִשָּׁה, וְכַעֲבֹר שָׁנִים יָרַשׁ אֶת הַמַּמְלָכָה כֻּלָּהּ. וְכָךְ, בְּזִכוּת לֵב טוֹב וּמְנַת אֹכֶל קְטַנָּה שֶׁחָלַק בְּשִׂמְחָה, הָפַךְ הַנַּעַר שֶׁכֻּלָּם לָעֲגוּ לוֹ לְנָסִיךְ מְאֻשָּׁר וְאָהוּב.

מְסֻפָּר עַל־פִּי "הָאַוָּזָה הַמֻּזְהֶבֶת" (Die goldene Gans) מֵאֵת הָאַחִים גְרִים (גֶּרְמַנְיָה, פֻּרְסְמָה לָרִאשׁוֹנָה בִּשְׁנַת 1819) - מְעֻבָּד וּמְסֻפָּר מֵחָדָשׁ.

כָּל מִלָּה בְּעִבְרִית אֶפְשָׁר לִכְתֹּב בִּשְׁתֵּי דְּרָכִים. כְּשֶׁיֵּשׁ נִיקוּד - כְּמוֹ בְּכָל הַסִּפּוּרִים בַּסִּדְרָה הַזֹּאת - כּוֹתְבִים אֶת הַמִּלָּה בְּ"כְּתִיב חָסֵר": בְּלִי אוֹתִיּוֹת נוֹסָפוֹת, כִּי הַנְּקֻדּוֹת וְהַקַּוִּים כְּבָר מַרְאִים אֵיךְ לְהַגּוֹת אוֹתָהּ נָכוֹן. אֲבָל כְּשֶׁאֵין נִיקוּד - כְּמוֹ בְּעִתּוֹן, בְּהוֹדָעַת טֶלֶפוֹן אוֹ בְּשֶׁלֶט בָּרְחוֹב - צָרִיךְ עֶזְרָה: מוֹסִיפִים אוֹתִיּוֹת "אִמּוֹת קְרִיאָה" (בְּעִקָּר וָי"ו) כְּדֵי שֶׁהַקּוֹרֵא יֵדַע אֵיךְ לְהַגּוֹת אֶת הַמִּלָּה נָכוֹן. כְּתִיב כָּזֶה, עִם הָאוֹתִיּוֹת הַנּוֹסָפוֹת, נִקְרָא "כְּתִיב מָלֵא".

לֹא כָּל מִלָּה מְקַבֶּלֶת וָי"ו נוֹסֶפֶת. כַּמָּה מִלִּים קְצָרוֹת וְנֶפוֹצוֹת מְאֹד נִשְׁאָרוֹת בְּאוֹתוֹ כֶּתֶב, גַּם בְּלִי נִיקוּד:

🔑 "לֹא" נִשְׁאֶרֶת "לא" (וְלֹא הוֹפֶכֶת ל"לוא"), וְ"כֹּל" נִשְׁאֶרֶת "כל" (וְלֹא הוֹפֶכֶת ל"כול") - גַּם בְּלִי נִיקוּד. אֵלֶּה שְׁתֵּי מִלִּים קְצָרוֹת וְנֶפוֹצוֹת שֶׁחוֹרְגוֹת מֵהַכְּלָל הָרָגִיל.
