📖 קְרִיאָה

בִּכְפָר קָטָן לְמַרְגְּלוֹת הַר גָּבוֹהַּ גָּר נַעַר בְּשֵׁם עוֹמֶר. כָּל בּוֹקֶר יָצָא עוֹמֶר עִם עֵדֶר הַכְּבָשִׂים שֶׁל אָבִיו אֶל מִדְרוֹן הָהָר, וְשָׁמַר עָלָיו לְבַדּוֹ עַד הָעֶרֶב. זוֹ הָיְיתָה עֲבוֹדָה חֲשׁוּבָה - אֲבָל גַּם מְשַׁעֲמֶמֶת מְאוֹד, שָׁעוֹת אֲרוּכּוֹת לְבַד מוּל הַכְּבָשִׂים הַשְּׁקֵטוֹת וְהַשָּׁמַיִם הָרֵיקִים.

יוֹם אֶחָד, אַחֲרֵי שָׁעוֹת שֶׁל שִׁעֲמוּם, עָלָה בְּמוֹחוֹ שֶׁל עוֹמֶר רַעְיוֹן "מְשַׁעֲשֵׁעַ". הוּא רָץ בְּמוֹרַד הָהָר אֶל הַכְּפָר, וְצָעַק בְּקוֹל רָם וּמְבוֹהָל: "זְאֵב! זְאֵב! זְאֵב תּוֹקֵף אֶת הַכְּבָשִׂים! בּוֹאוּ מַהֵר!"

הַכַּפְרִיִּים נִבְהֲלוּ. הֵם עָזְבוּ אֶת הָעֲבוֹדָה שֶׁלָּהֶם, תָּפְסוּ מַקְלוֹת וַאֲבָנִים, וְרָצוּ בְּמַעֲלֵה הָהָר כְּדֵי לְהַצִּיל אֶת הָעֵדֶר. אֲבָל כְּשֶׁהִגִּיעוּ, הֵם מָצְאוּ רַק אֶת עוֹמֶר, יוֹשֵׁב עַל אֶבֶן וְצוֹחֵק בְּכָל פֶּה. לֹא הָיָה שׁוּם זְאֵב.

"אַתָּה לֹא יָכוֹל לִצְחוֹק עָלֵינוּ כָּכָה!" אָמַר אֶחָד הַכּוֹרְמִים בְּכַעַס, כְּשֶׁהוּא מְנַגֵּב אֶת הַזֵּיעָה מִמִּצְחוֹ. עוֹמֶר רַק הִמְשִׁיךְ לִצְחוֹק, וְאָמַר שֶׁזּוֹ הָיְיתָה בְּדִיחָה קְטַנָּה, בְּלִי כַּוָּונָה רָעָה.

כַּמָּה יָמִים אַחַר כָּךְ, עוֹמֶר הִשְׁתַּעֲמֵם שׁוּב - וְחָזַר עַל אוֹתוֹ תַּעֲלוּל בְּדִיּוּק. הוּא צָעַק שׁוּב: "זְאֵב! זְאֵב!" וְהַכַּפְרִיִּים, שֶׁעֲדַיִין לֹא שָׁכְחוּ לְגַמְרֵי, רָצוּ שׁוּב בְּמַעֲלֵה הָהָר, מַזִּיעִים וּמוּדְאָגִים. שׁוּב לֹא הָיָה זְאֵב, וְשׁוּב עוֹמֶר צָחַק.

הַפַּעַם הַכַּפְרִיִּים כְּבָר לֹא צָחֲקוּ. "עוֹמֶר," אָמַר לוֹ הַזָּקֵן שֶׁבַּכְּפָר בְּקוֹל חָמוּר, "אִם תַּעֲשֶׂה אֶת זֶה שׁוּב, אִישׁ מֵאִיתָּנוּ לֹא יָבוֹא, גַּם אִם תִּצְעַק בְּכָל הַכּוֹחַ שֶׁלְּךָ." עוֹמֶר הִנְהֵן, אֲבָל בְּלִבּוֹ לֹא מַמָּשׁ הֶאֱמִין שֶׁזֶּה יִקְרֶה.

וְאָז, יוֹם אֶחָד, זְאֵב אֲמִיתִּי - גָּדוֹל, אָפוֹר וְרָעֵב - יָצָא פִּתְאוֹם מִבֵּין הָעֵצִים וְהִתְקָרֵב אֶל הָעֵדֶר. עוֹמֶר נִבְהַל מְאוֹד, וְצָעַק בְּכָל כּוֹחוֹ: "זְאֵב! זְאֵב! הַפַּעַם זֶה אֲמִיתִּי, תַּאֲמִינוּ לִי!"

אֲבָל בַּכְּפָר, כְּשֶׁשָּׁמְעוּ אֶת הַצְּעָקָה, אִישׁ לֹא זָז מִמְּקוֹמוֹ. "זֶה עוֹמֶר שׁוּב," אָמַר אֶחָד הַשְּׁכֵנִים בַּעֲיֵיפוּת, "בּוֹאוּ נִתְעַלֵּם, זוֹ בֶּטַח עוֹד בְּדִיחָה." כּוּלָּם הִמְשִׁיכוּ בַּעֲבוֹדָתָם, וְאַף אֶחָד לֹא טָרַח לַעֲלוֹת אֶל הָהָר.

עוֹמֶר נִשְׁאַר לְבַדּוֹ מוּל הַזְּאֵב. הוּא רָץ בֵּין הַכְּבָשִׂים, צָעַק וְנִיפְנֵף בְּמַקְלוֹ בְּכָל הַכּוֹחַ, עַד שֶׁהַזְּאֵב, שֶׁנִּבְהַל מֵהָרַעַשׁ, בָּרַח בַּחֲזָרָה אֶל הַיַּעַר. הַכְּבָשִׂים נִבְהֲלוּ וְנִמְלְטוּ לְכָל עֵבֶר, וְעוֹמֶר בִּילָּה אֶת כָּל הָעֶרֶב לְבַדּוֹ, רוֹעֵד וּמְפוּחָד, כְּשֶׁהוּא מְחַפֵּשׂ אוֹתָן אַחַת-אַחַת בֵּין הַסְּלָעִים.

רַק בַּלַּיְלָה, כְּשֶׁרָאוּ אֶת עוֹמֶר לֹא חוֹזֵר, כַּמָּה כַּפְרִיִּים עָלוּ לָהָר לִבְדּוֹק מָה קָרָה - וּמָצְאוּ אוֹתוֹ יוֹשֵׁב, עָיֵיף וּמְפוּחָד, אוֹסֵף אֶת הַכְּבָשִׂים הָאַחֲרוֹנוֹת לְבַדּוֹ. "טָעִיתִי," הוֹדָה עוֹמֶר בְּשֶׁקֶט. "שִׁיקַּרְתִּי לָכֶם פַּעֲמַיִים, וְלָכֵן בַּפַּעַם הַשְּׁלִישִׁית, כְּשֶׁאָמַרְתִּי אֶת הָאֱמֶת, אַף אֶחָד לֹא הֶאֱמִין לִי."

הַכַּפְרִיִּים עָזְרוּ לוֹ לֶאֱסוֹף אֶת שְׁאָר הָעֵדֶר, וְסָלְחוּ לוֹ - אֲבָל הַלֶּקַח נִשְׁאַר אִיתּוֹ לְתָמִיד: מִי שֶׁמְּשַׁקֵּר שׁוּב וָשׁוּב, מְאַבֵּד אֶת הָאֵמוּן שֶׁל כּוּלָּם - וְגַם כְּשֶׁהוּא בַּסּוֹף אוֹמֵר אֶת הָאֱמֶת, אִישׁ כְּבָר לֹא מַאֲמִין לוֹ.

מספרים מה מישהו אמר - בשתי דרכים

כשמספרים שמישהו אמר משהו, אפשר לעשות זאת בשתי דרכים: דיבור ישר - מצטטים את המילים המדויקות שהוא אמר, בתוך מירכאות (" "), בדרך כלל אחרי נקודתיים. או דיבור עקיף - מספרים מה מישהו אמר, במילים שלנו, בלי מירכאות, בעזרת המילה "ש-".

כשהופכים דיבור ישר לעקיף, כינויי הגוף והפעלים בגוף ראשון (אני, אנחנו) הופכים לגוף שלישי (הוא, היא, הם) - כי עכשיו מישהו אחר מספר את מה שנאמר, ולא הדובר עצמו.
