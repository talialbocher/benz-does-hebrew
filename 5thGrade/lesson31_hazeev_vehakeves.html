📖 קְרִיאָה

בְּיוֹם חַם אֶחָד יָרַד כֶּבֶשׂ צָעִיר וְקָטָן אֶל נַחַל צָלוּל כְּדֵי לִשְׁתּוֹת מַיִם, לִפְנֵי שֶׁיַּחֲזֹר אֶל הָעֵדֶר שֶׁלּוֹ.

בְּאוֹתוֹ רֶגַע יָרַד גַּם זְאֵב רָעֵב אֶל אוֹתוֹ נַחַל, קְצָת יוֹתֵר לְמַעְלָה בַּזֶּרֶם, וְרָאָה אֶת הַכֶּבֶשׂ שׁוֹתֶה בְּשֶׁקֶט.

הַזְּאֵב חָשַׁב לְעַצְמוֹ: "כֶּבֶשׂ רַךְ וְטָעִים כָּזֶה יִהְיֶה אֲרוּחָה מְצוּיֶנֶת." אֲבָל הוּא רָצָה לִמְצֹא תֵּרוּץ, כְּדֵי שֶׁיִּהְיֶה נִרְאֶה שֶׁהוּא צוֹדֵק לִטְרֹף אוֹתוֹ.

אָמַר הַזְּאֵב בְּקוֹל רָם: "אֵיךְ אַתָּה מֵעֵז לְהַעְכִּיר לִי אֶת הַמַּיִם שֶׁאֲנִי שׁוֹתֶה מֵהֶם?"

אֲבָל הַכֶּבֶשׂ, שֶׁהָיָה חַף מִפֶּשַׁע לְגַמְרֵי, עָנָה: "אֲדוֹנִי הַזְּאֵב, אֲנִי שׁוֹתֶה בְּמוֹרַד הַנַּחַל, הַרְחֵק מִמְּךָ - אֵיךְ אֲנִי יָכוֹל לְהַעְכִּיר לְךָ מַיִם שֶׁנִּמְצָאִים לְמַעְלָה, לִפְנֵי שֶׁהֵם מַגִּיעִים אֵלַי?"

הַזְּאֵב, שֶׁלֹּא יָדַע מָה לַעֲנוֹת עַל כָּךְ, מִיהֵר לִמְצֹא תֵּרוּץ אַחֵר: "אִם כָּךְ, אַתָּה בֶּטַח זֶה שֶׁדִּיבֵּר עָלַי רָעוֹת בַּשָּׁנָה שֶׁעָבְרָה, לְיַד הַמִּרְעֶה!"

עָנָה הַכֶּבֶשׂ, עֲדַיִן רוֹעֵד: "זֶה גַּם לֹא יָכוֹל לִהְיוֹת נָכוֹן, כִּי לִפְנֵי שָׁנָה עוֹד לֹא נוֹלַדְתִּי בִּכְלָל."

הַזְּאֵב, שֶׁכְּבָר אִיבֵּד אֶת הַסַּבְלָנוּת, אָמַר: "טוֹב מְאֹד, אָז בֶּטַח אָחִיךָ הוּא זֶה שֶׁדִּיבֵּר עָלַי רָעוֹת - וְזֶה בְּדִיּוּק אוֹתוֹ דָּבָר!"

לִפְנֵי שֶׁהַכֶּבֶשׂ הִסְפִּיק לַעֲנוֹת עוֹד מִלָּה, קָפַץ עָלָיו הַזְּאֵב וְטָרַף אוֹתוֹ, בְּלִי שׁוּם תֵּרוּץ אֲמִתִּי.

הַסִּפּוּר מְלַמֵּד אוֹתָנוּ: מִי שֶׁרוֹצֶה לִפְגֹּעַ בְּמִישֶׁהוּ חַלָּשׁ מִמֶּנּוּ יִמְצָא תָּמִיד תֵּרוּץ - וְגַם תֵּרוּץ הֶגְיוֹנִי לֹא בֶּאֱמֶת יְשַׁנֶּה דָּבָר, כִּי הַהַחְלָטָה כְּבָר הִתְקַבְּלָה מֵרֹאשׁ.

📚 דִּקְדּוּק — דיבור ישר ועקיף (חזרה)

מה ההבדל בין דיבור ישר לדיבור עקיף?

בשיעור 20 למדנו: דיבור ישר הוא ציטוט מדויק, מילה במילה, של מה שמישהו אמר - בתוך מירכאות. דיבור עקיף הוא סיפור מה מישהו אמר, במילים שלנו, בלי מירכאות, בעזרת מילת החיבור "ש-". כשעוברים מדיבור ישר לעקיף, כינויי הגוף חייבים להשתנות: אני ← הוא/היא, אתה/את ← הוא/היא, אנחנו ← הם/הן, שלי ← שלו/שלה.

איך מזהים דיבור ישר בטקסט כתוב?

דיבור ישר תמיד מופיע בתוך מירכאות, בדיוק כמו שלמדנו בשיעור 23 - כי המירכאות מסמנות שאלה המילים המדויקות שנאמרו, בלי שום שינוי. בדיבור עקיף אין מירכאות בכלל, ובמקום זאת יש את מילת החיבור "ש-" שמחברת בין מי שאמר לבין תוכן הדברים.
