📖 קְרִיאָה

צָב זָקֵן גָּר עַל גְּדַת נַחַל. בְּכָל יוֹם הוּא הִסְתַּכֵּל אֶל הַשָּׁמַיִם וְרָאָה צִפֳּרִים עָפוֹת גָּבוֹהַּ. ״כַּמָּה הֵן חָפְשִׁיּוֹת!״ אָמַר בְּאַנְחָה. ״אֲנִי כָּבֵד וְאִטִּי, וְהֵן מְרַחֲפוֹת בָּאֲוִיר. אִלּוּ יָכֹלְתִּי לָעוּף!״

בְּבֹקֶר אֶחָד נָחַת נֶשֶׁר גָּדוֹל עַל סֶלַע לְיַד הַנַּחַל. הַצָּב הִתְקָרֵב אֵלָיו וְאָמַר: ״אֲדוֹנִי הַנֶּשֶׁר, אֲנִי מְבַקֵּשׁ שֶׁתִּתֵּן לִי לָעוּף. בִּתְמוּרָה אֶתֵּן לְךָ אֶת הַצְּדָפִים הַיָּפִים בְּיוֹתֵר מִקַּרְקָעִית הַנַּחַל.״

הַנֶּשֶׁר צָחַק. ״לָעוּף? אַתָּה נוֹלַדְתָּ לָלֶכֶת עַל הָאֲדָמָה וְלִשְׂחוֹת בַּמַּיִם. הַשִּׁרְיוֹן שֶׁלְּךָ כָּבֵד, וְהָרַגְלַיִם שֶׁלְּךָ קְצָרוֹת.״ אֲבָל הַצָּב לֹא וִתֵּר, וְחָזַר עַל הַבַּקָּשָׁה יוֹם אַחֲרֵי יוֹם.

לְבַסּוֹף הִסְכִּים הַנֶּשֶׁר. ״טוֹב״, אָמַר, ״אֶשָּׂא אוֹתְךָ אֶל הָעֲנָנִים. אֲבָל אַל תִּגַּע בַּכְּנָפַיִם שֶׁלִּי וְהִשָּׁאֵר שָׁקֵט.״ הוּא אָחַז בַּצָּב בְּצִפָּרְנָיו וְעָלָה גָּבוֹהַּ. הַצָּב רָאָה אֶת הַנַּחַל נַעֲשֶׂה חוּט דַּק וְאֶת הָעֵצִים קְטַנִּים כְּמוֹ אֲפוּנִים.

״אֲנִי עָף!״ צָעַק הַצָּב בְּשִׂמְחָה, וְשָׁכַח מִיָּד אֶת הַהוֹרָאוֹת. הוּא פָּשַׁט אֶת רַגְלָיו וְנָגַע בַּכָּנָף שֶׁל הַנֶּשֶׁר. בְּאוֹתוֹ רֶגַע נָשְׁבָה רוּחַ חֲזָקָה. הַנֶּשֶׁר הִתְנַדְנֵד, וְהַצָּב נִשְׁמַט מִן הַצִּפָּרְנַיִם וְנָפַל. הוּא נָפַל וְנָפַל, וְהָרוּחַ שָׁרְקָה בְּאָזְנָיו.

לְמַזָּלוֹ נָחַת הַצָּב בַּאֲגַם עָמֹק, וּפְלוּץ! הוּא צָף עַל הַמַּיִם וְהִשְׁתַּעֵל קְצָת. הַנֶּשֶׁר יָרַד אֵלָיו וְאָמַר: ״עַכְשָׁו אַתָּה מֵבִין. הָאֲוִיר אֵינוֹ הַבַּיִת שֶׁלְּךָ, אֲבָל הַמַּיִם כֵּן.״ מֵאָז הַצָּב כְּבָר לֹא בִּקֵּשׁ לָעוּף. הוּא הִבִּיט בַּשָּׁמַיִם בְּחִיּוּךְ וְשָׂחָה בְּשִׂמְחָה בַּנַּחַל.

אֵיזוֹפּוֹס (יָוָן הָעַתִּיקָה), ״הַצָּב וְהַנֶּשֶׁר״, בְּעִבּוּד חָפְשִׁי עִם סוֹף מְרֻכָּךְ.

📚 דִּקְדּוּק — פְּעָלִים בְּגִזְרַת פ״נ

הָאוֹת נ׳ שֶׁנֶּעֱלֶמֶת

בַּסִּפּוּר פָּגַשְׁנוּ פְּעָלִים כְּמוֹ נָפַל, נָגַע, נָחַת, נָשְׁבָה. הָאוֹת הָרִאשׁוֹנָה בַּשֹּׁרֶשׁ שֶׁלָּהֶם הִיא נ׳ (נ־פ־ל, נ־ג־ע, נ־ח־ת, נ־ש־ב), וְלָכֵן הֵם נִקְרָאִים פְּעָלִים בְּגִּזְרַת פ״נ. בֶּעָבָר הַנּ׳ נִשְׁמֶרֶת וְרוֹאִים אוֹתָהּ: נָפַל נָפְלָה נָפַלְתִּי. אֲבָל בֶּעָתִיד, בַּצִּוּוּי וּבְשֵׁם הַפֹּעַל הַנּ׳ נֶעֱלֶמֶת, וּבֶעָתִיד הָאוֹת שֶׁאַחֲרֶיהָ מְקַבֶּלֶת דָּגֵשׁ.

🔑 הַכְּלָל: בֶּעָבָר רוֹאִים אֶת הַנּ׳ (נָפַל). בֶּעָתִיד הַנּ׳ נֶעֱלֶמֶת, וְהָאוֹת הַבָּאָה אַחֲרֶיהָ מְקַבֶּלֶת דָּגֵשׁ: נָפַל ← אֶפֹּל, נָגַע ← אֶגַּע, נָשָׂא ← אֶשָּׂא. כָּךְ גַּם בְּשֵׁם הַפֹּעַל: לִפֹּל, לָתֵת, לָשֵׂאת.

יוֹצְאִים מִן הַכְּלָל

• יֵשׁ פְּעָלִים בְּגִזְרַת פ״נ שֶׁהַנּ׳ שֶׁלָּהֶם נִשְׁאֶרֶת גַּם בֶּעָתִיד: נָהַג ← יִנְהַג נָשַׁב ← תִּנְשֹׁב נָחַת ← יִנְחַת. כְּשֶׁרוֹאִים נ׳ בֶּעָתִיד, כְּדַאי לִבְדֹּק אִם זֶה פֹּעַל כָּזֶה.

• הַצִּוּוּי בְּפ״נ קָצָר: מֵתֵּן עוֹשִׂים תֵּן!, מֵנָסַע עוֹשִׂים סַע!. וּלְצִוּוּי שְׁלִילִי מוֹסִיפִים אַל + עָתִיד: אַל תִּפֹּל!

• אִם רוֹאִים אֶת הַשֹּׁרֶשׁ נ־פ־ל אוֹ נ־ג־ע, אֶפְשָׁר לָדַעַת שֶׁזֶּה פ״נ. אֲבָל אַל תִּתְבַּלְבְּלוּ: רָאָה, קָרָא וְקָם אֵינָם פ״נ.
