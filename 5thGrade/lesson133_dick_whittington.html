דִּיק וִיטִינְגְטוֹן וְהַחֲתוּלָה

אַגָּדַת עַם אַנְגְּלִית עַל יֶלֶד עָנִי וַחֲתוּלָה אַחַת

📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת, בִּכְפָר קָטָן בְּאַנְגְּלִיָּה, חַי יֶלֶד יָתוֹם וְעָנִי בְּשֵׁם דִּיק וִיטִינְגְטוֹן. אֲנָשִׁים בַּכְּפָר סִפְּרוּ שֶׁבָּעִיר הַגְּדוֹלָה לוֹנְדוֹן הָרְחוֹבוֹת מְרֻצָּפִים בְּזָהָב. "שָׁם אֶמְצָא עֲבוֹדָה," אָמַר דִּיק לְעַצְמוֹ, וְיָצָא לַדֶּרֶךְ בָּרֶגֶל.

אַחֲרֵי הֲלִיכָה אֲרֻכָּה הִגִּיעַ דִּיק לְלוֹנְדוֹן. אֲבָל הָרְחוֹבוֹת לֹא הָיוּ מְרֻצָּפִים בְּזָהָב — הֵם הָיוּ מְלֵאִים בֹּץ. דִּיק הָיָה רָעֵב וְעָיֵף, וְנִרְדַּם עַל סַף הַדֶּלֶת שֶׁל בַּיִת גָּדוֹל. בַּבַּיִת הַזֶּה גָּר סוֹחֵר עָשִׁיר בְּשֵׁם מַר פִיצְווֹרֶן.

הַסּוֹחֵר יָצָא מִן הַמִּשְׂרָד שֶׁלּוֹ וְרָאָה אֶת הַיֶּלֶד. "מִי אַתָּה?" שָׁאַל. דִּיק סִפֵּר לוֹ עַל הַדֶּרֶךְ הָאֲרֻכָּה. הַסּוֹחֵר רִחֵם עָלָיו וְאָמַר: "לֵךְ לַמִּטְבָּח. שָׁם יִתְּנוּ לְךָ לֶאֱכֹל, וְתוּכַל לַעֲבֹד אֶצְלֵנוּ."

בַּמִּטְבָּח עָבְדָה טַבָּחִית רַגְזָנִית. הִיא צָעֲקָה עַל דִּיק כָּל הַיּוֹם: "קַרְצֵף אֶת הַסִּירִים! סְחַב אֶת שַׂקֵּי הַקֶּמַח מִן הַמַּחְסָן! קַח אֶת הַסְּדִינִים לַמִּכְבָּסָה!" רַק אֶלִיס, בִּתּוֹ שֶׁל הַסּוֹחֵר, דִּבְּרָה אֵלָיו בְּנֹעַם.

בַּלַּיְלָה יָשַׁן דִּיק בַּעֲלִיַּת גַּג קְטַנָּה, מְלֵאָה עַכְבָּרִים. הָעַכְבָּרִים רָצוּ עַל הַמִּטָּה וְלֹא נָתְנוּ לוֹ לִישֹׁן. יוֹם אֶחָד קִבֵּל דִּיק מַטְבֵּעַ קָטָן, וְקָנָה בּוֹ חֲתוּלָה. מֵאָז הָיוּ הַלֵּילוֹת שֶׁלּוֹ שְׁקֵטִים, וְהוּא אָהַב אֶת הַחֲתוּלָה אַהֲבָה גְּדוֹלָה.

בְּאוֹתָם יָמִים בָּנוּ פּוֹעֲלִים בַּמִּסְפָּנָה שֶׁעַל שְׂפַת הַנָּהָר סְפִינָה חֲדָשָׁה לַסּוֹחֵר. לִפְנֵי שֶׁהַסְּפִינָה יָצְאָה לְאֶרֶץ רְחוֹקָה, קָרָא הַסּוֹחֵר לְכָל אַנְשֵׁי הַבַּיִת. "כָּל אֶחָד יָכוֹל לִשְׁלֹחַ מַשֶּׁהוּ לִמְכִירָה," אָמַר. לְדִיק לֹא הָיָה דָּבָר מִלְּבַד הַחֲתוּלָה, וּבְלֵב כָּבֵד הוּא נָתַן אוֹתָהּ.

הַטַּבָּחִית צָחֲקָה עָלָיו: "חֲתוּלָה? מִי יִקְנֶה חֲתוּלָה?" וְהַחַיִּים בַּמִּטְבָּח נַעֲשׂוּ קָשִׁים עוֹד יוֹתֵר. לַיְלָה אֶחָד דִּיק כְּבָר לֹא יָכוֹל הָיָה לִסְבֹּל. הוּא אָרַז אֶת הַחֲפָצִים הַמְּעַטִּים שֶׁלּוֹ וּבָרַח מִן הָעִיר.

עַל גִּבְעָה מִחוּץ לָעִיר יָשַׁב דִּיק לָנוּחַ. פִּתְאוֹם שָׁמַע אֶת פַּעֲמוֹנֵי הָעִיר מְצַלְצְלִים מֵרָחוֹק. נִדְמֶה הָיָה לוֹ שֶׁהֵם שָׁרִים: "שׁוּב, וִיטִינְגְטוֹן, שׁוּב! רֹאשׁ הָעִיר לוֹנְדוֹן תִּהְיֶה!" דִּיק קָם, חָזַר עַל עִקְבוֹתָיו, וְהִסְפִּיק לְהַגִּיעַ לַמִּטְבָּח לִפְנֵי שֶׁהַטַּבָּחִית הִתְעוֹרְרָה.

בֵּינְתַיִם הִגִּיעָה הַסְּפִינָה לְמַמְלָכָה רְחוֹקָה. הַמֶּלֶךְ הִזְמִין אֶת הַמַּלָּחִים לִסְעוּדָה בָּאַרְמוֹן, אֲבָל כְּשֶׁהֻגַּשׁ הָאֹכֶל, זִנְּקוּ מִכָּל פִּנָּה מֵאוֹת עַכְבָּרִים וְאָכְלוּ הַכֹּל. "אֵין לָנוּ שׁוּם דֶּרֶךְ לְהִפָּטֵר מֵהֶם," נֶאֱנַח הַמֶּלֶךְ. רַב הַחוֹבֵל הֵבִיא אֶת הַחֲתוּלָה שֶׁל דִּיק, וְתוֹךְ דַּקּוֹת סְפוּרוֹת בָּרְחוּ כָּל הָעַכְבָּרִים.

הַמֶּלֶךְ שָׂמַח כָּל כָּךְ, עַד שֶׁשִּׁלֵּם עַל הַחֲתוּלָה תֵּבָה מְלֵאָה זָהָב וַאֲבָנִים טוֹבוֹת — יוֹתֵר מִכָּל הַסְּחוֹרָה שֶׁעַל הַסְּפִינָה. כְּשֶׁחָזְרָה הַסְּפִינָה לְלוֹנְדוֹן, קָרָא הַסּוֹחֵר לְדִיק אֶל הַמִּשְׂרָד. "הָאוֹצָר הַזֶּה שֶׁלְּךָ," אָמַר לוֹ, "כִּי אֶת הַחֲתוּלָה אַתָּה שָׁלַחְתָּ."

דִּיק לֹא שָׁכַח אֶת מִי שֶׁעָזַר לוֹ. הוּא נָתַן מַתָּנוֹת לְכָל אַנְשֵׁי הַבַּיִת — אֲפִלּוּ לַטַּבָּחִית הָרַגְזָנִית. בַּפַּעַם הָרִאשׁוֹנָה בְּחַיָּיו הָלַךְ לַמִּסְפָּרָה וְקָנָה בְּגָדִים חֲדָשִׁים, וְאַחֲרֵי כַּמָּה שָׁנִים נָשָׂא לְאִשָּׁה אֶת אֶלִיס. בַּכֶּסֶף שֶׁלּוֹ בָּנָה מִרְפָּאָה לַחוֹלִים הָעֲנִיִּים. וּכְמוֹ שֶׁשָּׁרוּ הַפַּעֲמוֹנִים — הוּא נִבְחַר שָׁלוֹשׁ פְּעָמִים לְרֹאשׁ הָעִיר לוֹנְדוֹן.

עַל פִּי אַגָּדַת עַם אַנְגְּלִית, ״דִּיק וִיטִינְגְטוֹן וְהֶחָתוּל שֶׁלּוֹ״ (Dick Whittington and His Cat), בְּעִבּוּד חָפְשִׁי. רִיצַ׳רְד וִיטִינְגְטוֹן הָיָה אִישׁ אֲמִתִּי, שֶׁהָיָה רֹאשׁ הָעִיר לוֹנְדוֹן לִפְנֵי יוֹתֵר מִשֵּׁשׁ מֵאוֹת שָׁנָה; הַסִּפּוּר עַל הֶחָתוּל הוּא אַגָּדָה.

📚 דִּקְדּוּק — שְׁמוֹת מְקוֹמוֹת בְּמִשְׁקַל מִקְטָל וּמִקְטָלָה

מִקְטָל וּמִקְטָלָה — הַמָּקוֹם שֶׁבּוֹ עוֹשִׂים אֶת הַפְּעֻלָּה

בְּשִׁיעוּר 130 לָמַדְנוּ אֶת מִשְׁקַל קַטָּל — הָאָדָם שֶׁעוֹשֶׂה אֶת הַפְּעֻלָּה (טַבָּח, סַפָּר). בְּשִׁיעוּר 132 לָמַדְנוּ אֶת מִשְׁקַל מַקְטֵל — הַכְּלִי (מַפְתֵּחַ, מַגְרֵפָה). הַיּוֹם נִלְמַד מִשְׁקָל שֶׁל הַמָּקוֹם שֶׁבּוֹ עוֹשִׂים אֶת הַפְּעֻלָּה: מִקְטָל וּמִקְטָלָה.

אֵיךְ בּוֹנִים? מוֹסִיפִים מִ (עִם חִירִיק) לִפְנֵי הַשֹּׁרֶשׁ. הָאוֹת הָרִאשׁוֹנָה בַּשֹּׁרֶשׁ מְקַבֶּלֶת שְׁוָא, וְהָאוֹת הַשְּׁנִיָּה מְקַבֶּלֶת קָמָץ: ט־ב־ח ← מִטְבָּח. בְּמִקְטָלָה מוֹסִיפִים גַּם ־ָה בַּסּוֹף: ס־ע־ד ← מִסְעָדָה.

שִׂימוּ לֵב:

1. מִקְטָל הוּא בְּדֶרֶךְ כְּלָל שֵׁם בְּזָכָר (מִטְבָּח, מִשְׂרָד), וּמִקְטָלָה — שֵׁם בִּנְקֵבָה (מִסְעָדָה, מִרְפָּאָה).

2. כְּשֶׁהָאוֹת הַשְּׁנִיָּה בַּשֹּׁרֶשׁ הִיא בּ, כּ אוֹ פּ, הִיא מְקַבֶּלֶת דָּגֵשׁ (כִּי לְפָנֶיהָ יֵשׁ שְׁוָא): מִטְבָּח מִסְפָּרָה מִרְפָּאָה

3. כְּשֶׁהָאוֹת הָרִאשׁוֹנָה בַּשֹּׁרֶשׁ הִיא גְּרוֹנִית (ע אוֹ ח), הַ־מ׳ מְקַבֶּלֶת פַּתָּח בִּמְקוֹם חִירִיק: מַחְסָן (ח־ס־ן) מַעְבָּדָה (ע־ב־ד)

4. רַבִּים: הַקָּמָץ נִשְׁאָר. מִקְטָל ← ־ִים: מִטְבָּחִים מִשְׂרָדִים. מִקְטָלָה ← ־וֹת: מִסְעָדוֹת מִרְפָּאוֹת

5. שֹׁרֶשׁ אֶחָד — אָדָם וּמָקוֹם: הַטַּבָּח (מִי?) מְבַשֵּׁל בַּמִּטְבָּח (אֵיפֹה?). הַסַּפָּר (מִי?) עוֹבֵד בַּמִּסְפָּרָה (אֵיפֹה?).

מַלְכּוֹדוֹת: לֹא כָּל מִלָּה בְּמִשְׁקַל מִקְטָל הִיא מָקוֹם — מִכְתָּב מִשְׂחָק מִסְפָּר. וְלֹא כָּל מָקוֹם בָּנוּי בְּמִשְׁקָל הַזֶּה: חֲנוּת נָמָל אַרְמוֹן. גַּם הַסִּיּוֹמֶת ־ִיָּה מְצַיֶּנֶת לִפְעָמִים מָקוֹם, אֲבָל זוֹ דֶּרֶךְ אַחֶרֶת לִבְנוֹת שֵׁם מָקוֹם: סִפְרִיָּה מַאֲפִיָּה

טִיפּ: כְּדֵי לִמְצֹא אֶת הַשֹּׁרֶשׁ — מוֹחֲקִים אֶת הַ־מִ בַּהַתְחָלָה (וְאֶת הַ־ָה בַּסּוֹף): מִכְבָּסָה ← כ־ב־ס.

✏️ מִשְׂחַק דִּקְדּוּק — רְחוֹב הַסּוֹחֲרִים בְּלוֹנְדוֹן
🏘️ רְחוֹב הַסּוֹחֲרִים בְּלוֹנְדוֹן
עַל כָּל תְּשׁוּבָה נְכוֹנָה נִבְנֶה בָּרְחוֹב עוֹד בִּנְיָן — וְעָלָיו שֵׁם הַמָּקוֹם.
🏘️ הָרְחוֹב שֶׁל מַר פִיצְווֹרֶן
✍️ כְּתִיבָה
✍️ מַטְלַת כְּתִיבָה

דַּמְיְנוּ שֶׁדִּיק מַרְאֶה לְאֶלִיס אֶת לוֹנְדוֹן. כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים עִם שְׁמוֹת מְקוֹמוֹת בְּמִשְׁקַל מִקְטָל אוֹ מִקְטָלָה, וְהַרְאוּ בַּמִּשְׁפָּט מָה עוֹשִׂים בְּכָל מָקוֹם. דֻּגְמָה: בַּמִּכְבָּסָה מְכַבְּסִים אֶת הַסְּדִינִים שֶׁל כָּל הַבַּיִת.
