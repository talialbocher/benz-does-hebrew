שִׁלְגִּיָּה

שִׁלְגִּיָּה בּוֹרַחַת מִן הַמַּלְכָּה הַמְּקַנֵּאת אֶל הַיַּעַר, מוֹצֵאת מִקְלָט אֵצֶל שִׁבְעָה גַּמָּדִים - וְלוֹמְדִים לְהַכִּיר אֶת שֵׁם הַפֹּעַל

📖 קְרִיאָה

לִפְנֵי שָׁנִים רַבּוֹת חָיְתָה מַלְכָּה שֶׁיָּשְׁבָה לְיַד חַלּוֹן בְּיוֹם חֹרֶף מֻשְׁלָג, וְתָפְרָה בֶּגֶד לְתִינוֹקְתָּהּ הָעֲתִידִית. הִיא נִדְקְרָה בְּמַחַט, וּשְׁלוֹשָׁה טִפּוֹת דָּם אֲדֻמּוֹת נָפְלוּ עַל הַשֶּׁלֶג הַלָּבָן שֶׁמֵּעֵבֶר לַחַלּוֹן. "הַלְוַאי שֶׁתִּהְיֶה לִי בַּת," חָשְׁבָה, "עוֹר לָבָן כַּשֶּׁלֶג, שְׂפָתַיִם אֲדֻמּוֹת כַּדָּם, וְשֵׂעָר שָׁחֹר כְּעֵץ הַהַבְנֶה שֶׁל הַחַלּוֹן." מִשְׁאֲלָתָהּ הִתְגַּשְּׁמָה, וְהִיא קָרְאָה לְבִתָּהּ הַיָּפָה שִׁלְגִּיָּה. אַךְ הַמַּלְכָּה נִפְטְרָה זְמַן קָצָר אַחֲרֵי הַלֵּדָה.

הַמֶּלֶךְ נָשָׂא לְאִשָּׁה מַלְכָּה חֲדָשָׁה - יָפָה מְאֹד, אֲבָל גַּאֲוְתָנִית וּמְקַנֵּאת. לַמַּלְכָּה הַחֲדָשָׁה הָיְתָה מַרְאָה קְסוּמָה, וְהִיא נָהֲגָה לִשְׁאֹל אוֹתָהּ בְּכָל בֹּקֶר: "מַרְאָה, מַרְאָה עַל הַקִּיר, מִי הַיָּפָה בָּאָרֶץ כֻּלָּהּ?" וְהַמַּרְאָה תָּמִיד עָנְתָה: "אַתְּ, הַמַּלְכָּה, הַיָּפָה מִכֻּלָּן." הַמַּלְכָּה שָׂמְחָה לִשְׁמֹעַ זֹאת שׁוּב וָשׁוּב.

שִׁלְגִּיָּה גָּדְלָה וְהָפְכָה יָפָה יוֹתֵר וְיוֹתֵר בְּכָל שָׁנָה. יוֹם אֶחָד עָנְתָה הַמַּרְאָה תְּשׁוּבָה אַחֶרֶת: "אַתְּ יָפָה, הַמַּלְכָּה - אֲבָל שִׁלְגִּיָּה יָפָה מִמֵּךְ פִּי אֶלֶף." הַמַּלְכָּה הִתְמַלְּאָה קִנְאָה גְּדוֹלָה, וּמֵאוֹתוֹ הָרֶגַע לֹא יָכְלָה לִסְבֹּל אֶת הַנַּעֲרָה. הִיא קָרְאָה לְצַיָּד וְצִוְּתָה עָלָיו לְהוֹלִיךְ אֶת שִׁלְגִּיָּה עָמֹק לְתוֹךְ הַיַּעַר וְלֹא לְהַחְזִירָהּ הַבַּיְתָה לְעוֹלָם.

הַצַּיָּד הוֹבִיל אֶת שִׁלְגִּיָּה בֵּין הָעֵצִים, אֲבָל בָּרֶגַע הָאַחֲרוֹן לֹא הִצְלִיחַ לִפְגֹּעַ בָּהּ - לִבּוֹ הָיָה רַךְ מִדַּי. "בִּרְחִי מַהֵר וְאַל תַּחְזְרִי," לָחַשׁ לָהּ, "אֲנִי אֹמַר לַמַּלְכָּה שֶׁהַמְּלָאכָה נַעֲשְׂתָה." שִׁלְגִּיָּה רָצְתָה לִבְרֹחַ הֲכִי רָחוֹק שֶׁאֶפְשָׁר, וְרָצָה בֵּין הָעֵצִים הַגְּבוֹהִים עַד שֶׁנֶּעֶצְרָה מֵרֹב עֲיֵפוּת, פּוֹחֶדֶת וּלְבַדָּהּ בַּיַּעַר הָאָפֵל.

לְקַרַאת עֶרֶב הִצְלִיחָה שִׁלְגִּיָּה לִמְצֹא בַּיִת קָטָן וְנָעִים בֵּין הָעֵצִים. בְּתוֹכוֹ עָמְדוּ שֻׁלְחָן קָטָן עִם שִׁבְעָה כִּסְאוֹת זְעִירִים, וְשֶׁבַע מִטּוֹת קְטַנְטַנּוֹת בַּחֶדֶר הַפְּנִימִי. שִׁלְגִּיָּה, רְעֵבָה וְעֲיֵפָה, טָעֲמָה קְצָת מֵהָאֹכֶל שֶׁעַל הַשֻּׁלְחָן, וְהָיְתָה כֹּה עֲיֵפָה שֶׁלֹּא יָכְלָה אֶלָּא לִישׁוֹן - וְשָׁכְבָה עַל אַחַת הַמִּטּוֹת.

כְּשֶׁחָזְרוּ הַבַּיְתָה שִׁבְעָה גַּמָּדִים קְטַנִּים, שֶׁעָבְדוּ כָּל הַיּוֹם בְּמִכְרֶה עָמֹק בָּהָר, גִּלּוּ אֶת שִׁלְגִּיָּה יְשֵׁנָה בְּמִטַּת הַגַּמָּד הַשְּׁבִיעִי. הֵם הֵעִירוּ אוֹתָהּ בְּעֲדִינוּת, וְהִיא סִפְּרָה לָהֶם עַל הַמַּלְכָּה הָרָעָה וְעַל בְּרִיחָתָהּ. הַגַּמָּדִים רִחֲמוּ עָלֶיהָ וְהִצִּיעוּ לָהּ לְהִשָּׁאֵר: "תּוּכְלִי לָגוּר אִתָּנוּ וּלְבַשֵּׁל וּלְנַקּוֹת בִּשְׁבִילֵנוּ, אֲבָל אַל תִּפְתְּחִי אֶת הַדֶּלֶת לְאִישׁ כְּשֶׁאֲנַחְנוּ בַּמִּכְרֶה, וְהִשְׁתַּדְּלִי לְהִתְחַבֵּא אִם מִישֶׁהוּ זָר מִתְקָרֵב."

בֵּינְתַיִם, בָּאַרְמוֹן, שָׁאֲלָה הַמַּלְכָּה שׁוּב אֶת מַרְאָתָהּ - וּלְתַדְהֵמָתָהּ עָנְתָה הַמַּרְאָה שֶׁשִּׁלְגִּיָּה עֲדַיִן חַיָּה, וְגָרָה עִם שִׁבְעָה גַּמָּדִים מֵעֵבֶר לֶהָרִים. הַמַּלְכָּה הִשְׁתּוֹלְלָה מִזַּעַם, וְהֶחְלִיטָה לְטַפֵּל בָּעִנְיָן בְּעַצְמָהּ. הִיא הִתְחַפְּשָׂה לְאִשָּׁה זְקֵנָה וּמוֹכֶרֶת פֵּרוֹת, וְיָצְאָה בְּעַצְמָהּ אֶל הַיַּעַר.

הַזְּקֵנָה הָיְתָה בְּעֶצֶם הַמַּלְכָּה, וּבְיָדָהּ תַּפּוּחַ אָדֹם וְיָפֶה - חֶצְיוֹ מָרוּחַ בְּאֶרֶס קָטְלָנִי. הִיא דָּפְקָה בַּדֶּלֶת, וְשִׁלְגִּיָּה, שֶׁזָּכְרָה אֶת אַזְהָרַת הַגַּמָּדִים, סֵרְבָה בַּתְּחִלָּה לִפְתֹּחַ. אַךְ הַזְּקֵנָה הִצְלִיחָה לְפַתּוֹת אוֹתָהּ בְּחִיּוּךְ נָעִים וּבְמַתָּנָה - "רַק בּוֹאִי טַעֲמִי בֵּיסָה אַחַת, יַלְדָּתִי הַיָּפָה" - וְשִׁלְגִּיָּה, שֶׁלֹּא חָשְׁדָה בִּמְאוּמָה, נָשְׂכָה בַּתַּפּוּחַ וְנָפְלָה מִיָּד אַרְצָה, נְטוּלַת הַכָּרָה.

כְּשֶׁחָזְרוּ הַגַּמָּדִים מָצְאוּ אֶת שִׁלְגִּיָּה שׁוֹכֶבֶת עַל הָרִצְפָּה, וְלֹא הִצְלִיחוּ לְהָעִיר אוֹתָהּ בְּשׁוּם דֶּרֶךְ. עוֹרָהּ נִשְׁאַר וָרֹד וְיָפֶה כְּמוֹ בְּחַיִּים, וְלָכֵן לֹא הֵבִינוּ שֶׁמֵּתָה - הֵם הֶחְלִיטוּ לְהַנִּיחַ אוֹתָהּ בְּאֲרוֹן זְכוּכִית שָׁקוּף, כְּדֵי שֶׁיּוּכְלוּ לְהַמְשִׁיךְ לִרְאוֹת אֶת פָּנֶיהָ הַיָּפוֹת, וְהִצִּיבוּ אוֹתוֹ בְּפִסְגַּת הָהָר.

יוֹם אֶחָד עָבַר בָּאֵזוֹר נָסִיךְ צָעִיר, וְרָאָה אֶת שִׁלְגִּיָּה יְשֵׁנָה בְּתוֹךְ הָאָרוֹן הַשָּׁקוּף. הוּא הִתְאַהֵב בְּיָפְיָהּ מִיָּד, וְהֵבִין שֶׁהוּא חַיָּב לְנַסּוֹת לְהַצִּיל אוֹתָהּ - וּבִקֵּשׁ מֵהַגַּמָּדִים רְשׁוּת לָקַחַת אֶת הָאָרוֹן עִמּוֹ. בַּדֶּרֶךְ מָעֲדוּ מְשָׁרְתָיו, וְהָאָרוֹן זָז בְּטַלְטֵלָה - וְאָז נָפְלָה חֲתִיכַת הַתַּפּוּחַ הָאֲרוּרָה מִגְּרוֹנָהּ שֶׁל שִׁלְגִּיָּה. הִיא פָּקְחָה אֶת עֵינֶיהָ וְהִתְעוֹרְרָה, מֻפְתַּעַת לִמְצֹא אֶת עַצְמָהּ חַיָּה וְנָסִיךְ מִתְבּוֹנֵן בָּהּ בְּשִׂמְחָה.

שִׁלְגִּיָּה וְהַנָּסִיךְ הִתְחַתְּנוּ בְּחֲתֻנָּה גְּדוֹלָה, וְהִזְמִינוּ אֶת שִׁבְעַת הַגַּמָּדִים כְּאוֹרְחֵי כָּבוֹד. כְּשֶׁשָּׁמְעָה הַמַּלְכָּה הָרָעָה שֶׁשִּׁלְגִּיָּה חַיָּה וּמְאֻשֶּׁרֶת, הִתְבַּיְּשָׁה כָּל כָּךְ בְּמַעֲשֶׂיהָ שֶׁבָּרְחָה מִן הָאַרְמוֹן וְאִישׁ לֹא רָאָה אוֹתָהּ שׁוּב. שִׁלְגִּיָּה חָיְתָה בְּאֹשֶׁר, וְלֹא שָׁכְחָה לְעוֹלָם לְבַקֵּר אֶת חֲבֵרֶיהָ הַגַּמָּדִים בַּבַּיִת הַקָּטָן בַּיַּעַר.

מָה זֶה שֵׁם הַפֹּעַל?

שֵׁם הַפֹּעַל הוּא צוּרַת הַיְּסוֹד שֶׁל הַפֹּעַל - "הַשֵּׁם" שֶׁל הַפְּעֻלָּה עַצְמָהּ, לְמָשָׁל "לָרוּץ", "לִכְתֹּב", "לִישׁוֹן". בְּשׁוֹנֶה מִ"אֲנִי רָץ" אוֹ "הִיא כָּתְבָה", שֵׁם הַפֹּעַל לֹא אוֹמֵר מִי עוֹשֶׂה אֶת הַפְּעֻלָּה - הוּא רַק מְצַיֵּן מָה הַפְּעֻלָּה עַצְמָהּ. בְּעִבְרִית בּוֹנִים אוֹתוֹ עִם הָאוֹת ל בִּתְחִלַּת הַמִּלָּה, וְתַבְנִית קְבוּעָה שֶׁמַּתְאִימָה לַבִּנְיָן שֶׁל הַפֹּעַל.

יוֹצְאִים מִן הַכְּלָל — שֵׁמוֹת פֹּעַל בְּלִי תַּבְנִית קְבוּעָה

כַּמָּה פְּעָלִים נְפוֹצִים מְאֹד לֹא הוֹלְכִים לְפִי הַתַּבְנִית הָרְגִילָה, וְצָרִיךְ פָּשׁוּט לִזְכֹּר אוֹתָם:

הָלַךְ ← לָלֶכֶת בָּא ← לָבוֹא נָתַן ← לָתֵת לָקַח ← לָקַחַת רָאָה ← לִרְאוֹת יָדַע ← לָדַעַת עָשָׂה ← לַעֲשׂוֹת
🔑 כְּלָל זָהָב: שֵׁם הַפֹּעַל מוֹפִיעַ הַרְבֵּה אַחֲרֵי פְּעָלִים כְּמוֹ רוֹצֶה / יָכוֹל / צָרִיךְ / אוֹהֵב / מַתְחִיל / מַפְסִיק - לְמָשָׁל: "הִיא רָצְתָה לִבְרֹחַ", "הוּא יָכוֹל לְהַצִּיל אוֹתָהּ".
