📖 קְרִיאָה

בְּיוֹם קַיִץ חַם, אַחֲרֵי הַצָּהֳרַיִם, הָלַךְ שׁוּעָל רָעֵב בִּשְׁבִיל עָפָר צַר. הַשֶּׁמֶשׁ צָרְבָה אֶת גַּבּוֹ, אָבָק עָלָה מִתַּחַת לְכַפּוֹת רַגְלָיו, וּבִטְנוֹ קִרְקְרָה בְּלִי הֶפְסֵק.

פִּתְאוֹם הֵרִים הַשּׁוּעָל אֶת אַפּוֹ. רֵיחַ מָתוֹק וְחָזָק הִגִּיעַ אֵלָיו מֵעֵבֶר לַגָּדֵר. הוּא הֵצִיץ בֵּין הַקְּרָשִׁים וְרָאָה כֶּרֶם. עַל גֶּפֶן גְּדוֹלָה, בֵּין עָלִים רְחָבִים וִירֻקִּים, הִתְנַדְנְדוּ אֶשְׁכּוֹלוֹת שֶׁל עֲנָבִים סְגֻלִּים וּמַבְרִיקִים.

הָעֲנָבִים נִרְאוּ בְּשֵׁלִים וַעֲסִיסִיִּים. הַשֶּׁמֶשׁ נִצְנְצָה עַל הַקְּלִפּוֹת הַחֲלָקוֹת שֶׁלָּהֶם, וְנִדְמָה שֶׁעוֹד רֶגַע יַתְחִיל הַמִּיץ לִנְטֹף מִכָּל גַּרְגִּיר.

הַשּׁוּעָל הִתְגַּנֵּב דֶּרֶךְ חוֹר בַּגָּדֵר וְעָמַד מִתַּחַת לַגֶּפֶן. ״אֵיזֶה יוֹם בָּרוּךְ!״ לָחַשׁ וְלָקַק אֶת שְׂפָתָיו. אֲבָל הָאֶשְׁכּוֹל הַיָּפֶה בְּיוֹתֵר הָיָה תָּלוּי גָּבוֹהַּ, כִּשְׁנֵי מֶטְרִים מֵעַל הָאָרֶץ.

הַקְּפִיצָה הָרִאשׁוֹנָה הָיְתָה קְצָרָה מִדַּי, וְהָעֲנָבִים נִשְׁאֲרוּ רְחוֹקִים. בַּקְּפִיצָה הַשְּׁנִיָּה נָגַע הַשּׁוּעָל בְּעָלֶה, וְרַק רַשְׁרוּשׁ קָטָן נִשְׁמַע.

בַּקְּפִיצָה הַשְּׁלִישִׁית כִּמְעַט תָּפַס אֶת הָאֶשְׁכּוֹל, אֲבָל נָפַל עַל הָאֲדָמָה הַיְּבֵשָׁה וְהִרְגִּישׁ כְּאֵב חַד בְּכַף רַגְלוֹ.

הוּא נָשַׁם עָמֹק, הִסְתַּכֵּל שׁוּב עַל הַגַּרְגְּרִים הַמַּבְרִיקִים וְאָמַר לְעַצְמוֹ: ״לָמָּה לִי לְהִתְיָאֵשׁ? אֲנַסֶּה עוֹד פַּעַם!״ אֲבָל גַּם הַקְּפִיצָה הָרְבִיעִית לֹא עָזְרָה.

בַּסּוֹף יָשַׁב עַל הָאֲדָמָה, מֵזִיעַ וְעָיֵף, וְהַלָּשׁוֹן שֶׁלּוֹ הָיְתָה יְבֵשָׁה. הוּא הִסְתַּכֵּל עַל הָעֲנָבִים וְאָמַר בְּקוֹל רָם, כְּדֵי שֶׁכֻּלָּם יִשְׁמְעוּ: ״מִי בִּכְלָל רוֹצֶה אֶת הָעֲנָבִים הָאֵלֶּה? בֶּטַח הֵם חֲמוּצִים וְקָשִׁים!״

הוּא זָקַף אֶת הַזָּנָב, הִסְתּוֹבֵב וְהָלַךְ בַּשְּׁבִיל בְּלִי לְהַבִּיט לַאֲחוֹר. אֲבָל הָרֵיחַ הַמָּתוֹק עוֹד עָמַד בָּאֲוִיר, וּבִטְנוֹ עֲדַיִן קִרְקְרָה.

אֵיזוֹפּוֹס (יָוָן הָעַתִּיקָה), ״הַשּׁוּעָל וְהָעֲנָבִים״, בְּעִבּוּד חָפְשִׁי.

📚 דִּקְדּוּק — כְּתִיבַת תֵּאוּר

אֵיךְ גּוֹרְמִים לַקּוֹרֵא לִרְאוֹת אֶת מַה שֶּׁמְּתָאֲרִים?

הַסּוֹפֵר לֹא כָּתַב ״הַכֶּרֶם הָיָה יָפֶה״. הוּא כָּתַב: ״עַל גֶּפֶן גְּדוֹלָה, בֵּין עָלִים רְחָבִים וִירֻקִּים, הִתְנַדְנְדוּ אֶשְׁכּוֹלוֹת שֶׁל עֲנָבִים סְגֻלִּים וּמַבְרִיקִים״. הַמִּלָּה ״יָפֶה״ אוֹמֶרֶת מַה הַסּוֹפֵר חוֹשֵׁב, אֲבָל הַפְּרָטִים הַמּוּחָשִׁיִּים נוֹתְנִים לַקּוֹרֵא לִרְאוֹת בְּעַצְמוֹ. כָּךְ בּוֹנִים תֵּאוּר טוֹב: בּוֹחֲרִים פְּרָטִים מִן הַחוּשִׁים.

• מִלִּים כְּלָלִיּוֹת כְּמוֹ יָפֶה נֶחְמָד טוֹב נוֹרָא אוֹמְרוֹת מָה הַכּוֹתֵב חוֹשֵׁב. כְּדַאי לְהַחֲלִיף אוֹתָן בִּפְרָט שֶׁאֶפְשָׁר לִרְאוֹת אוֹ לִשְׁמֹעַ.

• הַתֹּאַר מִתְאִים לְשֵׁם הָעֶצֶם בְּמִין וּבְמִסְפָּר: גֶּפֶן גְּדוֹלָה עָלִים רְחָבִים קְלִפּוֹת חֲלָקוֹת

• אֵין צֹרֶךְ בְּעֹמֶס. שְׁלוֹשָׁה פְּרָטִים בְּרוּרִים, מֵחוּשִׁים שׁוֹנִים, טוֹבִים יוֹתֵר מֵעֶשְׂרִים תֹּאָרִים.

בַּחֲרוּ מָקוֹם שֶׁאַתֶּם מַכִּירִים הֵיטֵב (הַמִּטְבָּח בַּבַּיִת, הַחָצֵר, הַפַּארְק, כִּתָּה). כִּתְבוּ עָלָיו תֵּאוּר בְּחָמֵשׁ שׁוּרוֹת, שׁוּרָה לְכָל חוּשׁ. הִשְׁתַּמְּשׁוּ בִּפְרָטִים מוּחָשִׁיִּים וְלֹא בְּמִלִּים כְּלָלִיּוֹת. לְמָשָׁל: ״בַּמִּטְבָּח עוֹמֶדֶת קְעָרָה שֶׁל תַּפּוּחִים אֲדֻמִּים וּמַבְרִיקִים.״
