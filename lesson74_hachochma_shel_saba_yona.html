הַחָכְמָה שֶׁל סָבָא יוֹנָה

עִידוֹ עוֹזֵר לְסָבָא לְתַקֵּן סַפְסָל, וְלוֹמֵד פִּתְגָּמִים יְשָׁנִים

📖 קְרִיאָה

בְּכָל יוֹם שִׁשִּׁי בָּא עִידוֹ אֶל בֵּית סָבָא יוֹנָה בַּכְּפָר, וְהַפַּעַם הֵם תִּכְנְנוּ לְתַקֵּן יַחַד אֶת הַסַּפְסָל הַיָּשָׁן בַּגִּנָּה. עִידוֹ הִתְרַגֵּשׁ מְאֹד, כִּי הוּא אוֹהֵב לִבְנוֹת דְּבָרִים בַּיָּדַיִם.

כְּשֶׁהֵבִיא סָבָא יוֹנָה אֶת תֵּבַת הַכֵּלִים, עִידוֹ רָצָה לְהַתְחִיל לְנַסֵּר מִיָּד, בְּלִי לִמְדֹּד כְּלוּם. סָבָא יוֹנָה עָצַר אוֹתוֹ בְּחִיּוּךְ וְאָמַר: "סוֹף מַעֲשֶׂה בְּמַחְשָׁבָה תְּחִלָּה" — קֹדֶם נְתַכְנֵן עַל הַנְּיָר, וְרַק אַחַר כָּךְ נַתְחִיל לְנַסֵּר.

בִּזְמַן הָעֲבוֹדָה הִגִּיעָה הַשְּׁכֵנָה וְסִפְּרָה שֶׁשָּׁמְעָה שְׁמוּעָה: יֶלֶד מֵהָרְחוֹב לָקַח עוּגִיּוֹת מֵהַמִּטְבָּח שֶׁל בֵּית הַסֵּפֶר בְּלִי רְשׁוּת. כְּשֶׁבָּדְקוּ, הִתְבָּרֵר שֶׁהַיֶּלֶד בֶּאֱמֶת אָכַל עוּגִיָּה אַחַת. "רוֹאִים?" אָמַר סָבָא יוֹנָה, "אֵין עָשָׁן בְּלִי אֵשׁ" — תָּמִיד יֵשׁ קְצָת אֱמֶת מֵאֲחוֹרֵי שְׁמוּעָה.

כְּשֶׁהִגִּיעַ הַזְּמַן לְסַדֵּר אֶת הַכֵּלִים בַּחֲזָרָה, עִידוֹ הִתְחַמֵּק וּבִקֵּשׁ מֵאֲחוֹתוֹ הַקְּטַנָּה לְסַדֵּר בִּמְקוֹמוֹ. סָבָא יוֹנָה הִבְחִין בַּדָּבָר וְאָמַר בְּרֹגֶז קַל: "אַתָּה טָמַנְתָּ אֶת יָדְךָ בַּצַּלַּחַת — אַל תַּשְׁאִיר אֶת הָעֲבוֹדָה לַאֲחֵרִים." עִידוֹ הִתְבַּיֵּשׁ וְחָזַר לְסַדֵּר בְּעַצְמוֹ.

בַּצָּהֳרַיִם סִפֵּר סָבָא יוֹנָה שֶׁהוּא הִבְטִיח לַשְּׁכֵנָה לַעֲזֹר לָהּ לְהָזִיז אֲרוֹן כָּבֵד, אֲבָל בַּסּוֹף לֹא הִגִּיעַ וְלֹא הִתְקַשֵּׁר לְהַסְבִּיר. "זֶה הָיָה מִן הַשָּׂפָה וְלַחוּץ," אָמַר סָבָא בַּחֲרָטָה, "הִבְטַחְתִּי בְּמִלִּים, אֲבָל לֹא עָמַדְתִּי בְּהַבְטָחָתִי בַּמַּעֲשֶׂה."

עִידוֹ שָׁאַל אֶת סָבָא אִם הוּא בֶּאֱמֶת שָׁכַח, אוֹ פָּשׁוּט לֹא רָצָה לָבוֹא. סָבָא יוֹנָה לֹא עָנָה דָּבָר, רַק הִשְׁפִּיל אֶת מַבָּטוֹ. "שְׁתִיקָה כְּהוֹדָאָה," אָמַר עִידוֹ בְּחָכְמָה, "אַתָּה לֹא מַכְחִישׁ, אָז כְּבָר אֲנִי יוֹדֵעַ אֶת הָאֱמֶת."

לִפְנֵי הַשְּׁקִיעָה הֵם חָזְרוּ לַעֲבוֹד עַל הַסַּפְסָל, וְעִידוֹ רָצָה לְהַפְסִיק כְּשֶׁנִּשְׁאֲרוּ רַק שְׁנֵי בְּרָגִים לְחַבֵּר. סָבָא יוֹנָה חִיֵּךְ וְאָמַר: "כָּל הַמַּתְחִיל בְּמִצְוָה אוֹמְרִים לוֹ גְּמוֹר" — נִשְׁאַר רַק קְצָת, וְנִסְגֹּר אֶת הָעֲבוֹדָה כְּמוֹ שֶׁצָּרִיךְ.

כְּשֶׁסִּיְּמוּ, יָשְׁבוּ שְׁנֵיהֶם עַל הַסַּפְסָל הֶחָדָשׁ וְצָפוּ בַּשְּׁקִיעָה הָאָדֻמָּה. עִידוֹ הֵבִין שֶׁכָּל הַפִּתְגָּמִים שֶׁסָּבָא אָמַר בְּמֶשֶׁךְ הַיּוֹם הֵם חָכְמָה יְשָׁנָה שֶׁעוֹבֶרֶת מִדּוֹר לְדוֹר.

"סָבָא," אָמַר עִידוֹ בְּשֶׁקֶט, "אֲנִי רוֹצֶה לִזְכֹּר אֶת כָּל הַפִּתְגָּמִים שֶׁלָּמַדְתִּי הַיּוֹם." סָבָא יוֹנָה חִבֵּק אוֹתוֹ וְאָמַר: "אִם תִּזְכֹּר אֲפִלּוּ אֶחָד מֵהֶם בְּרֶגַע הַנָּכוֹן — זֶה יִהְיֶה מַתָּנָה גְּדוֹלָה מִמֶּנִּי לְךָ."

📚 דִּקְדּוּק — נִיבִים וּפִתְגָּמִים

מָה הַהֶבְדֵּל בֵּין נִיב לְפִתְגָּם?

נִיב הוּא צֵרוּף מִלִּים קָבוּעַ שֶׁמְּתָאֵר פְּעֻלָּה אוֹ הַרְגָּשָׁה, וּמַשְׁמָעוּתוֹ אֵינָהּ מִלּוּלִית — לְמָשָׁל "הֵרִים יָדַיִם" (וִתֵּר). פִּתְגָּם הוּא מִשְׁפָּט שָׁלֵם שֶׁמְּלַמֵּד עֲצָה אוֹ חָכְמַת חַיִּים, וְעָבַר מִדּוֹר לְדוֹר.

כִּתְבוּ חֲמִשָּׁה מִשְׁפָּטִים שְׁלֵמִים. בְּכָל מִשְׁפָּט הִשְׁתַּמְּשׁוּ בְּפִתְגָּם אֶחָד מֵהַשִּׁיעוּר, כְּמוֹ בַּדֻּגְמָה: לִפְנֵי הַמִּבְחָן יָשַׁבְתִּי וְתִכְנַנְתִּי מָה לִלְמֹד, כִּי סוֹף מַעֲשֶׂה בְּמַחְשָׁבָה תְּחִלָּה.
